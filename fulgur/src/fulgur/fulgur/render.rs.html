28impl Render for Fulgur {
29    /// Render the Fulgur instance
30    ///
31    /// ### Arguments
32    /// - `window`: The window to render the Fulgur instance in
33    /// - `cx`: The application context
34    ///
35    /// ### Returns
36    /// - `impl IntoElement`: The rendered Fulgur instance
37    fn render(&mut self, window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
38        self.process_pending_initial_active_tab_activation(window, cx);
39        self.process_window_state_updates(window, cx);
40        Self::process_update_notifications(window, cx);
41        self.process_pending_files_from_macos(window, cx);
42        #[cfg(target_os = "windows")]
43        self.process_pending_ipc_commands(window, cx);
44        self.process_shared_files_from_sync(window, cx);
45        self.process_pending_remote_files(window, cx);
46        self.process_pending_share_sheet(window, cx);
47        if self.tabs.is_empty() {
48            self.active_tab_id = None;
49        }
50        self.handle_pending_transfer_scroll(window, cx);
51        self.handle_pending_tab_transfer(window, cx);
52        self.handle_pending_tab_removal(window, cx);
53        self.handle_pending_jump_to_line(window, cx);
54        self.refresh_window_title(cx);
55        let active_tab_index = self.active_tab_index(cx);
56        let app_content = self.build_app_content_with_actions(active_tab_index, window, cx);
57        self.assemble_ui_tree(app_content, window, cx)
58    }
59}

62    /// Process update notifications from the background update checker
63    ///
64    /// ### Arguments
65    /// - `window`: The window to display the notification in
66    /// - `cx`: The application context
67    fn process_update_notifications(window: &mut Window, cx: &mut Context<Self>) {
68        let update_info = {
69            let shared = Fulgur::shared_state(cx);
70            shared.update_info.lock().take()
71        };
72        if let Some(update_info) = update_info {
73            let notification = make_update_notification(&update_info);
74            window.push_notification(notification, cx);
75        }
76    }

160    /// Assemble the final UI tree with all layers
161    ///
162    /// ### Arguments
163    /// - `app_content`: The main content area (from `build_app_content_with_actions()`)
164    /// - `window`: The window to assemble the UI for
165    /// - `cx`: The application context
166    ///
167    /// ### Returns
168    /// - `impl IntoElement`: The complete UI tree ready to be rendered
169    fn assemble_ui_tree(
170        &self,
171        app_content: impl IntoElement,
172        window: &mut Window,
173        cx: &mut Context<Self>,
174    ) -> impl IntoElement {
175        // Create root layout: TitleBar OUTSIDE of focus-tracked content
176        // This is critical for Windows hit-testing to work!
177        let root_content = v_flex()
178            .size_full()
179            .child(self.title_bar.clone())
180            .child(app_content);
181        let mut root = div()
182            .size_full()
183            .child(root_content)
184            .children(Root::render_sheet_layer(window, cx))
185            .children(Root::render_notification_layer(window, cx))
186            .children(Root::render_dialog_layer(window, cx));
187        if let Some((position, menu)) = self
188            .editor_context_menu
189            .as_ref()
190            .map(|(pos, menu)| (*pos, menu.clone()))
191        {
192            root = root.child(
193                gpui::deferred(
194                    gpui::anchored()
195                        .position(position)
196                        .snap_to_window_with_margin(px(8.))
197                        .anchor(Anchor::TopLeft)
198                        .child(
199                            div()
200                                .font_family(cx.theme().font_family.clone())
201                                .cursor_default()
202                                .child(menu),
203                        ),
204                )
205                .with_priority(1),
206            );
207        }
208        root
209    }

211    /// Activate the initially restored tab after the first render pass.
212    ///
213    /// The first render builds the window's `Root` layers. Startup flows that can open
214    /// dialogs (like remote password prompts triggered by `set_active_tab`) must wait
215    /// until that initial render has completed.
216    ///
217    /// ### Arguments
218    /// - `window`: The window context
219    /// - `cx`: The application context
220    fn process_pending_initial_active_tab_activation(
221        &mut self,
222        window: &mut Window,
223        cx: &mut Context<Self>,
224    ) {
225        if !self.has_rendered_once {
226            self.has_rendered_once = true;
227            if self.pending_initial_active_tab.is_some() {
228                cx.notify();
229            }
230            return;
231        }
232
233        if let Some(tab_id) = self.pending_initial_active_tab.take()
234            && let Some(index) = self.tab_index_of(tab_id, cx)
235        {
236            self.set_active_tab(index, window, cx);
237        }
238    }
