20    /// Load app state from disk and restore tabs
21    ///
22    /// ### Arguments
23    /// - `window`: The window to load the state from
24    /// - `cx`: The application context
25    /// - `window_index`: Index of the window state to restore (0 = first window, etc.)
26    pub fn load_state(&mut self, window: &mut Window, cx: &mut Context<Self>, window_index: usize) {
27        log::debug!("Loading application state for window {window_index}...");
28        // Temporarily disable indent guides during restoration to prevent crash
29        let original_indent_guides = self.settings.editor_settings.show_indent_guides;
30        self.settings.editor_settings.show_indent_guides = false;
31        let window_state = cx
32            .global::<crate::fulgur::shared_state::SharedAppState>()
33            .restore_state
34            .lock()
35            .as_ref()
36            .and_then(|ws| ws.windows.get(window_index).cloned());
37        if let Some(window_state) = window_state {
38            log::debug!(
39                "Restoring {} tabs from startup snapshot",
40                window_state.tabs.len()
41            );
42            self.tabs.clear();
43            self.pending_remote_restore.clear();
44            self.inflight_remote_restore.clear();
45            let mut tab_id = TabId(0);
46            for tab_state in &window_state.tabs {
47                let tab = self.restore_tab_from_state(tab_state.clone(), tab_id, window, cx);
48                if let Some(editor_tab) = tab {
49                    self.tabs.push(Tab::Editor(editor_tab).into_entity(cx));
50                    tab_id = tab_id.next();
51                }
52            }
53            let saved_active_editor_id: Option<TabId> = window_state
54                .active_tab_index
55                .and_then(|idx| self.tabs.get(idx))
56                .and_then(|t| t.read(cx).as_editor())
57                .map(|et| et.id);
58            self.next_tab_id = tab_id;
59            self.insert_preview_tabs_for_markdown(cx);
60            self.active_tab_id =
61                saved_active_editor_id.or_else(|| self.tabs.first().map(|t| t.read(cx).id()));
62
63            let active_log_tab_id = self
64                .active_tab(cx)
65                .and_then(Tab::as_editor)
66                .filter(|editor| editor.log_view)
67                .map(|editor| editor.id);
68            if let Some(active_id) = active_log_tab_id {
69                self.activate_log_view(active_id, window, cx);
70            }
71
72            cx.notify();
73        } else {
74            log::warn!("No saved state for window {window_index}, starting fresh");
75        }
76        if self.tabs.is_empty() {
77            log::debug!("No tabs restored, creating initial empty tab");
78            let initial_tab = Tab::Editor(EditorTab::new(
79                TabId(0),
80                UNTITLED,
81                window,
82                cx,
83                &self.settings.editor_settings,
84            ));
85            self.active_tab_id = Some(initial_tab.id());
86            self.tabs.push(initial_tab.into_entity(cx));
87            self.next_tab_id = TabId(1);
88        }
89        self.settings.editor_settings.show_indent_guides = original_indent_guides;
90        if original_indent_guides {
91            // Indent guides were disabled during restoration; push the real
92            // settings into every restored tab once this update cycle is done.
93            cx.defer_in(window, |this, window, cx| {
94                this.apply_editor_settings_to_tabs(window, cx);
95            });
96        }
97    }

99    /// Restore a single tab from saved state:
100    ///
101    /// - If a file exists, it will be loaded from the file.
102    /// - If a file exists and was modified but unsaved in the last state save and not modified after externally, it'll be loaded from the saved state
103    /// - If a file does not exist, the saved content will be used.
104    /// - If no path and no content is provided, the tab will be skipped.
105    ///
106    /// ### Arguments
107    /// - `tab_state`: The saved state of the tab
108    /// - `tab_id`: The ID to assign to this tab (based on position)
109    /// - `window`: The window to restore the tab to
110    /// - `cx`: The application context
111    ///
112    /// ### Returns
113    /// - `Some(EditorTab)`: The restored tab
114    /// - `None`: If the tab could not be restored
115    fn restore_tab_from_state(
116        &mut self,
117        tab_state: TabState,
118        tab_id: TabId,
119        window: &mut Window,
120        cx: &mut App,
121    ) -> Option<EditorTab> {
122        log::debug!("Restoring tab: {}", tab_state.title);
123
124        let color_tag = tab_state.color_tag.as_deref().and_then(ColorTag::from_key);
125        let file_exists = tab_state.file_path.as_ref().is_some_and(|p| p.exists());
126        let file_modified_time = tab_state
127            .file_path
128            .as_ref()
129            .and_then(get_file_modified_time);
130        let mut readable_file = tab_state
131            .file_path
132            .as_ref()
133            .and_then(|path| fs::File::open(path).ok());
134        let can_read_file = readable_file.is_some();
135        let decision = determine_tab_restore_strategy(
136            tab_state.file_path.clone(),
137            tab_state.remote.clone(),
138            tab_state.content.clone(),
139            tab_state.last_saved,
140            file_exists,
141            file_modified_time,
142            can_read_file,
143        );
144        let (content, path, encoding, is_modified, lossy_decode) = match decision {
145            TabRestoreDecision::RestoreRemote { remote, content } => {
146                let is_modified = content.is_some();
147                let restored_content = content.unwrap_or_default();
148                let mut tab = EditorTab::from_remote_loaded(
149                    tab_id,
150                    RemoteFileResult {
151                        spec: remote.to_remote_spec(),
152                        file_size: restored_content.len(),
153                        content: restored_content,
154                        encoding: UTF_8.to_string(),
155                        lossy: false,
156                    },
157                    window,
158                    cx,
159                    &self.settings.editor_settings,
160                );
161                tab.modified = is_modified;
162                tab.color_tag = color_tag;
163                self.pending_remote_restore.insert(tab_id);
164                return Some(tab);
165            }
166            TabRestoreDecision::LoadFromFile { path } => {
167                let mut bytes = Vec::new();
168                let mut file = readable_file.take()?;
169                file.read_to_end(&mut bytes).ok()?;
170                let decoded = detect_encoding_and_decode(bytes);
171                (
172                    decoded.content,
173                    Some(path),
174                    decoded.encoding,
175                    false,
176                    decoded.lossy,
177                )
178            }
179            TabRestoreDecision::UseSavedContentWithPath { path, content } => {
180                (content, Some(path), UTF_8.to_string(), true, false)
181            }
182            TabRestoreDecision::UseSavedContentNoPath { content } => {
183                (content, None, UTF_8.to_string(), true, false)
184            }
185            TabRestoreDecision::Skip => return None,
186        };
187        let mut tab = if let Some(file_path) = path {
188            let mut tab = EditorTab::from_file(
189                FromFileParams {
190                    id: tab_id,
191                    path: file_path,
192                    contents: content,
193                    encoding,
194                    is_modified,
195                },
196                window,
197                cx,
198                &self.settings.editor_settings,
199            );
200            tab.lossy_decode = lossy_decode;
201            tab
202        } else {
203            let language = language_from_content(&tab_state.title, &content);
204            let large_file = crate::fulgur::ui::tabs::editor_tab::is_large_file(content.len());
205            // In large-file mode, substitute a language with no registered grammar so the background tree-sitter parser never runs.
206            let language_name = if large_file {
207                language_registry_name(&SupportedLanguage::Plain)
208            } else {
209                language_registry_name(&language)
210            };
211            let (csv_view_mode, csv_delimiter) =
212                crate::fulgur::ui::tabs::editor_tab::initial_csv_state(language, &content);
213            let content_entity = cx.new(|cx| {
214                gpui_component::input::InputState::new(window, cx)
215                    .code_editor(language_name)
216                    .line_number(self.settings.editor_settings.show_line_numbers)
217                    .indent_guides(self.settings.editor_settings.show_indent_guides)
218                    .tab_size(TabSize {
219                        tab_size: self.settings.editor_settings.tab_size,
220                        hard_tabs: false,
221                    })
222                    .soft_wrap(self.settings.editor_settings.soft_wrap && !large_file)
223                    .show_whitespaces(self.settings.editor_settings.show_whitespaces)
224                    .default_value(content)
225            });
226            EditorTab {
227                id: tab_id,
228                title: tab_state.title.into(),
229                content: content_entity,
230                location: TabLocation::Untitled,
231                modified: true,
232                original_content_hash:
233                    crate::fulgur::ui::tabs::editor_tab::content_fingerprint_from_str("").0,
234                original_content_len: 0,
235                encoding: "UTF-8".to_string(),
236                lossy_decode: false,
237                language,
238                show_markdown_toolbar: self
239                    .settings
240                    .editor_settings
241                    .markdown_settings
242                    .show_markdown_toolbar,
243                show_markdown_preview: self
244                    .settings
245                    .editor_settings
246                    .markdown_settings
247                    .show_markdown_preview,
248                file_size_bytes: None,
249                file_last_modified: None,
250                large_file,
251                csv_view_mode,
252                csv_delimiter,
253                csv_table: None,
254                csv_table_source_hash: 0,
255                color_tag: None,
256                log_view: false,
257                log_follow: true,
258                log_full: false,
259                log_content: None,
260                content_subscription: None,
261            }
262        };
263
264        let opens_as_log = tab
265            .file_path()
266            .is_some_and(|path| crate::fulgur::ui::log_view::opens_as_log_by_default(path));
267        if tab_state.log_view || opens_as_log {
268            tab.log_view = true;
269        }
270        tab.color_tag = color_tag;
271        Some(tab)
272    }
