20    /// Load app state from disk and restore tabs
21    ///
22    /// ### Arguments
23    /// - `window`: The window to load the state from
24    /// - `cx`: The application context
25    /// - `window_index`: Index of the window state to restore (0 = first window, etc.)
26    pub fn load_state(&mut self, window: &mut Window, cx: &mut Context<Self>, window_index: usize) {
27        log::debug!("Loading application state for window {window_index}...");
28        // Temporarily disable indent guides during restoration to prevent crash
29        let original_indent_guides = self.settings.editor_settings.show_indent_guides;
30        self.settings.editor_settings.show_indent_guides = false;
31        let window_state = cx
32            .global::<crate::fulgur::shared_state::SharedAppState>()
33            .restore_state
34            .lock()
35            .as_ref()
36            .and_then(|ws| ws.windows.get(window_index).cloned());
37        if let Some(window_state) = window_state {
38            log::debug!(
39                "Restoring {} tabs from startup snapshot",
40                window_state.tabs.len()
41            );
42            self.tabs.clear();
43            self.pending_remote_restore.clear();
44            self.inflight_remote_restore.clear();
45            // Restored tabs keep the identity they were persisted under, so the
46            // first save of the session updates their rows in place rather than
47            // deleting and reinserting every buffer under a fresh identity.
48            let mut highest_tab_id = 0;
49            for tab_state in &window_state.tabs {
50                let tab_id = TabId(tab_state.tab_id);
51                let tab = self.restore_tab_from_state(tab_state.clone(), tab_id, window, cx);
52                if let Some(editor_tab) = tab {
53                    self.tabs.push(Tab::Editor(editor_tab).into_entity(cx));
54                    highest_tab_id = highest_tab_id.max(tab_state.tab_id);
55                }
56            }
57            let saved_active_editor_id: Option<TabId> = window_state
58                .active_tab_index
59                .and_then(|idx| self.tabs.get(idx))
60                .and_then(|t| t.read(cx).as_editor())
61                .map(|et| et.id);
62            self.next_tab_id = TabId(highest_tab_id.saturating_add(1));
63            self.insert_preview_tabs_for_markdown(cx);
64            self.active_tab_id =
65                saved_active_editor_id.or_else(|| self.tabs.first().map(|t| t.read(cx).id()));
66
67            let active_log_tab_id = self
68                .active_tab(cx)
69                .and_then(Tab::as_editor)
70                .filter(|editor| editor.log_view)
71                .map(|editor| editor.id);
72            if let Some(active_id) = active_log_tab_id {
73                self.activate_log_view(active_id, window, cx);
74            }
75
76            cx.notify();
77        } else {
78            log::warn!("No saved state for window {window_index}, starting fresh");
79        }
80        if self.tabs.is_empty() {
81            log::debug!("No tabs restored, creating initial empty tab");
82            let initial_tab = Tab::Editor(EditorTab::new(
83                TabId(0),
84                UNTITLED,
85                window,
86                cx,
87                &self.settings.editor_settings,
88            ));
89            self.active_tab_id = Some(initial_tab.id());
90            self.tabs.push(initial_tab.into_entity(cx));
91            self.next_tab_id = TabId(1);
92        }
93        self.settings.editor_settings.show_indent_guides = original_indent_guides;
94        if original_indent_guides {
95            // Indent guides were disabled during restoration; push the real
96            // settings into every restored tab once this update cycle is done.
97            cx.defer_in(window, |this, window, cx| {
98                this.apply_editor_settings_to_tabs(window, cx);
99            });
100        }
101    }

103    /// Restore a single tab from saved state:
104    ///
105    /// - If a file exists, it will be loaded from the file.
106    /// - If a file exists and was modified but unsaved in the last state save and not modified after externally, it'll be loaded from the saved state
107    /// - If a file does not exist, the saved content will be used.
108    /// - If no path and no content is provided, the tab will be skipped.
109    ///
110    /// ### Arguments
111    /// - `tab_state`: The saved state of the tab
112    /// - `tab_id`: The ID to assign to this tab (based on position)
113    /// - `window`: The window to restore the tab to
114    /// - `cx`: The application context
115    ///
116    /// ### Returns
117    /// - `Some(EditorTab)`: The restored tab
118    /// - `None`: If the tab could not be restored
119    fn restore_tab_from_state(
120        &mut self,
121        tab_state: TabState,
122        tab_id: TabId,
123        window: &mut Window,
124        cx: &mut App,
125    ) -> Option<EditorTab> {
126        log::debug!("Restoring tab: {}", tab_state.title);
127
128        let color_tag = tab_state.color_tag.as_deref().and_then(ColorTag::from_key);
129        let file_exists = tab_state.file_path.as_ref().is_some_and(|p| p.exists());
130        let file_modified_time = tab_state
131            .file_path
132            .as_ref()
133            .and_then(get_file_modified_time);
134        let mut readable_file = tab_state
135            .file_path
136            .as_ref()
137            .and_then(|path| fs::File::open(path).ok());
138        let can_read_file = readable_file.is_some();
139        let decision = determine_tab_restore_strategy(
140            tab_state.file_path.clone(),
141            tab_state.remote.clone(),
142            tab_state.content.map(TabContent::into_string),
143            tab_state.last_saved,
144            file_exists,
145            file_modified_time,
146            can_read_file,
147        );
148        let (content, path, encoding, is_modified, lossy_decode) = match decision {
149            TabRestoreDecision::RestoreRemote { remote, content } => {
150                let is_modified = content.is_some();
151                let restored_content = content.unwrap_or_default();
152                let mut tab = EditorTab::from_remote_loaded(
153                    tab_id,
154                    RemoteFileResult {
155                        spec: remote.to_remote_spec(),
156                        file_size: restored_content.len(),
157                        content: restored_content,
158                        encoding: UTF_8.to_string(),
159                        lossy: false,
160                    },
161                    window,
162                    cx,
163                    &self.settings.editor_settings,
164                );
165                tab.modified = is_modified;
166                tab.color_tag = color_tag;
167                self.pending_remote_restore.insert(tab_id);
168                return Some(tab);
169            }
170            TabRestoreDecision::LoadFromFile { path } => {
171                let mut bytes = Vec::new();
172                let mut file = readable_file.take()?;
173                file.read_to_end(&mut bytes).ok()?;
174                let decoded = detect_encoding_and_decode(bytes);
175                (
176                    decoded.content,
177                    Some(path),
178                    decoded.encoding,
179                    false,
180                    decoded.lossy,
181                )
182            }
183            TabRestoreDecision::UseSavedContentWithPath { path, content } => {
184                (content, Some(path), UTF_8.to_string(), true, false)
185            }
186            TabRestoreDecision::UseSavedContentNoPath { content } => {
187                (content, None, UTF_8.to_string(), true, false)
188            }
189            TabRestoreDecision::Skip => return None,
190        };
191        let mut tab = if let Some(file_path) = path {
192            let mut tab = EditorTab::from_file(
193                FromFileParams {
194                    id: tab_id,
195                    path: file_path,
196                    contents: content,
197                    encoding,
198                    is_modified,
199                },
200                window,
201                cx,
202                &self.settings.editor_settings,
203            );
204            tab.lossy_decode = lossy_decode;
205            tab
206        } else {
207            let language = language_from_content(&tab_state.title, &content);
208            let large_file = crate::fulgur::ui::tabs::editor_tab::is_large_file(content.len());
209            // In large-file mode, substitute a language with no registered grammar so the background tree-sitter parser never runs.
210            let language_name = if large_file {
211                language_registry_name(&SupportedLanguage::Plain)
212            } else {
213                language_registry_name(&language)
214            };
215            let (csv_view_mode, csv_delimiter) =
216                crate::fulgur::ui::tabs::editor_tab::initial_csv_state(language, &content);
217            let content_entity = cx.new(|cx| {
218                gpui_component::input::EditorState::new(window, cx)
219                    .language(language_name)
220                    .line_number(self.settings.editor_settings.show_line_numbers)
221                    .indent_guides(self.settings.editor_settings.show_indent_guides)
222                    .tab_size(TabSize {
223                        tab_size: self.settings.editor_settings.tab_size,
224                        hard_tabs: false,
225                    })
226                    .soft_wrap(self.settings.editor_settings.soft_wrap && !large_file)
227                    .show_whitespaces(self.settings.editor_settings.show_whitespaces)
228                    .default_value(content)
229            });
230            EditorTab {
231                id: tab_id,
232                title: tab_state.title.into(),
233                content: content_entity,
234                location: TabLocation::Untitled,
235                modified: true,
236                original_content_hash:
237                    crate::fulgur::ui::tabs::editor_tab::content_fingerprint_from_str("").0,
238                original_content_len: 0,
239                encoding: "UTF-8".to_string(),
240                lossy_decode: false,
241                language,
242                show_markdown_toolbar: self
243                    .settings
244                    .editor_settings
245                    .markdown_settings
246                    .show_markdown_toolbar,
247                show_markdown_preview: self
248                    .settings
249                    .editor_settings
250                    .markdown_settings
251                    .show_markdown_preview,
252                file_size_bytes: None,
253                file_last_modified: None,
254                large_file,
255                csv_view_mode,
256                csv_delimiter,
257                csv_table: None,
258                csv_table_source_hash: 0,
259                color_tag: None,
260                log_view: false,
261                log_follow: true,
262                log_full: false,
263                log_content: None,
264                highlight_colors: self.settings.editor_settings.highlight_colors && !large_file,
265                content_subscription: None,
266            }
267        };
268
269        let opens_as_log = tab
270            .file_path()
271            .is_some_and(|path| crate::fulgur::ui::log_view::opens_as_log_by_default(path));
272        if tab_state.log_view || opens_as_log {
273            tab.log_view = true;
274        }
275        tab.color_tag = color_tag;
276        Some(tab)
277    }
