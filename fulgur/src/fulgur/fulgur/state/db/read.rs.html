1//! Reading the persisted session back into an in-memory snapshot.
2
3use super::StateDb;
4use super::paths::path_from_bytes;
5use crate::fulgur::state::persistence::{
6    SerializedRemoteSpec, SerializedWindowBounds, TabContent, TabState, WindowState, WindowsState,
7};
8use anyhow::anyhow;
9use rusqlite::Row;
10
11impl StateDb {
12    /// Read every persisted window, in restore order, with its tabs.
13    ///
14    /// ### Errors
15    /// - Returns an error if a query fails or if a stored row cannot be decoded.
16    ///
17    /// ### Returns
18    /// - `Ok(WindowsState)`: The persisted session, empty when nothing is stored
19    /// - `Err(anyhow::Error)`: The session could not be read
20    pub fn load(&self) -> anyhow::Result<WindowsState> {
21        let mut window_stmt = self
22            .conn
23            .prepare(
24                "SELECT id, active_tab_id, bounds_state, bounds_x, bounds_y, bounds_width,
25                        bounds_height, display_id
26                 FROM windows
27                 ORDER BY position, id",
28            )
29            .map_err(|e| anyhow!("Failed to prepare the window query: {e}"))?;
30        let windows: Vec<(i64, Option<i64>, SerializedWindowBounds)> = window_stmt
31            .query_map([], |row| {
32                Ok((row.get(0)?, row.get(1)?, window_bounds_from_row(row)?))
33            })
34            .map_err(|e| anyhow!("Failed to query persisted windows: {e}"))?
35            .collect::<Result<Vec<_>, _>>()
36            .map_err(|e| anyhow!("Failed to decode a persisted window: {e}"))?;
37
38        let mut state = WindowsState {
39            windows: Vec::with_capacity(windows.len()),
40        };
41        for (window_id, active_tab_id, window_bounds) in windows {
42            let tabs = self.load_tabs(window_id)?;
43            let active_tab_index = active_tab_id.and_then(|active| {
44                let active = u64::try_from(active).ok()?;
45                tabs.iter().position(|tab| tab.tab_id == active)
46            });
47            state.windows.push(WindowState {
48                window_id,
49                tabs,
50                active_tab_index,
51                window_bounds,
52            });
53        }
54        Ok(state)
55    }
56
57    /// Read the tabs of one window, in display order.
58    ///
59    /// ### Arguments
60    /// - `window_id`: Identity of the owning window
61    ///
62    /// ### Errors
63    /// - Returns an error if the query fails or a row cannot be decoded.
64    ///
65    /// ### Returns
66    /// - `Ok(Vec<TabState>)`: The window's tabs in display order
67    /// - `Err(anyhow::Error)`: The tabs could not be read
68    fn load_tabs(&self, window_id: i64) -> anyhow::Result<Vec<TabState>> {
69        let mut stmt = self
70            .conn
71            .prepare(
72                "SELECT id, title, file_path, content, last_saved, log_view, color_tag,
73                        remote_host, remote_port, remote_user, remote_path
74                 FROM tabs
75                 WHERE window_id = ?1
76                 ORDER BY position, id",
77            )
78            .map_err(|e| anyhow!("Failed to prepare the tab query: {e}"))?;
79        let tabs = stmt
80            .query_map([window_id], tab_state_from_row)
81            .map_err(|e| anyhow!("Failed to query persisted tabs: {e}"))?
82            .collect::<Result<Vec<_>, _>>()
83            .map_err(|e| anyhow!("Failed to decode a persisted tab: {e}"))?;
84        Ok(tabs)
85    }
86}

88/// Decode the window bounds columns of a window row.
89///
90/// ### Arguments
91/// - `row`: The window row, positioned on the bounds columns
92///
93/// ### Errors
94/// - Returns a `rusqlite::Error` if a column cannot be read.
95///
96/// ### Returns
97/// - `Ok(SerializedWindowBounds)`: The decoded bounds
98/// - `Err(rusqlite::Error)`: A column could not be read
99fn window_bounds_from_row(row: &Row<'_>) -> rusqlite::Result<SerializedWindowBounds> {
100    Ok(SerializedWindowBounds {
101        state: row.get(2)?,
102        x: row.get(3)?,
103        y: row.get(4)?,
104        width: row.get(5)?,
105        height: row.get(6)?,
106        display_id: row.get(7)?,
107    })
108}

110/// Decode one tab row into a `TabState`.
111///
112/// ### Arguments
113/// - `row`: The tab row
114///
115/// ### Errors
116/// - Returns a `rusqlite::Error` if a column cannot be read.
117///
118/// ### Returns
119/// - `Ok(TabState)`: The decoded tab
120/// - `Err(rusqlite::Error)`: A column could not be read
121fn tab_state_from_row(row: &Row<'_>) -> rusqlite::Result<TabState> {
122    let stored_id: i64 = row.get(0)?;
123    let file_path: Option<Vec<u8>> = row.get(2)?;
124    let content: Option<String> = row.get(3)?;
125    let remote_host: Option<String> = row.get(7)?;
126    let remote = match remote_host {
127        Some(host) => Some(SerializedRemoteSpec {
128            host,
129            port: row.get(8)?,
130            user: row.get::<_, Option<String>>(9)?.unwrap_or_default(),
131            path: row.get::<_, Option<String>>(10)?.unwrap_or_default(),
132        }),
133        None => None,
134    };
135    Ok(TabState {
136        tab_id: u64::try_from(stored_id).unwrap_or_default(),
137        title: row.get(1)?,
138        file_path: file_path.as_deref().map(path_from_bytes),
139        content: content.map(TabContent::Text),
140        last_saved: row.get(4)?,
141        remote,
142        log_view: row.get(5)?,
143        color_tag: row.get(6)?,
144    })
145}
