1//! Connection pragmas and versioned schema migrations for the state database.
2
3use anyhow::anyhow;
4use rusqlite::{Connection, TransactionBehavior};
5
6/// Schema version this build expects. Bumping it requires appending a step to
7/// `MIGRATIONS`; the existing steps must never be edited.
8pub const SCHEMA_VERSION: i64 = 1;
9
10/// Ordered schema migrations. Index `n` upgrades `user_version` from `n` to
11/// `n + 1`, so a fresh database runs every step in order.
12const MIGRATIONS: &[&str] = &[include_str!("migrations/001_initial.sql")];
13
14/// How long a connection waits for a lock held by another connection.
15const BUSY_TIMEOUT_MS: i64 = 5_000;
16
17/// Apply the connection-scoped pragmas the state store relies on.
18///
19/// ### Description
20/// `busy_timeout` is set first, before any statement that takes a lock. Switching
21/// a database to WAL needs a brief exclusive lock, and without a busy handler
22/// already installed `SQLITE_BUSY` is returned immediately instead of being
23/// waited out, which happens whenever two processes open a not-yet-WAL database
24/// at the same time.
25///
26/// ### Arguments
27/// - `conn`: The connection to configure
28///
29/// ### Errors
30/// - Returns an error if any pragma other than `journal_mode` cannot be applied.
31///   Failing to reach WAL is only logged, because the store stays correct in the
32///   rollback-journal mode the database already has, and giving up here would
33///   cost the whole session its persistence.
34///
35/// ### Returns
36/// - `Ok(())`: The connection is configured
37/// - `Err(anyhow::Error)`: A pragma could not be applied
38pub fn apply_pragmas(conn: &Connection) -> anyhow::Result<()> {
39    conn.pragma_update(None, "busy_timeout", BUSY_TIMEOUT_MS)
40        .map_err(|e| anyhow!("Failed to set busy timeout: {e}"))?;
41    // `journal_mode` returns the resulting mode as a row, so it needs a query.
42    match conn.query_row("PRAGMA journal_mode = WAL", [], |row| {
43        row.get::<_, String>(0)
44    }) {
45        Ok(mode) if mode.eq_ignore_ascii_case("wal") => {}
46        Ok(mode) => log::warn!("State database journal mode is '{mode}' instead of WAL"),
47        Err(e) => log::warn!("Failed to enable WAL journal mode, keeping the current one: {e}"),
48    }
49    conn.pragma_update(None, "synchronous", "NORMAL")
50        .map_err(|e| anyhow!("Failed to set synchronous mode: {e}"))?;
51    conn.pragma_update(None, "foreign_keys", true)
52        .map_err(|e| anyhow!("Failed to enable foreign key enforcement: {e}"))?;
53    // Checkpoint every ~4 MB of WAL so the sidecar cannot outgrow the database
54    // itself in a long-running process.
55    conn.pragma_update(None, "wal_autocheckpoint", 1000)
56        .map_err(|e| anyhow!("Failed to set WAL autocheckpoint: {e}"))?;
57    Ok(())
58}

60/// Read the schema version recorded in the database.
61///
62/// ### Arguments
63/// - `conn`: The connection to query
64///
65/// ### Errors
66/// - Returns an error if `PRAGMA user_version` cannot be read.
67///
68/// ### Returns
69/// - `Ok(i64)`: The recorded version, `0` for a database that has never been migrated
70/// - `Err(anyhow::Error)`: The version could not be read
71pub fn schema_version(conn: &Connection) -> anyhow::Result<i64> {
72    conn.query_row("PRAGMA user_version", [], |row| row.get(0))
73        .map_err(|e| anyhow!("Failed to read schema version: {e}"))
74}
75
76/// Bring the database up to `SCHEMA_VERSION`, running every pending step in one
77/// transaction.
78///
79/// ### Arguments
80/// - `conn`: The connection to migrate
81///
82/// ### Errors
83/// - Returns an error if the write lock cannot be taken, if the version cannot be
84///   read, if the database was written by a newer build, or if a migration step fails.
85///
86/// ### Returns
87/// - `Ok(bool)`: `true` when at least one migration step ran
88/// - `Err(anyhow::Error)`: The database could not be migrated
89pub fn migrate(conn: &mut Connection) -> anyhow::Result<bool> {
90    let tx = conn
91        .transaction_with_behavior(TransactionBehavior::Immediate)
92        .map_err(|e| anyhow!("Failed to lock the state database for migration: {e}"))?;
93    let current = schema_version(&tx)?;
94    if current > SCHEMA_VERSION {
95        return Err(anyhow!(
96            "State database schema version {current} is newer than the supported version {SCHEMA_VERSION}"
97        ));
98    }
99    if current == SCHEMA_VERSION {
100        return Ok(false);
101    }
102    let pending = usize::try_from(current)
103        .map_err(|e| anyhow!("Invalid schema version {current} in state database: {e}"))?;
104    for (offset, migration) in MIGRATIONS.iter().enumerate().skip(pending) {
105        let version = i64::try_from(offset + 1)
106            .map_err(|e| anyhow!("Migration index {offset} does not fit a schema version: {e}"))?;
107        log::info!("Migrating state database to schema version {version}");
108        tx.execute_batch(migration).map_err(|e| {
109            anyhow!("Failed to migrate state database to schema version {version}: {e}")
110        })?;
111        tx.pragma_update(None, "user_version", version)
112            .map_err(|e| anyhow!("Failed to record schema version {version}: {e}"))?;
113    }
114    tx.commit()
115        .map_err(|e| anyhow!("Failed to commit the state database migration: {e}"))?;
116    Ok(true)
117}

119#[cfg(test)]
120mod tests {
121    use super::{SCHEMA_VERSION, apply_pragmas, migrate, schema_version};
122    use rusqlite::Connection;
123
124    fn migrated_connection() -> Connection {
125        let mut conn = Connection::open_in_memory().expect("open in-memory database");
126        apply_pragmas(&conn).expect("apply pragmas");
127        migrate(&mut conn).expect("migrate");
128        conn
129    }
130
131    #[test]
132    fn migration_count_matches_the_declared_schema_version() {
133        assert_eq!(
134            super::MIGRATIONS.len(),
135            usize::try_from(SCHEMA_VERSION).expect("schema version fits usize"),
136            "every schema version must have exactly one migration step"
137        );
138    }
139
140    #[test]
141    fn fresh_database_is_migrated_to_the_current_version() {
142        let conn = migrated_connection();
143        assert_eq!(schema_version(&conn).expect("read version"), SCHEMA_VERSION);
144    }
145
146    #[test]
147    fn migrating_an_up_to_date_database_is_a_no_op() {
148        let mut conn = migrated_connection();
149        assert!(!migrate(&mut conn).expect("second migrate"));
150    }
151
152    #[test]
153    fn expected_tables_exist_after_migration() {
154        let conn = migrated_connection();
155        for table in ["windows", "tabs"] {
156            let count: i64 = conn
157                .query_row(
158                    "SELECT count(*) FROM sqlite_master WHERE type = 'table' AND name = ?1",
159                    [table],
160                    |row| row.get(0),
161                )
162                .expect("query sqlite_master");
163            assert_eq!(count, 1, "table {table} must exist");
164        }
165    }
166
167    #[test]
168    fn tables_are_declared_strict() {
169        let conn = migrated_connection();
170        // A STRICT table rejects a value whose type does not match the column.
171        conn.execute_batch(
172            "INSERT INTO windows (id, position, bounds_state, bounds_x, bounds_y, bounds_width, bounds_height)
173             VALUES (1, 0, 'Windowed', 0.0, 0.0, 100.0, 100.0);",
174        )
175        .expect("insert window");
176        let result = conn.execute(
177            "INSERT INTO tabs (window_id, id, position, title, log_view) VALUES (1, 1, 'not-an-integer', 'x', 0)",
178            [],
179        );
180        assert!(
181            result.is_err(),
182            "a STRICT table must reject a text value in an INTEGER column"
183        );
184    }
185
186    #[test]
187    fn deleting_a_window_cascades_to_its_tabs() {
188        let conn = migrated_connection();
189        conn.execute_batch(
190            "INSERT INTO windows (id, position, bounds_state, bounds_x, bounds_y, bounds_width, bounds_height)
191             VALUES (1, 0, 'Windowed', 0.0, 0.0, 100.0, 100.0);
192             INSERT INTO tabs (window_id, id, position, title, log_view) VALUES (1, 0, 0, 'a.txt', 0);",
193        )
194        .expect("seed window and tab");
195        conn.execute("DELETE FROM windows WHERE id = 1", [])
196            .expect("delete window");
197        let tabs: i64 = conn
198            .query_row("SELECT count(*) FROM tabs", [], |row| row.get(0))
199            .expect("count tabs");
200        assert_eq!(tabs, 0, "tabs must be removed with their window");
201    }
202
203    #[test]
204    fn a_newer_schema_version_is_refused_instead_of_downgraded() {
205        let mut conn = Connection::open_in_memory().expect("open in-memory database");
206        apply_pragmas(&conn).expect("apply pragmas");
207        conn.pragma_update(None, "user_version", SCHEMA_VERSION + 1)
208            .expect("set future version");
209        assert!(migrate(&mut conn).is_err());
210    }
