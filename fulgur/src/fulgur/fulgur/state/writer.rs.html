8/// Bounded channel capacity for pending save requests.
9///
10/// Kept small because writes are serialized through a single thread and blocking
11/// callers typically wait on the reply before issuing the next save. A full queue
12/// indicates an abnormal backlog and makes callers back off naturally.
13const CHANNEL_CAPACITY: usize = 16;
14
15/// Maximum time a dropped `StateWriter` waits for the worker to flush queued
16/// saves before detaching it.
17const WRITER_JOIN_TIMEOUT: Duration = Duration::from_secs(5);
18
19/// Minimum delay between two consecutive asynchronous state writes.
20const SAVE_THROTTLE: Duration = Duration::from_millis(200);
21
22/// A request sent from a UI thread to the writer thread: a serialized snapshot,
23/// the destination path, and an optional reply channel the writer uses to report
24/// the I/O result.
25struct WriteRequest {
26    state: WindowsState,
27    path: PathBuf,
28    reply: Option<mpsc::Sender<anyhow::Result<()>>>,
29}
30
31/// An asynchronous request held back by the throttle until its window elapses.
32struct PendingWrite {
33    state: WindowsState,
34    path: PathBuf,
35    deadline: Instant,
36}
37
38/// Writer-thread bookkeeping for throttling and de-duplicating writes.
39struct WriterState {
40    throttle: Duration,
41    pending: Option<PendingWrite>,
42    last_write_at: Option<Instant>,
43    last_written: Option<(PathBuf, [u8; 32])>,
44}
45
46impl WriterState {
47    /// Create the bookkeeping for a writer thread.
48    ///
49    /// ### Arguments
50    /// - `throttle`: Minimum delay between two consecutive asynchronous writes.
51    ///
52    /// ### Returns
53    /// - `Self`: Bookkeeping with no pending request and no write history.
54    fn new(throttle: Duration) -> Self {
55        Self {
56            throttle,
57            pending: None,
58            last_write_at: None,
59            last_written: None,
60        }
61    }
62
63    /// Persist a snapshot unless it is identical to the last one written.
64    ///
65    /// ### Arguments
66    /// - `state`: The snapshot to persist.
67    /// - `path`: Destination file path.
68    ///
69    /// ### Errors
70    /// - Returns an error if the snapshot cannot be serialized or written.
71    ///
72    /// ### Returns
73    /// - `Ok(())`: The file holds the snapshot, whether it was just written or
74    ///   already up to date.
75    /// - `Err(anyhow::Error)`: Serialization or I/O failed.
76    fn write(&mut self, state: &WindowsState, path: &Path) -> anyhow::Result<()> {
77        let result = self.write_if_changed(state, path);
78        self.last_write_at = Some(Instant::now());
79        result
80    }
81
82    /// Serialize, compare against the last write, and persist when it differs.
83    ///
84    /// ### Arguments
85    /// - `state`: The snapshot to persist.
86    /// - `path`: Destination file path.
87    ///
88    /// ### Errors
89    /// - Returns an error if the snapshot cannot be serialized or written.
90    ///
91    /// ### Returns
92    /// - `Ok(())`: The file holds the snapshot.
93    /// - `Err(anyhow::Error)`: Serialization or I/O failed.
94    fn write_if_changed(&mut self, state: &WindowsState, path: &Path) -> anyhow::Result<()> {
95        let json = state.to_json()?;
96        let digest: [u8; 32] = Sha256::digest(json.as_bytes()).into();
97        let unchanged = self
98            .last_written
99            .as_ref()
100            .is_some_and(|(written_path, written_digest)| {
101                written_path.as_path() == path && *written_digest == digest
102            });
103        if unchanged {
104            log::debug!("State unchanged since the last write, skipping");
105            return Ok(());
106        }
107        WindowsState::write_json_to_path(&json, path)?;
108        self.last_written = Some((path.to_path_buf(), digest));
109        Ok(())
110    }
111
112    /// Write the held-back request, if there is one, and clear it.
113    fn flush_pending(&mut self) {
114        let Some(pending) = self.pending.take() else {
115            return;
116        };
117        if let Err(e) = self.write(&pending.state, &pending.path) {
118            log::error!("State writer failed to save state: {e}");
119        }
120    }
121
122    /// Process one request from the channel.
123    ///
124    /// ### Arguments
125    /// - `request`: The request to process.
126    fn handle(&mut self, request: WriteRequest) {
127        let Some(reply) = request.reply else {
128            self.enqueue_throttled(request.state, request.path);
129            return;
130        };
131        if self
132            .pending
133            .as_ref()
134            .is_some_and(|pending| pending.path == request.path)
135        {
136            self.pending = None;
137        } else {
138            self.flush_pending();
139        }
140        let result = self.write(&request.state, &request.path);
141        if let Err(ref e) = result {
142            log::error!("State writer failed to save state: {e}");
143        }
144        if reply.send(result).is_err() {
145            log::warn!("State writer reply channel dropped before result was read");
146        }
147    }
148
149    /// Hold an asynchronous request back, or write it if the throttle allows.
150    ///
151    /// ### Arguments
152    /// - `state`: The snapshot to persist.
153    /// - `path`: Destination file path.
154    fn enqueue_throttled(&mut self, state: WindowsState, path: PathBuf) {
155        if let Some(pending) = self.pending.as_mut() {
156            if pending.path == path {
157                pending.state = state;
158                return;
159            }
160            self.flush_pending();
161        }
162        let ready_at = self.last_write_at.map(|last| last + self.throttle);
163        match ready_at {
164            Some(deadline) if deadline > Instant::now() => {
165                self.pending = Some(PendingWrite {
166                    state,
167                    path,
168                    deadline,
169                });
170            }
171            _ => {
172                if let Err(e) = self.write(&state, &path) {
173                    log::error!("State writer failed to save state: {e}");
174                }
175            }
176        }
177    }
178}

180/// Dedicated background writer that serializes all `WindowsState` persistence.
181pub struct StateWriter {
182    sender: mpsc::SyncSender<WriteRequest>,
183    _worker: Worker,
184}
185
186impl StateWriter {
187    /// Spawn the writer thread and return a handle for submitting save requests.
188    ///
189    /// ### Returns
190    /// - `Self`: A writer handle whose `save_blocking` method dispatches work to
191    ///   the worker thread.
192    ///
193    /// ### Panics
194    /// Panics if the OS refuses to spawn the background writer thread.
195    #[must_use]
196    pub fn new() -> Self {
197        Self::with_throttle(SAVE_THROTTLE)
198    }
199
200    /// Spawn the writer thread with an explicit throttle period.
201    ///
202    /// ### Arguments
203    /// - `throttle`: Minimum delay between two consecutive asynchronous writes.
204    ///
205    /// ### Returns
206    /// - `Self`: A writer handle bound to the freshly spawned worker thread.
207    fn with_throttle(throttle: Duration) -> Self {
208        let (sender, receiver) = mpsc::sync_channel::<WriteRequest>(CHANNEL_CAPACITY);
209        let worker = Worker::spawn(
210            "fulgur-state-writer",
211            WRITER_JOIN_TIMEOUT,
212            move |_shutdown| {
213                Self::run(&receiver, throttle);
214            },
215        );
216        Self {
217            sender,
218            _worker: worker,
219        }
220    }
221
222    /// Worker-thread loop that processes save requests one at a time.
223    ///
224    /// ### Arguments
225    /// - `receiver`: Channel of pending write requests.
226    /// - `throttle`: Minimum delay between two consecutive asynchronous writes.
227    fn run(receiver: &mpsc::Receiver<WriteRequest>, throttle: Duration) {
228        let mut writer_state = WriterState::new(throttle);
229        loop {
230            let next_request = match writer_state.pending.as_ref().map(|p| p.deadline) {
231                Some(deadline) => {
232                    let wait = deadline.saturating_duration_since(Instant::now());
233                    match receiver.recv_timeout(wait) {
234                        Ok(request) => Some(request),
235                        Err(mpsc::RecvTimeoutError::Timeout) => None,
236                        Err(mpsc::RecvTimeoutError::Disconnected) => break,
237                    }
238                }
239                None => match receiver.recv() {
240                    Ok(request) => Some(request),
241                    Err(mpsc::RecvError) => break,
242                },
243            };
244            match next_request {
245                Some(request) => writer_state.handle(request),
246                None => writer_state.flush_pending(),
247            }
248        }
249        writer_state.flush_pending();
250        log::debug!("State writer thread exiting (no more senders)");
251    }
252
253    /// Enqueue a snapshot and block the caller until the writer has persisted it.
254    ///
255    /// ### Description
256    /// Guarantees FIFO ordering with all other save requests: any save started
257    /// before this call completes before this one begins, and any save issued
258    /// after this call returns sees the result of this save on disk. Blocking
259    /// saves are never throttled, and they supersede any asynchronous save still
260    /// held back for the same path, which is by construction older.
261    ///
262    /// A snapshot identical to the last one written to `path` is reported as
263    /// persisted without touching the file, since the file already holds it.
264    ///
265    /// ### Arguments
266    /// - `state`: The fully-assembled windows state snapshot to persist.
267    /// - `path`: Destination file path (typically the user config `state.json`).
268    ///
269    /// ### Errors
270    /// Returns an error if the writer thread has exited, if the reply channel is
271    /// dropped, or if the underlying write reports an I/O or serialization error.
272    ///
273    /// ### Returns
274    /// - `Ok(())`: The writer successfully persisted the snapshot.
275    /// - `Err(anyhow::Error)`: The writer reported an I/O or serialization error,
276    ///   or the writer thread has exited before the request could be processed.
277    pub fn save_blocking(&self, state: WindowsState, path: PathBuf) -> anyhow::Result<()> {
278        let (reply_tx, reply_rx) = mpsc::channel();
279        self.sender
280            .send(WriteRequest {
281                state,
282                path,
283                reply: Some(reply_tx),
284            })
285            .map_err(|_| anyhow::anyhow!("state writer thread has exited"))?;
286        reply_rx
287            .recv()
288            .map_err(|_| anyhow::anyhow!("state writer reply channel closed before result"))?
289    }
290
291    /// Enqueue a snapshot without waiting for the write to complete.
292    ///
293    /// ### Arguments
294    /// - `state`: The fully-assembled windows state snapshot to persist.
295    /// - `path`: Destination file path (typically the user config `state.json`).
296    pub fn save_async(&self, state: WindowsState, path: PathBuf) {
297        if self
298            .sender
299            .send(WriteRequest {
300                state,
301                path,
302                reply: None,
303            })
304            .is_err()
305        {
306            log::error!("State writer thread has exited; dropped async save request");
307        }
308    }
309}
