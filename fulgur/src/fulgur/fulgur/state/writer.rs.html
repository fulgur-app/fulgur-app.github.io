9/// Bounded channel capacity for writer messages.
10///
11/// Asynchronous snapshots never travel through this channel: they are coalesced
12/// in the `SnapshotMailbox`, so the queue only ever holds blocking requests,
13/// whose callers wait on the reply before issuing another one, plus zero-sized
14/// wakeups. A full queue therefore indicates an abnormal backlog of blocking
15/// saves and makes those callers back off naturally.
16const CHANNEL_CAPACITY: usize = 16;
17
18/// Maximum time a dropped `StateWriter` waits for the worker to flush queued
19/// saves before detaching it.
20const WRITER_JOIN_TIMEOUT: Duration = Duration::from_secs(5);
21
22/// Minimum delay between two consecutive asynchronous state writes.
23const SAVE_THROTTLE: Duration = Duration::from_millis(200);
24
25/// A message sent from a UI thread to the writer thread.
26enum WriterMessage {
27    /// A snapshot whose caller blocks until the writer reports the I/O result.
28    Blocking {
29        state: WindowsState,
30        path: PathBuf,
31        reply: mpsc::Sender<anyhow::Result<()>>,
32    },
33    /// Signals that the mailbox now holds a snapshot waiting to be written.
34    MailboxFilled,
35}
36
37/// A snapshot waiting in the mailbox, together with its destination.
38struct PendingWrite {
39    state: WindowsState,
40    path: PathBuf,
41}
42
43/// Coalescing hand-off for asynchronous snapshots.
44#[derive(Default)]
45struct SnapshotMailbox {
46    pending: Mutex<Vec<PendingWrite>>,
47}
48
49impl SnapshotMailbox {
50    /// Store a snapshot, replacing any snapshot already waiting for the same path.
51    ///
52    /// ### Arguments
53    /// - `state`: The snapshot to hand to the writer thread.
54    /// - `path`: Destination file path.
55    fn put(&self, state: WindowsState, path: PathBuf) {
56        let mut pending = self.pending.lock();
57        match pending.iter_mut().find(|entry| entry.path == path) {
58            Some(existing) => existing.state = state,
59            None => pending.push(PendingWrite { state, path }),
60        }
61    }
62
63    /// Drop the snapshot waiting for `path`, if any.
64    ///
65    /// ### Arguments
66    /// - `path`: Destination whose waiting snapshot is superseded.
67    fn discard(&self, path: &Path) {
68        self.pending.lock().retain(|entry| entry.path != path);
69    }
70
71    /// Take every waiting snapshot, oldest destination first.
72    ///
73    /// ### Returns
74    /// - `Vec<PendingWrite>`: The waiting snapshots, leaving the mailbox empty.
75    fn drain(&self) -> Vec<PendingWrite> {
76        std::mem::take(&mut *self.pending.lock())
77    }
78
79    /// Report whether at least one snapshot is waiting.
80    ///
81    /// ### Returns
82    /// - `bool`: `true` when the mailbox holds a snapshot to write.
83    fn has_pending(&self) -> bool {
84        !self.pending.lock().is_empty()
85    }
86}

88/// Writer-thread bookkeeping for throttling and de-duplicating writes.
89struct WriterState {
90    throttle: Duration,
91    mailbox: Arc<SnapshotMailbox>,
92    last_write_at: Option<Instant>,
93    last_written: Option<(PathBuf, [u8; 32])>,
94}
95
96impl WriterState {
97    /// Create the bookkeeping for a writer thread.
98    ///
99    /// ### Arguments
100    /// - `throttle`: Minimum delay between two consecutive asynchronous writes.
101    /// - `mailbox`: Shared mailbox asynchronous snapshots are handed through.
102    ///
103    /// ### Returns
104    /// - `Self`: Bookkeeping with no write history.
105    fn new(throttle: Duration, mailbox: Arc<SnapshotMailbox>) -> Self {
106        Self {
107            throttle,
108            mailbox,
109            last_write_at: None,
110            last_written: None,
111        }
112    }
113
114    /// Persist a snapshot unless it is identical to the last one written.
115    ///
116    /// ### Arguments
117    /// - `state`: The snapshot to persist.
118    /// - `path`: Destination file path.
119    ///
120    /// ### Errors
121    /// - Returns an error if the snapshot cannot be serialized or written.
122    ///
123    /// ### Returns
124    /// - `Ok(())`: The file holds the snapshot, whether it was just written or
125    ///   already up to date.
126    /// - `Err(anyhow::Error)`: Serialization or I/O failed.
127    fn write(&mut self, state: &WindowsState, path: &Path) -> anyhow::Result<()> {
128        let result = self.write_if_changed(state, path);
129        self.last_write_at = Some(Instant::now());
130        result
131    }
132
133    /// Serialize, compare against the last write, and persist when it differs.
134    ///
135    /// ### Arguments
136    /// - `state`: The snapshot to persist.
137    /// - `path`: Destination file path.
138    ///
139    /// ### Errors
140    /// - Returns an error if the snapshot cannot be serialized or written.
141    ///
142    /// ### Returns
143    /// - `Ok(())`: The file holds the snapshot.
144    /// - `Err(anyhow::Error)`: Serialization or I/O failed.
145    fn write_if_changed(&mut self, state: &WindowsState, path: &Path) -> anyhow::Result<()> {
146        let json = state.to_json()?;
147        let digest: [u8; 32] = Sha256::digest(json.as_bytes()).into();
148        let unchanged = self
149            .last_written
150            .as_ref()
151            .is_some_and(|(written_path, written_digest)| {
152                written_path.as_path() == path && *written_digest == digest
153            });
154        if unchanged {
155            log::debug!("State unchanged since the last write, skipping");
156            return Ok(());
157        }
158        WindowsState::write_json_to_path(&json, path)?;
159        self.last_written = Some((path.to_path_buf(), digest));
160        Ok(())
161    }
162
163    /// Compute the earliest time the mailbox may be flushed.
164    ///
165    /// ### Returns
166    /// - `Some(Instant)`: A snapshot is waiting; the value is when the throttle
167    ///   allows writing it, which is in the past when it may be written now.
168    /// - `None`: The mailbox is empty, so there is nothing to wait for.
169    fn mailbox_deadline(&self) -> Option<Instant> {
170        if !self.mailbox.has_pending() {
171            return None;
172        }
173        Some(
174            self.last_write_at
175                .map_or_else(Instant::now, |last| last + self.throttle),
176        )
177    }
178
179    /// Write every snapshot waiting in the mailbox, ignoring the throttle.
180    fn flush_mailbox(&mut self) {
181        for pending in self.mailbox.drain() {
182            if let Err(e) = self.write(&pending.state, &pending.path) {
183                log::error!("State writer failed to save state: {e}");
184            }
185        }
186    }
187
188    /// Write a snapshot whose caller is waiting for the result.
189    ///
190    /// ### Arguments
191    /// - `state`: The snapshot to persist.
192    /// - `path`: Destination file path.
193    /// - `reply`: Channel the I/O result is reported on.
194    fn handle_blocking(
195        &mut self,
196        state: &WindowsState,
197        path: &Path,
198        reply: &mpsc::Sender<anyhow::Result<()>>,
199    ) {
200        self.mailbox.discard(path);
201        self.flush_mailbox();
202        let result = self.write(state, path);
203        if let Err(ref e) = result {
204            log::error!("State writer failed to save state: {e}");
205        }
206        if reply.send(result).is_err() {
207            log::warn!("State writer reply channel dropped before result was read");
208        }
209    }
210}

212/// Dedicated background writer that serializes all `WindowsState` persistence.
213pub struct StateWriter {
214    sender: mpsc::SyncSender<WriterMessage>,
215    mailbox: Arc<SnapshotMailbox>,
216    _worker: Worker,
217}
218
219impl StateWriter {
220    /// Spawn the writer thread and return a handle for submitting save requests.
221    ///
222    /// ### Returns
223    /// - `Self`: A writer handle whose `save_blocking` method dispatches work to
224    ///   the worker thread.
225    ///
226    /// ### Panics
227    /// Panics if the OS refuses to spawn the background writer thread.
228    #[must_use]
229    pub fn new() -> Self {
230        Self::with_throttle(SAVE_THROTTLE)
231    }
232
233    /// Spawn the writer thread with an explicit throttle period.
234    ///
235    /// ### Arguments
236    /// - `throttle`: Minimum delay between two consecutive asynchronous writes.
237    ///
238    /// ### Returns
239    /// - `Self`: A writer handle bound to the freshly spawned worker thread.
240    fn with_throttle(throttle: Duration) -> Self {
241        let (sender, receiver) = mpsc::sync_channel::<WriterMessage>(CHANNEL_CAPACITY);
242        let mailbox = Arc::new(SnapshotMailbox::default());
243        let writer_state = WriterState::new(throttle, Arc::clone(&mailbox));
244        let worker = Worker::spawn(
245            "fulgur-state-writer",
246            WRITER_JOIN_TIMEOUT,
247            move |_shutdown| {
248                Self::run(&receiver, writer_state);
249            },
250        );
251        Self {
252            sender,
253            mailbox,
254            _worker: worker,
255        }
256    }
257
258    /// Worker-thread loop that processes save requests one at a time.
259    ///
260    /// ### Arguments
261    /// - `receiver`: Channel of blocking requests and mailbox wakeups.
262    /// - `writer_state`: Throttle and de-duplication bookkeeping, owning the
263    ///   writer's handle on the shared mailbox.
264    fn run(receiver: &mpsc::Receiver<WriterMessage>, mut writer_state: WriterState) {
265        loop {
266            let next_message = match writer_state.mailbox_deadline() {
267                Some(deadline) => {
268                    let wait = deadline.saturating_duration_since(Instant::now());
269                    match receiver.recv_timeout(wait) {
270                        Ok(message) => Some(message),
271                        Err(mpsc::RecvTimeoutError::Timeout) => None,
272                        Err(mpsc::RecvTimeoutError::Disconnected) => break,
273                    }
274                }
275                None => match receiver.recv() {
276                    Ok(message) => Some(message),
277                    Err(mpsc::RecvError) => break,
278                },
279            };
280            match next_message {
281                Some(WriterMessage::Blocking { state, path, reply }) => {
282                    writer_state.handle_blocking(&state, &path, &reply);
283                }
284                Some(WriterMessage::MailboxFilled) => {}
285                None => writer_state.flush_mailbox(),
286            }
287        }
288        writer_state.flush_mailbox();
289        log::debug!("State writer thread exiting (no more senders)");
290    }
291
292    /// Enqueue a snapshot and block the caller until the writer has persisted it.
293    ///
294    /// ### Description
295    /// Guarantees FIFO ordering with all other save requests: any save started
296    /// before this call completes before this one begins, and any save issued
297    /// after this call returns sees the result of this save on disk. Blocking
298    /// saves are never throttled, and they supersede any asynchronous save still
299    /// held back for the same path, which is by construction older.
300    ///
301    /// A snapshot identical to the last one written to `path` is reported as
302    /// persisted without touching the file, since the file already holds it.
303    ///
304    /// ### Arguments
305    /// - `state`: The fully-assembled windows state snapshot to persist.
306    /// - `path`: Destination file path (typically the user config `state.json`).
307    ///
308    /// ### Errors
309    /// Returns an error if the writer thread has exited, if the reply channel is
310    /// dropped, or if the underlying write reports an I/O or serialization error.
311    ///
312    /// ### Returns
313    /// - `Ok(())`: The writer successfully persisted the snapshot.
314    /// - `Err(anyhow::Error)`: The writer reported an I/O or serialization error,
315    ///   or the writer thread has exited before the request could be processed.
316    pub fn save_blocking(&self, state: WindowsState, path: PathBuf) -> anyhow::Result<()> {
317        let (reply_tx, reply_rx) = mpsc::channel();
318        self.sender
319            .send(WriterMessage::Blocking {
320                state,
321                path,
322                reply: reply_tx,
323            })
324            .map_err(|_| anyhow::anyhow!("state writer thread has exited"))?;
325        reply_rx
326            .recv()
327            .map_err(|_| anyhow::anyhow!("state writer reply channel closed before result"))?
328    }
329
330    /// Hand a snapshot to the writer without waiting for the write to complete.
331    ///
332    /// ### Arguments
333    /// - `state`: The fully-assembled windows state snapshot to persist.
334    /// - `path`: Destination file path (typically the user config `state.json`).
335    pub fn save_async(&self, state: WindowsState, path: PathBuf) {
336        self.mailbox.put(state, path);
337        match self.sender.try_send(WriterMessage::MailboxFilled) {
338            Ok(()) | Err(mpsc::TrySendError::Full(_)) => {}
339            Err(mpsc::TrySendError::Disconnected(_)) => {
340                self.mailbox.drain();
341                log::error!("State writer thread has exited; dropped async save request");
342            }
343        }
344    }
345}
