5/// Persisted SSH/SFTP tab location metadata.
6///
7/// This representation intentionally excludes any credential material.
8#[derive(Serialize, Deserialize, Debug, Clone, PartialEq, Eq)]
9pub struct SerializedRemoteSpec {
10    /// Remote hostname
11    pub host: String,
12    /// SSH port
13    pub port: u16,
14    /// Remote username. Empty means "prompt user".
15    pub user: String,
16    /// Remote file path
17    pub path: String,
18}
19
20impl SerializedRemoteSpec {
21    /// Build a persisted remote spec from a runtime `RemoteSpec`.
22    ///
23    /// ### Arguments
24    /// - `spec`: Runtime remote spec to persist.
25    ///
26    /// ### Returns
27    /// - `SerializedRemoteSpec`: Persistable remote spec with no password field.
28    #[must_use]
29    pub fn from_remote_spec(spec: &RemoteSpec) -> Self {
30        Self {
31            host: spec.host.clone(),
32            port: spec.port,
33            user: spec.user.clone().unwrap_or_default(),
34            path: spec.path.clone(),
35        }
36    }
37
38    /// Convert persisted remote metadata back into a runtime `RemoteSpec`.
39    ///
40    /// ### Returns
41    /// - `RemoteSpec`: Runtime remote spec with `password_in_url` cleared.
42    #[must_use]
43    pub fn to_remote_spec(&self) -> RemoteSpec {
44        RemoteSpec {
45            host: self.host.clone(),
46            port: self.port,
47            user: (!self.user.trim().is_empty()).then_some(self.user.clone()),
48            path: self.path.clone(),
49            password_in_url: None,
50        }
51    }
52}

54/// Persisted state of a single editor tab
55///
56/// Tab IDs are not persisted as they are assigned at runtime based on position.
57#[derive(Serialize, Deserialize, Debug, Clone)]
58pub struct TabState {
59    /// Display title shown in the tab bar (usually the filename)
60    pub title: String,
61    /// Path to the file on disk, if the tab has an associated file. `None` for unsaved/new tabs.
62    pub file_path: Option<PathBuf>,
63    /// The text content of the tab, stored for unsaved tabs or when the file may have been modified since last save
64    pub content: Option<String>,
65    /// ISO 8601 timestamp of when the content was last saved to disk. Used to detect if the file has been modified externally.
66    pub last_saved: Option<String>,
67    /// Serialized remote location metadata for SSH/SFTP tabs.
68    #[serde(default)]
69    pub remote: Option<SerializedRemoteSpec>,
70    /// Whether the tab was in log view mode and should reopen in it.
71    #[serde(default)]
72    pub log_view: bool,
73}

75#[cfg(test)]
76mod tests {
77    use super::SerializedRemoteSpec;
78
79    #[test]
80    fn test_serialized_remote_spec_roundtrip_omits_password() {
81        let spec = crate::fulgur::sync::ssh::url::RemoteSpec {
82            host: "example.com".to_string(),
83            port: 22,
84            user: Some("alice".to_string()),
85            path: "/tmp/test.txt".to_string(),
86            password_in_url: Some(zeroize::Zeroizing::new("secret".to_string())),
87        };
88
89        let serialized = SerializedRemoteSpec::from_remote_spec(&spec);
90        assert_eq!(serialized.host, "example.com");
91        assert_eq!(serialized.user, "alice");
92
93        let json = serde_json::to_string(&serialized).expect("serialize serialized remote spec");
94        assert!(
95            !json.contains("password"),
96            "serialized remote spec must not include a password field"
97        );
98
99        let restored = serialized.to_remote_spec();
100        assert_eq!(restored.host, "example.com");
101        assert_eq!(restored.port, 22);
102        assert_eq!(restored.user.as_deref(), Some("alice"));
103        assert_eq!(restored.path, "/tmp/test.txt");
104        assert!(restored.password_in_url.is_none());
105    }
