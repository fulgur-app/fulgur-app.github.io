3/// Serialized window bounds that can be saved to JSON
4#[derive(Serialize, Deserialize, Debug, Clone)]
5pub struct SerializedWindowBounds {
6    /// Window state: "Windowed", "Maximized", or "Fullscreen"
7    pub state: String,
8    /// X position of window origin in pixels
9    pub x: f32,
10    /// Y position of window origin in pixels
11    pub y: f32,
12    /// Window width in pixels
13    pub width: f32,
14    /// Window height in pixels
15    pub height: f32,
16    /// Display ID (monitor) where the window was located
17    #[serde(default)]
18    pub display_id: Option<u32>,
19}
20
21impl Default for SerializedWindowBounds {
22    /// Default values for serialized window bounds
23    ///
24    /// ### Returns
25    /// - `SerializedWindowBounds`: The default serialized window bounds
26    fn default() -> Self {
27        Self {
28            state: "Windowed".to_string(),
29            x: 100.0,
30            y: 100.0,
31            width: 1200.0,
32            height: 800.0,
33            display_id: None,
34        }
35    }
36}
37
38impl SerializedWindowBounds {
39    /// Convert GPUI `WindowBounds` to `SerializedWindowBounds`
40    ///
41    /// ### Arguments
42    /// - `bounds`: The GPUI `WindowBounds` to convert
43    /// - `display_id`: Optional display ID (monitor) for the window
44    ///
45    /// ### Returns
46    /// - `SerializedWindowBounds`: The serialized window bounds
47    #[must_use]
48    pub fn from_gpui_bounds(bounds: gpui::WindowBounds, display_id: Option<u32>) -> Self {
49        use gpui::WindowBounds;
50        match bounds {
51            WindowBounds::Windowed(rect) => Self {
52                state: "Windowed".to_string(),
53                x: rect.origin.x.into(),
54                y: rect.origin.y.into(),
55                width: rect.size.width.into(),
56                height: rect.size.height.into(),
57                display_id,
58            },
59            WindowBounds::Maximized(rect) => Self {
60                state: "Maximized".to_string(),
61                x: rect.origin.x.into(),
62                y: rect.origin.y.into(),
63                width: rect.size.width.into(),
64                height: rect.size.height.into(),
65                display_id,
66            },
67            WindowBounds::Fullscreen(rect) => Self {
68                state: "Fullscreen".to_string(),
69                x: rect.origin.x.into(),
70                y: rect.origin.y.into(),
71                width: rect.size.width.into(),
72                height: rect.size.height.into(),
73                display_id,
74            },
75        }
76    }
77
78    /// Convert `SerializedWindowBounds` to GPUI `WindowBounds`
79    ///
80    /// ### Returns
81    /// - `gpui::WindowBounds`: The GPUI window bounds
82    #[must_use]
83    pub fn to_gpui_bounds(&self) -> gpui::WindowBounds {
84        use gpui::{Bounds, WindowBounds, point, px, size};
85        let bounds = Bounds {
86            origin: point(px(self.x), px(self.y)),
87            size: size(px(self.width), px(self.height)),
88        };
89        match self.state.as_str() {
90            "Maximized" => WindowBounds::Maximized(bounds),
91            "Fullscreen" => WindowBounds::Fullscreen(bounds),
92            _ => WindowBounds::Windowed(bounds), // Default to Windowed for unknown states
93        }
94    }
95}

101    /// Assert the geometry values of a GPUI window bounds rectangle.
102    ///
103    /// ### Parameters
104    /// - `bounds`: The GPUI window bounds to inspect.
105    /// - `expected_x`: Expected x origin in pixels.
106    /// - `expected_y`: Expected y origin in pixels.
107    /// - `expected_width`: Expected width in pixels.
108    /// - `expected_height`: Expected height in pixels.
109    fn assert_gpui_bounds_geometry(
110        bounds: &gpui::WindowBounds,
111        expected_x: f32,
112        expected_y: f32,
113        expected_width: f32,
114        expected_height: f32,
115    ) {
116        use gpui::WindowBounds;
117        let rect = match bounds {
118            WindowBounds::Windowed(rect)
119            | WindowBounds::Maximized(rect)
120            | WindowBounds::Fullscreen(rect) => rect,
121        };
122        assert!((f32::from(rect.origin.x) - expected_x).abs() < f32::EPSILON);
123        assert!((f32::from(rect.origin.y) - expected_y).abs() < f32::EPSILON);
124        assert!((f32::from(rect.size.width) - expected_width).abs() < f32::EPSILON);
125        assert!((f32::from(rect.size.height) - expected_height).abs() < f32::EPSILON);
126    }

128    #[test]
129    fn test_serialized_window_bounds_from_gpui_windowed_preserves_geometry_and_display() {
130        use gpui::{Bounds, WindowBounds, point, px, size};
131        let gpui_bounds = WindowBounds::Windowed(Bounds {
132            origin: point(px(120.0), px(80.0)),
133            size: size(px(1440.0), px(900.0)),
134        });
135        let serialized = SerializedWindowBounds::from_gpui_bounds(gpui_bounds, Some(7));
136        assert_eq!(serialized.state, "Windowed");
137        assert!((serialized.x - 120.0_f32).abs() < f32::EPSILON);
138        assert!((serialized.y - 80.0_f32).abs() < f32::EPSILON);
139        assert!((serialized.width - 1440.0_f32).abs() < f32::EPSILON);
140        assert!((serialized.height - 900.0_f32).abs() < f32::EPSILON);
141        assert_eq!(serialized.display_id, Some(7));
142    }

224                serialized.x,
225                serialized.y,
226                serialized.width,
227                serialized.height,
