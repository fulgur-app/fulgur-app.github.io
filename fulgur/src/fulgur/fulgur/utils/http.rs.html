1//! HTTP client used by the GPUI image loader for Markdown preview images.
2
3use futures::future::BoxFuture;
4use http_client::{AsyncBody, HttpClient, Request, Response, StatusCode, Url};
5use reqwest_client::ReqwestClient;
6
7/// An [`HttpClient`] that serves `file://` URLs from the local filesystem and
8/// delegates every other request to an inner [`ReqwestClient`].
9pub struct FileAwareHttpClient {
10    inner: ReqwestClient,
11}
12
13impl FileAwareHttpClient {
14    /// Build a client with a Fulgur user agent.
15    ///
16    /// ### Errors
17    /// Returns an error if the inner [`ReqwestClient`] cannot be constructed
18    /// (for example when the user agent header is invalid).
19    ///
20    /// ### Returns
21    /// - `Ok(FileAwareHttpClient)`: A ready-to-use client.
22    /// - `Err(anyhow::Error)`: The inner client could not be created.
23    pub fn new() -> anyhow::Result<Self> {
24        let user_agent = concat!("Fulgur/", env!("CARGO_PKG_VERSION"));
25        let inner = ReqwestClient::user_agent(user_agent)?;
26        Ok(Self { inner })
27    }
28}
29
30/// Build a synthetic HTTP response carrying a local file read from `uri`.
31///
32/// ### Arguments
33/// - `uri`: A `file://` URI pointing at a local file.
34///
35/// ### Returns
36/// - `Ok(Response<AsyncBody>)`: A `200 OK` response whose body is the file bytes.
37/// - `Err(anyhow::Error)`: The URI could not be parsed, mapped to a path, or read.
38fn read_file_uri(uri: &str) -> anyhow::Result<Response<AsyncBody>> {
39    let url = Url::parse(uri).map_err(|e| anyhow::anyhow!("invalid file URL {uri}: {e}"))?;
40    let path = url
41        .to_file_path()
42        .map_err(|()| anyhow::anyhow!("file URL is not a local path: {uri}"))?;
43    let bytes =
44        std::fs::read(&path).map_err(|e| anyhow::anyhow!("reading {}: {e}", path.display()))?;
45    Response::builder()
46        .status(StatusCode::OK)
47        .body(AsyncBody::from(bytes))
48        .map_err(|e| anyhow::anyhow!("building file response for {uri}: {e}"))
49}
50
51impl HttpClient for FileAwareHttpClient {
52    fn user_agent(&self) -> Option<&http_client::http::HeaderValue> {
53        self.inner.user_agent()
54    }
55
56    fn proxy(&self) -> Option<&Url> {
57        self.inner.proxy()
58    }
59
60    fn send(
61        &self,
62        req: Request<AsyncBody>,
63    ) -> BoxFuture<'static, anyhow::Result<Response<AsyncBody>>> {
64        // `file://` is served through `get`, which the GPUI image loader calls;
65        // `http::Uri` cannot represent a `file://` target, so it never reaches
66        // `send`. Every real `send` request is therefore remote.
67        self.inner.send(req)
68    }
69
70    fn get(
71        &self,
72        uri: &str,
73        body: AsyncBody,
74        follow_redirects: bool,
75    ) -> BoxFuture<'static, anyhow::Result<Response<AsyncBody>>> {
76        if uri.starts_with("file://") {
77            let uri = uri.to_string();
78            return Box::pin(async move { read_file_uri(&uri) });
79        }
80        self.inner.get(uri, body, follow_redirects)
81    }
82}

84#[cfg(test)]
85mod tests {
86    use super::*;
87    use futures::AsyncReadExt as _;
88    use std::io::Write as _;
89
90    #[test]
91    fn read_file_uri_serves_local_bytes() {
92        let mut file = tempfile::NamedTempFile::new().expect("temp file");
93        file.write_all(b"local-image-bytes").expect("write");
94        let url = Url::from_file_path(file.path()).expect("file url");
95
96        let response = read_file_uri(url.as_str()).expect("response");
97        assert_eq!(response.status(), StatusCode::OK);
98
99        let mut body = Vec::new();
100        futures::executor::block_on(response.into_body().read_to_end(&mut body))
101            .expect("read body");
102        assert_eq!(body, b"local-image-bytes");
103    }
104
105    #[test]
106    fn read_file_uri_rejects_missing_file() {
107        let missing = std::env::temp_dir().join("fulgur-nonexistent-image.png");
108        let _ = std::fs::remove_file(&missing);
109        let url = Url::from_file_path(&missing).expect("file url");
110        assert!(read_file_uri(url.as_str()).is_err());
111    }
