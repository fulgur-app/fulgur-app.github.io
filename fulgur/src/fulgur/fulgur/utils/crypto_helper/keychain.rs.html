32/// Saves a profile's private key in the keychain. Accepts `Zeroizing<String>`
33/// to ensure the key material is zeroed on drop.
34///
35/// ### Arguments
36/// - `profile_id`: The profile id used to namespace the keychain entry.
37/// - `private_key`: The key to save, wrapped in `Zeroizing` for secure memory
38///   handling. `None` deletes the entry.
39///
40/// ### Errors
41/// - Returns an error if the keychain entry cannot be created, written, odeleted.
42///
43/// ### Returns
44/// - `Ok(())`: The private key was saved (or removed) successfully.
45/// - `Err(anyhow::Error)`: If the keychain operation failed.
46pub fn save_private_key_to_keychain(
47    profile_id: &str,
48    private_key: Option<&Zeroizing<String>>,
49) -> anyhow::Result<()> {
50    save_or_remove_to_keychain(
51        &private_key_user(profile_id),
52        private_key.map(|k| k.as_str()),
53    )
54}

75/// Saves or removes a value from the keychain. If the value is `None`, the entry is removed from the keychain.
76///
77/// ### Arguments
78/// - `user`: the user name (the entry to look for in the keychain)
79/// - `value`: the value to save (borrowed to avoid taking ownership of sensitive data)
80///
81/// ### Returns
82/// - `Ok()`: The value has been succesfully saved in the keychain
83/// - `Err(anyhow::Error)`: If the value could not be saved
84pub(super) fn save_or_remove_to_keychain(user: &str, value: Option<&str>) -> anyhow::Result<()> {
85    if should_use_in_memory_keychain() {
86        save_or_remove_to_in_memory_keychain(user, value);
87        return Ok(());
88    }
89    init_keychain_backend()?;
90    let entry = Entry::new(SERVICE_NAME, user)?;
91    if let Some(value) = value
92        && !value.is_empty()
93    {
94        entry.set_password(value)?;
95        return Ok(());
96    }
97    match entry.delete_credential() {
98        Ok(()) | Err(keyring_core::Error::NoEntry) => Ok(()),
99        Err(e) => Err(anyhow::anyhow!(
100            "Failed to remove '{user}' from keychain: {e}"
101        )),
102    }
103}

105/// Loads a profile's private key from the keychain, wrapped in `Zeroizing`
106/// to ensure the key material is zeroed when dropped.
107///
108/// ### Arguments
109/// - `profile_id`: The profile id used to namespace the keychain entry.
110///
111/// ### Errors
112/// - Returns an error if the keychain access fails for a reason other than a
113///   missing entry.
114///
115/// ### Returns
116/// - `Ok(Some(Zeroizing<String>))`: The private key when present.
117/// - `Ok(None)`: The keychain has no entry for this profile.
118/// - `Err(anyhow::Error)`: If the keychain access failed.
119pub fn load_private_key_from_keychain(
120    profile_id: &str,
121) -> anyhow::Result<Option<Zeroizing<String>>> {
122    load_from_keychain(&private_key_user(profile_id)).map(|opt| opt.map(Zeroizing::new))
123}

142/// Loads a value from the keychain
143///
144/// ### Arguments
145/// - `user`: the user name (the entry to look for in the keychain)
146///
147/// ### Returns
148/// - `Ok(Some(String))`: The stored non-empty value
149/// - `Ok(None)`: No entry exists, or the entry was empty and has been removed
150/// - `Err(anyhow::Error)`: If the value could not be loaded or the empty entry
151///   could not be removed
152pub(super) fn load_from_keychain(user: &str) -> anyhow::Result<Option<String>> {
153    if should_use_in_memory_keychain() {
154        return Ok(load_from_in_memory_keychain(user));
155    }
156    init_keychain_backend()?;
157    let entry = Entry::new(SERVICE_NAME, user)?;
158    match entry.get_password() {
159        Ok(value) if value.is_empty() => {
160            log::warn!(
161                "Keychain entry '{user}' is empty; treating as missing and removing stale credential"
162            );
163            match entry.delete_credential() {
164                Ok(()) | Err(keyring_core::Error::NoEntry) => Ok(None),
165                Err(e) => Err(anyhow::anyhow!(
166                    "Failed to clean up empty '{user}' keychain entry: {e}"
167                )),
168            }
169        }
170        Ok(value) => Ok(Some(value)),
171        Err(keyring_core::Error::NoEntry) => Ok(None),
172        Err(e) => Err(anyhow::anyhow!(
173            "Failed to load '{user}' from keychain: {e}"
174        )),
175    }
176}
