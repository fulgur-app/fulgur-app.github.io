11/// Generate a matching pair of private/public keys
12///
13/// ### Returns
14/// - `(Identity, Recipient)`: The generated (private, public) key pair
15pub fn generate_key_pair() -> (Identity, Recipient) {
16    let private_key = age::x25519::Identity::generate();
17    let public_key = private_key.to_public();
18    log::info!("New public and private keys generated!");
19    (private_key, public_key)
20}

22/// Serializes the private key to a zeroizing string. The returned `Zeroizing<String>`
23/// ensures the key material is overwritten with zeros when dropped.
24///
25/// ### Arguments
26/// - `private_key`: the private key to serialize
27///
28/// ### Returns
29/// - `Zeroizing<String>`: The serialized private key, zeroed on drop
30pub fn serialize(private_key: &Identity) -> Zeroizing<String> {
31    let secret = private_key.to_string();
32    Zeroizing::new(secret.expose_secret().to_string())
33}

35/// Ensure a single profile has a valid X25519 keypair in the keychain and a paired `public_key` set on the profile struct.
36///
37/// The function does not save settings:  the caller is responsible for
38/// persistence so the new public key is broadcast atomically with any
39/// other change.
40///
41/// ### Arguments
42/// - `profile`: The profile that may need a keypair.
43///
44/// ### Errors
45/// - Returns an error if reading from the keychain fails, if the stored private
46///   key cannot be parsed, or if saving a newly generated key to the keychain fails.
47///
48/// ### Returns
49/// - `Ok(true)`: The profile's `public_key` was updated (either generated or
50///   recovered); the caller must persist settings.
51/// - `Ok(false)`: The keypair already exists in full; no change was made.
52/// - `Err(anyhow::Error)`: A keychain operation or key parsing/generation
53///   failed.
54pub fn ensure_profile_keypair(profile: &mut ServerProfile) -> anyhow::Result<bool> {
55    let private_key = load_private_key_from_keychain(&profile.id)?;
56    if let Some(private_key_str) = private_key {
57        if profile.public_key.is_some() {
58            log::debug!("Profile '{}' already has a keypair", profile.name);
59            return Ok(false);
60        }
61        log::info!(
62            "Profile '{}' has a stored private key but no public key; recovering public key",
63            profile.name
64        );
65        let identity: Identity = private_key_str
66            .parse()
67            .map_err(|e| anyhow::anyhow!("Failed to parse stored private key: {e}"))?;
68        profile.public_key = Some(identity.to_public().to_string());
69        return Ok(true);
70    }
71    log::debug!(
72        "Profile '{}' has no keypair, generating a new one",
73        profile.name
74    );
75    let (new_private_key, new_public_key) = generate_key_pair();
76    save_private_key_to_keychain(&profile.id, Some(&serialize(&new_private_key)))?;
77    profile.public_key = Some(new_public_key.to_string());
78    Ok(true)
79}

81/// Ensure each active profile has a valid X25519 keypair available in the
82/// keychain and a paired public key in settings.
83///
84/// ### Arguments
85/// - `settings`: The application settings; profiles whose keys had to be
86///   generated will be mutated to carry the new public key.
87///
88/// ### Errors
89/// - Returns an error if keychain access or key generation fails for any active
90///   profile, or if persisting the updated settings fails.
91///
92/// ### Returns
93/// - `Ok(())`: All active profiles have valid keys.
94/// - `Err(anyhow::Error)`: If keychain access or key generation failed for
95///   any profile (the operation stops at the first failure).
96pub fn check_private_public_keys(settings: &mut Settings) -> anyhow::Result<()> {
97    if !settings
98        .app_settings
99        .synchronization_settings
100        .is_synchronization_activated
101    {
102        return Ok(());
103    }
104    let mut settings_changed = false;
105    let profile_ids: Vec<String> = settings
106        .app_settings
107        .synchronization_settings
108        .profiles
109        .iter()
110        .filter(|p| p.is_active)
111        .map(|p| p.id.clone())
112        .collect();
113    for profile_id in profile_ids {
114        let Some(profile) = settings
115            .app_settings
116            .synchronization_settings
117            .profiles
118            .iter_mut()
119            .find(|p| p.id == profile_id)
120        else {
121            continue;
122        };
123        if ensure_profile_keypair(profile)? {
124            settings_changed = true;
125        }
126    }
127    if settings_changed {
128        log::debug!("Saving updated settings after key generation");
129        settings.save()?;
130    }
131    Ok(())
132}

134/// Checks whether a string is a valid age x25519 public key.
135///
136/// ### Arguments
137/// - `key`: The string to validate (expected format: `age1...`)
138///
139/// ### Returns
140/// - `bool`: `true` if the string parses as a valid `age::x25519::Recipient`, `false` otherwise
141pub fn is_valid_public_key(key: &str) -> bool {
142    key.parse::<Recipient>().is_ok()

145/// Encrypt bytes (e.g., compressed data) for file sharing using age encryption
146///
147/// ### Arguments
148/// - `content_bytes`: The bytes to encrypt
149/// - `recipient_public_key`: The recipient's age x25519 public key (format: "age1...")
150///
151/// ### Errors
152/// - Returns an error if the recipient public key cannot be parsed, the
153///   encryptor cannot be created, or writing the encrypted payload fails.
154///
155/// ### Returns
156/// - `Ok(String)`: The base64-encoded encrypted content
157/// - `Err(anyhow::Error)`: If the encryption failed
158pub fn encrypt_bytes(content_bytes: &[u8], recipient_public_key: &str) -> anyhow::Result<String> {
159    let recipient: Recipient = recipient_public_key
160        .parse()
161        .map_err(|e| anyhow::anyhow!("Failed to parse recipient public key: {e}"))?;
162    let recipients: Vec<Box<dyn age::Recipient>> = vec![Box::new(recipient)];
163    let encryptor =
164        age::Encryptor::with_recipients(recipients.iter().map(std::convert::AsRef::as_ref))
165            .map_err(|e| anyhow::anyhow!("Failed to create encryptor: {e}"))?;
166    let mut encrypted = vec![];
167    let mut writer = encryptor
168        .wrap_output(&mut encrypted)
169        .map_err(|e| anyhow::anyhow!("Failed to create encryption writer: {e}"))?;
170    std::io::Write::write_all(&mut writer, content_bytes)
171        .map_err(|e| anyhow::anyhow!("Failed to write encrypted data: {e}"))?;
172    writer
173        .finish()
174        .map_err(|e| anyhow::anyhow!("Failed to finish encryption: {e}"))?;
175    // Encode to base64 for transmission
176    Ok(BASE64.encode(encrypted))
177}

179/// Decrypt bytes (e.g., compressed data) received from another device using age decryption
180///
181/// ### Arguments
182/// - `encrypted_b64`: Base64-encoded encrypted content
183/// - `private_key_str`: The recipient's age x25519 private key
184///
185/// ### Errors
186/// - Returns an error if the input is not valid base64, the private key cannot
187///   be parsed, the decryptor cannot be created, or reading the decrypted
188///   payload fails.
189///
190/// ### Returns
191/// - `Ok(Vec<u8>)`: The decrypted bytes
192/// - `Err(anyhow::Error)`: If the decryption failed
193pub fn decrypt_bytes(encrypted_b64: &str, private_key_str: &str) -> anyhow::Result<Vec<u8>> {
194    let encrypted = BASE64
195        .decode(encrypted_b64)
196        .map_err(|e| anyhow::anyhow!("Failed to decode base64: {e}"))?;
197    let identity: Identity = private_key_str
198        .parse()
199        .map_err(|e| anyhow::anyhow!("Failed to parse private key: {e}"))?;
200    let decryptor = age::Decryptor::new(&encrypted[..])
201        .map_err(|e| anyhow::anyhow!("Failed to create decryptor: {e}"))?;
202    let mut decrypted = vec![];
203    let mut reader = decryptor
204        .decrypt(std::iter::once(&identity as &dyn age::Identity))
205        .map_err(|e| anyhow::anyhow!("Failed to decrypt: {e}"))?;
206    std::io::Read::read_to_end(&mut reader, &mut decrypted)
207        .map_err(|e| anyhow::anyhow!("Failed to read decrypted data: {e}"))?;
208    Ok(decrypted)
209}
