11/// Generate a matching pair of private/public keys
12///
13/// ### Returns
14/// - `(Identity, Recipient)`: The generated (private, public) key pair
15#[must_use]
16pub fn generate_key_pair() -> (Identity, Recipient) {
17    let private_key = age::x25519::Identity::generate();
18    let public_key = private_key.to_public();
19    log::info!("New public and private keys generated!");
20    (private_key, public_key)
21}

23/// Serializes the private key to a zeroizing string. The returned `Zeroizing<String>`
24/// ensures the key material is overwritten with zeros when dropped.
25///
26/// ### Arguments
27/// - `private_key`: the private key to serialize
28///
29/// ### Returns
30/// - `Zeroizing<String>`: The serialized private key, zeroed on drop
31#[must_use]
32pub fn serialize(private_key: &Identity) -> Zeroizing<String> {
33    let secret = private_key.to_string();
34    Zeroizing::new(secret.expose_secret().to_string())
35}

37/// Ensure a single profile has a valid X25519 keypair in the keychain and a paired `public_key` set on the profile struct.
38///
39/// The function does not save settings:  the caller is responsible for
40/// persistence so the new public key is broadcast atomically with any
41/// other change.
42///
43/// ### Arguments
44/// - `profile`: The profile that may need a keypair.
45///
46/// ### Errors
47/// - Returns an error if reading from the keychain fails, if the stored private
48///   key cannot be parsed, or if saving a newly generated key to the keychain fails.
49///
50/// ### Returns
51/// - `Ok(true)`: The profile's `public_key` was updated (either generated or
52///   recovered); the caller must persist settings.
53/// - `Ok(false)`: The keypair already exists in full; no change was made.
54/// - `Err(anyhow::Error)`: A keychain operation or key parsing/generation
55///   failed.
56pub fn ensure_profile_keypair(profile: &mut ServerProfile) -> anyhow::Result<bool> {
57    let private_key = load_private_key_from_keychain(&profile.id)?;
58    if let Some(private_key_str) = private_key {
59        if profile.public_key.is_some() {
60            log::debug!("Profile '{}' already has a keypair", profile.name);
61            return Ok(false);
62        }
63        log::info!(
64            "Profile '{}' has a stored private key but no public key; recovering public key",
65            profile.name
66        );
67        let identity: Identity = private_key_str
68            .parse()
69            .map_err(|e| anyhow::anyhow!("Failed to parse stored private key: {e}"))?;
70        profile.public_key = Some(identity.to_public().to_string());
71        return Ok(true);
72    }
73    log::debug!(
74        "Profile '{}' has no keypair, generating a new one",
75        profile.name
76    );
77    let (new_private_key, new_public_key) = generate_key_pair();
78    save_private_key_to_keychain(&profile.id, Some(&serialize(&new_private_key)))?;
79    profile.public_key = Some(new_public_key.to_string());
80    Ok(true)
81}

83/// Ensure each active profile has a valid X25519 keypair available in the
84/// keychain and a paired public key in settings.
85///
86/// ### Arguments
87/// - `settings`: The application settings; profiles whose keys had to be
88///   generated will be mutated to carry the new public key.
89///
90/// ### Errors
91/// - Returns an error if keychain access or key generation fails for any active
92///   profile, or if persisting the updated settings fails.
93///
94/// ### Returns
95/// - `Ok(())`: All active profiles have valid keys.
96/// - `Err(anyhow::Error)`: If keychain access or key generation failed for
97///   any profile (the operation stops at the first failure).
98pub fn check_private_public_keys(settings: &mut Settings) -> anyhow::Result<()> {
99    if !settings
100        .app_settings
101        .synchronization_settings
102        .is_synchronization_activated
103    {
104        return Ok(());
105    }
106    let mut settings_changed = false;
107    let profile_ids: Vec<String> = settings
108        .app_settings
109        .synchronization_settings
110        .profiles
111        .iter()
112        .filter(|p| p.is_active)
113        .map(|p| p.id.clone())
114        .collect();
115    for profile_id in profile_ids {
116        let Some(profile) = settings
117            .app_settings
118            .synchronization_settings
119            .profiles
120            .iter_mut()
121            .find(|p| p.id == profile_id)
122        else {
123            continue;
124        };
125        if ensure_profile_keypair(profile)? {
126            settings_changed = true;
127        }
128    }
129    if settings_changed {
130        log::debug!("Saving updated settings after key generation");
131        settings.save()?;
132    }
133    Ok(())
134}

136/// Checks whether a string is a valid age x25519 public key.
137///
138/// ### Arguments
139/// - `key`: The string to validate (expected format: `age1...`)
140///
141/// ### Returns
142/// - `bool`: `true` if the string parses as a valid `age::x25519::Recipient`, `false` otherwise
143#[must_use]
144pub fn is_valid_public_key(key: &str) -> bool {
145    key.parse::<Recipient>().is_ok()

148/// Encrypt bytes (e.g., compressed data) for file sharing using age encryption
149///
150/// ### Arguments
151/// - `content_bytes`: The bytes to encrypt
152/// - `recipient_public_key`: The recipient's age x25519 public key (format: "age1...")
153///
154/// ### Errors
155/// - Returns an error if the recipient public key cannot be parsed, the
156///   encryptor cannot be created, or writing the encrypted payload fails.
157///
158/// ### Returns
159/// - `Ok(String)`: The base64-encoded encrypted content
160/// - `Err(anyhow::Error)`: If the encryption failed
161pub fn encrypt_bytes(content_bytes: &[u8], recipient_public_key: &str) -> anyhow::Result<String> {
162    let recipient: Recipient = recipient_public_key
163        .parse()
164        .map_err(|e| anyhow::anyhow!("Failed to parse recipient public key: {e}"))?;
165    let recipients: Vec<Box<dyn age::Recipient>> = vec![Box::new(recipient)];
166    let encryptor =
167        age::Encryptor::with_recipients(recipients.iter().map(std::convert::AsRef::as_ref))
168            .map_err(|e| anyhow::anyhow!("Failed to create encryptor: {e}"))?;
169    let mut encrypted = vec![];
170    let mut writer = encryptor
171        .wrap_output(&mut encrypted)
172        .map_err(|e| anyhow::anyhow!("Failed to create encryption writer: {e}"))?;
173    std::io::Write::write_all(&mut writer, content_bytes)
174        .map_err(|e| anyhow::anyhow!("Failed to write encrypted data: {e}"))?;
175    writer
176        .finish()
177        .map_err(|e| anyhow::anyhow!("Failed to finish encryption: {e}"))?;
178    // Encode to base64 for transmission
179    Ok(BASE64.encode(encrypted))
180}

182/// Decrypt bytes (e.g., compressed data) received from another device using age decryption
183///
184/// ### Arguments
185/// - `encrypted_b64`: Base64-encoded encrypted content
186/// - `private_key_str`: The recipient's age x25519 private key
187///
188/// ### Errors
189/// - Returns an error if the input is not valid base64, the private key cannot
190///   be parsed, the decryptor cannot be created, or reading the decrypted
191///   payload fails.
192///
193/// ### Returns
194/// - `Ok(Vec<u8>)`: The decrypted bytes
195/// - `Err(anyhow::Error)`: If the decryption failed
196pub fn decrypt_bytes(encrypted_b64: &str, private_key_str: &str) -> anyhow::Result<Vec<u8>> {
197    let encrypted = BASE64
198        .decode(encrypted_b64)
199        .map_err(|e| anyhow::anyhow!("Failed to decode base64: {e}"))?;
200    let identity: Identity = private_key_str
201        .parse()
202        .map_err(|e| anyhow::anyhow!("Failed to parse private key: {e}"))?;
203    let decryptor = age::Decryptor::new(&encrypted[..])
204        .map_err(|e| anyhow::anyhow!("Failed to create decryptor: {e}"))?;
205    let mut decrypted = vec![];
206    let mut reader = decryptor
207        .decrypt(std::iter::once(&identity as &dyn age::Identity))
208        .map_err(|e| anyhow::anyhow!("Failed to decrypt: {e}"))?;
209    std::io::Read::read_to_end(&mut reader, &mut decrypted)
210        .map_err(|e| anyhow::anyhow!("Failed to read decrypted data: {e}"))?;
211    Ok(decrypted)
212}

351        // Reproduces the Begin-Synchronization-then-Save flow: a prior call
352        // wrote the private key to the keychain but the in-memory profile

367            Some(expected_public.as_str()),
