10/// Convert `ThreadId(7)` debug output to plain `7`.
11///
12/// ### Returns
13/// - `String`: The current thread id without the `ThreadId(...)` wrapper
14fn current_thread_id() -> String {
15    let thread_id = format!("{:?}", std::thread::current().id());
16    thread_id
17        .strip_prefix("ThreadId(")
18        .and_then(|s| s.strip_suffix(')'))
19        .unwrap_or(&thread_id)
20        .to_string()
21}

23/// File log formatter keeping the legacy style:
24/// `2026-04-06T15:53:54.698025+02:00 [DEBUG] (2) target: message`
25///
26/// ### Arguments
27/// - `w`: The output writer used by `flexi_logger`
28/// - `now`: Deferred timestamp provider for the current record
29/// - `record`: The `log` record to format
30///
31/// ### Returns
32/// - `Ok(())`: If the log line was written successfully
33/// - `Err(std::io::Error)`: If writing to the output fails
34fn file_log_format(
35    w: &mut dyn std::io::Write,
36    now: &mut flexi_logger::DeferredNow,
37    record: &log::Record,
38) -> Result<(), std::io::Error> {
39    let timestamp = now.format_rfc3339();
40    let thread_id = current_thread_id();
41    write!(
42        w,
43        "{} [{}] ({}) {}: {}",
44        timestamp,
45        record.level(),
46        thread_id,
47        record.target(),
48        record.args()
49    )
50}

52/// Get the directory where log files are stored.
53///
54/// ### Returns
55/// - `Ok(PathBuf)`: The path to the log directory
56/// - `Err(anyhow::Error)`: If the log file path could not be determined or created
57fn log_directory() -> anyhow::Result<PathBuf> {
58    paths::config_dir()
59}

61/// Initialize the file logger
62///
63/// The logger is always initialized with `Debug` level capacity so that the effective
64/// level can be changed at runtime via `set_debug_mode`. In release builds the
65/// effective level starts at `Info`; debug builds always use `Debug`.
66///
67/// ### Errors
68/// Returns an error if the log directory cannot be resolved or created, or if
69/// the underlying `flexi_logger` initialization fails.
70///
71/// ### Returns
72/// - `Ok(())`: If the logger was initialized successfully
73/// - `Err(anyhow::Error)`: If the logger could not be initialized
74pub fn init() -> anyhow::Result<()> {
75    let log_dir = log_directory()?;
76    let logger_handle = Logger::with(LevelFilter::Debug)
77        .log_to_file(
78            FileSpec::default()
79                .directory(log_dir.clone())
80                .basename("Fulgur")
81                .suffix("log")
82                .suppress_timestamp(),
83        )
84        .format_for_files(file_log_format)
85        .append()
86        .rotate(
87            Criterion::Age(Age::Day),
88            Naming::Timestamps,
89            Cleanup::KeepLogFiles(10),
90        )
91        .start()?;
92    LOGGER_HANDLE
93        .set(logger_handle)
94        .map_err(|_| anyhow!("logger already initialized"))?;
95    #[cfg(not(debug_assertions))]
96    log::set_max_level(LevelFilter::Info);
97    log::info!("Logger initialized in: {}", log_dir.display());
98    Ok(())
99}

101/// Update the effective log level based on the `debug_mode` setting.
102///
103/// In debug builds this is a no-op: the level is always `Debug`.
104/// In release builds, `debug_mode = true` enables `Debug`-level output;
105/// `false` restores `Info`.
106///
107/// ### Arguments
108/// - `debug_mode`: Whether debug-level logging should be active
109pub fn set_debug_mode(debug_mode: bool) {
110    let level = if debug_mode || cfg!(debug_assertions) {
111        LevelFilter::Debug
112    } else {
113        LevelFilter::Info
114    };
115    log::set_max_level(level);
116}

125    // All assertions live in a single test function to avoid races on the
126    // process-wide `log::max_level()` global when tests run in parallel.
127    #[test]
128    fn set_debug_mode_updates_max_level() {
129        // true always yields Debug, in both debug and release builds.
130        set_debug_mode(true);
131        assert_eq!(log::max_level(), LevelFilter::Debug);
132
133        // false: release → Info, debug build → still Debug (cfg! guards inside
134        // set_debug_mode keep it at Debug so dev builds are never silenced).
135        set_debug_mode(false);
136        #[cfg(debug_assertions)]
137        assert_eq!(log::max_level(), LevelFilter::Debug);
138        #[cfg(not(debug_assertions))]
139        assert_eq!(log::max_level(), LevelFilter::Info);
140
141        // Toggling back to true always restores Debug.
142        set_debug_mode(true);
143        assert_eq!(log::max_level(), LevelFilter::Debug);
144    }

173        let line = String::from_utf8(output).expect("Formatted log line should be valid UTF-8");
174        assert!(

178        assert!(

182        assert!(
