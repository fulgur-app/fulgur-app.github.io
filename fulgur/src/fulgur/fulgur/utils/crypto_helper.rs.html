24/// Registers the platform credential store as the `keyring-core` default.
25///
26/// ### Errors
27/// Returns an error if the platform store cannot be created or the current
28/// platform has no supported native store.
29///
30/// ### Returns
31/// - `Ok(())`: The default store was registered.
32/// - `Err(anyhow::Error)`: The store could not be created or is unsupported.
33#[cfg(target_os = "macos")]
34fn register_platform_store() -> anyhow::Result<()> {
35    let store = apple_native_keyring_store::keychain::Store::new()
36        .map_err(|e| anyhow::anyhow!("Failed to create macOS keychain store: {e}"))?;
37    keyring_core::set_default_store(store);
38    Ok(())
39}

67/// Ensures the platform credential store is registered exactly once.
68///
69/// ### Errors
70/// Returns an error if the platform store could not be registered.
71///
72/// ### Returns
73/// - `Ok(())`: The default store is registered and ready.
74/// - `Err(anyhow::Error)`: The default store could not be registered.
75pub fn init_keychain_backend() -> anyhow::Result<()> {
76    static STORE_INIT: Once = Once::new();
77    static INIT_OK: AtomicBool = AtomicBool::new(false);
78    STORE_INIT.call_once(|| match register_platform_store() {
79        Ok(()) => INIT_OK.store(true, Ordering::SeqCst),
80        Err(e) => log::error!("Failed to register keychain backend: {e}"),
81    });
82    if INIT_OK.load(Ordering::SeqCst) {
83        Ok(())
84    } else {
85        Err(anyhow::anyhow!("Keychain backend is not available"))
86    }
87}

114///
115/// ### Arguments

131/// Determines whether keychain operations should use the in-memory backend.
132///
133/// This avoids interactive keychain prompts during `cargo test` and in CI.
134/// Set `FULGUR_USE_REAL_KEYCHAIN=1` to force real keychain access.
135///
136/// Precedence:
137/// 1. `FULGUR_USE_REAL_KEYCHAIN=1` always forces the real keychain backend.
138/// 2. `FULGUR_DISABLE_KEYCHAIN=1` forces the in-memory backend.
139/// 3. `CI=1` forces the in-memory backend.
140/// 4. Test binary heuristics (`target/*/deps/*-<hash>`) use the in-memory backend.
141///
142/// ### Returns
143/// - `true`: Use in-memory keychain storage.
144/// - `false`: Use the platform keychain backend.
145fn should_use_in_memory_keychain() -> bool {
146    if env_var_is_truthy("FULGUR_USE_REAL_KEYCHAIN") {
147        return false;
148    }
149    if env_var_is_truthy("FULGUR_DISABLE_KEYCHAIN") {
150        return true;
151    }
152    if env_var_is_truthy("CI") {
153        return true;
154    }
155    // `cargo test` binaries are typically emitted under `target/*/deps/`.
156    if let Ok(exe) = std::env::current_exe() {
157        let in_deps_dir = exe.parent().is_some_and(|parent| parent.ends_with("deps"));
158        let has_hashed_test_name = exe
159            .file_name()
160            .and_then(OsStr::to_str)
161            .is_some_and(|name| name.contains('-'));
162        if in_deps_dir && has_hashed_test_name {
163            return true;
164        }
165    }
166    false
167}

189/// Saves or removes a value in the in-memory keychain backend.
190///
191/// If `value` is `None` or empty, the entry is removed.
192///
193/// ### Arguments
194/// - `user`: The keychain entry name.
195/// - `value`: The value to save.
196///
197/// ### Returns
198/// - `Ok(())`: The in-memory operation succeeded.
199/// - `Err(anyhow::Error)`: The in-memory store lock failed.
200fn save_or_remove_to_in_memory_keychain(user: &str, value: Option<&str>) -> anyhow::Result<()> {
201    let mut keychain = in_memory_keychain()
202        .lock()
203        .map_err(|_| anyhow::anyhow!("Failed to lock in-memory keychain"))?;
204    let key = in_memory_key(user);
205    if let Some(value) = value
206        && !value.is_empty()
207    {
208        keychain.insert(key, value.to_string());
209    } else {
210        keychain.remove(&key);
211    }
212    Ok(())
213}

215/// Loads a value from the in-memory keychain backend.
216///
217/// ### Arguments
218/// - `user`: The keychain entry name.
219///
220/// ### Returns
221/// - `Ok(Some(String))`: The value exists.
222/// - `Ok(None)`: The value does not exist.
223/// - `Err(anyhow::Error)`: The in-memory store lock failed.
224fn load_from_in_memory_keychain(user: &str) -> anyhow::Result<Option<String>> {

229}

287/// Saves or removes a value from the keychain. If the value is `None`, the entry is removed from the keychain.
288///
289/// ### Arguments
290/// - `user`: the user name (the entry to look for in the keychain)
291/// - `value`: the value to save (borrowed to avoid taking ownership of sensitive data)
292///
293/// ### Returns
294/// - `Ok()`: The value has been succesfully saved in the keychain
295/// - `Err(anyhow::Error)`: If the value could not be saved
296fn save_or_remove_to_keychain(user: &str, value: Option<&str>) -> anyhow::Result<()> {
297    if should_use_in_memory_keychain() {
298        return save_or_remove_to_in_memory_keychain(user, value);
299    }
300    init_keychain_backend()?;
301    let entry = Entry::new(SERVICE_NAME, user)?;
302    if let Some(value) = value
303        && !value.is_empty()
304    {
305        entry.set_password(value)?;
306        return Ok(());
307    }
308    match entry.delete_credential() {
309        Ok(()) | Err(keyring_core::Error::NoEntry) => Ok(()),
310        Err(e) => Err(anyhow::anyhow!(
311            "Failed to remove '{user}' from keychain: {e}"
312        )),
313    }
314}

353/// Migrate legacy single-server keychain entries into per-profile entries
354/// for the given profile id.
355///
356/// ### Arguments
357/// - `profile_id`: The id of the profile that should receive the migrated
358///   credentials (typically the migrated "Fulgurant" profile).
359///
360/// ### Errors
361/// Returns an error if any of the keychain read, write, or delete operations
362/// fail; legacy entries are left in place so the migration can be retried.
363///
364/// ### Returns
365/// - `Ok(())`: Migration completed (or had nothing to migrate).
366/// - `Err(anyhow::Error)`: If a keychain operation failed; the legacy entries
367///   are left in place so a future startup can retry.
368pub fn migrate_legacy_keychain_to_profile(profile_id: &str) -> anyhow::Result<()> {
369    if let Some(legacy_private) = load_from_keychain(PRIVATE_KEY_PREFIX)? {
370        let target_user = private_key_user(profile_id);
371        if load_from_keychain(&target_user)?.is_none() {
372            log::info!("Migrating legacy private key to profile '{profile_id}'");
373            save_or_remove_to_keychain(&target_user, Some(&legacy_private))?;
374        }
375        save_or_remove_to_keychain(PRIVATE_KEY_PREFIX, None)?;
376    }
377    if let Some(legacy_api) = load_from_keychain(DEVICE_API_KEY_PREFIX)? {
378        let target_user = device_api_key_user(profile_id);
379        if load_from_keychain(&target_user)?.is_none() {
380            log::info!("Migrating legacy device API key to profile '{profile_id}'");
381            save_or_remove_to_keychain(&target_user, Some(&legacy_api))?;
382        }
383        save_or_remove_to_keychain(DEVICE_API_KEY_PREFIX, None)?;
384    }
385    Ok(())
386}

399/// Migrate legacy single-server keychain entries into the first configured profile, regardless of whether sync is currently activated.
400///
401/// ### Arguments
402/// - `settings`: Application settings used to locate the target profile.
403///
404/// ### Errors
405/// Returns an error if any keychain read, write, or delete operation fails.
406///
407/// ### Returns
408/// - `Ok(())`: Migration completed or there was nothing to migrate.
409/// - `Err(anyhow::Error)`: A keychain operation failed.
410pub fn migrate_legacy_keychain_entries_if_present(settings: &Settings) -> anyhow::Result<()> {
411    if !legacy_keychain_entries_present()? {
412        return Ok(());
413    }
414    let Some(target_profile_id) = settings
415        .app_settings
416        .synchronization_settings
417        .profiles
418        .first()
419        .map(|profile| profile.id.clone())
420    else {
421        log::warn!(
422            "Legacy keychain entries detected but no profiles are configured; leaving entries in place"
423        );
424        return Ok(());
425    };
426    log::info!(
427        "Migrating legacy keychain entries to profile '{target_profile_id}' (first configured profile)"
428    );
429    migrate_legacy_keychain_to_profile(&target_profile_id)
430}

432/// Loads a value from the keychain
433///
434/// ### Arguments
435/// - `user`: the user name (the entry to look for in the keychain)
436///
437/// ### Returns
438/// - `Ok(Option<String>)`: The value if it exists, otherwise `None`
439/// - `Err(anyhow::Error)`: If the value could not be loaded
440fn load_from_keychain(user: &str) -> anyhow::Result<Option<String>> {
441    if should_use_in_memory_keychain() {
442        return load_from_in_memory_keychain(user);
443    }
444    init_keychain_backend()?;
445    let entry = Entry::new(SERVICE_NAME, user)?;
446    match entry.get_password() {
447        Ok(value) if value.is_empty() => {
448            // Legacy behavior stored empty strings instead of deleting credentials.
449            // TODO: remove this in further version.
450            log::warn!(
451                "Keychain entry '{user}' is empty; treating as missing and removing stale credential"
452            );
453            match entry.delete_credential() {
454                Ok(()) | Err(keyring_core::Error::NoEntry) => Ok(None),
455                Err(e) => Err(anyhow::anyhow!(
456                    "Failed to clean up empty '{user}' keychain entry: {e}"
457                )),
458            }
459        }
460        Ok(value) => Ok(Some(value)),
461        Err(keyring_core::Error::NoEntry) => Ok(None),
462        Err(e) => Err(anyhow::anyhow!(
463            "Failed to load '{user}' from keychain: {e}"
464        )),
465    }
466}

481/// Ensure a single profile has a valid X25519 keypair in the keychain and a paired `public_key` set on the profile struct.
482///
483/// The function does not save settings:  the caller is responsible for
484/// persistence so the new public key is broadcast atomically with any
485/// other change.
486///
487/// ### Arguments
488/// - `profile`: The profile that may need a keypair.
489///
490/// ### Errors
491/// Returns an error if reading from the keychain fails, if the stored private
492/// key cannot be parsed, or if saving a newly generated key to the keychain
493/// fails.
494///
495/// ### Returns
496/// - `Ok(true)`: The profile's `public_key` was updated (either generated or
497///   recovered); the caller must persist settings.
498/// - `Ok(false)`: The keypair already exists in full; no change was made.
499/// - `Err(anyhow::Error)`: A keychain operation or key parsing/generation
500///   failed.
501pub fn ensure_profile_keypair(profile: &mut ServerProfile) -> anyhow::Result<bool> {
502    let private_key = load_private_key_from_keychain(&profile.id)?;
503    if let Some(private_key_str) = private_key {
504        if profile.public_key.is_some() {
505            log::debug!("Profile '{}' already has a keypair", profile.name);
506            return Ok(false);
507        }
508        log::info!(
509            "Profile '{}' has a stored private key but no public key; recovering public key",
510            profile.name
511        );
512        let identity: Identity = private_key_str
513            .parse()
514            .map_err(|e| anyhow::anyhow!("Failed to parse stored private key: {e}"))?;
515        profile.public_key = Some(identity.to_public().to_string());
516        return Ok(true);
517    }
518    log::debug!(
519        "Profile '{}' has no keypair, generating a new one",
520        profile.name
521    );
522    let (new_private_key, new_public_key) = generate_key_pair();
523    save_private_key_to_keychain(&profile.id, Some(&serialize(&new_private_key)))?;
524    profile.public_key = Some(new_public_key.to_string());
525    Ok(true)
526}

528/// Ensure each active profile has a valid X25519 keypair available in the
529/// keychain and a paired public key in settings.
530///
531/// ### Arguments
532/// - `settings`: The application settings; profiles whose keys had to be
533///   generated will be mutated to carry the new public key.
534///
535/// ### Errors
536/// Returns an error if keychain access or key generation fails for any active
537/// profile, or if persisting the updated settings fails.
538///
539/// ### Returns
540/// - `Ok(())`: All active profiles have valid keys.
541/// - `Err(anyhow::Error)`: If keychain access or key generation failed for
542///   any profile (the operation stops at the first failure).
543pub fn check_private_public_keys(settings: &mut Settings) -> anyhow::Result<()> {
544    if !settings
545        .app_settings
546        .synchronization_settings
547        .is_synchronization_activated
548    {
549        return Ok(());
550    }
551    let mut settings_changed = false;
552    let profile_ids: Vec<String> = settings
553        .app_settings
554        .synchronization_settings
555        .profiles
556        .iter()
557        .filter(|p| p.is_active)
558        .map(|p| p.id.clone())
559        .collect();
560    for profile_id in profile_ids {
561        let Some(profile) = settings
562            .app_settings
563            .synchronization_settings
564            .profiles
565            .iter_mut()
566            .find(|p| p.id == profile_id)
567        else {
568            continue;
569        };
570        if ensure_profile_keypair(profile)? {
571            settings_changed = true;
572        }
573    }
574    if settings_changed {
575        log::debug!("Saving updated settings after key generation");
576        settings.save()?;
577    }
578    Ok(())
579}

592/// Encrypt bytes (e.g., compressed data) for file sharing using age encryption
593///
594/// ### Arguments
595/// - `content_bytes`: The bytes to encrypt
596/// - `recipient_public_key`: The recipient's age x25519 public key (format: "age1...")
597///
598/// ### Errors
599/// Returns an error if the recipient public key cannot be parsed, the
600/// encryptor cannot be created, or writing the encrypted payload fails.
601///
602/// ### Returns
603/// - `Ok(String)`: The base64-encoded encrypted content
604/// - `Err(anyhow::Error)`: If the encryption failed
605pub fn encrypt_bytes(content_bytes: &[u8], recipient_public_key: &str) -> anyhow::Result<String> {
606    let recipient: Recipient = recipient_public_key
607        .parse()
608        .map_err(|e| anyhow::anyhow!("Failed to parse recipient public key: {e}"))?;
609    let recipients: Vec<Box<dyn age::Recipient>> = vec![Box::new(recipient)];
610    let encryptor =
611        age::Encryptor::with_recipients(recipients.iter().map(std::convert::AsRef::as_ref))
612            .map_err(|e| anyhow::anyhow!("Failed to create encryptor: {e}"))?;
613    let mut encrypted = vec![];
614    let mut writer = encryptor
615        .wrap_output(&mut encrypted)
616        .map_err(|e| anyhow::anyhow!("Failed to create encryption writer: {e}"))?;
617    std::io::Write::write_all(&mut writer, content_bytes)
618        .map_err(|e| anyhow::anyhow!("Failed to write encrypted data: {e}"))?;
619    writer
620        .finish()
621        .map_err(|e| anyhow::anyhow!("Failed to finish encryption: {e}"))?;
622    // Encode to base64 for transmission
623    Ok(BASE64.encode(encrypted))
624}

626/// Decrypt bytes (e.g., compressed data) received from another device using age decryption
627///
628/// ### Arguments
629/// - `encrypted_b64`: Base64-encoded encrypted content
630/// - `private_key_str`: The recipient's age x25519 private key
631///
632/// ### Errors
633/// Returns an error if the input is not valid base64, the private key cannot
634/// be parsed, the decryptor cannot be created, or reading the decrypted
635/// payload fails.
636///
637/// ### Returns
638/// - `Ok(Vec<u8>)`: The decrypted bytes
639/// - `Err(anyhow::Error)`: If the decryption failed
640pub fn decrypt_bytes(encrypted_b64: &str, private_key_str: &str) -> anyhow::Result<Vec<u8>> {
641    let encrypted = BASE64
642        .decode(encrypted_b64)
643        .map_err(|e| anyhow::anyhow!("Failed to decode base64: {e}"))?;
644    let identity: Identity = private_key_str
645        .parse()
646        .map_err(|e| anyhow::anyhow!("Failed to parse private key: {e}"))?;
647    let decryptor = age::Decryptor::new(&encrypted[..])
648        .map_err(|e| anyhow::anyhow!("Failed to create decryptor: {e}"))?;
649    let mut decrypted = vec![];
650    let mut reader = decryptor
651        .decrypt(std::iter::once(&identity as &dyn age::Identity))
652        .map_err(|e| anyhow::anyhow!("Failed to decrypt: {e}"))?;
653    std::io::Read::read_to_end(&mut reader, &mut decrypted)
654        .map_err(|e| anyhow::anyhow!("Failed to read decrypted data: {e}"))?;
655    Ok(decrypted)
656}
