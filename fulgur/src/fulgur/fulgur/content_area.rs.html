21    /// Handle a right-click in the editor area to show a custom context menu.
22    ///
23    /// Called during the capture phase so propagation can be stopped before
24    /// the editor's built-in context menu fires.
25    ///
26    /// ### Arguments
27    /// - `event`: The mouse-down event
28    /// - `window`: The window context
29    /// - `cx`: The application context
30    fn on_editor_right_click(
31        &mut self,
32        event: &MouseDownEvent,
33        window: &mut Window,
34        cx: &mut Context<Self>,
35    ) {
36        if event.button != MouseButton::Right {
37            return;
38        }
39        cx.stop_propagation();
40
41        let Some(active_index) = self.active_tab_index else {
42            return;
43        };
44        let Some(Tab::Editor(editor_tab)) = self.tabs.get(active_index) else {
45            return;
46        };
47        let editor_focus = editor_tab.content.focus_handle(cx);
48        let has_file = editor_tab.file_path().is_some();
49        let position = event.position;
50
51        let menu = PopupMenu::build(window, cx, {
52            let editor_focus = editor_focus.clone();
53            move |mut menu, _window, _cx| {
54                menu = menu.action_context(editor_focus);
55                if has_file {
56                    menu = menu
57                        .menu(
58                            crate::fulgur::ui::components_utils::reveal_in_file_manager_label(),
59                            Box::new(ui::tabs::tab_bar::ShowInFileManager(active_index)),
60                        )
61                        .separator();
62                }
63                menu.menu("Cut", Box::new(gpui_component::input::Cut))
64                    .menu("Copy", Box::new(gpui_component::input::Copy))
65                    .menu("Paste", Box::new(gpui_component::input::Paste))
66                    .separator()
67                    .menu("Select All", Box::new(gpui_component::input::SelectAll))
68            }
69        });
70
71        let subscription = cx.subscribe_in(
72            &menu,
73            window,
74            |this: &mut Self, _, _: &DismissEvent, _, cx| {
75                this.editor_context_menu = None;
76                this.editor_context_menu_subscription = None;
77                cx.notify();
78            },
79        );
80
81        self.editor_context_menu = Some((position, menu));
82        self.editor_context_menu_subscription = Some(subscription);
83        cx.notify();
84    }

86    /// Render the content area (editor or settings)
87    ///
88    /// ### Arguments
89    /// - `active_tab_index`: The index of the active tab (if any)
90    /// - `window`: The window context
91    /// - `cx`: The application context
92    ///
93    /// ### Returns
94    /// - `AnyElement`: The rendered content area element (wrapped in `AnyElement`)
95    pub(super) fn render_content_area(
96        &mut self,
97        active_tab_index: Option<usize>,
98        window: &mut Window,
99        cx: &mut Context<Self>,
100    ) -> AnyElement {
101        enum ActiveTabRenderData {
102            Editor {
103                tab_id: usize,
104                language: SupportedLanguage,
105                show_markdown_preview: bool,
106                content: Entity<InputState>,
107                csv_view_mode: editor_tab::CsvViewMode,
108                csv_table: Option<Entity<TableState<editor_tab::CsvTableDelegate>>>,
109            },
110            Settings,
111            MarkdownPreview {
112                source_tab_id: usize,
113                content: Entity<InputState>,
114                view_state: Entity<gpui_component::text::TextViewState>,
115            },
116        }
117
118        // A CSV tab in table mode needs its grid (re)built from the canonical
119        // text before we snapshot the tab read-only below. When the parse is
120        // lossy, `ensure_csv_table` falls back to text mode and returns a
121        // warning to surface to the user.
122        let csv_table_warning = if let Some(active_index) = active_tab_index
123            && let Some(Tab::Editor(editor_tab)) = self.tabs.get_mut(active_index)
124            && editor_tab.language == SupportedLanguage::Csv
125            && editor_tab.csv_view_mode == editor_tab::CsvViewMode::Table
126        {
127            editor_tab.ensure_csv_table(window, cx)
128        } else {
129            None
130        };
131        if let Some(message) = csv_table_warning {
132            self.pending_notification = Some((NotificationType::Warning, message.into()));
133        }
134
135        let tabs_ref = &self.tabs;
136        let active_tab = active_tab_index.and_then(|active_index| {
137            tabs_ref.get(active_index).map(|tab| match tab {
138                Tab::Editor(editor_tab) => ActiveTabRenderData::Editor {
139                    tab_id: editor_tab.id,
140                    language: editor_tab.language,
141                    show_markdown_preview: editor_tab.show_markdown_preview,
142                    content: editor_tab.content.clone(),
143                    csv_view_mode: editor_tab.csv_view_mode,
144                    csv_table: editor_tab.csv_table.clone(),
145                },
146                Tab::Settings(_) => ActiveTabRenderData::Settings,
147                Tab::MarkdownPreview(preview_tab) => ActiveTabRenderData::MarkdownPreview {
148                    source_tab_id: preview_tab.source_tab_id,
149                    content: tabs_ref
150                        .iter()
151                        .find_map(|t| match t {
152                            Tab::Editor(editor_tab)
153                                if editor_tab.id == preview_tab.source_tab_id =>
154                            {
155                                Some(editor_tab.content.clone())
156                            }
157                            _ => None,
158                        })
159                        .unwrap_or_else(|| preview_tab.content.clone()),
160                    view_state: preview_tab.view_state.clone(),
161                },
162            })
163        });
164
165        if let Some(tab) = active_tab {
166            match tab {
167                ActiveTabRenderData::Editor {
168                    tab_id,
169                    language,
170                    show_markdown_preview,
171                    content,
172                    csv_view_mode,
173                    csv_table,
174                } => {
175                    if language == SupportedLanguage::Csv
176                        && csv_view_mode == editor_tab::CsvViewMode::Table
177                        && let Some(table) = csv_table
178                    {
179                        return v_flex()
180                            .w_full()
181                            .flex_1()
182                            .child(DataTable::new(&table).bordered(true).stripe(true))
183                            .into_any_element();
184                    }
185                    let editor_input = Input::new(&content)
186                        .bordered(false)
187                        .p_0()
188                        .h_full()
189                        .font_family(self.settings.editor_settings.font_family.clone())
190                        .text_size(px(self.settings.editor_settings.font_size))
191                        .focus_bordered(false);
192                    let capture_right_click =
193                        cx.listener(|this, event: &MouseDownEvent, window, cx| {
194                            this.on_editor_right_click(event, window, cx);
195                        });
196                    if language == SupportedLanguage::Markdown
197                        && show_markdown_preview
198                        && self.settings.editor_settings.markdown_settings.preview_mode
199                            == crate::fulgur::settings::MarkdownPreviewMode::Panel
200                    {
201                        let preview_text = self.markdown_preview_text_for(tab_id, &content, cx);
202                        return v_flex()
203                            .w_full()
204                            .flex_1()
205                            .child(
206                                h_resizable("markdown-preview-container")
207                                    .child(
208                                        resizable_panel().child(
209                                            div()
210                                                .id("markdown-editor")
211                                                .size_full()
212                                                .capture_any_mouse_down(capture_right_click)
213                                                .child(editor_input),
214                                        ),
215                                    )
216                                    .child(
217                                        resizable_panel().child(
218                                            TextView::markdown("markdown-preview", preview_text)
219                                                .flex_none()
220                                                .py_0()
221                                                .px_2()
222                                                .scrollable(true)
223                                                .selectable(true)
224                                                .bg(cx.theme().muted),
225                                        ),
226                                    ),
227                            )
228                            .into_any_element();
229                    }
230                    return v_flex()
231                        .w_full()
232                        .flex_1()
233                        .capture_any_mouse_down(capture_right_click)
234                        .child(editor_input)
235                        .into_any_element();
236                }
237                ActiveTabRenderData::Settings => {
238                    return v_flex()
239                        .id("settings-tab-scrollable")
240                        .w_full()
241                        .flex_1()
242                        .overflow_y_scrollbar()
243                        .child(self.render_settings(window, cx))
244                        .into_any_element();
245                }
246                ActiveTabRenderData::MarkdownPreview {
247                    source_tab_id,
248                    content,
249                    view_state,
250                } => {
251                    let preview_text = self.markdown_preview_text_for(source_tab_id, &content, cx);
252                    view_state.update(cx, |state, cx| {
253                        state.set_text(preview_text.as_ref(), cx);
254                    });
255                    return v_flex()
256                        .w_full()
257                        .flex_1()
258                        .child(
259                            TextView::new(&view_state)
260                                .py_2()
261                                .px_4()
262                                .scrollable(true)
263                                .selectable(true),
264                        )
265                        .into_any_element();
266                }
267            }
268        }
269        v_flex().w_full().flex_1().into_any_element()
270    }
