25    /// Handle a right-click in the editor area to show a custom context menu.
26    ///
27    /// Called during the capture phase so propagation can be stopped before
28    /// the editor's built-in context menu fires.
29    ///
30    /// ### Arguments
31    /// - `event`: The mouse-down event
32    /// - `window`: The window context
33    /// - `cx`: The application context
34    fn on_editor_right_click(
35        &mut self,
36        event: &MouseDownEvent,
37        window: &mut Window,
38        cx: &mut Context<Self>,
39    ) {
40        if event.button != MouseButton::Right {
41            return;
42        }
43        cx.stop_propagation();
44
45        let Some(Tab::Editor(editor_tab)) = self.active_tab(cx) else {
46            return;
47        };
48        let active_tab_id = editor_tab.id;
49        let editor_focus = editor_tab.content.focus_handle(cx);
50        let has_file = editor_tab.file_path().is_some();
51        let position = event.position;
52
53        let menu = PopupMenu::build(window, cx, {
54            let editor_focus = editor_focus.clone();
55            move |mut menu, _window, _cx| {
56                menu = menu.action_context(editor_focus);
57                if has_file {
58                    menu = menu
59                        .menu(
60                            crate::fulgur::ui::components_utils::reveal_in_file_manager_label(),
61                            Box::new(ui::tabs::tab_bar::ShowInFileManager(active_tab_id)),
62                        )
63                        .separator();
64                }
65                menu.menu("Cut", Box::new(gpui_component::input::Cut))
66                    .menu("Copy", Box::new(gpui_component::input::Copy))
67                    .menu("Paste", Box::new(gpui_component::input::Paste))
68                    .separator()
69                    .menu("Select All", Box::new(gpui_component::input::SelectAll))
70            }
71        });
72
73        let subscription = cx.subscribe_in(
74            &menu,
75            window,
76            |this: &mut Self, _, _: &DismissEvent, _, cx| {
77                this.editor_context_menu = None;
78                this.editor_context_menu_subscription = None;
79                cx.notify();
80            },
81        );
82
83        self.editor_context_menu = Some((position, menu));
84        self.editor_context_menu_subscription = Some(subscription);
85        cx.notify();
86    }

88    /// Handle a right-click on the markdown preview to show its context menu.
89    ///
90    /// ### Arguments
91    /// - `event`: The mouse-down event
92    /// - `window`: The window context
93    /// - `cx`: The application context
94    fn on_preview_right_click(
95        &mut self,
96        event: &MouseDownEvent,
97        window: &mut Window,
98        cx: &mut Context<Self>,
99    ) {
100        if event.button != MouseButton::Right {
101            return;
102        }
103        cx.stop_propagation();
104
105        let position = event.position;
106        let preview_focus = self.markdown_preview_focus.clone();
107
108        // The replacement, `gpui_base::TextSelection::selected_text`, lives in the
109        // `gpui-base` crate, which gpui-component does not re-export and which Fulgur
110        // does not depend on directly. `WindowExt::selected_text` forwards to it.
111        #[allow(deprecated)]
112        let selection = window.selected_text(cx).trim().to_string();
113        let has_selection = !selection.is_empty();
114        self.markdown_preview_pending_copy = has_selection.then_some(selection);
115
116        let menu = PopupMenu::build(window, cx, move |menu, _window, _cx| {
117            menu.action_context(preview_focus)
118                .menu_with_enable("Copy", Box::new(gpui_component::input::Copy), has_selection)
119                .menu("Select All", Box::new(gpui_component::input::SelectAll))
120        });
121
122        let subscription = cx.subscribe_in(
123            &menu,
124            window,
125            |this: &mut Self, _, _: &DismissEvent, _, cx| {
126                this.editor_context_menu = None;
127                this.editor_context_menu_subscription = None;
128                cx.notify();
129            },
130        );
131
132        self.editor_context_menu = Some((position, menu));
133        self.editor_context_menu_subscription = Some(subscription);
134        cx.notify();
135    }

137    /// Resolve the text view state backing the currently visible markdown preview.
138    ///
139    /// ### Arguments
140    /// - `cx`: The application context
141    ///
142    /// ### Returns
143    /// - `Some(Entity<TextViewState>)`: The state of the active preview, either a
144    ///   dedicated preview tab or the inline panel of a markdown editor tab.
145    /// - `None`: If no markdown preview is currently displayed.
146    fn active_markdown_preview_state(&self, cx: &App) -> Option<Entity<TextViewState>> {
147        match self.active_tab(cx)? {
148            Tab::MarkdownPreview(preview) => Some(preview.view_state.clone()),
149            Tab::Editor(_) => self.markdown_panel_view_state.clone(),
150            Tab::Settings(_) => None,
151        }
152    }

154    /// Ensure the inline preview panel owns a persistent text view state.
155    ///
156    /// ### Arguments
157    /// - `text`: The rendered markdown source to display
158    /// - `cx`: The application context
159    ///
160    /// ### Returns
161    /// - `Entity<TextViewState>`: The persistent state for the inline preview.
162    fn ensure_markdown_panel_state(
163        &mut self,
164        text: &str,
165        cx: &mut Context<Self>,
166    ) -> Entity<TextViewState> {
167        let state = self
168            .markdown_panel_view_state
169            .get_or_insert_with(|| cx.new(|cx| TextViewState::markdown(text, cx)))
170            .clone();
171        state.update(cx, |state, cx| state.set_text(text, cx));
172        state
173    }

175    /// Lay out a markdown preview inside its container, honouring the width limit setting.
176    ///
177    /// ### Arguments
178    /// - `preview`: The preview text view to lay out
179    ///
180    /// ### Returns
181    /// - `Div`: A full-size container holding the preview, capped at
182    ///   `MARKDOWN_PREVIEW_MAX_WIDTH` and centered when the limit is enabled.
183    fn layout_markdown_preview(&self, preview: TextView) -> Div {
184        let limited = self
185            .settings
186            .editor_settings
187            .markdown_settings
188            .limit_preview_width;
189        h_flex()
190            .size_full()
191            .justify_center()
192            .child(preview.when(limited, |view| view.max_w(px(MARKDOWN_PREVIEW_MAX_WIDTH))))
193    }

195    /// Wrap a markdown preview element with its context-menu affordances.
196    ///
197    /// ### Arguments
198    /// - `child`: The preview element to wrap
199    /// - `cx`: The application context
200    ///
201    /// ### Returns
202    /// - `impl IntoElement`: The wrapped, right-clickable preview element.
203    fn wrap_markdown_preview(&self, child: AnyElement, cx: &mut Context<Self>) -> impl IntoElement {
204        let right_click = cx.listener(|this, event: &MouseDownEvent, window, cx| {
205            this.on_preview_right_click(event, window, cx);
206        });
207        let copy = cx.listener(|this, _: &gpui_component::input::Copy, _window, cx| {
208            if let Some(text) = this.markdown_preview_pending_copy.take() {
209                cx.write_to_clipboard(gpui::ClipboardItem::new_string(text));
210            }
211        });
212        let select_all = cx.listener(|this, _: &gpui_component::input::SelectAll, _window, cx| {
213            if let Some(state) = this.active_markdown_preview_state(cx) {
214                state.update(cx, TextViewState::select_all);
215            }
216        });
217
218        div()
219            .id("markdown-preview-context")
220            .track_focus(&self.markdown_preview_focus)
221            .size_full()
222            .capture_any_mouse_down(right_click)
223            .on_action(copy)
224            .on_action(select_all)
225            .child(child)
226    }

228    /// Render the content area (editor or settings)
229    ///
230    /// ### Arguments
231    /// - `active_tab_index`: The index of the active tab (if any)
232    /// - `window`: The window context
233    /// - `cx`: The application context
234    ///
235    /// ### Returns
236    /// - `AnyElement`: The rendered content area element (wrapped in `AnyElement`)
237    pub(super) fn render_content_area(
238        &mut self,
239        active_tab_index: Option<usize>,
240        window: &mut Window,
241        cx: &mut Context<Self>,
242    ) -> AnyElement {
243        enum ActiveTabRenderData {
244            Editor {
245                language: SupportedLanguage,
246                show_markdown_preview: bool,
247                large_file: bool,
248                content: Entity<EditorState>,
249                path: Option<std::path::PathBuf>,
250                csv_view_mode: editor_tab::CsvViewMode,
251                csv_table: Option<Entity<TableState<editor_tab::CsvTableDelegate>>>,
252                log_view: bool,
253                log_content: Option<Entity<EditorState>>,
254            },
255            Settings,
256            MarkdownPreview {
257                content: Entity<EditorState>,
258                source_path: Option<std::path::PathBuf>,
259                view_state: Entity<gpui_component::text::TextViewState>,
260            },
261        }
262
263        // A CSV tab in table mode needs its grid (re)built from the canonical
264        // text before we snapshot the tab read-only below. When the parse is
265        // lossy, `ensure_csv_table` falls back to text mode and returns a
266        // warning to surface to the user.
267        let csv_table_warning = if let Some(active_index) = active_tab_index
268            && let Some(tab_entity) = self.tabs.get(active_index).cloned()
269        {
270            tab_entity.update(cx, |tab, cx| match tab {
271                Tab::Editor(editor_tab)
272                    if editor_tab.language == SupportedLanguage::Csv
273                        && editor_tab.csv_view_mode == editor_tab::CsvViewMode::Table =>
274                {
275                    editor_tab.ensure_csv_table(window, cx)
276                }
277                _ => None,
278            })
279        } else {
280            None
281        };
282        if let Some(message) = csv_table_warning {
283            window.push_notification((NotificationType::Warning, SharedString::from(message)), cx);
284        }
285
286        let tabs_ref = &self.tabs;
287        let active_tab = active_tab_index.and_then(|active_index| {
288            tabs_ref.get(active_index).map(|tab| match tab.read(cx) {
289                Tab::Editor(editor_tab) => ActiveTabRenderData::Editor {
290                    language: editor_tab.language,
291                    show_markdown_preview: editor_tab.show_markdown_preview,
292                    large_file: editor_tab.large_file,
293                    content: editor_tab.content.clone(),
294                    path: editor_tab.location.local_path().cloned(),
295                    csv_view_mode: editor_tab.csv_view_mode,
296                    csv_table: editor_tab.csv_table.clone(),
297                    log_view: editor_tab.log_view,
298                    log_content: editor_tab.log_content.clone(),
299                },
300                Tab::Settings(_) => ActiveTabRenderData::Settings,
301                Tab::MarkdownPreview(preview_tab) => ActiveTabRenderData::MarkdownPreview {
302                    content: tabs_ref
303                        .iter()
304                        .find_map(|t| match t.read(cx) {
305                            Tab::Editor(editor_tab)
306                                if editor_tab.id == preview_tab.source_tab_id =>
307                            {
308                                Some(editor_tab.content.clone())
309                            }
310                            _ => None,
311                        })
312                        .unwrap_or_else(|| preview_tab.content.clone()),
313                    source_path: tabs_ref.iter().find_map(|t| match t.read(cx) {
314                        Tab::Editor(editor_tab) if editor_tab.id == preview_tab.source_tab_id => {
315                            editor_tab.location.local_path().cloned()
316                        }
317                        _ => None,
318                    }),
319                    view_state: preview_tab.view_state.clone(),
320                },
321            })
322        });
323
324        if let Some(tab) = active_tab {
325            match tab {
326                ActiveTabRenderData::Editor {
327                    language,
328                    show_markdown_preview,
329                    large_file,
330                    content,
331                    path,
332                    csv_view_mode,
333                    csv_table,
334                    log_view,
335                    log_content,
336                } => {
337                    if log_view && let Some(log_content) = log_content {
338                        let log_input = Editor::new(&log_content)
339                            .disabled(true)
340                            .bordered(false)
341                            .p_0()
342                            .h_full()
343                            .font_family(self.settings.editor_settings.font_family.clone())
344                            .text_size(px(self.settings.editor_settings.font_size));
345                        return v_flex()
346                            .w_full()
347                            .flex_1()
348                            .child(log_input)
349                            .into_any_element();
350                    }
351                    if language == SupportedLanguage::Csv
352                        && csv_view_mode == editor_tab::CsvViewMode::Table
353                        && let Some(table) = csv_table
354                    {
355                        return v_flex()
356                            .w_full()
357                            .flex_1()
358                            .child(DataTable::new(&table).bordered(true).stripe(true))
359                            .into_any_element();
360                    }
361                    let editor_input = Editor::new(&content)
362                        .bordered(false)
363                        .p_0()
364                        .h_full()
365                        .font_family(self.settings.editor_settings.font_family.clone())
366                        .text_size(px(self.settings.editor_settings.font_size));
367                    let capture_right_click =
368                        cx.listener(|this, event: &MouseDownEvent, window, cx| {
369                            this.on_editor_right_click(event, window, cx);
370                        });
371                    if language == SupportedLanguage::Markdown
372                        && show_markdown_preview
373                        && !large_file
374                        && self.settings.editor_settings.markdown_settings.preview_mode
375                            == crate::fulgur::settings::MarkdownPreviewMode::Panel
376                    {
377                        // Reading the content entity here tracks it for this
378                        // window, so edits re-render the panel automatically.
379                        // Multi-line raw-HTML blocks are collapsed first so the
380                        // Markdown renderer never shapes a run containing a
381                        // newline (see `sanitize_markdown_preview`).
382                        let preview_text =
383                            crate::fulgur::utils::sanitize::sanitize_markdown_preview(
384                                &crate::fulgur::utils::markdown_images::rewrite_markdown_image_paths(
385                                    content.read(cx).value().as_ref(),
386                                    path.as_deref().and_then(std::path::Path::parent),
387                                ),
388                            );
389                        let preview_state = self.ensure_markdown_panel_state(&preview_text, cx);
390                        let preview = self
391                            .layout_markdown_preview(
392                                TextView::new(&preview_state)
393                                    .flex_none()
394                                    .py_0()
395                                    .px_2()
396                                    .scrollable(true)
397                                    .selectable(true),
398                            )
399                            .bg(cx.theme().muted)
400                            .into_any_element();
401                        return v_flex()
402                            .w_full()
403                            .flex_1()
404                            .child(
405                                h_resizable("markdown-preview-container")
406                                    .child(
407                                        resizable_panel().child(
408                                            div()
409                                                .id("markdown-editor")
410                                                .size_full()
411                                                .capture_any_mouse_down(capture_right_click)
412                                                .child(editor_input),
413                                        ),
414                                    )
415                                    .child(
416                                        resizable_panel()
417                                            .child(self.wrap_markdown_preview(preview, cx)),
418                                    ),
419                            )
420                            .into_any_element();
421                    }
422                    return v_flex()
423                        .w_full()
424                        .flex_1()
425                        .capture_any_mouse_down(capture_right_click)
426                        .child(editor_input)
427                        .into_any_element();
428                }
429                ActiveTabRenderData::Settings => {
430                    return v_flex()
431                        .id("settings-tab-scrollable")
432                        .w_full()
433                        .flex_1()
434                        .overflow_y_scrollbar()
435                        .child(self.render_settings(window, cx))
436                        .into_any_element();
437                }
438                ActiveTabRenderData::MarkdownPreview {
439                    content,
440                    source_path,
441                    view_state,
442                } => {
443                    let base_dir = source_path.as_deref().and_then(std::path::Path::parent);
444                    let preview_text = crate::fulgur::utils::sanitize::sanitize_markdown_preview(
445                        &crate::fulgur::utils::markdown_images::rewrite_markdown_image_paths(
446                            content.read(cx).value().as_ref(),
447                            base_dir,
448                        ),
449                    );
450                    view_state.update(cx, |state, cx| {
451                        state.set_text(&preview_text, cx);
452                    });
453                    let preview = self
454                        .layout_markdown_preview(
455                            TextView::new(&view_state)
456                                .py_2()
457                                .px_4()
458                                .scrollable(true)
459                                .selectable(true),
460                        )
461                        .into_any_element();
462                    return v_flex()
463                        .w_full()
464                        .flex_1()
465                        .child(self.wrap_markdown_preview(preview, cx))
466                        .into_any_element();
467                }
468            }
469        }
470        v_flex().w_full().flex_1().into_any_element()
471    }
