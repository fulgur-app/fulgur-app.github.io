30    /// Handle a right-click in the editor area to show a custom context menu.
31    ///
32    /// Called during the capture phase so propagation can be stopped before
33    /// the editor's built-in context menu fires.
34    ///
35    /// ### Arguments
36    /// - `event`: The mouse-down event
37    /// - `window`: The window context
38    /// - `cx`: The application context
39    fn on_editor_right_click(
40        &mut self,
41        event: &MouseDownEvent,
42        window: &mut Window,
43        cx: &mut Context<Self>,
44    ) {
45        if event.button != MouseButton::Right {
46            return;
47        }
48        cx.stop_propagation();
49
50        let Some(Tab::Editor(editor_tab)) = self.active_tab(cx) else {
51            return;
52        };
53        let active_tab_id = editor_tab.id;
54        let editor_focus = editor_tab.content.focus_handle(cx);
55        let has_file = editor_tab.file_path().is_some();
56        let position = event.position;
57
58        let menu = PopupMenu::build(window, cx, {
59            let editor_focus = editor_focus.clone();
60            move |mut menu, _window, _cx| {
61                menu = menu.action_context(editor_focus);
62                if has_file {
63                    menu = menu
64                        .menu(
65                            crate::fulgur::ui::components_utils::reveal_in_file_manager_label(),
66                            Box::new(ui::tabs::tab_bar::ShowInFileManager(active_tab_id)),
67                        )
68                        .separator();
69                }
70                menu.menu("Cut", Box::new(gpui_component::input::Cut))
71                    .menu("Copy", Box::new(gpui_component::input::Copy))
72                    .menu("Paste", Box::new(gpui_component::input::Paste))
73                    .separator()
74                    .menu("Select All", Box::new(gpui_component::input::SelectAll))
75            }
76        });
77
78        let subscription = cx.subscribe_in(
79            &menu,
80            window,
81            |this: &mut Self, _, _: &DismissEvent, _, cx| {
82                this.editor_context_menu = None;
83                this.editor_context_menu_subscription = None;
84                cx.notify();
85            },
86        );
87
88        self.editor_context_menu = Some((position, menu));
89        self.editor_context_menu_subscription = Some(subscription);
90        cx.notify();
91    }

93    /// Handle a right-click on the markdown preview to show its context menu.
94    ///
95    /// ### Arguments
96    /// - `event`: The mouse-down event
97    /// - `window`: The window context
98    /// - `cx`: The application context
99    fn on_preview_right_click(
100        &mut self,
101        event: &MouseDownEvent,
102        window: &mut Window,
103        cx: &mut Context<Self>,
104    ) {
105        if event.button != MouseButton::Right {
106            return;
107        }
108        cx.stop_propagation();
109
110        let position = event.position;
111        let preview_focus = self.markdown_preview_focus.clone();
112
113        // The replacement, `gpui_base::TextSelection::selected_text`, lives in the
114        // `gpui-base` crate, which gpui-component does not re-export and which Fulgur
115        // does not depend on directly. `WindowExt::selected_text` forwards to it.
116        #[allow(deprecated)]
117        let selection = window.selected_text(cx).trim().to_string();
118        let has_selection = !selection.is_empty();
119        self.markdown_preview_pending_copy = has_selection.then_some(selection);
120
121        let menu = PopupMenu::build(window, cx, move |menu, _window, _cx| {
122            menu.action_context(preview_focus)
123                .menu_with_enable("Copy", Box::new(gpui_component::input::Copy), has_selection)
124                .menu("Select All", Box::new(gpui_component::input::SelectAll))
125        });
126
127        let subscription = cx.subscribe_in(
128            &menu,
129            window,
130            |this: &mut Self, _, _: &DismissEvent, _, cx| {
131                this.editor_context_menu = None;
132                this.editor_context_menu_subscription = None;
133                cx.notify();
134            },
135        );
136
137        self.editor_context_menu = Some((position, menu));
138        self.editor_context_menu_subscription = Some(subscription);
139        cx.notify();
140    }

148    /// - `Some(Entity<TextViewState>)`: The state of the active preview, either a

159    /// Ensure the inline preview panel owns a persistent text view state.
160    ///
161    /// ### Arguments
162    /// - `text`: The rendered markdown source to display
163    /// - `cx`: The application context
164    ///
165    /// ### Returns
166    /// - `Entity<TextViewState>`: The persistent state for the inline preview.
167    fn ensure_markdown_panel_state(
168        &mut self,
169        text: &str,
170        cx: &mut Context<Self>,
171    ) -> Entity<TextViewState> {
172        let state = self
173            .markdown_panel_view_state
174            .get_or_insert_with(|| cx.new(|cx| TextViewState::markdown(text, cx)))
175            .clone();
176        state.update(cx, |state, cx| state.set_text(text, cx));
177        state
178    }

180    /// Build the link activation handler for a markdown preview.
181    ///
182    /// ### Arguments
183    /// - `base_dir`: The directory of the previewed file, used to resolve
184    ///   relative links. `None` for a buffer that has never been saved.
185    /// - `cx`: The application context
186    ///
187    /// ### Returns
188    /// - `impl Fn(...)`: A handler that hands remote links to the system and
189    ///   opens local files in a tab of this window.
190    fn markdown_link_handler(
191        base_dir: Option<PathBuf>,
192        cx: &Context<Self>,
193    ) -> impl Fn(&SharedString, &ClickEvent, &mut Window, &mut App) + use<> {
194        let view = cx.entity().downgrade();
195        move |target, _event, window, cx| {
196            match resolve_markdown_link(target, base_dir.as_deref()) {
197                Some(MarkdownLinkTarget::External(url)) => {
198                    log::debug!("Opening markdown preview link {url} in the system handler");
199                    cx.open_url(&url);
200                }
201                Some(MarkdownLinkTarget::LocalFile(path)) => {
202                    if !path.is_file() {
203                        log::warn!(
204                            "Markdown preview link points at a missing file: {}",
205                            path.display()
206                        );
207                        window.push_notification(
208                            (
209                                NotificationType::Warning,
210                                SharedString::from(format!(
211                                    "Cannot open '{}': the file does not exist",
212                                    path.display()
213                                )),
214                            ),
215                            cx,
216                        );
217                        return;
218                    }
219                    view.update(cx, |this, cx| this.do_open_file(window, cx, path))
220                        .ok();
221                }
222                // An anchor has no scroll target in the preview, and an
223                // unresolvable reference has nowhere to go.
224                Some(MarkdownLinkTarget::Anchor) | None => {}
225            }
226        }
227    }

276    /// Lay out a markdown preview inside its container, honouring the width limit setting.
277    ///
278    /// ### Arguments
279    /// - `preview`: The preview text view to lay out
280    ///
281    /// ### Returns
282    /// - `Div`: A full-size container holding the preview, capped at
283    ///   `MARKDOWN_PREVIEW_MAX_WIDTH` and centered when the limit is enabled.
284    fn layout_markdown_preview(&self, preview: TextView) -> Div {
285        let limited = self
286            .settings
287            .editor_settings
288            .markdown_settings
289            .limit_preview_width;
290        h_flex()
291            .size_full()
292            .justify_center()
293            .child(preview.when(limited, |view| view.max_w(px(MARKDOWN_PREVIEW_MAX_WIDTH))))
294    }

296    /// Wrap a markdown preview element with its context-menu affordances.
297    ///
298    /// ### Arguments
299    /// - `child`: The preview element to wrap
300    /// - `cx`: The application context
301    ///
302    /// ### Returns
303    /// - `impl IntoElement`: The wrapped, right-clickable preview element.
304    fn wrap_markdown_preview(&self, child: AnyElement, cx: &mut Context<Self>) -> impl IntoElement {
305        let right_click = cx.listener(|this, event: &MouseDownEvent, window, cx| {
306            this.on_preview_right_click(event, window, cx);
307        });
308        let copy = cx.listener(|this, _: &gpui_component::input::Copy, _window, cx| {
309            if let Some(text) = this.markdown_preview_pending_copy.take() {
310                cx.write_to_clipboard(gpui::ClipboardItem::new_string(text));
311            }
312        });
313        let select_all = cx.listener(|this, _: &gpui_component::input::SelectAll, _window, cx| {
314            if let Some(state) = this.active_markdown_preview_state(cx) {
315                state.update(cx, TextViewState::select_all);
316            }
317        });
318
319        div()
320            .id("markdown-preview-context")
321            .track_focus(&self.markdown_preview_focus)
322            .size_full()
323            .capture_any_mouse_down(right_click)
324            .on_action(copy)
325            .on_action(select_all)
326            .child(child)
327    }

329    /// Render the content area (editor or settings)
330    ///
331    /// ### Arguments
332    /// - `active_tab_index`: The index of the active tab (if any)
333    /// - `window`: The window context
334    /// - `cx`: The application context
335    ///
336    /// ### Returns
337    /// - `AnyElement`: The rendered content area element (wrapped in `AnyElement`)
338    pub(super) fn render_content_area(
339        &mut self,
340        active_tab_index: Option<usize>,
341        window: &mut Window,
342        cx: &mut Context<Self>,
343    ) -> AnyElement {
344        enum ActiveTabRenderData {
345            Editor {
346                language: SupportedLanguage,
347                show_markdown_preview: bool,
348                large_file: bool,
349                content: Entity<EditorState>,
350                path: Option<std::path::PathBuf>,
351                csv_view_mode: editor_tab::CsvViewMode,
352                csv_table: Option<Entity<TableState<editor_tab::CsvTableDelegate>>>,
353                log_view: bool,
354                log_content: Option<Entity<EditorState>>,
355            },
356            Settings,
357            MarkdownPreview {
358                content: Entity<EditorState>,
359                source_path: Option<std::path::PathBuf>,
360                view_state: Entity<gpui_component::text::TextViewState>,
361            },
362        }
363
364        // A CSV tab in table mode needs its grid (re)built from the canonical
365        // text before we snapshot the tab read-only below. When the parse is
366        // lossy, `ensure_csv_table` falls back to text mode and returns a
367        // warning to surface to the user.
368        let csv_table_warning = if let Some(active_index) = active_tab_index
369            && let Some(tab_entity) = self.tabs.get(active_index).cloned()
370        {
371            tab_entity.update(cx, |tab, cx| match tab {
372                Tab::Editor(editor_tab)
373                    if editor_tab.language == SupportedLanguage::Csv
374                        && editor_tab.csv_view_mode == editor_tab::CsvViewMode::Table =>
375                {
376                    editor_tab.ensure_csv_table(window, cx)
377                }
378                _ => None,
379            })
380        } else {
381            None
382        };
383        if let Some(message) = csv_table_warning {
384            window.push_notification((NotificationType::Warning, SharedString::from(message)), cx);
385        }
386
387        let tabs_ref = &self.tabs;
388        let active_tab = active_tab_index.and_then(|active_index| {
389            tabs_ref.get(active_index).map(|tab| match tab.read(cx) {
390                Tab::Editor(editor_tab) => ActiveTabRenderData::Editor {
391                    language: editor_tab.language,
392                    show_markdown_preview: editor_tab.show_markdown_preview,
393                    large_file: editor_tab.large_file,
394                    content: editor_tab.content.clone(),
395                    path: editor_tab.location.local_path().cloned(),
396                    csv_view_mode: editor_tab.csv_view_mode,
397                    csv_table: editor_tab.csv_table.clone(),
398                    log_view: editor_tab.log_view,
399                    log_content: editor_tab.log_content.clone(),
400                },
401                Tab::Settings(_) => ActiveTabRenderData::Settings,
402                Tab::MarkdownPreview(preview_tab) => ActiveTabRenderData::MarkdownPreview {
403                    content: tabs_ref
404                        .iter()
405                        .find_map(|t| match t.read(cx) {
406                            Tab::Editor(editor_tab)
407                                if editor_tab.id == preview_tab.source_tab_id =>
408                            {
409                                Some(editor_tab.content.clone())
410                            }
411                            _ => None,
412                        })
413                        .unwrap_or_else(|| preview_tab.content.clone()),
414                    source_path: tabs_ref.iter().find_map(|t| match t.read(cx) {
415                        Tab::Editor(editor_tab) if editor_tab.id == preview_tab.source_tab_id => {
416                            editor_tab.location.local_path().cloned()
417                        }
418                        _ => None,
419                    }),
420                    view_state: preview_tab.view_state.clone(),
421                },
422            })
423        });
424
425        if let Some(tab) = active_tab {
426            match tab {
427                ActiveTabRenderData::Editor {
428                    language,
429                    show_markdown_preview,
430                    large_file,
431                    content,
432                    path,
433                    csv_view_mode,
434                    csv_table,
435                    log_view,
436                    log_content,
437                } => {
438                    if log_view && let Some(log_content) = log_content {
439                        let log_input = Editor::new(&log_content)
440                            .disabled(true)
441                            .bordered(false)
442                            .p_0()
443                            .h_full()
444                            .font_family(self.settings.editor_settings.font_family.clone())
445                            .text_size(px(self.settings.editor_settings.font_size));
446                        return v_flex()
447                            .w_full()
448                            .flex_1()
449                            .child(log_input)
450                            .into_any_element();
451                    }
452                    if language == SupportedLanguage::Csv
453                        && csv_view_mode == editor_tab::CsvViewMode::Table
454                        && let Some(table) = csv_table
455                    {
456                        return v_flex()
457                            .w_full()
458                            .flex_1()
459                            .child(DataTable::new(&table).bordered(true).stripe(true))
460                            .into_any_element();
461                    }
462                    let editor_input = Editor::new(&content)
463                        .bordered(false)
464                        .p_0()
465                        .h_full()
466                        .font_family(self.settings.editor_settings.font_family.clone())
467                        .text_size(px(self.settings.editor_settings.font_size));
468                    let capture_right_click =
469                        cx.listener(|this, event: &MouseDownEvent, window, cx| {
470                            this.on_editor_right_click(event, window, cx);
471                        });
472                    if language == SupportedLanguage::Markdown
473                        && show_markdown_preview
474                        && !large_file
475                        && self.settings.editor_settings.markdown_settings.preview_mode
476                            == crate::fulgur::settings::MarkdownPreviewMode::Panel
477                    {
478                        // Reading the content entity here tracks it for this
479                        // window, so edits re-render the panel automatically.
480                        // Multi-line raw-HTML blocks are collapsed first so the
481                        // Markdown renderer never shapes a run containing a
482                        // newline (see `sanitize_markdown_preview`).
483                        let preview_text =
484                            crate::fulgur::utils::sanitize::sanitize_markdown_preview(
485                                &crate::fulgur::utils::markdown_images::rewrite_markdown_image_paths(
486                                    content.read(cx).value().as_ref(),
487                                    path.as_deref().and_then(std::path::Path::parent),
488                                ),
489                            );
490                        let preview_state = self.ensure_markdown_panel_state(&preview_text, cx);
491                        let link_handler = Self::markdown_link_handler(
492                            path.as_deref()
493                                .and_then(std::path::Path::parent)
494                                .map(std::path::Path::to_path_buf),
495                            cx,
496                        );
497                        let preview = self
498                            .layout_markdown_preview(
499                                TextView::new(&preview_state)
500                                    .flex_none()
501                                    .py_0()
502                                    .px_2()
503                                    .scrollable(true)
504                                    .selectable(true)
505                                    .on_link_click(link_handler)
506                                    .code_block_actions(|code_block, _window, _cx| {
507                                        Self::markdown_code_block_copy(code_block.code())
508                                    })
509                                    .table_actions(Self::markdown_table_actions),
510                            )
511                            .bg(cx.theme().muted)
512                            .into_any_element();
513                        return v_flex()
514                            .w_full()
515                            .flex_1()
516                            .child(
517                                h_resizable("markdown-preview-container")
518                                    .child(
519                                        resizable_panel().child(
520                                            div()
521                                                .id("markdown-editor")
522                                                .size_full()
523                                                .capture_any_mouse_down(capture_right_click)
524                                                .child(editor_input),
525                                        ),
526                                    )
527                                    .child(
528                                        resizable_panel()
529                                            .child(self.wrap_markdown_preview(preview, cx)),
530                                    ),
531                            )
532                            .into_any_element();
533                    }
534                    return v_flex()
535                        .w_full()
536                        .flex_1()
537                        .capture_any_mouse_down(capture_right_click)
538                        .child(editor_input)
539                        .into_any_element();
540                }
541                ActiveTabRenderData::Settings => {
542                    return v_flex()
543                        .id("settings-tab-scrollable")
544                        .w_full()
545                        .flex_1()
546                        .overflow_y_scrollbar()
547                        .child(self.render_settings(window, cx))
548                        .into_any_element();
549                }
550                ActiveTabRenderData::MarkdownPreview {
551                    content,
552                    source_path,
553                    view_state,
554                } => {
555                    let base_dir = source_path.as_deref().and_then(std::path::Path::parent);
556                    let preview_text = crate::fulgur::utils::sanitize::sanitize_markdown_preview(
557                        &crate::fulgur::utils::markdown_images::rewrite_markdown_image_paths(
558                            content.read(cx).value().as_ref(),
559                            base_dir,
560                        ),
561                    );
562                    view_state.update(cx, |state, cx| {
563                        state.set_text(&preview_text, cx);
564                    });
565                    let link_handler =
566                        Self::markdown_link_handler(base_dir.map(std::path::Path::to_path_buf), cx);
567                    let preview = self
568                        .layout_markdown_preview(
569                            TextView::new(&view_state)
570                                .py_2()
571                                .px_4()
572                                .scrollable(true)
573                                .selectable(true)
574                                .on_link_click(link_handler)
575                                .code_block_actions(|code_block, _window, _cx| {
576                                    Self::markdown_code_block_copy(code_block.code())
577                                })
578                                .table_actions(Self::markdown_table_actions),
579                        )
580                        .into_any_element();
581                    return v_flex()
582                        .w_full()
583                        .flex_1()
584                        .child(self.wrap_markdown_preview(preview, cx))
585                        .into_any_element();
586                }
587            }
588        }
589        v_flex().w_full().flex_1().into_any_element()
590    }
591}

593/// Serialize a markdown preview table to CSV text.
594///
595/// ### Arguments
596/// - `table`: The table snapshot taken from the rendered document
597///
598/// ### Returns
599/// - `Some(String)`: The table as CSV, with the delimiter Fulgur's own CSV
600///   view defaults to
601/// - `None`: The rows could not be serialized; the reason is logged
602fn markdown_table_as_csv(table: &TableData) -> Option<String> {
603    serialize_csv(&table.headers, &table.rows, DEFAULT_DELIMITER)
604        .inspect_err(|error| log::warn!("Cannot offer the preview table as CSV: {error}"))
605        .ok()
606}
