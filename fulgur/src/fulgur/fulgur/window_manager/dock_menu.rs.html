12    /// Compute a lightweight fingerprint for this window's dock/jump-list inputs.
13    ///
14    /// Includes all local tabs (id/title/path) and recent files to detect when
15    /// system menu state may need rebuilding.
16    ///
17    /// ### Returns
18    /// - `u64`: Fingerprint of this window's menu-relevant state
19    #[cfg(any(target_os = "macos", target_os = "windows"))]
20    fn compute_local_menu_fingerprint(&mut self) -> u64 {
21        let mut hasher = std::collections::hash_map::DefaultHasher::new();
22        for tab in &self.tabs {
23            tab.id().hash(&mut hasher);
24            tab.title().hash(&mut hasher);
25            if let Some(path) = tab.as_editor().and_then(|editor| editor.file_path.as_ref()) {
26                path.hash(&mut hasher);
27            }
28        }
29        for file in self.settings.get_recent_files() {
30            file.hash(&mut hasher);
31        }
32        hasher.finish()
33    }

35    /// Publish this window's local menu fingerprint to the `WindowManager`.
36    ///
37    /// ### Arguments
38    /// - `cx`: The application context
39    #[cfg(any(target_os = "macos", target_os = "windows"))]
40    pub(super) fn publish_window_menu_fingerprint_if_changed(&mut self, cx: &mut Context<Self>) {
41        let fingerprint = self.compute_local_menu_fingerprint();
42        let already_published = cx
43            .global::<WindowManager>()
44            .get_window_menu_fingerprint(self.window_id)
45            == Some(fingerprint);
46        if fingerprint == self.local_window_menu_fingerprint && already_published {
47            return;
48        }
49        self.local_window_menu_fingerprint = fingerprint;
50        cx.update_global::<WindowManager, _>(|manager, _| {
51            manager.update_window_menu_fingerprint(self.window_id, fingerprint);
52        });
53    }

55    /// Update the macOS dock menu if the open tabs or recent files have changed.
56    ///
57    /// Computes a hash of the current state (all tabs across all windows and recent files)
58    /// and only rebuilds the dock menu when the hash differs from the last known state.
59    ///
60    /// ### Arguments
61    /// - `cx`: The application context
62    #[cfg(target_os = "macos")]
63    pub(super) fn update_dock_menu_if_changed(&mut self, cx: &mut Context<Self>) {
64        use gpui::{SharedString, WeakEntity};
65
66        use crate::fulgur::ui::menus::{DockMenuTab, build_dock_menu};
67        let menu_state_revision = cx.global::<WindowManager>().menu_state_revision();
68        if menu_state_revision == self.last_dock_menu_revision {
69            return;
70        }
71        self.last_dock_menu_revision = menu_state_revision;
72        let mut hasher = std::collections::hash_map::DefaultHasher::new();
73
74        // Collect raw tab data from all windows: (file_path_or_none, title, window_group_index)
75        // We need all file paths upfront to compute cross-window duplicate filenames.
76        struct RawTab {
77            path: Option<PathBuf>,
78            title: SharedString,
79        }
80
81        let mut all_windows_raw: Vec<Vec<RawTab>> = Vec::new();
82        let current_window_raw: Vec<RawTab> = self
83            .tabs
84            .iter()
85            .map(|tab| {
86                let path = tab.as_editor().and_then(|e| e.file_path.clone());
87                if let Some(ref p) = path {
88                    p.hash(&mut hasher);
89                }
90                tab.title().hash(&mut hasher);
91                RawTab {
92                    path,
93                    title: tab.title(),
94                }
95            })
96            .collect();
97        all_windows_raw.push(current_window_raw);
98        let manager = cx.global::<WindowManager>();
99        let other_windows: Vec<WeakEntity<Fulgur>> = manager
100            .get_all_window_ids()
101            .into_iter()
102            .filter(|id| *id != self.window_id)
103            .filter_map(|id| manager.get_window(id))
104            .collect();
105        for weak in other_windows {
106            if let Some(entity) = weak.upgrade() {
107                let other = entity.read(cx);
108                let window_raw: Vec<RawTab> = other
109                    .tabs
110                    .iter()
111                    .map(|tab| {
112                        let path = tab.as_editor().and_then(|e| e.file_path.clone());
113                        if let Some(ref p) = path {
114                            p.hash(&mut hasher);
115                        }
116                        tab.title().hash(&mut hasher);
117                        RawTab {
118                            path,
119                            title: tab.title(),
120                        }
121                    })
122                    .collect();
123                all_windows_raw.push(window_raw);
124            }
125        }
126        let recent_files = self.settings.get_recent_files();
127        for file in &recent_files {
128            file.hash(&mut hasher);
129        }
130        let hash = hasher.finish();
131        if hash == self.last_dock_menu_hash {
132            return;
133        }
134        self.last_dock_menu_hash = hash;
135        let all_file_paths: Vec<PathBuf> = all_windows_raw
136            .iter()
137            .flat_map(|w| w.iter())
138            .filter_map(|t| t.path.clone())
139            .collect();
140        let display_name_for_path = |path: &PathBuf| -> SharedString {
141            let filename = path
142                .file_name()
143                .and_then(|n| n.to_str())
144                .unwrap_or("Untitled");
145            let has_duplicate = all_file_paths.iter().any(|other_path| {
146                other_path != path
147                    && (other_path.file_name().and_then(|n| n.to_str()) == Some(filename))
148            });
149            if has_duplicate
150                && let Some(parent_name) = path
151                    .parent()
152                    .and_then(|p| p.file_name())
153                    .and_then(|n| n.to_str())
154            {
155                return SharedString::from(format!("{} (../{})", filename, parent_name));
156            }
157            SharedString::from(filename.to_string())
158        };
159        let windows: Vec<Vec<DockMenuTab>> = all_windows_raw
160            .into_iter()
161            .map(|window_raw| {
162                window_raw
163                    .into_iter()
164                    .map(|raw| match raw.path {
165                        Some(path) => DockMenuTab::File {
166                            name: display_name_for_path(&path),
167                            path,
168                        },
169                        None => DockMenuTab::Titled {
170                            name: raw.title.clone(),
171                            title: raw.title,
172                        },
173                    })
174                    .collect()
175            })
176            .collect();
177        let dock_items = build_dock_menu(&windows, &recent_files);
178        cx.set_dock_menu(dock_items);
179    }

304    /// Handle the DockActivateTab action: focus the tab with the given file path,
305    /// switching to the correct window if necessary.
306    ///
307    /// ### Arguments
308    /// - `action`: The DockActivateTab action containing the file path
309    /// - `window`: The current window
310    /// - `cx`: The application context
311    pub fn handle_dock_activate_tab(
312        &mut self,
313        action: &crate::fulgur::ui::menus::DockActivateTab,
314        window: &mut Window,
315        cx: &mut Context<Self>,
316    ) {
317        let path = action.0.clone();
318        if let Some(tab_index) = self.find_tab_by_path(&path) {
319            self.set_active_tab(tab_index, window, cx);
320            self.focus_active_tab(window, cx);
321            cx.notify();
322            return;
323        }
324        let manager = cx.global::<WindowManager>();
325        let other_windows: Vec<WeakEntity<Fulgur>> = manager
326            .get_all_window_ids()
327            .into_iter()
328            .filter(|id| *id != self.window_id)
329            .filter_map(|id| manager.get_window(id))
330            .collect();
331        let mut target: Option<(WindowId, Entity<Fulgur>)> = None;
332        for weak in other_windows {
333            if let Some(entity) = weak.upgrade() {
334                let other = entity.read(cx);
335                if other.find_tab_by_path(&path).is_some() {
336                    target = Some((other.window_id, entity.clone()));
337                    break;
338                }
339            }
340        }
341        if let Some((target_wid, target_entity)) = target {
342            cx.spawn_in(window, {
343                let path = path.clone();
344                async move |_this, async_window| {
345                    async_window
346                        .update(|_window, cx| {
347                            target_entity.update(cx, |fulgur, cx| {
348                                if let Some(tab_index) = fulgur.find_tab_by_path(&path) {
349                                    fulgur.tab_scroll_handle.scroll_to_item(tab_index);
350                                    fulgur.active_tab_index = Some(tab_index);
351                                    cx.notify();
352                                }
353                            });
354                            for handle in cx.windows() {
355                                if handle.window_id() == target_wid {
356                                    handle
357                                        .update(cx, |_, target_window, _| {
358                                            target_window.activate_window();
359                                        })
360                                        .ok();
361                                    break;
362                                }
363                            }
364                        })
365                        .ok();
366                }
367            })
368            .detach();
369        }
370    }

441    /// Process pending IPC commands from the Windows jump list listener
442    ///
443    /// Drains the `pending_ipc_commands` queue and executes each command
444    /// in-process. Only the last-focused window processes the queue, mirroring
445    /// the behaviour of `process_pending_files_from_macos`.
446    ///
447    /// ### Arguments
448    /// - `window`: The window being rendered
449    /// - `cx`: The application context
450    #[cfg(target_os = "windows")]
451    pub fn process_pending_ipc_commands(&mut self, window: &mut Window, cx: &mut Context<Self>) {
452        let should_process = cx
453            .global::<WindowManager>()
454            .get_last_focused()
455            .map(|id| id == self.window_id)
456            .unwrap_or(true);
457        if !should_process {
458            return;
459        }
460        let commands: Vec<String> = {
461            let shared = self.shared_state(cx);
462            if let Some(mut q) = shared.pending_ipc_commands.try_lock() {
463                q.drain(..).collect()
464            } else {
465                Vec::new()
466            }
467        };
468        for cmd in commands {
469            match cmd.as_str() {
470                "new-tab" => {
471                    log::info!("IPC: opening new tab");
472                    self.new_tab(window, cx);
473                }
474                "new-window" => {
475                    log::info!("IPC: opening new window");
476                    self.open_new_window(cx);
477                }
478                other => {
479                    log::warn!("IPC: unknown command '{other}'");
480                }
481            }
482        }
483    }
