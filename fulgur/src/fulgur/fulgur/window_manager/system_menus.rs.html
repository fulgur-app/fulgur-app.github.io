1//! App-scope rebuilding of the OS-level menus (macOS dock menu, Windows jump list).
2
3use gpui::SharedString;
4use std::path::PathBuf;
5
6/// Menu-relevant snapshot of one tab, published per window to the `WindowManager`
7#[derive(Clone, Debug, PartialEq, Eq)]
8pub struct WindowMenuTab {
9    /// File backing the tab, when it has one (used for reliable cross-window lookup)
10    pub path: Option<PathBuf>,
11    /// Tab title (used for lookup of non-file tabs)
12    pub title: SharedString,
13}
14
15/// Register the app-scope observers that keep the OS menus in sync.
16///
17/// ### Arguments
18/// - `cx`: The application context
19pub fn init(cx: &mut gpui::App) {
20    #[cfg(any(target_os = "macos", target_os = "windows"))]
21    {
22        use crate::fulgur::shared_state::SharedAppState;
23        use crate::fulgur::window_manager::WindowManager;
24        use std::cell::RefCell;
25        use std::rc::Rc;
26
27        let last_inputs: Rc<RefCell<Option<SystemMenuInputs>>> = Rc::new(RefCell::new(None));
28        let inputs_for_manager = last_inputs.clone();
29        cx.observe_global::<WindowManager>(move |cx| {
30            rebuild_if_inputs_changed(cx, &inputs_for_manager);
31        })
32        .detach();
33        cx.observe_global::<SharedAppState>(move |cx| {
34            rebuild_if_inputs_changed(cx, &last_inputs);
35        })
36        .detach();
37    }
38    #[cfg(not(any(target_os = "macos", target_os = "windows")))]
39    {
40        let _ = cx;
41    }
42}

44/// Combined inputs of the system menus, compared between rebuilds
45#[cfg(any(target_os = "macos", target_os = "windows"))]
46#[derive(PartialEq, Eq)]
47struct SystemMenuInputs {
48    /// Published tab snapshots of every window, in registration order
49    windows: Vec<Vec<WindowMenuTab>>,
50    /// Recent file paths, most recent first
51    recent_files: Vec<PathBuf>,
52}
53
54/// Rebuild the dock menu / jump list when the menu inputs changed.
55///
56/// ### Arguments
57/// - `cx`: The application context
58/// - `last_inputs`: Inputs of the previous build, used to skip no-op rebuilds
59#[cfg(any(target_os = "macos", target_os = "windows"))]
60fn rebuild_if_inputs_changed(
61    cx: &mut gpui::App,
62    last_inputs: &std::cell::RefCell<Option<SystemMenuInputs>>,
63) {
64    use crate::fulgur::shared_state::SharedAppState;
65    use crate::fulgur::window_manager::WindowManager;
66
67    let inputs = SystemMenuInputs {
68        windows: cx.global::<WindowManager>().ordered_window_menu_tabs(),
69        recent_files: cx.global::<SharedAppState>().settings.get_recent_files(),
70    };
71    if last_inputs.borrow().as_ref() == Some(&inputs) {
72        return;
73    }
74    let menu_tabs = to_dock_menu_tabs(&inputs.windows);
75    #[cfg(target_os = "macos")]
76    cx.set_dock_menu(crate::fulgur::ui::menus::build_dock_menu(
77        &menu_tabs,
78        &inputs.recent_files,
79    ));
80    #[cfg(target_os = "windows")]
81    crate::fulgur::utils::jump_list::update_windows_jump_list(&menu_tabs, &inputs.recent_files);
82    *last_inputs.borrow_mut() = Some(inputs);
83}

85/// Convert published tab snapshots to dock menu entries with display names.
86///
87/// ### Arguments
88/// - `windows`: Published tab snapshots of every window
89///
90/// ### Returns
91/// - `Vec<Vec<DockMenuTab>>`: Menu entries grouped by window
92#[cfg(any(target_os = "macos", target_os = "windows"))]
93fn to_dock_menu_tabs(
94    windows: &[Vec<WindowMenuTab>],
95) -> Vec<Vec<crate::fulgur::ui::menus::DockMenuTab>> {
96    use crate::fulgur::ui::menus::DockMenuTab;
97
98    let all_file_paths: Vec<&PathBuf> = windows
99        .iter()
100        .flat_map(|window| window.iter())
101        .filter_map(|tab| tab.path.as_ref())
102        .collect();
103    let display_name_for_path = |path: &PathBuf| -> SharedString {
104        let filename = path
105            .file_name()
106            .and_then(|n| n.to_str())
107            .unwrap_or("Untitled");
108        let has_duplicate = all_file_paths.iter().any(|other_path| {
109            *other_path != path
110                && (other_path.file_name().and_then(|n| n.to_str()) == Some(filename))
111        });
112        if has_duplicate
113            && let Some(parent_name) = path
114                .parent()
115                .and_then(|p| p.file_name())
116                .and_then(|n| n.to_str())
117        {
118            return SharedString::from(format!("{filename} (../{parent_name})"));
119        }
120        SharedString::from(filename.to_string())
121    };
122    windows
123        .iter()
124        .map(|window| {
125            window
126                .iter()
127                .map(|tab| match &tab.path {
128                    Some(path) => DockMenuTab::File {
129                        name: display_name_for_path(path),
130                        path: path.clone(),
131                    },
132                    None => DockMenuTab::Titled {
133                        name: tab.title.clone(),
134                        title: tab.title.clone(),
135                    },
136                })
137                .collect()
138        })
139        .collect()
140}

142#[cfg(all(test, any(target_os = "macos", target_os = "windows")))]
143mod tests {
144    use super::{WindowMenuTab, to_dock_menu_tabs};
145    use crate::fulgur::ui::menus::DockMenuTab;
146    use std::path::PathBuf;
147
148    /// Build a file-backed snapshot tab for tests.
149    fn file_tab(path: &str) -> WindowMenuTab {
150        WindowMenuTab {
151            path: Some(PathBuf::from(path)),
152            title: gpui::SharedString::from(
153                PathBuf::from(path)
154                    .file_name()
155                    .and_then(|n| n.to_str())
156                    .unwrap_or_default()
157                    .to_string(),
158            ),
159        }
160    }
161
162    #[test]
163    fn test_to_dock_menu_tabs_uses_plain_filename_when_unique() {
164        let windows = vec![vec![file_tab("/projects/alpha/main.rs")]];
165        let converted = to_dock_menu_tabs(&windows);
166        match &converted[0][0] {
167            DockMenuTab::File { name, .. } => {
168                assert_eq!(
169                    name.as_ref(),
170                    "main.rs",
171                    "unique filenames should not carry a parent-directory suffix"
172                );
173            }
174            DockMenuTab::Titled { .. } => panic!("file-backed tab should convert to File entry"),
175        }
176    }
177
178    #[test]
179    fn test_to_dock_menu_tabs_disambiguates_duplicate_filenames_across_windows() {
180        let windows = vec![
181            vec![file_tab("/projects/alpha/main.rs")],
182            vec![file_tab("/projects/beta/main.rs")],
183        ];
184        let converted = to_dock_menu_tabs(&windows);
185        match (&converted[0][0], &converted[1][0]) {
186            (DockMenuTab::File { name: first, .. }, DockMenuTab::File { name: second, .. }) => {
187                assert_eq!(
188                    first.as_ref(),
189                    "main.rs (../alpha)",
190                    "duplicate filenames should be disambiguated with the parent directory"
191                );
192                assert_eq!(
193                    second.as_ref(),
194                    "main.rs (../beta)",
195                    "duplicate filenames should be disambiguated with the parent directory"
196                );
197            }
198            _ => panic!("file-backed tabs should convert to File entries"),
199        }
200    }
201
202    #[test]
203    fn test_to_dock_menu_tabs_maps_titled_tabs_to_title_lookup() {
204        let windows = vec![vec![WindowMenuTab {
205            path: None,
206            title: gpui::SharedString::from("Settings"),
207        }]];
208        let converted = to_dock_menu_tabs(&windows);
209        match &converted[0][0] {
210            DockMenuTab::Titled { name, title } => {
211                assert_eq!(name.as_ref(), "Settings");
212                assert_eq!(title.as_ref(), "Settings");
213            }
214            DockMenuTab::File { .. } => panic!("path-less tab should convert to Titled entry"),
215        }
216    }
