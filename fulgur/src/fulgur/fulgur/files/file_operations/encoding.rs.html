4/// Number of leading bytes inspected by the binary-content heuristic.
5const BINARY_SNIFF_LEN: usize = 8_000;

7/// Outcome of decoding raw file bytes into editor text.
8pub struct DecodedContents {
9    pub encoding: String,
10    pub content: String,
11    pub lossy: bool,
12}
13
14/// Outcome of re-encoding editor text for writing back to disk.
15pub enum EncodedContents {
16    Encoded(Vec<u8>),
17    Lossy,
18}
19
20/// Detect encoding from file bytes
21///
22/// ### Arguments
23/// - `bytes`: The bytes to detect encoding from
24///
25/// ### Returns
26/// - `DecodedContents`: The detected encoding, decoded text, and whether the decode was lossy
27pub fn detect_encoding_and_decode(bytes: &[u8]) -> DecodedContents {
28    if let Ok(text) = std::str::from_utf8(bytes) {
29        log::debug!("File encoding detected as UTF-8");
30        return DecodedContents {
31            encoding: UTF_8.to_string(),
32            content: text.to_string(),
33            lossy: false,
34        };
35    }
36    let mut detector = EncodingDetector::new(Iso2022JpDetection::Allow);
37    detector.feed(bytes, true);
38    let encoding = detector.guess(None, Utf8Detection::Allow);
39    let (decoded, _, had_errors) = encoding.decode(bytes);
40    if had_errors {
41        log::warn!("File encoding detection failed, using UTF-8 lossy conversion");
42        return DecodedContents {
43            encoding: UTF_8.to_string(),
44            content: String::from_utf8_lossy(bytes).to_string(),
45            lossy: true,
46        };
47    }
48    let encoding_name = encoding.name().to_string();
49    log::debug!("File encoding detected as: {encoding_name}");
50    DecodedContents {
51        encoding: encoding_name,
52        content: decoded.to_string(),
53        lossy: false,
54    }
55}

57/// Re-encode editor text into bytes using the tab's stored encoding label.
58///
59/// ### Arguments
60/// - `contents`: The editor text to encode
61/// - `label`: The target encoding label (the tab's `encoding` field)
62///
63/// ### Returns
64/// - `EncodedContents::Encoded`: The encoded bytes ready to write
65/// - `EncodedContents::Lossy`: The text cannot be represented losslessly in the
66///   target encoding
67pub fn encode_for_save(contents: &str, label: &str) -> EncodedContents {
68    if label.eq_ignore_ascii_case(UTF_8) {
69        return EncodedContents::Encoded(contents.as_bytes().to_vec());
70    }
71    let Some(encoding) = encoding_rs::Encoding::for_label(label.as_bytes()) else {
72        log::warn!("Unknown encoding label '{label}', saving as UTF-8");
73        return EncodedContents::Encoded(contents.as_bytes().to_vec());
74    };
75    if encoding == encoding_rs::UTF_16LE
76        || encoding == encoding_rs::UTF_16BE
77        || encoding == encoding_rs::REPLACEMENT
78    {
79        log::warn!("Cannot encode to '{label}', saving as UTF-8");
80        return EncodedContents::Encoded(contents.as_bytes().to_vec());
81    }
82    let (encoded, _, had_unmappable) = encoding.encode(contents);
83    if had_unmappable {
84        EncodedContents::Lossy
85    } else {
86        EncodedContents::Encoded(encoded.into_owned())
87    }
88}

90/// Heuristically determine whether bytes represent a binary (non-text) file.
91///
92/// ### Arguments
93/// - `bytes`: The file bytes to inspect
94///
95/// ### Returns
96/// - `bool`: `true` if the prefix contains a NUL byte
97pub fn looks_binary(bytes: &[u8]) -> bool {
98    let prefix = &bytes[..bytes.len().min(BINARY_SNIFF_LEN)];
99    prefix.contains(&0)
100}

107    #[test]
108    fn test_detect_encoding_returns_utf8_for_valid_utf8_text() {
109        let text = "Hello, world! Fulgur rocks.";
110        let decoded = detect_encoding_and_decode(text.as_bytes());
111        assert_eq!(decoded.encoding, UTF_8);
112        assert_eq!(decoded.content, text);
113        assert!(!decoded.lossy);
114    }

116    #[test]
117    fn test_detect_encoding_returns_utf8_for_ascii_content() {
118        let text = "fn main() { println!(\"hi\"); }";
119        let decoded = detect_encoding_and_decode(text.as_bytes());
120        assert_eq!(decoded.encoding, UTF_8);
121        assert_eq!(decoded.content, text);
122        assert!(!decoded.lossy);
123    }

125    #[test]
126    fn test_detect_encoding_detects_non_utf8_encoding() {
127        // 0xE9 is 'é' in Latin-1 but not a valid UTF-8 byte sequence on its own
128        let bytes: &[u8] = &[0x63, 0x61, 0x66, 0xE9]; // "café" in Latin-1
129        let decoded = detect_encoding_and_decode(bytes);
130        assert_ne!(decoded.encoding, UTF_8);
131        assert!(!decoded.content.is_empty());
132    }

134    #[test]
135    fn test_encode_for_save_roundtrips_latin1() {
136        // "café" decoded from Latin-1, re-encoded must restore the original bytes.
137        let original: &[u8] = &[0x63, 0x61, 0x66, 0xE9];
138        let decoded = detect_encoding_and_decode(original);
139        let EncodedContents::Encoded(bytes) = encode_for_save(&decoded.content, &decoded.encoding)
140        else {
141            panic!("expected lossless re-encode for Latin-1 content");
142        };
143        assert_eq!(bytes, original);
144    }

146    #[test]
147    fn test_encode_for_save_utf8_passthrough() {

149            panic!("UTF-8 must always encode losslessly");

154    #[test]
155    fn test_encode_for_save_reports_lossy_for_unrepresentable_chars() {
156        // The euro sign maps to 0x80 in windows-1252, so it encodes cleanly.
