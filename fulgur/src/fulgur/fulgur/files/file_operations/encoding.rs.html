4/// Detect encoding from file bytes
5///
6/// ### Arguments
7/// - `bytes`: The bytes to detect encoding from
8///
9/// ### Returns
10/// - `(String, String)`: The detected encoding and decoded string
11pub fn detect_encoding_and_decode(bytes: &[u8]) -> (String, String) {
12    if let Ok(text) = std::str::from_utf8(bytes) {
13        log::debug!("File encoding detected as UTF-8");
14        return (UTF_8.to_string(), text.to_string());
15    }
16    let mut detector = EncodingDetector::new(Iso2022JpDetection::Allow);
17    detector.feed(bytes, true);
18    let encoding = detector.guess(None, Utf8Detection::Allow);
19    let (decoded, _, had_errors) = encoding.decode(bytes);
20    let encoding_name = if had_errors {
21        if let Ok(text) = std::str::from_utf8(bytes) {
22            log::debug!("File encoding detected as UTF-8 (after error recovery)");
23            return (UTF_8.to_string(), text.to_string());
24        }
25        let text = String::from_utf8_lossy(bytes).to_string();
26        log::warn!("File encoding detection failed, using UTF-8 lossy conversion");
27        return (UTF_8.to_string(), text);
28    } else {
29        encoding.name().to_string()
30    };
31    log::debug!("File encoding detected as: {encoding_name}");
32    (encoding_name, decoded.to_string())
33}

35#[cfg(test)]
36mod tests {
37    use super::detect_encoding_and_decode;
38    use crate::fulgur::ui::components_utils::UTF_8;
39
40    #[test]
41    fn test_detect_encoding_returns_utf8_for_valid_utf8_text() {
42        let text = "Hello, world! Fulgur rocks.";
43        let (encoding, decoded) = detect_encoding_and_decode(text.as_bytes());
44        assert_eq!(encoding, UTF_8);
45        assert_eq!(decoded, text);
46    }
47
48    #[test]
49    fn test_detect_encoding_returns_utf8_for_ascii_content() {
50        let text = "fn main() { println!(\"hi\"); }";
51        let (encoding, decoded) = detect_encoding_and_decode(text.as_bytes());
52        assert_eq!(encoding, UTF_8);
53        assert_eq!(decoded, text);
54    }
55
56    #[test]
57    fn test_detect_encoding_detects_non_utf8_encoding() {
58        // 0xE9 is 'é' in Latin-1 but not a valid UTF-8 byte sequence on its own
59        let bytes: &[u8] = &[0x63, 0x61, 0x66, 0xE9]; // "café" in Latin-1
60        let (encoding, decoded) = detect_encoding_and_decode(bytes);
61        assert_ne!(encoding, UTF_8);
62        assert!(!decoded.is_empty());
63    }
