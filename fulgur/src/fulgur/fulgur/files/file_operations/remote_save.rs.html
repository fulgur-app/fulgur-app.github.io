20    /// Save a remote tab by resolving credentials then spawning an SSH/SFTP worker.
21    ///
22    /// ### Arguments
23    /// - `window`: The window used to spawn dialog and monitoring tasks
24    /// - `cx`: The application context
25    /// - `tab_id`: Stable editor-tab id used to apply completion updates
26    /// - `spec`: Remote file specification for the tab
27    /// - `contents`: Snapshot of editor text, used to reset the modified baseline
28    /// - `bytes`: The encoded file contents to write to the remote host
29    pub(super) fn save_remote_file(
30        &mut self,
31        window: &mut gpui::Window,
32        cx: &mut gpui::Context<Self>,
33        tab_id: TabId,
34        mut spec: RemoteSpec,
35        contents: String,
36        bytes: Vec<u8>,
37    ) {
38        let ssh_session_cache = Arc::clone(&Fulgur::shared_state(cx).ssh_session_cache);
39        let ssh_session_pool = Arc::clone(&Fulgur::shared_state(cx).ssh_session_pool);
40        if let (Some(user), Some(password)) = (spec.user.clone(), spec.password_in_url.take()) {
41            let key = SshCredKey::new(spec.host.clone(), spec.port, user);
42            ssh_session_cache.lock().insert(key, password);
43        }
44
45        let saved_content = Arc::new(contents);
46        let saved_bytes = Arc::new(bytes);
47        if let Some(user) = spec.user.clone() {
48            let cache_key = SshCredKey::new(spec.host.clone(), spec.port, user.clone());
49            if let Some(cached_password) = ssh_session_cache.lock().get(&cache_key).cloned() {
50                spec.password_in_url = None;
51                let request_id = self.next_remote_request_id;
52                self.next_remote_request_id = self.next_remote_request_id.wrapping_add(1);
53                self.latest_remote_save_request_by_tab
54                    .insert(tab_id, request_id);
55                Self::spawn_ssh_save_task(
56                    window,
57                    cx,
58                    RemoteSaveTaskParams {
59                        tab_id,
60                        request_id,
61                        spec,
62                        saved_content: Arc::clone(&saved_content),
63                        saved_bytes: Arc::clone(&saved_bytes),
64                        password: cached_password,
65                        credential_key: cache_key,
66                        ssh_session_cache: Arc::clone(&ssh_session_cache),
67                        ssh_session_pool: Arc::clone(&ssh_session_pool),
68                    },
69                );
70                return;
71            }
72        }
73
74        let host = spec.host.clone();
75        let port = spec.port;
76        let user = spec.user.clone();
77        let entity = cx.entity().downgrade();
78        let cache_for_callback = Arc::clone(&ssh_session_cache);
79        let pool_for_callback = Arc::clone(&ssh_session_pool);
80
81        self.show_ssh_password_dialog(
82            window,
83            cx,
84            &host,
85            port,
86            user,
87            move |resolved_user, password, window, cx| {
88                let mut spec_with_user = spec.clone();
89                spec_with_user.user = Some(resolved_user.clone());
90                spec_with_user.password_in_url = None;
91                let cache_key = SshCredKey::new(
92                    spec_with_user.host.clone(),
93                    spec_with_user.port,
94                    resolved_user.clone(),
95                );
96                if let Some(entity) = entity.upgrade() {
97                    entity.update(cx, |fulgur, cx| {
98                        cache_for_callback
99                            .lock()
100                            .insert(cache_key.clone(), password.clone());
101                        if let Some(tab_entity) = fulgur.tab_entity_of(tab_id, cx) {
102                            tab_entity.update(cx, |tab, _| {
103                                if let Some(editor_tab) = tab.as_editor_mut()
104                                    && let TabLocation::Remote(remote_spec) =
105                                        &mut editor_tab.location
106                                {
107                                    remote_spec.user = Some(resolved_user.clone());
108                                }
109                            });
110                        }
111                        let request_id = fulgur.next_remote_request_id;
112                        fulgur.next_remote_request_id =
113                            fulgur.next_remote_request_id.wrapping_add(1);
114                        fulgur
115                            .latest_remote_save_request_by_tab
116                            .insert(tab_id, request_id);
117                        Self::spawn_ssh_save_task(
118                            window,
119                            cx,
120                            RemoteSaveTaskParams {
121                                tab_id,
122                                request_id,
123                                spec: spec_with_user,
124                                saved_content: Arc::clone(&saved_content),
125                                saved_bytes: Arc::clone(&saved_bytes),
126                                password,
127                                credential_key: cache_key,
128                                ssh_session_cache: Arc::clone(&cache_for_callback),
129                                ssh_session_pool: Arc::clone(&pool_for_callback),
130                            },
131                        );
132                    });
133                }
134            },
135        );
136    }

138    /// Spawn a blocking SSH/SFTP save worker with host-key UI monitoring.
139    ///
140    /// ### Arguments
141    /// - `window`: The window context used to spawn the async monitor task
142    /// - `cx`: The application context
143    /// - `params`: All data required to run the remote save operation
144    fn spawn_ssh_save_task(
145        window: &mut gpui::Window,
146        cx: &mut gpui::Context<Self>,
147        params: RemoteSaveTaskParams,
148    ) {
149        let RemoteSaveTaskParams {
150            tab_id,
151            request_id,
152            spec,
153            saved_content,
154            saved_bytes,
155            password,
156            credential_key,
157            ssh_session_cache,
158            ssh_session_pool,
159        } = params;
160        let pending_save_result: Arc<Mutex<Option<Result<(), String>>>> =
161            Arc::new(Mutex::new(None));
162        let pending_save_for_publish = Arc::clone(&pending_save_result);
163        let content_for_thread = Arc::clone(&saved_bytes);
164
165        let entity_weak = cx.entity().downgrade();
166        let cancel_callback: Option<CancelCallback> = Some(Box::new(move |_window, cx| {
167            if let Some(entity) = entity_weak.upgrade() {
168                entity.update(cx, |fulgur, _cx| {
169                    if fulgur
170                        .latest_remote_save_request_by_tab
171                        .get(&tab_id)
172                        .copied()
173                        == Some(request_id)
174                    {
175                        fulgur.latest_remote_save_request_by_tab.remove(&tab_id);
176                    }
177                });
178            }
179        }));
180        spawn_ssh_task(
181            window,
182            cx,
183            SshTaskContext {
184                spec,
185                password,
186                credential_key,
187                ssh_session_cache,
188                ssh_session_pool,
189                progress_prefix: "Saving to ",
190                timeout_label: SSH_SAVE_TIMEOUT_LABEL,
191                cancel_callback,
192            },
193            move |session, spec| {
194                ssh::sftp::write_remote_file(session, &spec.path, content_for_thread.as_slice())
195            },
196            move |save_finished, outcome| {
197                Self::publish_remote_save_outcome(save_finished, &pending_save_for_publish, outcome)
198            },
199            move |_save_finished| pending_save_result.lock().take(),
200            move |fulgur, result, window, cx| {
201                fulgur.handle_remote_save_result(
202                    tab_id,
203                    request_id,
204                    saved_content.as_str(),
205                    result,
206                    window,
207                    cx,
208                );
209            },
210        );
211    }

213    /// Publish a remote-save outcome exactly once for a single save operation.
214    ///
215    /// ### Arguments
216    /// - `save_finished`: Per-operation completion flag shared by worker and monitor
217    /// - `pending_save`: Shared slot consumed by the monitor task
218    /// - `outcome`: Save result to publish
219    ///
220    /// ### Returns
221    /// - `true`: The outcome was accepted and stored
222    /// - `false`: Another outcome already won the race and this one was ignored
223    pub(super) fn publish_remote_save_outcome(
224        save_finished: &AtomicBool,
225        pending_save: &Mutex<Option<Result<(), String>>>,
226        outcome: Result<(), String>,
227    ) -> bool {
228        if save_finished.swap(true, Ordering::AcqRel) {
229            false
230        } else {
231            *pending_save.lock() = Some(outcome);
232            true
233        }
234    }

236    /// Apply a completed remote-save result on the UI thread.
237    ///
238    /// ### Arguments
239    /// - `tab_id`: Stable editor-tab id associated with the save request
240    /// - `request_id`: Monotonic save request token used to ignore stale completions
241    /// - `saved_content`: Snapshot that was successfully sent to the remote host
242    /// - `result`: Save outcome from the worker task
243    /// - `cx`: The application context
244    pub(super) fn handle_remote_save_result(
245        &mut self,
246        tab_id: TabId,
247        request_id: u64,
248        saved_content: &str,
249        result: Result<(), String>,
250        window: &mut gpui::Window,
251        cx: &mut gpui::Context<Self>,
252    ) {
253        if self.latest_remote_save_request_by_tab.get(&tab_id).copied() != Some(request_id) {
254            return;
255        }
256        self.latest_remote_save_request_by_tab.remove(&tab_id);
257
258        match result {
259            Ok(()) => {
260                let updated = self
261                    .update_editor_tab(tab_id, cx, |editor_tab, cx| {
262                        // Keep async save semantics correct: if content changed after dispatch,
263                        // this remains dirty because baseline is set to the persisted snapshot.
264                        editor_tab.set_original_content_from_str(saved_content);
265                        editor_tab.modified = false;
266                        editor_tab.modified = editor_tab.content_differs_from_original(cx);
267                        editor_tab.update_file_tooltip_cache(saved_content.len());
268                        cx.notify();
269                    })
270                    .is_some();
271                if updated {
272                    self.pending_remote_restore.remove(&tab_id);
273                    self.inflight_remote_restore.remove(&tab_id);
274                    cx.notify();
275                }
276            }
277            Err(msg) => {
278                window.push_notification(
279                    (
280                        NotificationType::Error,
281                        gpui::SharedString::from(format!("Failed to save: {msg}")),
282                    ),
283                    cx,
284                );
285            }
286        }
287    }

296    #[test]
297    fn test_publish_remote_save_outcome_ignores_timeout_after_success() {
298        let save_finished = AtomicBool::new(false);
299        let pending_save = Mutex::new(None);
300
301        let published_success =
302            Fulgur::publish_remote_save_outcome(&save_finished, &pending_save, Ok(()));
303        let published_timeout = Fulgur::publish_remote_save_outcome(
304            &save_finished,
305            &pending_save,
306            Err("SSH save timed out (60 s)".to_string()),
307        );
308
309        assert!(published_success, "first save outcome should win");
310        assert!(!published_timeout, "timeout must be ignored after success");
311        let result = pending_save.lock();
312        assert!(result.is_some(), "one save outcome should be queued");
313        assert!(result.as_ref().is_some_and(Result::is_ok));
314    }
