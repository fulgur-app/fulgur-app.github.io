26    /// Open a remote file from a parsed `RemoteSpec`.
27    ///
28    /// ### Arguments
29    /// - `window`: The window to open the tab in
30    /// - `cx`: The application context
31    /// - `spec`: The parsed remote file specification
32    pub fn do_open_remote_file(
33        &mut self,
34        window: &mut gpui::Window,
35        cx: &mut gpui::Context<Self>,
36        spec: RemoteSpec,
37    ) {
38        if let Some(tab_index) = self.find_tab_by_remote_spec(&spec, cx) {
39            self.active_tab_id = self.tabs.get(tab_index).map(|t| t.read(cx).id());
40            self.focus_active_tab(window, cx);
41            cx.notify();
42            return;
43        }
44        self.last_failed_remote_open_url = Some(format_remote_url(&spec));
45        self.open_remote_file_with_target(window, cx, spec, None);
46    }

48    /// Start loading a remote file and apply the result either to a new tab or an existing tab.
49    ///
50    /// ### Arguments
51    /// - `window`: The target window
52    /// - `cx`: The application context
53    /// - `spec`: The remote specification
54    /// - `target_tab_id`: Existing tab to refresh, or `None` to open in a new tab
55    fn open_remote_file_with_target(
56        &mut self,
57        window: &mut gpui::Window,
58        cx: &mut gpui::Context<Self>,
59        mut spec: RemoteSpec,
60        target_tab_id: Option<TabId>,
61    ) {
62        let ssh_session_cache = Arc::clone(&Fulgur::shared_state(cx).ssh_session_cache);
63        let ssh_session_pool = Arc::clone(&Fulgur::shared_state(cx).ssh_session_pool);
64        let target_request_id = target_tab_id.map(|tab_id| {
65            let request_id = self.next_remote_request_id;
66            self.next_remote_request_id = self.next_remote_request_id.wrapping_add(1);
67            self.latest_remote_open_request_by_tab
68                .insert(tab_id, request_id);
69            request_id
70        });
71
72        // If the URL embeds a password, move it immediately into the session cache.
73        if let (Some(user), Some(password)) = (spec.user.clone(), spec.password_in_url.take()) {
74            let key = SshCredKey::new(spec.host.clone(), spec.port, user);
75            ssh_session_cache.lock().insert(key, password);
76        }
77
78        // Reuse a cached password when we already know the user.
79        if let Some(user) = spec.user.clone() {
80            let cache_key = SshCredKey::new(spec.host.clone(), spec.port, user.clone());
81            if let Some(cached_password) = ssh_session_cache.lock().get(&cache_key).cloned() {
82                spec.password_in_url = None;
83                self.spawn_ssh_open_task(
84                    window,
85                    cx,
86                    RemoteOpenTaskParams {
87                        spec,
88                        password: cached_password,
89                        credential_key: cache_key,
90                        ssh_session_cache: Arc::clone(&ssh_session_cache),
91                        ssh_session_pool: Arc::clone(&ssh_session_pool),
92                        target_tab_id,
93                        target_request_id,
94                    },
95                );
96                return;
97            }
98        }
99
100        let host = spec.host.clone();
101        let port = spec.port;
102        let user = spec.user.clone();
103        let entity = cx.entity().downgrade();
104        let cache_for_callback = Arc::clone(&ssh_session_cache);
105        let pool_for_callback = Arc::clone(&ssh_session_pool);
106
107        self.show_ssh_password_dialog(
108            window,
109            cx,
110            &host,
111            port,
112            user,
113            move |resolved_user, password, window, cx| {
114                let mut spec_with_user = spec.clone();
115                spec_with_user.user = Some(resolved_user);
116                spec_with_user.password_in_url = None;
117                let cache_key = SshCredKey::new(
118                    spec_with_user.host.clone(),
119                    spec_with_user.port,
120                    spec_with_user.user.clone().unwrap_or_default(),
121                );
122                if let Some(entity) = entity.upgrade() {
123                    entity.update(cx, |fulgur, cx| {
124                        cache_for_callback
125                            .lock()
126                            .insert(cache_key.clone(), password.clone());
127                        fulgur.spawn_ssh_open_task(
128                            window,
129                            cx,
130                            RemoteOpenTaskParams {
131                                spec: spec_with_user,
132                                password,
133                                credential_key: cache_key,
134                                ssh_session_cache: Arc::clone(&cache_for_callback),
135                                ssh_session_pool: Arc::clone(&pool_for_callback),
136                                target_tab_id,
137                                target_request_id,
138                            },
139                        );
140                    });
141                }
142            },
143        );
144    }

163    /// Resolve a remote open request into either file contents or a browse fallback.
164    ///
165    /// ### Arguments
166    /// - `session`: Established SSH session with SFTP subsystem.
167    /// - `spec`: Requested remote location.
168    ///
169    /// ### Returns
170    /// - `Ok(RemoteOpenResult::File)`: Target is a readable file.
171    /// - `Ok(RemoteOpenResult::Browse)`: Target is a directory or missing path.
172    /// - `Err(SshError)`: Remote classification or I/O failure.
173    fn resolve_remote_open_result(
174        session: &ssh::session::SshSession,
175        spec: &RemoteSpec,
176    ) -> Result<RemoteOpenResult, ssh::error::SshError> {
177        use ssh::sftp::{RemotePathKind, classify_remote_path, closest_existing_remote_directory};
178
179        match classify_remote_path(session, &spec.path)? {
180            RemotePathKind::File => {
181                let bytes = ssh::sftp::read_remote_file(session, &spec.path)?;
182                let decoded = detect_encoding_and_decode(bytes);
183                Ok(RemoteOpenResult::File(RemoteFileResult {
184                    spec: spec.clone(),
185                    file_size: decoded.byte_len,
186                    content: decoded.content,
187                    encoding: decoded.encoding,
188                    lossy: decoded.lossy,
189                }))
190            }
191            RemotePathKind::Directory => {
192                Self::build_remote_browse_result(session, spec, &spec.path, None)
193            }
194            RemotePathKind::Missing => {
195                let fallback_directory = closest_existing_remote_directory(session, &spec.path)?;
196                let notice = Some(format!(
197                    "Remote path '{}' was not found. Showing closest existing directory '{}'.",
198                    spec.path, fallback_directory
199                ));
200                Self::build_remote_browse_result(session, spec, &fallback_directory, notice)
201            }
202        }
203    }

205    /// Build remote browser entries for a directory fallback dialog.
206    ///
207    /// ### Arguments
208    /// - `session`: Established SSH session with SFTP subsystem.
209    /// - `base_spec`: Original open request spec.
210    /// - `directory_path`: Directory path to list.
211    /// - `notice`: Optional informational message for the UI.
212    ///
213    /// ### Returns
214    /// - `Ok(RemoteOpenResult::Browse)`: Browser payload with directory entries.
215    /// - `Err(SshError)`: Directory listing failed.
216    fn build_remote_browse_result(
217        session: &ssh::session::SshSession,
218        base_spec: &RemoteSpec,
219        directory_path: &str,
220        notice: Option<String>,
221    ) -> Result<RemoteOpenResult, ssh::error::SshError> {
222        let directory = if directory_path.trim().is_empty() {
223            ssh::REMOTE_ROOT_PATH.to_string()
224        } else {
225            directory_path.to_string()
226        };
227        let mut directory_spec = base_spec.clone();
228        directory_spec.path.clone_from(&directory);
229
230        let mut entries: Vec<ssh::sftp::RemoteDirectoryEntry> = Vec::new();
231        if directory != ssh::REMOTE_ROOT_PATH {
232            let parent = ssh::sftp::parent_remote_path(&directory);
233            entries.push(ssh::sftp::RemoteDirectoryEntry {
234                name: "..".to_string(),
235                is_dir: true,
236                full_path: parent,
237            });
238        }
239        entries.extend(ssh::sftp::list_remote_directory(session, &directory)?);
240
241        Ok(RemoteOpenResult::Browse(RemoteBrowseResult {
242            directory_spec,
243            entries,
244            notice,
245        }))
246    }

248    /// Spawn the blocking SSH connect + file-read task plus a GPUI monitoring task.
249    ///
250    /// ### Arguments
251    /// - `window`: The window context used to spawn the async monitoring task
252    /// - `cx`: The application context
253    /// - `params`: All data required to run the remote open operation
254    fn spawn_ssh_open_task(
255        &mut self,
256        window: &mut gpui::Window,
257        cx: &mut gpui::Context<Self>,
258        params: RemoteOpenTaskParams,
259    ) {
260        let RemoteOpenTaskParams {
261            spec,
262            password,
263            credential_key,
264            ssh_session_cache,
265            ssh_session_pool,
266            target_tab_id,
267            target_request_id,
268        } = params;
269        if let Some(tab_id) = target_tab_id {
270            self.inflight_remote_restore.insert(tab_id);
271        }
272        let pending_remote_open = Arc::clone(&self.pending_remote_open);
273
274        let entity_weak = cx.entity().downgrade();
275        let cancel_callback: Option<CancelCallback> = Some(Box::new(move |_window, cx| {
276            if let Some(entity) = entity_weak.upgrade() {
277                entity.update(cx, |fulgur, _cx| {
278                    if let Some(tab_id) = target_tab_id {
279                        fulgur.latest_remote_open_request_by_tab.remove(&tab_id);
280                        fulgur.inflight_remote_restore.remove(&tab_id);
281                    }
282                });
283            }
284        }));
285
286        spawn_ssh_task(
287            window,
288            cx,
289            SshTaskContext {
290                spec,
291                password,
292                credential_key,
293                ssh_session_cache,
294                ssh_session_pool,
295                progress_prefix: "Connecting to ",
296                timeout_label: SSH_CONNECTION_TIMEOUT_LABEL,
297                cancel_callback,
298            },
299            move |session, spec| {
300                if target_tab_id.is_some() {
301                    ssh::sftp::read_remote_file(session, &spec.path).map(|bytes| {
302                        let decoded = detect_encoding_and_decode(bytes);
303                        RemoteOpenResult::File(RemoteFileResult {
304                            spec: spec.clone(),
305                            file_size: decoded.byte_len,
306                            content: decoded.content,
307                            encoding: decoded.encoding,
308                            lossy: decoded.lossy,
309                        })
310                    })
311                } else {
312                    Self::resolve_remote_open_result(session, spec)
313                }
314            },
315            move |open_finished, outcome| {
316                Self::publish_remote_open_outcome(
317                    open_finished,
318                    &pending_remote_open,
319                    target_tab_id,
320                    target_request_id,
321                    outcome,
322                )
323            },
324            |open_finished| open_finished.load(Ordering::Acquire).then_some(()),
325            |_fulgur, (), _window, cx| cx.notify(),
326        );
327    }

329    /// Publish a remote-open outcome exactly once for a single open operation.
330    ///
331    /// ### Arguments
332    /// - `open_finished`: Per-operation completion flag shared by the worker and monitor task.
333    /// - `pending_remote`: Queue of remote-open outcomes drained by the render loop.
334    /// - `target_tab_id`: Existing tab id for a reload operation, or `None` for new-tab opens.
335    /// - `target_request_id`: Monotonic request token used to reject stale completions.
336    /// - `outcome`: Success or failure result to enqueue.
337    ///
338    /// ### Returns
339    /// - `true`: The outcome was accepted and enqueued.
340    /// - `false`: Another outcome already won the race and this one was ignored.
341    pub(super) fn publish_remote_open_outcome(
342        open_finished: &AtomicBool,
343        pending_remote: &Mutex<Vec<PendingRemoteOpenOutcome>>,
344        target_tab_id: Option<TabId>,
345        target_request_id: Option<u64>,
346        outcome: Result<RemoteOpenResult, String>,
347    ) -> bool {
348        if open_finished.swap(true, Ordering::AcqRel) {
349            false
350        } else {
351            pending_remote.lock().push(PendingRemoteOpenOutcome {
352                target_tab_id,
353                target_request_id,
354                result: outcome,
355            });
356            true
357        }
358    }

384    #[test]
385    fn test_publish_remote_open_outcome_ignores_timeout_after_success() {
386        let open_finished = AtomicBool::new(false);
387        let pending_remote = Mutex::new(Vec::new());
388
389        let published_success = Fulgur::publish_remote_open_outcome(
390            &open_finished,
391            &pending_remote,
392            None,
393            None,
394            Ok(make_remote_result()),
395        );
396        let published_timeout = Fulgur::publish_remote_open_outcome(
397            &open_finished,
398            &pending_remote,
399            None,
400            None,
401            Err("SSH connection timed out (60 s)".to_string()),
402        );
403
404        assert!(published_success, "first outcome should win");
405        assert!(!published_timeout, "timeout must be ignored after success");
406        let queue = pending_remote.lock();
407        assert_eq!(queue.len(), 1, "only one outcome must be queued");
408        assert!(
409            queue[0].result.is_ok(),
410            "queued outcome should be the success result"
411        );
412    }
