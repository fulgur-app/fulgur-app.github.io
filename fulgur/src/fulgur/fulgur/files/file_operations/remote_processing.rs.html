9    /// Drain pending remote file results and open loaded content in new tabs.
10    ///
11    /// Called every render pass. When SSH background threads deliver results
12    /// (success or error), this method consumes them and either opens new tabs with
13    /// loaded content or shows error notifications.
14    ///
15    /// ### Arguments
16    /// - `window`: The window context
17    /// - `cx`: The application context
18    pub(crate) fn process_pending_remote_files(
19        &mut self,
20        window: &mut Window,
21        cx: &mut Context<Self>,
22    ) {
23        let outcomes = std::mem::take(&mut *self.pending_remote_open.lock());
24        if outcomes.is_empty() {
25            return;
26        }
27        for outcome in outcomes {
28            let target_tab_id = outcome.target_tab_id;
29            let target_reload_guard = outcome.target_reload_guard;
30            if let Some(tab_id) = target_tab_id {
31                if let Some(request_id) = outcome.target_request_id
32                    && self.latest_remote_open_request_by_tab.get(&tab_id).copied()
33                        != Some(request_id)
34                {
35                    // A newer request for this tab is already in flight; ignore stale completion.
36                    continue;
37                }
38                if let Some(request_id) = outcome.target_request_id
39                    && self.latest_remote_open_request_by_tab.get(&tab_id).copied()
40                        == Some(request_id)
41                {
42                    self.latest_remote_open_request_by_tab.remove(&tab_id);
43                }
44                self.inflight_remote_restore.remove(&tab_id);
45            }
46
47            match outcome.result {
48                Ok(RemoteOpenResult::File(remote_file)) => {
49                    if let Some(tab_id) = target_tab_id {
50                        if target_reload_guard.is_some_and(|guard| {
51                            self.apply_remote_reload_to_existing_tab(
52                                tab_id,
53                                &guard,
54                                remote_file,
55                                window,
56                                cx,
57                            )
58                        }) {
59                            self.pending_remote_restore.remove(&tab_id);
60                        }
61                    } else {
62                        self.last_failed_remote_open_url = None;
63                        let recent_remote_url =
64                            crate::fulgur::sync::ssh::url::format_remote_url(&remote_file.spec);
65                        log::debug!(
66                            "Remote file loaded: {}:{}",
67                            remote_file.spec.host,
68                            remote_file.spec.path
69                        );
70                        let new_tab_id = self.allocate_tab_id();
71                        let editor_tab = editor_tab::EditorTab::from_remote_loaded(
72                            new_tab_id,
73                            remote_file,
74                            window,
75                            cx,
76                            &self.settings.editor_settings,
77                        );
78                        self.place_editor_tab_reusing_scratch(Tab::Editor(editor_tab), window, cx);
79                        self.focus_active_tab(window, cx);
80                        if let Err(e) = self.settings.add_file(PathBuf::from(recent_remote_url)) {
81                            log::error!("Failed to add remote file to recent files: {e}");
82                        }
83                        let update_link = Fulgur::shared_state(cx)
84                            .update_info
85                            .lock()
86                            .as_ref()
87                            .map(|info| info.download_url.clone());
88                        let menus =
89                            build_menus(&self.settings.get_recent_files(), update_link.as_deref());
90                        self.update_menus(menus, cx);
91                        self.save_state_async(cx, window);
92                        cx.notify();
93                    }
94                }
95                Ok(RemoteOpenResult::Browse(browse)) => {
96                    if let Some(tab_id) = target_tab_id {
97                        self.pending_remote_restore.insert(tab_id);
98                        window.push_notification(
99                            (
100                                NotificationType::Error,
101                                gpui_kit::SharedString::from(
102                                    "Restored remote tab path is no longer a file",
103                                ),
104                            ),
105                            cx,
106                        );
107                    } else {
108                        self.show_remote_path_browser_dialog(window, cx, &browse);
109                    }
110                }
111                Err(msg) => {
112                    if let Some(tab_id) = target_tab_id {
113                        self.pending_remote_restore.insert(tab_id);
114                    }
115                    window.push_notification(
116                        (NotificationType::Error, gpui_kit::SharedString::from(msg)),
117                        cx,
118                    );
119                }
120            }
121        }
122    }

124    /// Apply fresh remote contents to an already-restored tab after lazy reconnect.
125    ///
126    /// ### Arguments
127    /// - `tab_id`: Stable editor tab id to update
128    /// - `guard`: Content revision and source captured before the SSH read
129    /// - `remote_file`: Loaded remote payload from SSH worker
130    /// - `window`: The window context
131    /// - `cx`: The application context
132    /// ### Returns
133    /// - `true`: The original tab and content revision still matched and were updated
134    /// - `false`: The tab changed while the remote read was pending and was left untouched
135    fn apply_remote_reload_to_existing_tab(
136        &mut self,
137        tab_id: TabId,
138        guard: &RemoteReloadGuard,
139        remote_file: RemoteFileResult,
140        window: &mut Window,
141        cx: &mut Context<Self>,
142    ) -> bool {
143        let Some(tab_entity) = self.tab_entity_of(tab_id, cx) else {
144            return false;
145        };
146        let can_apply = tab_entity.read(cx).as_editor().is_some_and(|editor| {
147            editor.content_revision(cx) == guard.content_revision
148                && matches!(
149                    &editor.location,
150                    crate::fulgur::editor_tab::TabLocation::Remote(spec)
151                        if crate::fulgur::sync::ssh::url::format_remote_url(spec) == guard.source_url
152                )
153        });
154        if !can_apply {
155            log::warn!(
156                "Discarding stale remote reload for tab {tab_id} because its source or content changed"
157            );
158            tab_entity.update(cx, |tab, cx| {
159                if let Some(editor) = tab.as_editor_mut() {
160                    editor.check_modified(cx);
161                    cx.notify();
162                }
163            });
164            window.push_notification(
165                (
166                    NotificationType::Warning,
167                    gpui_kit::SharedString::from(
168                        "Remote reload skipped because the tab changed while loading; local edits were kept",
169                    ),
170                ),
171                cx,
172            );
173            return false;
174        }
175        tab_entity.update(cx, |tab, cx| {
176            let Some(editor_tab) = tab.as_editor_mut() else {
177                return;
178            };
179            editor_tab.content.update(cx, |input_state, cx| {
180                input_state.set_value(&remote_file.content, window, cx);
181            });
182            editor_tab.location =
183                crate::fulgur::editor_tab::TabLocation::Remote(remote_file.spec.clone());
184            editor_tab.encoding = remote_file.encoding;
185            editor_tab.set_original_content_from_str(&remote_file.content);
186            editor_tab.modified = false;
187            editor_tab.update_file_tooltip_cache(remote_file.file_size);
188            let filename = remote_file
189                .spec
190                .path
191                .rsplit('/')
192                .next()
193                .unwrap_or(&remote_file.spec.path)
194                .to_string();
195            editor_tab.title = filename.into();
196            let language = crate::fulgur::languages::supported_languages::language_from_content(
197                editor_tab.title.as_ref(),
198                &remote_file.content,
199            );
200            tab.force_language(cx, language);
201            cx.notify();
202        });
203        cx.notify();
204        true
205    }

222    /// Build the remote location shared by the restored tab and its delayed result.
223    ///
224    /// ### Returns
225    /// - `RemoteSpec`: A deterministic remote file location for the test
226    fn remote_spec() -> RemoteSpec {
227        RemoteSpec {
228            host: "example.com".to_string(),
229            port: 22,
230            user: Some("alice".to_string()),
231            path: "/tmp/reconnect.txt".to_string(),
232            password_in_url: None,
233        }

236    #[gpui_kit::test]
237    fn test_remote_reconnect_keeps_edits_made_while_loading(cx: &mut TestAppContext) {
238        let (fulgur, mut visual_cx) = setup_fulgur_with_root(cx);
239
240        visual_cx.update(|window, cx| {
241            fulgur.update(cx, |this, cx| {
242                let tab_entity = this.tabs.first().expect("expected an editor tab").clone();
243                let (tab_id, expected_revision) = tab_entity.update(cx, |tab, cx| {
244                    let editor = tab.as_editor_mut().expect("expected an editor tab");
245                    editor.location = TabLocation::Remote(remote_spec());
246                    editor.set_original_content_from_str("");
247                    editor.modified = false;
248                    (editor.id, editor.content_revision(cx))
249                });
250                this.pending_remote_restore.insert(tab_id);
251
252                tab_entity.update(cx, |tab, cx| {
253                    let editor = tab.as_editor_mut().expect("expected an editor tab");
254                    editor.content.update(cx, |state, cx| {
255                        state.set_value("local edit", window, cx);
256                    });
257                });
258                this.pending_remote_open
259                    .lock()
260                    .push(PendingRemoteOpenOutcome {
261                        target_tab_id: Some(tab_id),
262                        target_request_id: None,
263                        target_reload_guard: Some(RemoteReloadGuard {
264                            content_revision: expected_revision,
265                            source_url: crate::fulgur::sync::ssh::url::format_remote_url(
266                                &remote_spec(),
267                            ),
268                        }),
269                        result: Ok(RemoteOpenResult::File(RemoteFileResult {
270                            spec: remote_spec(),
271                            content: "remote content".to_string(),
272                            encoding: UTF_8.to_string(),
273                            lossy: false,
274                            file_size: 14,
275                        })),
276                    });
277                this.process_pending_remote_files(window, cx);
278
279                let editor = tab_entity
280                    .read(cx)
281                    .as_editor()
282                    .expect("expected an editor tab");
283                assert_eq!(editor.content.read(cx).text().to_string(), "local edit");
284                assert!(editor.modified, "the intervening edit must remain dirty");
285                assert!(
286                    this.pending_remote_restore.contains(&tab_id),
287                    "a rejected stale reload must not mark the restored tab as refreshed"
288                );
289            });
290        });
291    }
