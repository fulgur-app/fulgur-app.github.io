7    /// Open a file
8    ///
9    /// ### Arguments
10    /// - `window`: The window to open the file in
11    /// - `cx`: The application context
12    pub fn open_file(&mut self, window: &mut Window, cx: &mut Context<Self>) {
13        let path_future = cx.prompt_for_paths(PathPromptOptions {
14            files: true,
15            directories: false,
16            multiple: false,
17            prompt: None,
18        });
19        cx.spawn_in(window, async move |view, window| {
20            let paths = path_future.await.ok()?.ok()??;
21            let raw_path = paths.first()?.clone();
22            let path = std::fs::canonicalize(&raw_path).unwrap_or(raw_path);
23
24            // Check if tab already exists for this path
25            let should_open_new = window
26                .update(|window, cx| {
27                    view.update(cx, |this, cx| {
28                        if let Some(tab_index) = this.find_tab_by_path(&path, cx) {
29                            log::debug!(
30                                "Tab already exists for {} at index {tab_index}, focusing existing tab",
31                                path.display()
32                            );
33                            this.focus_existing_local_tab_for_open(&path, tab_index, window, cx);
34                            false // Don't open new tab
35                        } else {
36                            true // Open new tab
37                        }
38                    })
39                    .ok()
40                })
41                .ok()??;
42
43            if should_open_new {
44                Self::open_file_from_path(&view, window, &path).await
45            } else {
46                Some(())
47            }
48        })
49        .detach();
50    }

52    /// Open a file from a given path. First detects if the file is already open, and will focus on that tab if that's the case.
53    ///
54    /// ### Arguments
55    /// - `window`: The window to open the file in
56    /// - `cx`: The application context
57    /// - `path`: The path to the file to open
58    pub fn do_open_file(&mut self, window: &mut Window, cx: &mut Context<Self>, path: PathBuf) {
59        let path = std::fs::canonicalize(&path).unwrap_or(path);
60        if let Some(tab_index) = self.find_tab_by_path(&path, cx) {
61            log::debug!(
62                "Tab already exists for {} at index {tab_index}, focusing existing tab",
63                path.display()
64            );
65            self.focus_existing_local_tab_for_open(&path, tab_index, window, cx);
66            return;
67        }
68        let window_manager = cx.global::<crate::fulgur::window_manager::WindowManager>();
69        if let Some(existing_window_id) =
70            window_manager.find_window_with_file(&path, self.window_id, cx)
71        {
72            let file_name = path
73                .file_name()
74                .and_then(|n| n.to_str())
75                .unwrap_or("Unknown file");
76            let message = format!("File '{file_name}' is already open in another window");
77            window.push_notification((NotificationType::Info, SharedString::from(message)), cx);
78            log::debug!(
79                "File {} is already open in window {existing_window_id:?}",
80                path.display()
81            );
82            return;
83        }
84        cx.spawn_in(window, async move |view, window| {
85            Self::open_file_from_path(&view, window, &path).await
86        })
87        .detach();
88    }

90    /// Open a recent entry, dispatching to local or remote open logic.
91    ///
92    /// ### Arguments
93    /// - `window`: The target window
94    /// - `cx`: The application context
95    /// - `path`: The recent entry payload
96    pub fn do_open_recent_file(
97        &mut self,
98        window: &mut Window,
99        cx: &mut Context<Self>,
100        path: PathBuf,
101    ) {
102        let recent_value = path.to_string_lossy();
103        if recent_value.starts_with("ssh://") || recent_value.starts_with("sftp://") {
104            match parse_remote_url(recent_value.as_ref()) {
105                Ok(spec) => self.do_open_remote_file(window, cx, spec),
106                Err(error) => {
107                    window.push_notification(
108                        (
109                            NotificationType::Error,
110                            SharedString::from(format!(
111                                "Failed to open remote recent file: {}",
112                                error.user_message()
113                            )),
114                        ),
115                        cx,
116                    );
117                }
118            }
119            return;
120        }
121        self.do_open_file(window, cx, path);
122    }

228    fn test_do_open_file_reuses_empty_scratch_tab(cx: &mut TestAppContext) {
229        let (fulgur, mut visual_cx) = setup_fulgur(cx);
230        let dir = TempDir::new().expect("failed to create temp dir");
231        let path = dir.path().join("open_new_tab.rs");
232        std::fs::write(&path, "fn main() {}").expect("failed to write file");
233
234        // The default tab created on Fulgur::new is an empty, unsaved scratch buffer.
235        let count_before = fulgur.read_with(&visual_cx, |this, _| this.tabs.len());
236
237        visual_cx.update(|window, cx| {
238            fulgur.update(cx, |this, cx| {
239                this.do_open_file(window, cx, path.clone());
240            });
241        });
242        visual_cx.run_until_parked();
243
244        let count_after = fulgur.read_with(&visual_cx, |this, _| this.tabs.len());
245        assert_eq!(
246            count_after, count_before,
247            "opening a file should reuse the empty scratch tab instead of adding one"
248        );
249
250        let tab_path = fulgur.read_with(&visual_cx, |this, cx| {
251            this.tabs
252                .last()
253                .and_then(|t| t.read(cx).as_editor())
254                .and_then(|e| e.file_path().cloned())
255        });
256        // Canonicalize both sides since macOS may resolve /var/ -> /private/var/
257        let canonical_expected = std::fs::canonicalize(&path).unwrap_or(path.clone());
258        let canonical_actual = tab_path
259            .as_ref()
260            .and_then(|p| std::fs::canonicalize(p).ok())
261            .unwrap_or_else(|| tab_path.clone().unwrap_or_default());
262        assert_eq!(canonical_actual, canonical_expected);
263    }

352    fn test_do_open_file_reuses_only_the_last_tab_position(cx: &mut TestAppContext) {
353        let (fulgur, mut visual_cx) = setup_fulgur(cx);
354        let dir = TempDir::new().expect("failed to create temp dir");
355        let path = dir.path().join("reuse_last_position.rs");
356        std::fs::write(&path, "fn main() {}").expect("failed to write file");
357
358        // Two blank scratch tabs: only the last one in position should be replaced,
359        // leaving the earlier blank tab untouched.
360        visual_cx.update(|window, cx| {
361            fulgur.update(cx, |this, cx| {
362                this.new_tab(window, cx);
363            });
364        });
365
366        let count_before = fulgur.read_with(&visual_cx, |this, _| this.tabs.len());
367        let first_tab_id = fulgur.read_with(&visual_cx, |this, cx| this.tabs[0].read(cx).id());
368
369        visual_cx.update(|window, cx| {
370            fulgur.update(cx, |this, cx| {
371                this.do_open_file(window, cx, path.clone());
372            });
373        });
374        visual_cx.run_until_parked();
375
376        let count_after = fulgur.read_with(&visual_cx, |this, _| this.tabs.len());
377        assert_eq!(
378            count_after, count_before,
379            "the trailing scratch tab should be reused, keeping the tab count stable"
380        );
381
382        let (first_id_after, first_is_blank, last_has_path) =
383            fulgur.read_with(&visual_cx, |this, cx| {
384                let first_blank = this.tabs[0].read(cx).as_editor().is_some_and(|e| {
385                    e.location.is_untitled() && e.content.read(cx).text().len() == 0
386                });
387                let last_path = this
388                    .tabs
389                    .last()
390                    .and_then(|t| t.read(cx).as_editor())
391                    .and_then(|e| e.file_path().cloned())
392                    .is_some();
393                (this.tabs[0].read(cx).id(), first_blank, last_path)
394            });
395        assert_eq!(
396            first_id_after, first_tab_id,
397            "the earlier blank tab must be preserved, not the one replaced"
398        );
399        assert!(
400            first_is_blank,
401            "the earlier blank scratch tab must remain blank"
402        );
403        assert!(last_has_path, "the opened file must land in the last tab");
404    }
