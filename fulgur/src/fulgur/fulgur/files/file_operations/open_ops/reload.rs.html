6impl Fulgur {
7    /// Reload tab content from disk
8    ///
9    /// ### Arguments
10    /// - `tab_index`: The index of the tab to reload
11    /// - `window`: The window context
12    /// - `cx`: The application context
13    pub fn reload_tab_from_disk(
14        &mut self,
15        tab_index: usize,
16        window: &mut Window,
17        cx: &mut Context<Self>,
18    ) {
19        let path = if let Some(Tab::Editor(editor_tab)) =
20            self.tabs.get(tab_index).map(|tab| tab.read(cx))
21        {
22            editor_tab.file_path().cloned()
23        } else {
24            None
25        };
26        let Some(path) = path else {
27            return;
28        };
29        log::debug!("Reloading tab content from disk: {}", path.display());
30        cx.spawn_in(window, async move |view, window| {
31            let read_path = path.clone();
32            let read_result = window
33                .background_executor()
34                .spawn(async move { std::fs::read(&read_path).map(detect_encoding_and_decode) })
35                .await;
36            match read_result {
37                Ok(decoded) => {
38                    window
39                        .update(|window, cx| {
40                            _ = view.update(cx, |this, cx| {
41                                this.apply_reloaded_contents(&path, decoded, window, cx);
42                            });
43                        })
44                        .ok();
45                }
46                Err(e) => {
47                    log::error!("Failed to reload file {}: {e}", path.display());
48                }
49            }
50        })
51        .detach();
52    }
53
54    /// Apply freshly decoded file contents to the editor tab backing a path.
55    ///
56    /// ### Arguments
57    /// - `path`: The path whose tab should receive the reloaded content
58    /// - `decoded`: The decoded file contents produced off the UI thread
59    /// - `window`: The window context
60    /// - `cx`: The application context
61    fn apply_reloaded_contents(
62        &mut self,
63        path: &Path,
64        decoded: DecodedContents,
65        window: &mut Window,
66        cx: &mut Context<Self>,
67    ) {
68        let Some(tab_index) = self.find_tab_by_path(&path.to_path_buf(), cx) else {
69            return;
70        };
71        let Some(tab_entity) = self.tabs.get(tab_index).cloned() else {
72            return;
73        };
74        let settings = self.settings.editor_settings.clone();
75        tab_entity.update(cx, |tab, cx| {
76            let Some(editor_tab) = tab.as_editor_mut() else {
77                return;
78            };
79            editor_tab.content.update(cx, |input_state, cx| {
80                input_state.set_value(&decoded.content, window, cx);
81            });
82            editor_tab.set_original_content_from_str(&decoded.content);
83            editor_tab.encoding = decoded.encoding;
84            editor_tab.lossy_decode = decoded.lossy;
85            editor_tab.modified = false;
86            editor_tab.update_file_tooltip_cache(decoded.byte_len);
87            tab.update_language(window, cx, &settings);
88            log::debug!("Tab reloaded successfully from disk: {}", path.display());
89        });
90    }
91}

107    #[cfg(feature = "gpui-test-support")]
108    #[gpui::test]
109    fn test_reload_tab_from_disk_updates_content_from_file(cx: &mut TestAppContext) {
110        let (fulgur, mut visual_cx) = setup_fulgur(cx);
111        let dir = TempDir::new().expect("failed to create temp dir");
112        let path = dir.path().join("reload_test.txt");
113        std::fs::write(&path, "initial content").expect("failed to write initial file");
114
115        visual_cx.update(|_window, cx| {
116            fulgur.update(cx, |this, cx| {
117                this.tabs
118                    .last()
119                    .expect("expected at least one tab")
120                    .clone()
121                    .update(cx, |tab, _cx| {
122                        if let Some(editor_tab) = tab.as_editor_mut() {
123                            editor_tab.location = TabLocation::Local(path.clone());
124                            editor_tab.set_original_content_from_str("initial content");
125                        }
126                    });
127            });
128        });
129
130        std::fs::write(&path, "updated content").expect("failed to overwrite file");
131
132        visual_cx.update(|window, cx| {
133            fulgur.update(cx, |this, cx| {
134                this.reload_tab_from_disk(0, window, cx);
135            });
136        });
137        // The read and decode now run on the background executor and are applied
138        // asynchronously, so let the spawned task complete before asserting.
139        visual_cx.run_until_parked();
140
141        visual_cx.update(|_window, cx| {
142            fulgur.update(cx, |this, cx| {
143                let content = this
144                    .tabs
145                    .first()
146                    .and_then(|t| t.read(cx).as_editor())
147                    .map(|e| e.content.read(cx).text().to_string())
148                    .unwrap_or_default();
149                assert_eq!(content, "updated content");
150                let modified = this
151                    .tabs
152                    .first()
153                    .and_then(|t| t.read(cx).as_editor())
154                    .is_none_or(|e| e.modified);
155                assert!(!modified, "tab should not be marked modified after reload");
156            });
157        });
158    }

163        let (fulgur, mut visual_cx) = setup_fulgur(cx);
164
165        visual_cx.update(|window, cx| {

173                    .unwrap_or_default();
174                this.reload_tab_from_disk(0, window, cx);
175                let content_after = this
176                    .tabs

178                    .and_then(|t| t.read(cx).as_editor())
179                    .map(|e| e.content.read(cx).text().to_string())
