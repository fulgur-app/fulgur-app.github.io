26/// Interval at which the monitor task checks for completion and host-key prompts.
27const MONITOR_POLL_INTERVAL: Duration = Duration::from_millis(100);
28
29/// Connection inputs and presentation labels shared by every remote SSH operation.
30pub(super) struct SshTaskContext {
31    pub spec: RemoteSpec,
32    pub password: Zeroizing<String>,
33    pub credential_key: SshCredKey,
34    pub ssh_session_cache: Arc<Mutex<SshCredentialCache>>,
35    pub ssh_session_pool: Arc<SshSessionPool>,
36    /// Verb prefix of the progress label, ending with a space, e.g. `"Connecting to "`.
37    pub progress_prefix: &'static str,
38    /// Label used to build the timeout message, e.g. `SSH_CONNECTION_TIMEOUT_LABEL`.
39    pub timeout_label: &'static str,
40    pub cancel_callback: Option<CancelCallback>,
41}

43/// Run a remote SSH operation on a worker thread with host-key and timeout monitoring.
44///
45/// ### Arguments
46/// - `window`: The window used to show progress and spawn the monitor task
47/// - `cx`: The application context
48/// - `context`: Connection inputs and labels for this operation
49/// - `work`: Operation to run on the established session, given the session and the request spec
50/// - `publish`: Records an outcome exactly once, guarded by the shared completion flag
51/// - `poll_completion`: Returns the payload to deliver once the operation has completed
52/// - `on_complete`: Applies the delivered payload on the UI thread
53pub(super) fn spawn_ssh_task<Work, Payload>(
54    window: &mut gpui::Window,
55    cx: &mut gpui::Context<Fulgur>,
56    context: SshTaskContext,
57    work: impl FnOnce(&SshSession, &RemoteSpec) -> Result<Work, ssh::error::SshError> + Send + 'static,
58    publish: impl Fn(&AtomicBool, Result<Work, String>) -> bool + Send + Sync + 'static,
59    poll_completion: impl Fn(&AtomicBool) -> Option<Payload> + 'static,
60    on_complete: impl Fn(&mut Fulgur, Payload, &mut gpui::Window, &mut gpui::Context<Fulgur>) + 'static,
61) where
62    Work: Send + 'static,
63    Payload: 'static,
64{
65    let SshTaskContext {
66        spec,
67        password,
68        credential_key,
69        ssh_session_cache,
70        ssh_session_pool,
71        progress_prefix,
72        timeout_label,
73        cancel_callback,
74    } = context;
75
76    let pending_host_key: Arc<Mutex<Option<HostKeyRequest>>> = Arc::new(Mutex::new(None));
77    let pending_host_key_for_thread = Arc::clone(&pending_host_key);
78    let finished = Arc::new(AtomicBool::new(false));
79    let finished_for_thread = Arc::clone(&finished);
80    let host_key_decision_timed_out = Arc::new(AtomicBool::new(false));
81    let host_key_decision_timed_out_for_thread = Arc::clone(&host_key_decision_timed_out);
82
83    let timeout_message = format!("{timeout_label} ({SSH_HOST_KEY_APPROVAL_TIMEOUT_SECS} s)");
84    let timeout_message_for_thread = timeout_message.clone();
85
86    let user = spec.user.clone().unwrap_or_default();
87    let progress_label =
88        format_remote_endpoint_label(progress_prefix, &spec.host, spec.port, &user);
89    let progress = start_progress(window, cx, progress_label.into(), cancel_callback);
90    let cancel_flag = progress.cancel_flag();
91    let cancel_flag_for_thread = Arc::clone(&cancel_flag);
92
93    let publish = Arc::new(publish);
94    let publish_for_thread = Arc::clone(&publish);
95    let spec_for_thread = spec;
96    let cache_for_thread = ssh_session_cache;
97    let pool_for_thread = ssh_session_pool;
98
99    std::thread::spawn(move || {
100        let slot = pending_host_key_for_thread;
101        let host_key_decision_timed_out_for_callback =
102            Arc::clone(&host_key_decision_timed_out_for_thread);
103        let session_result = pool_for_thread.checkout_or_connect(
104            &spec_for_thread,
105            &user,
106            &password,
107            move |fingerprint, host, port| {
108                let (tx, rx) = std::sync::mpsc::channel();
109                *slot.lock() = Some(HostKeyRequest {
110                    fingerprint: fingerprint.to_string(),
111                    host: host.to_string(),
112                    port,
113                    decision_tx: tx,
114                });
115                wait_for_host_key_decision(&rx, &host_key_decision_timed_out_for_callback)
116            },
117        );
118        if let Err(ssh::error::SshError::AuthFailed) = &session_result {
119            cache_for_thread.lock().remove(&credential_key);
120        }
121
122        let mut outcome = session_result
123            .and_then(|pooled_session| {
124                let result = work(pooled_session.session(), &spec_for_thread);
125                if result.is_err() {
126                    pooled_session.invalidate();
127                }
128                result
129            })
130            .map_err(|e| e.user_message());
131        if host_key_decision_timed_out_for_thread.load(Ordering::Acquire) {
132            outcome = Err(timeout_message_for_thread);
133        }
134
135        if cancel_flag_for_thread.load(Ordering::Acquire) {
136            // User cancelled, discard the outcome and unblock the monitor task.
137            finished_for_thread.store(true, Ordering::Release);
138        } else {
139            (*publish_for_thread)(&finished_for_thread, outcome);
140        }
141    });
142
143    cx.spawn_in(window, async move |view, async_cx| {
144        let _progress = progress;
145        let deadline = Instant::now() + SSH_HOST_KEY_APPROVAL_TIMEOUT;
146        loop {
147            async_cx
148                .background_executor()
149                .timer(MONITOR_POLL_INTERVAL)
150                .await;
151
152            let completion = if let Some(payload) = poll_completion(&finished) {
153                Some(payload)
154            } else if cancel_flag.load(Ordering::Acquire) {
155                None
156            } else {
157                let hk_req = pending_host_key.lock().take();
158                if let Some(req) = hk_req {
159                    async_cx
160                        .update(|window, cx| {
161                            _ = view.update(cx, |fulgur, cx| {
162                                fulgur.show_ssh_host_fingerprint_dialog(window, cx, req);
163                            });
164                        })
165                        .ok();
166                }
167
168                if Instant::now() <= deadline {
169                    continue;
170                }
171
172                if let Some(request) = pending_host_key.lock().take() {
173                    let _ = request.decision_tx.send(HostKeyDecision::Reject);
174                }
175                (*publish)(&finished, Err(timeout_message.clone()));
176                poll_completion(&finished)
177            };
178
179            if let Some(payload) = completion {
180                async_cx
181                    .update(|window, cx| {
182                        _ = view.update(cx, |fulgur, cx| {
183                            on_complete(fulgur, payload, window, cx);
184                        });
185                    })
186                    .ok();
187            }
188            break;
189        }
190    })
191    .detach();
192}
