1use crate::fulgur::{Fulgur, tab::Tab};
2use gpui::{Context, SharedString, Window};
3use gpui_component::{WindowExt, notification::NotificationType};
4
5impl Fulgur {
6    /// Open the native OS print dialog for the current document
7    ///
8    /// Writes the active tab's content to a temporary HTML file and opens it with
9    /// the system's default browser, which automatically triggers the native print dialog.
10    /// This approach works cross-platform without requiring OS-specific print APIs.
11    ///
12    /// ### Arguments
13    /// - `window`: The window containing the editor
14    /// - `cx`: The application context
15    pub fn print_file(&mut self, window: &mut Window, cx: &mut Context<Self>) {
16        let Some(active_tab_index) = self.active_tab_index(cx) else {
17            return;
18        };
19        let (title, content) = match self.tabs[active_tab_index].read(cx) {
20            Tab::Editor(editor_tab) => {
21                let title = editor_tab.title.clone();
22                let content = editor_tab.content.read(cx).text().to_string();
23                (title, content)
24            }
25            Tab::Settings(_) | Tab::MarkdownPreview(_) => return,
26        };
27        let escaped_content = content
28            .replace('&', "&amp;")
29            .replace('<', "&lt;")
30            .replace('>', "&gt;");
31        let escaped_title = title
32            .replace('&', "&amp;")
33            .replace('<', "&lt;")
34            .replace('>', "&gt;");
35        let html = format!(
36            r#"<!DOCTYPE html>
37<html>
38<head>
39<meta charset="utf-8">
40<title>{escaped_title}</title>
41<style>
42  body {{ margin: 0; padding: 1em; font-family: monospace; white-space: pre-wrap; word-wrap: break-word; }}
43  @media print {{ body {{ margin: 0; }} }}
44</style>
45</head>
46<body>{escaped_content}</body>
47<script>window.onload = function() {{ window.print(); }};</script>
48</html>"#,
49        );
50        let temp_path =
51            std::env::temp_dir().join(format!("fulgur_print_{}.html", std::process::id()));
52        if let Err(e) = std::fs::write(&temp_path, html.as_bytes()) {
53            log::error!("Failed to write print temp file: {e}");
54            window.push_notification(
55                (
56                    NotificationType::Error,
57                    SharedString::from(format!("Failed to prepare print: {e}")),
58                ),
59                cx,
60            );
61            return;
62        }
63        if let Err(e) = open::that(&temp_path) {
64            log::error!("Failed to open print file: {e}");
65            window.push_notification(
66                (
67                    NotificationType::Error,
68                    SharedString::from(format!("Failed to open print dialog: {e}")),
69                ),
70                cx,
71            );
72        }
73    }
74}
