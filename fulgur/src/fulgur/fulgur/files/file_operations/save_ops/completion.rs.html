1use crate::fulgur::Fulgur;
2use crate::fulgur::ui::tabs::tab::TabId;
3use gpui::{Context, SharedString, Window};
4use gpui_component::{WindowExt, notification::NotificationType};
5use std::path::PathBuf;
6
7/// Snapshot of a tab's saved-content baseline, captured before an optimistic
8/// save so a failed background write can restore it.
9pub(super) struct SavedBaseline {
10    /// The tab's `original_content_hash` at dispatch time
11    pub(super) hash: u64,
12    /// The tab's `original_content_len` at dispatch time
13    pub(super) len: usize,
14    /// The tab's `modified` flag at dispatch time
15    pub(super) modified: bool,
16}
17
18/// Dispatch-time context of a background save, handed back to the completion
19/// handler that runs on the UI thread once the write resolves.
20pub(super) struct SaveCompletion {
21    /// Stable identifier of the editor tab being saved
22    pub(super) tab_id: TabId,
23    /// Destination path of the write, as requested at dispatch
24    pub(super) path: PathBuf,
25    /// Size of the written content in bytes
26    pub(super) byte_len: usize,
27    /// Saved baseline captured at dispatch, restored if the write fails
28    pub(super) previous_baseline: Option<SavedBaseline>,
29}
30
31impl Fulgur {
32    /// Capture a tab's saved-content baseline before an optimistic save.
33    ///
34    /// ### Arguments
35    /// - `tab_id`: Stable identifier of the editor tab
36    /// - `cx`: The application context
37    ///
38    /// ### Returns
39    /// - `Some(SavedBaseline)`: The tab's current baseline and modified flag
40    /// - `None`: If the tab no longer exists or is not an editor tab
41    pub(super) fn capture_saved_baseline(
42        &self,
43        tab_id: TabId,
44        cx: &Context<Self>,
45    ) -> Option<SavedBaseline> {
46        self.tab_entity_of(tab_id, cx).and_then(|tab| {
47            tab.read(cx).as_editor().map(|editor_tab| SavedBaseline {
48                hash: editor_tab.original_content_hash,
49                len: editor_tab.original_content_len,
50                modified: editor_tab.modified,
51            })
52        })
53    }
54
55    /// Report a failed background save and roll back the optimistic saved state.
56    ///
57    /// ### Arguments
58    /// - `completion`: Dispatch-time context of the save that failed
59    /// - `error`: The write error to report
60    /// - `window`: The window context
61    /// - `cx`: The application context
62    pub(super) fn handle_failed_save(
63        &mut self,
64        completion: SaveCompletion,
65        error: &anyhow::Error,
66        window: &mut Window,
67        cx: &mut Context<Self>,
68    ) {
69        log::error!("Failed to save file {}: {error}", completion.path.display());
70        if let Some(baseline) = completion.previous_baseline {
71            self.update_editor_tab(completion.tab_id, cx, |editor_tab, cx| {
72                let edited_during_save = editor_tab.modified;
73                editor_tab.original_content_hash = baseline.hash;
74                editor_tab.original_content_len = baseline.len;
75                if editor_tab.large_file {
76                    editor_tab.modified = baseline.modified || edited_during_save;
77                } else {
78                    editor_tab.check_modified(cx);
79                }
80                cx.notify();
81            });
82        }
83        let file_name = completion
84            .path
85            .file_name()
86            .and_then(|n| n.to_str())
87            .unwrap_or("file");
88        window.push_notification(
89            (
90                NotificationType::Error,
91                SharedString::from(format!("Failed to save '{file_name}': {error}")),
92            ),
93            cx,
94        );
95        cx.notify();
96    }
97}
