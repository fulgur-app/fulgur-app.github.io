50pub enum FileWatchEvent {
51    Modified(PathBuf),
52    Deleted(PathBuf),
53    Renamed { from: PathBuf, to: PathBuf },
54    Error(String),
55}

57pub struct FileWatcher {
58    watcher: Option<RecommendedWatcher>,
59    watched_paths: HashMap<PathBuf, SystemTime>,
60    event_tx: Sender<FileWatchEvent>,
61    /// Pending rename source path for Linux inotify, which splits rename events
62    /// into separate From and To notifications rather than a single two-path event.
63    pending_rename_from: Arc<Mutex<Option<PathBuf>>>,
64}
65
66impl FileWatcher {
67    /// Creates a new file watcher
68    ///
69    /// ### Returns
70    /// - `(FileWatcher, Receiver<FileWatchEvent>)`: A tuple containing the file watcher and the event receiver
71    ///   - `FileWatcher`: The file watcher instance
72    ///   - `Receiver<FileWatchEvent>`: The event receiver to receive the file watch events from the file watcher
73    pub fn new() -> (Self, Receiver<FileWatchEvent>) {
74        let (event_tx, event_rx) = channel();
75        let watcher = Self {
76            watcher: None,
77            watched_paths: HashMap::new(),
78            event_tx,
79            pending_rename_from: Arc::new(Mutex::new(None)),
80        };
81        (watcher, event_rx)
82    }
83
84    /// Starts the file watcher
85    ///
86    /// ### Returns
87    /// - `Ok(())`: If the file watcher was started successfully
88    /// - `Err(NotifyError)`: If the file watcher could not be started
89    pub fn start(&mut self) -> Result<(), NotifyError> {
90        if self.watcher.is_some() {
91            return Ok(());
92        }
93        let event_tx = self.event_tx.clone();
94        let pending_rename_from = Arc::clone(&self.pending_rename_from);
95        let watcher =
96            notify::recommended_watcher(move |res: Result<Event, NotifyError>| match res {
97                Ok(event) => {
98                    Self::handle_notify_event(event, &event_tx, &pending_rename_from);
99                }
100                Err(e) => {
101                    let _ = event_tx.send(FileWatchEvent::Error(e.to_string()));
102                }
103            })?;
104        self.watcher = Some(watcher);
105        Ok(())
106    }
107
108    /// Handles a notify event and converts it to a FileWatchEvent
109    ///
110    /// ### Description
111    /// - If the event is a modification, it sends a Modified event to the event sender
112    /// - If the event is a deletion, it sends a Deleted event to the event sender
113    /// - If the event is a creation after a rename or save, it sends a Modified event to the event sender
114    /// - If the event is other, it ignores it
115    /// - Ignore other event types (access, etc.)
116    ///
117    /// Rename events come in two shapes depending on the OS backend:
118    /// - **macOS/Windows**: A single event with two paths (`RenameMode::Both`)
119    /// - **Linux inotify**: Two consecutive single-path events (`RenameMode::From` then
120    ///   `RenameMode::To`). The `pending_rename_from` accumulator pairs them up.
121    ///
122    /// ### Arguments
123    /// - `event`: The notify event to handle
124    /// - `event_tx`: The event sender to send the events to
125    /// - `pending_rename_from`: Accumulator for the Linux split-rename `From` path
126    fn handle_notify_event(
127        event: Event,
128        event_tx: &Sender<FileWatchEvent>,
129        pending_rename_from: &Mutex<Option<PathBuf>>,
130    ) {
131        use notify::event::{ModifyKind, RenameMode};
132
133        match event.kind {
134            EventKind::Modify(ModifyKind::Name(rename_mode)) => {
135                if event.paths.len() == 2 {
136                    // macOS / Windows: both paths arrive in one event
137                    let from = event.paths[0].clone();
138                    let to = event.paths[1].clone();
139                    let _ = event_tx.send(FileWatchEvent::Renamed { from, to });
140                } else if event.paths.len() == 1 {
141                    match rename_mode {
142                        RenameMode::From => {
143                            // Linux inotify: first half — store and wait for the To event
144                            if let Ok(mut pending) = pending_rename_from.lock() {
145                                *pending = Some(event.paths[0].clone());
146                            }
147                        }
148                        RenameMode::To => {
149                            // Linux inotify: second half — pair with the stored From path
150                            let from = pending_rename_from.lock().ok().and_then(|mut p| p.take());
151                            match from {
152                                Some(from) => {
153                                    let _ = event_tx.send(FileWatchEvent::Renamed {
154                                        from,
155                                        to: event.paths[0].clone(),
156                                    });
157                                }
158                                None => {
159                                    // No matching From; treat as a new file appearing
160                                    let _ = event_tx
161                                        .send(FileWatchEvent::Modified(event.paths[0].clone()));
162                                }
163                            }
164                        }
165                        _ => {
166                            // Orphaned or unrecognised single-path rename; flush any pending From
167                            if let Ok(mut pending) = pending_rename_from.lock()
168                                && let Some(stale) = pending.take()
169                            {
170                                let _ = event_tx.send(FileWatchEvent::Deleted(stale));
171                            }
172                        }
173                    }
174                }
175            }
176            EventKind::Modify(_) => {
177                for path in event.paths {
178                    let _ = event_tx.send(FileWatchEvent::Modified(path));
179                }
180            }
181            EventKind::Remove(_) => {
182                for path in event.paths {
183                    let _ = event_tx.send(FileWatchEvent::Deleted(path));
184                }
185            }
186            EventKind::Create(_) => {
187                for path in event.paths {
188                    let _ = event_tx.send(FileWatchEvent::Modified(path));
189                }
190            }
191            _ => {}
192        }
193    }

195    /// Starts watching a file
196    ///
197    /// ### Arguments
198    /// - `path`: The path to the file to watch
199    ///
200    /// ### Returns
201    /// - `Ok(())`: If the file was watched successfully
202    /// - `Err(String)`: If the file could not be watched
203    pub fn watch_file(&mut self, path: PathBuf) -> Result<(), String> {
204        if self.watched_paths.contains_key(&path) {
205            return Ok(());
206        }
207        if self.watcher.is_none() {
208            self.start().map_err(|e| e.to_string())?;
209        }
210        let modified_time = std::fs::metadata(&path)
211            .and_then(|m| m.modified())
212            .unwrap_or(SystemTime::UNIX_EPOCH);
213        if let Some(watcher) = &mut self.watcher {
214            watcher
215                .watch(&path, RecursiveMode::NonRecursive)
216                .map_err(|e| format!("Failed to watch file {}: {}", path.display(), e))?;
217        }
218        self.watched_paths.insert(path.clone(), modified_time);
219        log::debug!("Started watching file: {}", path.display());
220        Ok(())
221    }
222
223    /// Stops watching a file
224    ///
225    /// ### Arguments
226    /// - `path`: The path to the file to stop watching
227    pub fn unwatch_file(&mut self, path: &PathBuf) {
228        if !self.watched_paths.contains_key(path) {
229            return;
230        }
231        if let Some(watcher) = &mut self.watcher
232            && let Err(e) = watcher.unwatch(path)
233        {
234            log::warn!("Failed to unwatch file {}: {}", path.display(), e);
235        }
236
237        self.watched_paths.remove(path);
238        log::debug!("Stopped watching file: {}", path.display());
239    }
240
241    /// Stops the file watcher completely
242    pub fn stop(&mut self) {
243        if let Some(mut watcher) = self.watcher.take() {
244            for path in self.watched_paths.keys() {
245                let _ = watcher.unwatch(path);
246            }
247        }
248        self.watched_paths.clear();
249        log::debug!("File watcher stopped");
250    }
251}

254    /// Stops the file watcher when the FileWatcher instance is dropped

256        self.stop();
257    }

260impl Fulgur {
261    /// Handle file watch events received from the file watcher
262    ///
263    /// ### Description
264    /// - If the event is a modification, it shows a conflict dialog if the file is modified and the tab is active
265    /// - If the event is a deletion, it shows a notification that the file was deleted
266    /// - If the event is a rename, it shows a notification that the file was renamed
267    /// - If the event is an error, it logs the error
268    ///
269    /// ### Arguments
270    /// - `event`: The file watch event to handle
271    /// - `window`: The window context
272    /// - `cx`: The application context
273    pub fn handle_file_watch_event(
274        &mut self,
275        event: FileWatchEvent,
276        window: &mut Window,
277        cx: &mut Context<Self>,
278    ) {
279        match event {
280            FileWatchEvent::Modified(path) => {
281                let now = Instant::now();
282                if let Some(&last_time) = self.file_watch_state.last_file_events.get(&path)
283                    && now.duration_since(last_time) < Duration::from_millis(500)
284                {
285                    return;
286                }
287                if let Some(&save_time) = self.file_watch_state.last_file_saves.get(&path)
288                    && now.duration_since(save_time) < Duration::from_millis(500)
289                {
290                    return;
291                }
292                self.file_watch_state
293                    .last_file_events
294                    .insert(path.clone(), now);
295                if let Some(tab_index) = self.find_tab_by_path(&path)
296                    && let Some(Tab::Editor(editor_tab)) = self.tabs.get(tab_index)
297                {
298                    if editor_tab.modified {
299                        let is_active = self.active_tab_index == Some(tab_index);
300
301                        if is_active {
302                            self.show_file_conflict_dialog(path, tab_index, window, cx);
303                        } else {
304                            self.file_watch_state
305                                .pending_conflicts
306                                .insert(path, tab_index);
307                        }
308                    } else {
309                        self.reload_tab_from_disk(tab_index, window, cx);
310                        self.show_notification_file_reloaded(&path, window, cx);
311                    }
312                }
313            }
314            FileWatchEvent::Deleted(path) => {
315                self.show_notification_file_deleted(&path, window, cx);
316            }
317            FileWatchEvent::Renamed { from, to } => {
318                if let Some(tab_index) = self.find_tab_by_path(&from) {
319                    self.unwatch_file(&from);
320                    self.watch_file(&to);
321                    if let Some(Tab::Editor(editor_tab)) = self.tabs.get_mut(tab_index) {
322                        editor_tab.file_path = Some(to.clone());
323                        editor_tab.title = to
324                            .file_name()
325                            .and_then(|n| n.to_str())
326                            .unwrap_or("Untitled")
327                            .to_string()
328                            .into();
329                    }
330                    self.show_notification_file_renamed(&from, &to, window, cx);
331                }
332            }
333            FileWatchEvent::Error(msg) => {
334                log::error!("File watcher error: {}", msg);
335            }
336        }
337    }
338
339    /// Start the file watcher and watch all open files
340    pub fn start_file_watcher(&mut self) {
341        let (mut watcher, receiver) = FileWatcher::new();
342        if let Err(e) = watcher.start() {
343            log::error!("Failed to start file watcher: {}", e);
344            return;
345        }
346        for tab in &self.tabs {
347            if let Tab::Editor(editor_tab) = tab
348                && let Some(path) = &editor_tab.file_path
349                && let Err(e) = watcher.watch_file(path.clone())
350            {
351                log::warn!("Failed to watch file {}: {}", path.display(), e);
352            }
353        }
354        self.file_watch_state.file_watcher = Some(watcher);
355        self.file_watch_state.file_watch_events = Some(receiver);
356    }
357
358    /// Stop the file watcher
359    pub fn stop_file_watcher(&mut self) {
360        if let Some(mut watcher) = self.file_watch_state.file_watcher.take() {
361            watcher.stop();
362        }
363        self.file_watch_state.file_watch_events = None;
364    }
365
366    /// Add a file to the watcher
367    ///
368    /// ### Arguments
369    /// - `path`: The path to the file to watch
370    pub fn watch_file(&mut self, path: &std::path::Path) {
371        if let Some(watcher) = &mut self.file_watch_state.file_watcher
372            && let Err(e) = watcher.watch_file(path.to_path_buf())
373        {
374            log::warn!("Failed to watch file {}: {}", path.display(), e);
375        }
376    }
377
378    /// Remove a file from the watcher
379    ///
380    /// ### Arguments
381    /// - `path`: The path to the file to unwatch
382    pub fn unwatch_file(&mut self, path: &PathBuf) {
383        if let Some(watcher) = &mut self.file_watch_state.file_watcher {
384            watcher.unwatch_file(path);
385        }
386    }

388    /// Collect and process file watch events:
389    /// - Modified: File content changed externally (may trigger auto-reload or conflict dialog)
390    /// - Deleted: File was deleted externally (shows notification)
391    /// - Renamed: File was moved/renamed (updates tab path and continues watching)
392    ///
393    /// ### Arguments
394    /// - `window`: The window containing the tabs with watched files
395    /// - `cx`: The application context
396    pub fn process_file_watch_events(&mut self, window: &mut Window, cx: &mut Context<Self>) {
397        let events = collect_events(&self.file_watch_state.file_watch_events);
398        for event in events {
399            self.handle_file_watch_event(event, window, cx);
400        }
401    }
