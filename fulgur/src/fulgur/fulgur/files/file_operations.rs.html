20/// Detect encoding from file bytes
21///
22/// ### Arguments
23/// - `bytes`: The bytes to detect encoding from
24///
25/// ### Returns
26/// - `(String, String)`: The detected encoding and decoded string
27pub fn detect_encoding_and_decode(bytes: &[u8]) -> (String, String) {
28    if let Ok(text) = std::str::from_utf8(bytes) {
29        log::debug!("File encoding detected as UTF-8");
30        return (UTF_8.to_string(), text.to_string());
31    }
32    let mut detector = EncodingDetector::new();
33    detector.feed(bytes, true);
34    let encoding = detector.guess(None, true);
35    let (decoded, _, had_errors) = encoding.decode(bytes);
36    let encoding_name = if had_errors {
37        match std::str::from_utf8(bytes) {
38            Ok(text) => {
39                log::debug!("File encoding detected as UTF-8 (after error recovery)");
40                return (UTF_8.to_string(), text.to_string());
41            }
42            Err(_) => {
43                let text = String::from_utf8_lossy(bytes).to_string();
44                log::warn!("File encoding detection failed, using UTF-8 lossy conversion");
45                return (UTF_8.to_string(), text);
46            }
47        }
48    } else {
49        encoding.name().to_string()
50    };
51    log::debug!("File encoding detected as: {}", encoding_name);
52    (encoding_name, decoded.to_string())
53}

56    /// Find the index of a tab with the given file path
57    ///
58    /// ### Arguments
59    /// - `path`: The path to search for
60    ///
61    /// ### Returns
62    /// - `Some(usize)`: The index of the tab if found
63    /// - `None`: If the tab was not found
64    pub fn find_tab_by_path(&self, path: &PathBuf) -> Option<usize> {
65        self.tabs.iter().position(|tab| {
66            if let Tab::Editor(editor_tab) = tab {
67                if let Some(ref tab_path) = editor_tab.file_path {
68                    tab_path == path
69                } else {
70                    false
71                }
72            } else {
73                false
74            }
75        })
76    }

78    /// Reload tab content from disk
79    ///
80    /// ### Arguments
81    /// - `tab_index`: The index of the tab to reload
82    /// - `window`: The window context
83    /// - `cx`: The application context
84    pub fn reload_tab_from_disk(
85        &mut self,
86        tab_index: usize,
87        window: &mut Window,
88        cx: &mut Context<Self>,
89    ) {
90        let path = if let Some(Tab::Editor(editor_tab)) = self.tabs.get(tab_index) {
91            editor_tab.file_path.clone()
92        } else {
93            None
94        };
95        if let Some(path) = path {
96            log::debug!("Reloading tab content from disk: {:?}", path);
97            match std::fs::read(&path) {
98                Ok(bytes) => {
99                    let (encoding, contents) = detect_encoding_and_decode(&bytes);
100                    if let Some(Tab::Editor(editor_tab)) = self.tabs.get_mut(tab_index) {
101                        editor_tab.content.update(cx, |input_state, cx| {
102                            input_state.set_value(&contents, window, cx);
103                        });
104                        editor_tab.set_original_content_from_str(&contents);
105                        editor_tab.encoding = encoding;
106                        editor_tab.modified = false;
107                        editor_tab.update_file_tooltip_cache(bytes.len());
108                        editor_tab.update_language(window, cx, &self.settings.editor_settings);
109                        log::debug!("Tab reloaded successfully from disk: {:?}", path);
110                    }
111                }
112                Err(e) => {
113                    log::error!("Failed to reload file {:?}: {}", path, e);
114                }
115            }
116        }
117    }

119    /// Internal helper function to open a file from a path. This function handles reading the file, detecting encoding, and creating the editor tab
120    ///
121    /// ### Arguments
122    /// - `view`: The view entity (WeakEntity)
123    /// - `window`: The async window context
124    /// - `path`: The path to the file to open
125    ///
126    /// ### Returns
127    /// - `None`: If the file could not be opened
128    /// - `Some(())`: If the file was opened successfully
129    async fn open_file_from_path(
130        view: WeakEntity<Self>,
131        window: &mut AsyncWindowContext,
132        path: PathBuf,
133    ) -> Option<()> {
134        log::debug!("Attempting to open file: {:?}", path);
135        let bytes = match std::fs::read(&path) {
136            Ok(bytes) => {
137                log::debug!("Successfully read file: {:?} ({} bytes)", path, bytes.len());
138                bytes
139            }
140            Err(e) => {
141                log::error!("Failed to read file {:?}: {}", path, e);
142                return None;
143            }
144        };
145        let (encoding, contents) = detect_encoding_and_decode(&bytes);
146        window
147            .update(|window, cx| {
148                _ = view.update(cx, |this, cx| {
149                    let editor_tab = EditorTab::from_file(
150                        FromFileParams {
151                            id: this.next_tab_id,
152                            path: path.clone(),
153                            contents,
154                            encoding,
155                            is_modified: false,
156                        },
157                        window,
158                        cx,
159                        &this.settings.editor_settings,
160                    );
161                    let editor_tab_index = this.tabs.len();
162                    this.tabs.push(Tab::Editor(editor_tab));
163                    this.active_tab_index = Some(editor_tab_index);
164                    this.pending_tab_scroll = Some(editor_tab_index);
165                    this.next_tab_id += 1;
166                    this.maybe_open_markdown_preview_for_editor(editor_tab_index);
167                    this.watch_file(&path);
168                    this.focus_active_tab(window, cx);
169                    if let Err(e) = this.settings.add_file(path.clone()) {
170                        log::error!("Failed to add file to recent files: {}", e);
171                    }
172                    let shared = this.shared_state(cx);
173                    let update_info = shared.update_info.lock().clone();
174                    let update_link = if let Some(info) = update_info {
175                        Some(info.download_url.clone())
176                    } else {
177                        None
178                    };
179                    let menus = menus::build_menus(&this.settings.get_recent_files(), update_link);
180                    this.update_menus(menus, cx);
181                    let title = path
182                        .file_name()
183                        .map(|file_name| file_name.to_string_lossy().to_string());
184                    this.set_title(title, cx);
185                    log::debug!("File opened successfully in new tab: {:?}", path);
186                    if let Err(e) = this.save_state(cx, window) {
187                        log::error!("Failed to save app state after opening file: {}", e);
188                        this.pending_notification = Some((
189                            NotificationType::Warning,
190                            format!("File opened but failed to save state: {}", e).into(),
191                        ));
192                    }
193                    cx.notify();
194                });
195            })
196            .ok();
197        Some(())
198    }

200    /// Open a file
201    ///
202    /// ### Arguments
203    /// - `window`: The window to open the file in
204    /// - `cx`: The application context
205    pub fn open_file(&mut self, window: &mut Window, cx: &mut Context<Self>) {
206        let path_future = cx.prompt_for_paths(PathPromptOptions {
207            files: true,
208            directories: false,
209            multiple: false,
210            prompt: None,
211        });
212        cx.spawn_in(window, async move |view, window| {
213            let paths = path_future.await.ok()?.ok()??;
214            let path = paths.first()?.clone();
215
216            // Check if tab already exists for this path
217            let should_open_new = window
218                .update(|window, cx| {
219                    view.update(cx, |this, cx| {
220                        if let Some(tab_index) = this.find_tab_by_path(&path) {
221                            log::debug!(
222                                "Tab already exists for {:?} at index {}, focusing and reloading",
223                                path,
224                                tab_index
225                            );
226                            if let Some(Tab::Editor(editor_tab)) = this.tabs.get(tab_index) {
227                                if editor_tab.modified {
228                                    log::debug!("Tab is modified, reloading content from disk");
229                                    this.reload_tab_from_disk(tab_index, window, cx);
230                                } else {
231                                    log::debug!("Tab is not modified, just focusing it");
232                                }
233                            }
234                            this.active_tab_index = Some(tab_index);
235                            this.focus_active_tab(window, cx);
236                            cx.notify();
237                            false // Don't open new tab
238                        } else {
239                            true // Open new tab
240                        }
241                    })
242                    .ok()
243                })
244                .ok()??;
245
246            if should_open_new {
247                Self::open_file_from_path(view, window, path).await
248            } else {
249                Some(())
250            }
251        })
252        .detach();
253    }

255    /// Open a file from a given path
256    ///
257    /// ### Behavior
258    /// First detects if the file is already open, and will focus on that tab if that's the case.
259    ///
260    /// ### Arguments
261    /// - `window`: The window to open the file in
262    /// - `cx`: The application context
263    /// - `path`: The path to the file to open
264    pub fn do_open_file(&mut self, window: &mut Window, cx: &mut Context<Self>, path: PathBuf) {
265        if let Some(tab_index) = self.find_tab_by_path(&path) {
266            log::debug!(
267                "Tab already exists for {:?} at index {}, focusing and reloading if modified",
268                path,
269                tab_index
270            );
271            if let Some(Tab::Editor(editor_tab)) = self.tabs.get(tab_index) {
272                if editor_tab.modified {
273                    log::debug!("Tab is modified, reloading content from disk");
274                    self.reload_tab_from_disk(tab_index, window, cx);
275                } else {
276                    log::debug!("Tab is not modified, just focusing it");
277                }
278            }
279            self.active_tab_index = Some(tab_index);
280            self.focus_active_tab(window, cx);
281            cx.notify();
282            return;
283        }
284        let window_manager = cx.global::<crate::fulgur::window_manager::WindowManager>();
285        if let Some(existing_window_id) =
286            window_manager.find_window_with_file(&path, self.window_id, cx)
287        {
288            let file_name = path
289                .file_name()
290                .and_then(|n| n.to_str())
291                .unwrap_or("Unknown file");
292            let message = format!("File '{}' is already open in another window", file_name);
293            window.push_notification((NotificationType::Info, SharedString::from(message)), cx);
294            log::debug!(
295                "File {:?} is already open in window {:?}",
296                path,
297                existing_window_id
298            );
299            return;
300        }
301        cx.spawn_in(window, async move |view, window| {
302            Self::open_file_from_path(view, window, path).await
303        })
304        .detach();
305    }

328    /// Handle dropping external file system paths into this window.
329    ///
330    /// ### Behavior
331    /// - Opens dropped files in new tabs (or focuses existing tabs via `do_open_file`)
332    /// - Ignores non-file entries (e.g. directories)
333    /// - Deduplicates duplicate paths within the same drop gesture
334    ///
335    /// ### Arguments
336    /// - `paths`: Paths provided by GPUI external file drop
337    /// - `window`: The target window
338    /// - `cx`: The application context
339    pub fn handle_external_paths_drop(
340        &mut self,
341        paths: &ExternalPaths,
342        window: &mut Window,
343        cx: &mut Context<Self>,
344    ) {
345        let mut dropped_files = Vec::new();
346        let mut seen = HashSet::new();
347        let mut skipped_non_files = 0usize;
348        for path in paths.paths() {
349            if !path.is_file() {
350                skipped_non_files += 1;
351                continue;
352            }
353            if seen.insert(path.clone()) {
354                dropped_files.push(path.clone());
355            }
356        }
357        if dropped_files.is_empty() {
358            if skipped_non_files > 0 {
359                window.push_notification(
360                    (
361                        NotificationType::Info,
362                        SharedString::from("Dropped items contain no files to open"),
363                    ),
364                    cx,
365                );
366            }
367            return;
368        }
369        log::info!(
370            "Opening {} dropped file(s) in window {:?}",
371            dropped_files.len(),
372            self.window_id
373        );
374        for file_path in dropped_files {
375            self.do_open_file(window, cx, file_path);
376        }
377        if skipped_non_files > 0 {
378            window.push_notification(
379                (
380                    NotificationType::Info,
381                    SharedString::from(format!(
382                        "Ignored {} dropped item(s) that are not files",
383                        skipped_non_files
384                    )),
385                ),
386                cx,
387            );
388        }
389    }

391    /// Save a file
392    ///
393    /// ### Arguments
394    /// - `window`: The window to save the file in
395    /// - `cx`: The application context
396    pub fn save_file(&mut self, window: &mut Window, cx: &mut Context<Self>) {
397        if self.tabs.is_empty() {
398            return;
399        }
400        let Some(active_tab_index) = self.active_tab_index else {
401            return;
402        };
403        let active_tab = &self.tabs[active_tab_index];
404        let (path, content_entity) = match active_tab {
405            Tab::Editor(editor_tab) => {
406                let Some(file_path) = editor_tab.file_path.clone() else {
407                    self.save_file_as(window, cx);
408                    return;
409                };
410                (file_path, editor_tab.content.clone())
411            }
412            Tab::Settings(_) | Tab::MarkdownPreview(_) => return,
413        };
414        let contents = content_entity.read(cx).text().to_string();
415        log::debug!("Saving file: {:?} ({} bytes)", path, contents.len());
416        if let Err(e) = atomic_write_file(&path, contents.as_bytes()) {
417            log::error!("Failed to save file {:?}: {}", path, e);
418            let file_name = path.file_name().and_then(|n| n.to_str()).unwrap_or("file");
419            window.push_notification(
420                (
421                    NotificationType::Error,
422                    SharedString::from(format!("Failed to save '{}': {}", file_name, e)),
423                ),
424                cx,
425            );
426            return;
427        }
428        log::debug!("File saved successfully: {:?}", path);
429        self.file_watch_state
430            .last_file_saves
431            .insert(path.clone(), std::time::Instant::now());
432        if let Tab::Editor(editor_tab) = &mut self.tabs[active_tab_index] {
433            editor_tab.mark_as_saved(cx);
434            editor_tab.update_file_tooltip_cache(contents.len());
435        }
436        cx.notify();
437    }

439    /// Save a file as
440    ///
441    /// ### Arguments
442    /// - `window`: The window to save the file as in
443    /// - `cx`: The application context
444    pub fn save_file_as(&mut self, window: &mut Window, cx: &mut Context<Self>) {
445        if self.tabs.is_empty() {
446            return;
447        }
448        let Some(active_tab_index) = self.active_tab_index else {
449            return;
450        };
451        let (content_entity, directory, suggested_filename) = match &self.tabs[active_tab_index] {
452            Tab::Editor(editor_tab) => {
453                let dir = if let Some(ref path) = editor_tab.file_path {
454                    path.parent()
455                        .unwrap_or(std::path::Path::new("."))
456                        .to_path_buf()
457                } else {
458                    std::env::current_dir().unwrap_or_default()
459                };
460                let suggested = editor_tab.get_suggested_filename();
461                (editor_tab.content.clone(), dir, suggested)
462            }
463            Tab::Settings(_) | Tab::MarkdownPreview(_) => return,
464        };
465        let path_future = cx.prompt_for_new_path(&directory, suggested_filename.as_deref());
466        cx.spawn_in(window, async move |view, window| {
467            let path = path_future.await.ok()?.ok()??;
468            let contents = window
469                .update(|_, cx| content_entity.read(cx).text().to_string())
470                .ok()?;
471            log::debug!("Saving file as: {:?} ({} bytes)", path, contents.len());
472            if let Err(e) = atomic_write_file(&path, contents.as_bytes()) {
473                log::error!("Failed to save file {:?}: {}", path, e);
474                let file_name = path.file_name().and_then(|n| n.to_str()).unwrap_or("file");
475                let message = SharedString::from(format!("Failed to save '{}': {}", file_name, e));
476                window
477                    .update(|_, cx| {
478                        _ = view.update(cx, |this, cx| {
479                            this.pending_notification = Some((NotificationType::Error, message));
480                            cx.notify();
481                        });
482                    })
483                    .ok()?;
484                return None;
485            }
486            log::debug!("File saved successfully as: {:?}", path);
487            window
488                .update(|window, cx| {
489                    _ = view.update(cx, |this, cx| {
490                        let old_path = if let Some(Tab::Editor(editor_tab)) =
491                            this.tabs.get(active_tab_index)
492                        {
493                            editor_tab.file_path.clone()
494                        } else {
495                            None
496                        };
497                        if let Some(old_path) = old_path {
498                            this.unwatch_file(&old_path);
499                        }
500                        this.file_watch_state
501                            .last_file_saves
502                            .insert(path.clone(), std::time::Instant::now());
503                        if let Some(Tab::Editor(editor_tab)) = this.tabs.get_mut(active_tab_index) {
504                            editor_tab.file_path = Some(path.clone());
505                            editor_tab.title = path
506                                .file_name()
507                                .and_then(|n| n.to_str())
508                                .unwrap_or(UNTITLED)
509                                .to_string()
510                                .into();
511                            editor_tab.mark_as_saved(cx);
512                            editor_tab.update_file_tooltip_cache(contents.len());
513                            editor_tab.update_language(window, cx, &this.settings.editor_settings);
514                            cx.notify();
515                        }
516                        this.watch_file(&path);
517                    });
518                })
519                .ok()?;
520            Some(())
521        })
522        .detach();
523    }

580    ///
581    /// ### Arguments
582    /// - `window`: The window to open files in
583    /// - `cx`: The application context
584    pub fn process_pending_files_from_macos(
585        &mut self,
586        window: &mut Window,
587        cx: &mut Context<Self>,
588    ) {
589        let shared = self.shared_state(cx);
590        let should_process_files = cx
591            .global::<window_manager::WindowManager>()
592            .get_last_focused()
593            .map(|id| id == self.window_id)
594            .unwrap_or(true); // If no last focused window, allow this one to process
595        let files_to_open = if should_process_files {
596            if let Some(mut pending) = shared.pending_files_from_macos.try_lock() {
597                if pending.is_empty() {
598                    Vec::new()
599                } else {
600                    log::info!(
601                        "Processing {} pending file(s) from macOS open event in window {:?}",
602                        pending.len(),
603                        self.window_id
604                    );
605                    pending.drain(..).collect()
606                }
607            } else {
608                Vec::new()
609            }
610        } else {
611            Vec::new()
612        };
613        for file_path in files_to_open {
614            self.handle_open_file_from_cli(window, cx, file_path);
615        }
616    }

635    /// Open the native OS print dialog for the current document
636    ///
637    /// Writes the active tab's content to a temporary HTML file and opens it with
638    /// the system's default browser, which automatically triggers the native print dialog.
639    /// This approach works cross-platform without requiring OS-specific print APIs.
640    ///
641    /// ### Arguments
642    /// - `window`: The window containing the editor
643    /// - `cx`: The application context
644    pub fn print_file(&mut self, window: &mut Window, cx: &mut Context<Self>) {
645        let Some(active_tab_index) = self.active_tab_index else {
646            return;
647        };
648        let (title, content) = match &self.tabs[active_tab_index] {
649            Tab::Editor(editor_tab) => {
650                let title = editor_tab.title.clone();
651                let content = editor_tab.content.read(cx).text().to_string();
652                (title, content)
653            }
654            Tab::Settings(_) | Tab::MarkdownPreview(_) => return,
655        };
656        let escaped_content = content
657            .replace('&', "&amp;")
658            .replace('<', "&lt;")
659            .replace('>', "&gt;");
660        let escaped_title = title
661            .replace('&', "&amp;")
662            .replace('<', "&lt;")
663            .replace('>', "&gt;");
664        let html = format!(
665            r#"<!DOCTYPE html>
666<html>
667<head>
668<meta charset="utf-8">
669<title>{title}</title>
670<style>
671  body {{ margin: 0; padding: 1em; font-family: monospace; white-space: pre-wrap; word-wrap: break-word; }}
672  @media print {{ body {{ margin: 0; }} }}
673</style>
674</head>
675<body>{content}</body>
676<script>window.onload = function() {{ window.print(); }};</script>
677</html>"#,
678            title = escaped_title,
679            content = escaped_content,
680        );
681        let temp_path =
682            std::env::temp_dir().join(format!("fulgur_print_{}.html", std::process::id()));
683        if let Err(e) = std::fs::write(&temp_path, html.as_bytes()) {
684            log::error!("Failed to write print temp file: {}", e);
685            window.push_notification(
686                (
687                    NotificationType::Error,
688                    SharedString::from(format!("Failed to prepare print: {}", e)),
689                ),
690                cx,
691            );
692            return;
693        }
694        if let Err(e) = open::that(&temp_path) {
695            log::error!("Failed to open print file: {}", e);
696            window.push_notification(
697                (
698                    NotificationType::Error,
699                    SharedString::from(format!("Failed to open print dialog: {}", e)),
700                ),
701                cx,
702            );
703        }
704    }

706    /// Handle pending jump-to-line action
707    ///
708    /// ### Arguments
709    /// - `window`: The window containing the editor
710    /// - `cx`: The application context
711    pub fn handle_pending_jump_to_line(&mut self, window: &mut Window, cx: &mut Context<Self>) {
712        if let Some(jump) = self.pending_jump.take()
713            && let Some(index) = self.active_tab_index
714            && let Some(Tab::Editor(editor_tab)) = self.tabs.get_mut(index)
715        {
716            editor_tab.jump_to_line(window, cx, jump);
717        }
718    }
