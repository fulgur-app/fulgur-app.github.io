13    /// Start the file watcher, watch all open files, and spawn the event consumer task
14    ///
15    /// ### Arguments
16    /// - `cx`: The application context used to spawn the consumer task
17    pub fn start_file_watcher(&mut self, cx: &mut Context<Self>) {
18        let (mut watcher, receiver) = FileWatcher::new();
19        if let Err(e) = watcher.start() {
20            log::error!("Failed to start file watcher: {e}");
21            return;
22        }
23        for tab in &self.tabs {
24            if let Tab::Editor(editor_tab) = tab.read(cx)
25                && let Some(path) = editor_tab.file_path()
26                && let Err(e) = watcher.watch_file(path)
27            {
28                log::warn!("Failed to watch file {}: {}", path.display(), e);
29            }
30        }
31        let pending_rename_from = Arc::clone(&watcher.pending_rename_from);
32        let flush_tx = watcher.event_tx.clone();
33        self.file_watch_state.file_watcher = Some(watcher);
34        self.file_watch_state.consumer_task =
35            Some(self.spawn_file_watch_consumer(receiver, pending_rename_from, flush_tx, cx));
36    }
37
38    /// Stop the file watcher and cancel the event consumer task
39    pub fn stop_file_watcher(&mut self) {
40        if let Some(mut watcher) = self.file_watch_state.file_watcher.take() {
41            watcher.stop();
42        }
43        self.file_watch_state.consumer_task = None;
44        self.clear_file_watch_bookkeeping();
45    }
46
47    /// Add a file to the watcher
48    ///
49    /// ### Arguments
50    /// - `path`: The path to the file to watch
51    pub fn watch_file(&mut self, path: &std::path::Path) {
52        if let Some(watcher) = &mut self.file_watch_state.file_watcher
53            && let Err(e) = watcher.watch_file(&path.to_path_buf())
54        {
55            log::warn!("Failed to watch file {}: {}", path.display(), e);
56        }
57    }
58
59    /// Remove a file from the watcher
60    ///
61    /// ### Arguments
62    /// - `path`: The path to the file to unwatch
63    pub fn unwatch_file(&mut self, path: &PathBuf) {
64        if let Some(watcher) = &mut self.file_watch_state.file_watcher {
65            watcher.unwatch_file(path);
66        }
67        self.prune_file_watch_bookkeeping_for_path(path);
68    }
69
70    /// Spawn the task that consumes file watch events for this window
71    ///
72    /// ### Arguments
73    /// - `events`: The receiver side of the watcher's event channel
74    /// - `pending_rename_from`: The watcher's pending split-rename accumulator
75    /// - `flush_tx`: Sender used to emit the synthesized `Deleted` event on expiry
76    /// - `cx`: The application context used to spawn the task
77    ///
78    /// ### Returns
79    /// - `Task<()>`: The consumer task; dropping it cancels the consumer
80    fn spawn_file_watch_consumer(
81        &self,
82        mut events: Receiver<FileWatchEvent>,
83        pending_rename_from: Arc<Mutex<Option<(PathBuf, Instant)>>>,
84        mut flush_tx: Sender<FileWatchEvent>,
85        cx: &mut Context<Self>,
86    ) -> Task<()> {
87        let window_id = self.window_id;
88        cx.spawn(async move |view, cx| {
89            loop {
90                let has_pending_rename = pending_rename_from
91                    .lock()
92                    .is_ok_and(|pending| pending.is_some());
93                let event = if has_pending_rename {
94                    let timer = cx.background_executor().timer(PENDING_RENAME_TIMEOUT);
95                    match futures::future::select(events.next(), std::pin::pin!(timer)).await {
96                        Either::Left((event, _)) => event,
97                        Either::Right(((), _)) => {
98                            FileWatcher::expire_pending_rename_from(
99                                &pending_rename_from,
100                                &mut flush_tx,
101                            );
102                            continue;
103                        }
104                    }
105                } else {
106                    events.next().await
107                };
108                let Some(event) = event else {
109                    break;
110                };
111                let handle = cx.update(|cx| {
112                    cx.windows()
113                        .into_iter()
114                        .find(|handle| handle.window_id() == window_id)
115                });
116                let Some(handle) = handle else {
117                    break;
118                };
119                let delivered = handle.update(cx, |_, window, cx| {
120                    view.update(cx, |this, cx| {
121                        this.handle_file_watch_event(event, window, cx);
122                        cx.notify();
123                    })
124                });
125                if !matches!(delivered, Ok(Ok(()))) {
126                    break;
127                }
128            }
129        })
130    }

139    #[gpui::test]
140    #[cfg_attr(
141        target_os = "macos",
142        ignore = "known upstream a11y panic on gpui TestWindow"
143    )]
144    fn test_unwatch_file_prunes_bookkeeping_maps(cx: &mut TestAppContext) {
145        let (fulgur, mut visual_cx) = setup_fulgur(cx);
146        let path = temp_test_path("fulgur_unwatch_cleanup.txt");
147        visual_cx.update(|_window, cx| {
148            fulgur.update(cx, |this, _cx| {
149                this.file_watch_state
150                    .last_file_events
151                    .insert(path.clone(), Instant::now());
152                this.file_watch_state
153                    .last_file_saves
154                    .insert(path.clone(), Instant::now());
155                this.file_watch_state
156                    .pending_conflicts
157                    .insert(path.clone(), 0);
158                this.unwatch_file(&path);
159                assert!(
160                    !this.file_watch_state.last_file_events.contains_key(&path),
161                    "unwatch must prune debounce bookkeeping"
162                );
163                assert!(
164                    !this.file_watch_state.last_file_saves.contains_key(&path),
165                    "unwatch must prune save bookkeeping"
166                );
167                assert!(
168                    !this.file_watch_state.pending_conflicts.contains_key(&path),
169                    "unwatch must prune deferred conflict bookkeeping"
170                );
171            });
172        });
173    }
