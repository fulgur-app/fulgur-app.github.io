38pub struct Fulgur {
39    // Identity of this window in the state database, stable across restarts
40    persistent_window_id: i64,
41    pub window_id: WindowId,                          // The ID of this window
42    focus_handle: FocusHandle,                        // The focus handle for the application
43    title_bar: Entity<CustomTitleBar>,                // The title bar of the application
44    tabs: Vec<Entity<Tab>>, // The tabs in the application, each its own entity
45    active_tab_id: Option<TabId>, // Stable ID of the active tab
46    next_tab_id: TabId,     // The next tab ID
47    search_bar: Entity<SearchBar>, // The search and replace bar view
48    _search_bar_subscription: Subscription, // Routes SearchBarEvent from the search bar to window-level handlers
49    markdown_toolbar: Entity<MarkdownToolbar>, // The markdown formatting toolbar view (acts directly on the active editor, emits no events)
50    csv_toolbar: Entity<CsvToolbar>, // The CSV structural-edit toolbar view (acts directly on the active tab's table, emits no events)
51    color_picker_bar: Entity<ColorPickerBar>, // The color picker bar view (owns the picker widget and its three text inputs)
52    _color_picker_bar_subscription: Subscription, // Routes ColorPickerBarEvent from the color picker bar to window-level handlers
53    pub jump_to_line_input: Entity<InputState>,   // Input for jumping to a line in the editor
54    pub rename_tab_input: Entity<InputState>,     // Input for renaming an unsaved tab
55    pending_jump: Option<editor_tab::Jump>,       // Pending jump to line action
56    pub settings: Settings, // The settings for the application (local snapshot, refreshed by the SharedAppState observer)
57    _shared_state_observation: Subscription, // Global observer keeping the local settings snapshot in sync with SharedAppState and applying editor settings to tabs
58    log_tail_state: HashMap<TabId, LogTailState>, // Per-log-tab tail bookkeeping (byte offset, dropped lines, pending text) keyed by tab id
59    log_tail_cancel: HashMap<TabId, Arc<AtomicBool>>, // Cancellation flag for the per-active-log-tab poll task keyed by tab id
60    pub file_watch_state: FileWatchState, // File watching state for external file change detection
61    save_failed_once: bool, // Flag: save already failed once, allow force-close on next attempt
62    pub share_sheet_state: Option<Arc<ui::sheets::share_file::ShareSheetState>>, // When Some, a share sheet is open and devices are being fetched per profile
63    cached_window_bounds: Option<state::SerializedWindowBounds>, // Cached window bounds for cross-window saves
64    font_select_subscription: Option<Subscription>, // Subscription for font family selection events (set when settings tab is opened)
65    editor_context_menu: Option<(Point<Pixels>, Entity<PopupMenu>)>, // Custom right-click context menu for the editor and markdown preview
66    editor_context_menu_subscription: Option<Subscription>, // Subscription to clear editor_context_menu on dismiss
67    markdown_preview_focus: FocusHandle, // Stable dispatch target for the markdown preview context menu (Copy / Select All)
68    markdown_panel_view_state: Option<Entity<gpui_component::text::TextViewState>>, // Owned text view state backing the inline markdown preview panel, created lazily on render
69    markdown_preview_pending_copy: Option<String>, // Preview selection captured at right-click time, before the menu's own click clears it, consumed by the Copy action
70    status_bar: Entity<StatusBar>, // The status bar view at the bottom of the window
71    _status_bar_subscription: Subscription, // Routes StatusBarEvent from the status bar to window-level handlers
72    tab_bar: Entity<TabBar>,                // The tab bar view at the top of the window
73    _tab_bar_subscription: Subscription, // Routes TabBarEvent from the tab bar to window-level handlers
74    pub pending_tab_transfer: Option<editor_tab::TabTransferData>, // Incoming tab state from another window, processed on next render
75    pending_tab_removal: Option<TabId>, // Tab ID to remove after it has been sent to another window
76    pending_transfer_scroll: Option<gpui_component::input::Position>, // Deferred scroll-to-cursor after tab transfer (needs one render cycle for layout)
77    pending_remote_open: Arc<parking_lot::Mutex<Vec<PendingRemoteOpenOutcome>>>, // Queue for SSH background threads to deliver loaded remote files
78    next_remote_request_id: u64, // Monotonic identifier for remote open/save operations targeting existing tabs
79    latest_remote_open_request_by_tab: HashMap<TabId, u64>, // Latest remote-open request id expected per tab id
80    latest_remote_save_request_by_tab: HashMap<TabId, u64>, // Latest remote-save request id expected per tab id
81    last_failed_remote_open_url: Option<String>, // Last attempted remote URL kept for retry prefill after connection/open failures
82    pending_remote_restore: HashSet<TabId>, // Restored remote tab ids that should lazily reconnect on first activation/save
83    inflight_remote_restore: HashSet<TabId>, // Restored remote tabs currently running a reconnect task
84    inflight_saves: HashMap<TabId, PathBuf>, // Destination path of each background local-file write in flight, keyed by tab id; guards against overlapping saves and suppresses self-save watcher events
85    pending_initial_active_tab: Option<TabId>, // Active tab to re-activate after first render so dialogs can open safely
86    has_rendered_once: bool, // Tracks first render completion for startup actions that require mounted Root layers
87}
