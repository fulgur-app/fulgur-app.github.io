38    pub window_id: WindowId,                         // The ID of this window

40    title_bar: Entity<CustomTitleBar>,               // The title bar of the application
41    tabs: Vec<Tab>,                                  // The tabs in the application
42    active_tab_index: Option<usize>,                 // Index of the active tab
43    next_tab_id: usize,                              // The next tab ID
44    pub search_state: SearchState,                   // Search and replace functionality state
45    pub color_picker_bar_state: ColorPickerBarState, // Color picker bar state
46    pub jump_to_line_input: Entity<InputState>,      // Input for jumping to a line in the editor
47    pending_jump: Option<editor_tab::Jump>,          // Pending jump to line action
48    pub settings: Settings, // The settings for the application (local copy for fast access)
49    settings_changed: bool, // Flag to indicate that the settings have been changed and need to be saved
50    local_settings_version: u64, // Track the version of settings this window has loaded
51    rendered_tabs: HashSet<usize>, // Track which tabs have been rendered
52    tabs_pending_update: HashSet<usize>, // Track tabs that need settings update on next render
53    editor_modified_subscriptions: HashMap<usize, (EntityId, Subscription)>, // Per-editor (subscribed content entity id, subscription) for incremental modified-state updates
54    markdown_preview_cache: HashMap<usize, SharedString>, // Cached markdown source text keyed by source editor tab id
55    markdown_preview_to_refresh: HashSet<usize>, // Source tab ids whose cached preview text is stale and must be refreshed on next read
56    markdown_preview_subscriptions: HashMap<usize, (EntityId, Subscription)>, // Per-source (subscribed content entity id, subscription) for markdown preview cache updates
57    log_tail_state: HashMap<usize, LogTailState>, // Per-log-tab tail bookkeeping (byte offset, dropped lines, pending text) keyed by tab id
58    log_tail_cancel: HashMap<usize, Arc<AtomicBool>>, // Cancellation flag for the per-active-log-tab poll task keyed by tab id
59    tab_scroll_handle: ScrollHandle, // Scroll handle for the tab bar to scroll active tab into view
60    pending_tab_scroll: Option<usize>, // Deferred scroll-to-tab request (needs one render cycle for layout)
61    pub file_watch_state: FileWatchState, // File watching state for external file change detection
62    pub pending_notification: Option<(NotificationType, SharedString)>, // Pending notification to display on next render
63    save_failed_once: bool, // Flag: save already failed once, allow force-close on next attempt
64    pub share_sheet_state: Option<Arc<ui::sheets::share_file::ShareSheetState>>, // When Some, a share sheet is open and devices are being fetched per profile
65    cached_window_bounds: Option<state::SerializedWindowBounds>, // Cached window bounds for cross-window saves
66    font_select_subscription: Option<Subscription>, // Subscription for font family selection events (set when settings tab is opened)
67    editor_context_menu: Option<(Point<Pixels>, Entity<PopupMenu>)>, // Custom right-click context menu for the editor
68    editor_context_menu_subscription: Option<Subscription>, // Subscription to clear editor_context_menu on dismiss
69    drag_ghost: Option<(usize, ui::tabs::tab_drag::DraggedTab)>, // Ghost tab shown at insertion point during tab drag
70    status_bar_cache: StatusBarCache, // Cached status bar label strings (refreshed each render)
71    cached_tab_filename_counts: HashMap<String, usize>, // Cached tab filename frequency map (refreshed when tabs change)
72    tab_filename_fp: u64, // Fingerprint of the tab list used to detect when cached_tab_filename_counts is stale
73    pub pending_tab_transfer: Option<editor_tab::TabTransferData>, // Incoming tab state from another window, processed on next render
74    pending_tab_removal: Option<usize>, // Tab ID to remove after it has been sent to another window
75    pending_transfer_scroll: Option<gpui_component::input::Position>, // Deferred scroll-to-cursor after tab transfer (needs one render cycle for layout)
76    pending_remote_open: Arc<parking_lot::Mutex<Vec<PendingRemoteOpenOutcome>>>, // Queue for SSH background threads to deliver loaded remote files
77    next_remote_request_id: u64, // Monotonic identifier for remote open/save operations targeting existing tabs
78    latest_remote_open_request_by_tab: HashMap<usize, u64>, // Latest remote-open request id expected per tab id
79    latest_remote_save_request_by_tab: HashMap<usize, u64>, // Latest remote-save request id expected per tab id
80    last_failed_remote_open_url: Option<String>, // Last attempted remote URL kept for retry prefill after connection/open failures
81    pending_remote_restore: HashSet<usize>, // Restored remote tab ids that should lazily reconnect on first activation/save
82    inflight_remote_restore: HashSet<usize>, // Restored remote tabs currently running a reconnect task
83    pending_initial_active_tab: Option<usize>, // Active tab to re-activate after first render so dialogs can open safely
84    has_rendered_once: bool, // Tracks first render completion for startup actions that require mounted Root layers
85    #[cfg(any(target_os = "macos", target_os = "windows"))]
86    local_window_menu_fingerprint: u64, // Cached local menu-state fingerprint published to WindowManager
87    #[cfg(target_os = "macos")]
88    last_dock_menu_revision: u64, // Last global menu-state revision processed by dock menu updater
89    #[cfg(target_os = "macos")]
90    last_dock_menu_hash: u64, // Hash of the last dock menu state to avoid unnecessary rebuilds
91    #[cfg(target_os = "windows")]
92    last_jump_list_revision: u64, // Last global menu-state revision processed by jump list updater
93    #[cfg(target_os = "windows")]
94    last_jump_list_hash: u64, // Hash of the last jump list state to avoid unnecessary rebuilds
