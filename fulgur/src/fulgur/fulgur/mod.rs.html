200    /// Create a new Fulgur instance
201    ///
202    /// ### Arguments
203    /// - `window`: The window to create the Fulgur instance in
204    /// - `cx`: The application context
205    /// - `window_id`: The window ID for this instance, obtained from `window.window_handle().window_id()`
206    /// - `window_index`: Index of this window in saved state (0 = first window, etc.). Use `usize::MAX` for new empty windows.
207    ///
208    /// ### Returns
209    /// - `Entity<Self>`: The new Fulgur instance
210    pub fn new(
211        window: &mut Window,
212        cx: &mut App,
213        window_id: WindowId,
214        window_index: usize,
215    ) -> Entity<Self> {
216        let title_bar = CustomTitleBar::new(window, cx);
217        let shared = cx.global::<shared_state::SharedAppState>();
218        let settings = shared.settings.lock().clone();
219        let search_input = cx.new(|cx| InputState::new(window, cx).placeholder("Search"));
220        let replace_input = cx.new(|cx| InputState::new(window, cx).placeholder("Replace"));
221        let jump_to_line_input =
222            cx.new(|cx| InputState::new(window, cx).placeholder("Jump to line or line:character"));
223        let entity = cx.new(|cx| {
224            let search_subscription =
225                cx.subscribe(&search_input, |this: &mut Self, _, ev: &InputEvent, cx| {
226                    if let InputEvent::Change = ev
227                        && this.search_state.show_search
228                    {
229                        cx.notify();
230                    }
231                });
232            Self {
233                window_id,
234                focus_handle: cx.focus_handle(),
235                title_bar,
236                tabs: vec![],
237                active_tab_index: None,
238                next_tab_id: 0,
239                search_state: SearchState::new(search_input, replace_input, search_subscription),
240                color_picker_bar_state: ColorPickerBarState::new(window, cx),
241                jump_to_line_input,
242                pending_jump: None,
243                jump_to_line_dialog_open: false,
244                settings,
245                settings_changed: false,
246                local_settings_version: 0,
247                rendered_tabs: HashSet::new(),
248                tabs_pending_update: HashSet::new(),
249                editor_modified_subscriptions: HashMap::new(),
250                markdown_preview_cache: HashMap::new(),
251                markdown_preview_to_refresh: HashSet::new(),
252                markdown_preview_subscriptions: HashMap::new(),
253                tab_scroll_handle: ScrollHandle::new(),
254                pending_tab_scroll: None,
255                file_watch_state: FileWatchState::new(),
256                sse_state: SseState::new(),
257                pending_notification: None,
258                save_failed_once: false,
259                pending_share_sheet: false,
260                cached_window_bounds: None,
261                _font_select_subscription: None,
262                editor_context_menu: None,
263                _editor_context_menu_subscription: None,
264                drag_ghost: None,
265                status_bar_cache: StatusBarCache::default(),
266                cached_tab_filename_counts: HashMap::new(),
267                tab_filename_fp: u64::MAX, // sentinel: differs from fingerprint of any real tab list
268                pending_tab_transfer: None,
269                pending_tab_removal: None,
270                pending_transfer_scroll: None,
271                pending_remote_open: Arc::new(parking_lot::Mutex::new(Vec::new())),
272                next_remote_request_id: 1,
273                latest_remote_open_request_by_tab: HashMap::new(),
274                latest_remote_save_request_by_tab: HashMap::new(),
275                last_failed_remote_open_url: None,
276                pending_remote_restore: HashSet::new(),
277                inflight_remote_restore: HashSet::new(),
278                pending_initial_active_tab: None,
279                has_rendered_once: false,
280                #[cfg(any(target_os = "macos", target_os = "windows"))]
281                local_window_menu_fingerprint: 0,
282                #[cfg(target_os = "macos")]
283                last_dock_menu_revision: 0,
284                #[cfg(target_os = "macos")]
285                last_dock_menu_hash: 0,
286                #[cfg(target_os = "windows")]
287                last_jump_list_revision: 0,
288                #[cfg(target_os = "windows")]
289                last_jump_list_hash: 0,
290            }
291        });
292        let (sse_tx, sse_rx) = std::sync::mpsc::channel();
293        let sse_shutdown_flag = Arc::new(AtomicBool::new(false));
294        entity.update(cx, |this, cx| {
295            this.sse_state.sse_events = Some(sse_rx);
296            this.sse_state.sse_event_tx = Some(sse_tx);
297            this.sse_state.sse_shutdown_flag = Some(sse_shutdown_flag);
298            let shared = cx.global::<shared_state::SharedAppState>();
299            if let Some(error_msg) = shared.sync_error.lock().as_ref() {
300                this.pending_notification =
301                    Some((NotificationType::Error, error_msg.clone().into()));
302            }
303            if window_index == usize::MAX {
304                let initial_tab = Tab::Editor(editor_tab::EditorTab::new(
305                    0,
306                    crate::fulgur::ui::components_utils::UNTITLED,
307                    window,
308                    cx,
309                    &this.settings.editor_settings,
310                ));
311                this.tabs.push(initial_tab);
312                this.active_tab_index = Some(0);
313                this.next_tab_id = 1;
314            } else if window_index < usize::MAX - 1 {
315                // usize::MAX - 1 means new window receiving a tab transfer: skip initial tab
316                this.load_state(window, cx, window_index);
317                this.pending_tab_scroll = this.active_tab_index;
318                this.pending_initial_active_tab = this.active_tab_index;
319            }
320            if this.settings.editor_settings.watch_files {
321                this.start_file_watcher();
322            }
323        });
324        sync::synchronization::begin_synchronization(&entity, cx);
325        entity
326    }

439    /// Assemble the final UI tree with all layers
440    ///
441    /// ### Arguments
442    /// - `app_content`: The main content area (from `build_app_content_with_actions()`)
443    /// - `window`: The window to assemble the UI for
444    /// - `cx`: The application context
445    ///
446    /// ### Returns
447    /// - `impl IntoElement`: The complete UI tree ready to be rendered
448    fn assemble_ui_tree(
449        &self,
450        app_content: impl IntoElement,
451        window: &mut Window,
452        cx: &mut Context<Self>,
453    ) -> impl IntoElement {
454        // Create root layout: TitleBar OUTSIDE of focus-tracked content
455        // This is critical for Windows hit-testing to work!
456        let root_content = v_flex()
457            .size_full()
458            .child(self.title_bar.clone())
459            .child(app_content);
460        let mut root = div()
461            .size_full()
462            .child(root_content)
463            .children(Root::render_sheet_layer(window, cx))
464            .children(Root::render_notification_layer(window, cx))
465            .children(Root::render_dialog_layer(window, cx));
466        if let Some((position, menu)) = self
467            .editor_context_menu
468            .as_ref()
469            .map(|(pos, menu)| (*pos, menu.clone()))
470        {
471            root = root.child(
472                gpui::deferred(
473                    gpui::anchored()
474                        .position(position)
475                        .snap_to_window_with_margin(px(8.))
476                        .anchor(Anchor::TopLeft)
477                        .child(
478                            div()
479                                .font_family(cx.theme().font_family.clone())
480                                .cursor_default()
481                                .child(menu),
482                        ),
483                )
484                .with_priority(1),
485            );
486        }
487        root
488    }

510    ///

513    fn render(&mut self, window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
514        self.process_pending_initial_active_tab_activation(window, cx);
515        self.process_window_state_updates(window, cx);
516        self.process_update_notifications(window, cx);
517        self.synchronize_settings_from_other_windows(cx);
518        self.process_pending_files_from_macos(window, cx);
519        #[cfg(target_os = "windows")]
520        self.process_pending_ipc_commands(window, cx);
521        self.process_shared_files_from_sync(window, cx);
522        self.process_file_watch_events(window, cx);
523        self.process_sse_events(window, cx);
524        self.process_pending_remote_files(window, cx);
525        self.process_pending_share_sheet(window, cx);
526        if self.tabs.is_empty() {
527            self.active_tab_index = None;
528        }
529        self.update_search_if_needed(window, cx);
530        self.propagate_settings_to_tabs(window, cx);
531        self.track_newly_rendered_tabs(cx);
532        self.handle_pending_transfer_scroll(window, cx);
533        self.handle_pending_tab_transfer(window, cx);
534        self.handle_pending_tab_removal(window, cx);
535        self.handle_pending_jump_to_line(window, cx);
536        if !self.jump_to_line_dialog_open {
537            window.close_dialog(cx);
538            self.jump_to_line_dialog_open = true;
539        }
540        self.update_modified_status(cx);
541        self.prune_markdown_preview_cache(cx);
542        self.process_pending_tab_scroll(cx);
543        self.refresh_tab_filename_counts();
544        self.refresh_status_bar_labels(cx);
545        let app_content = self.build_app_content_with_actions(self.active_tab_index, window, cx);
546        self.assemble_ui_tree(app_content, window, cx)
547    }

552    ///
553    /// The first render builds the window's `Root` layers. Startup flows that can open
554    /// dialogs (like remote password prompts triggered by `set_active_tab`) must wait
555    /// until that initial render has completed.
556    ///
557    /// ### Arguments
558    /// - `window`: The window context
559    /// - `cx`: The application context
560    fn process_pending_initial_active_tab_activation(
561        &mut self,
562        window: &mut Window,
563        cx: &mut Context<Self>,
564    ) {
565        if !self.has_rendered_once {
566            self.has_rendered_once = true;
567            if self.pending_initial_active_tab.is_some() {
568                cx.notify();
569            }
570            return;
571        }
572
573        if let Some(index) = self.pending_initial_active_tab.take() {
574            self.set_active_tab(index, window, cx);
575        }
576    }

578    /// Process a deferred scroll-to-tab request
579    ///
580    /// GPUI's `ScrollHandle` needs one render cycle to populate child bounds and overflow
581    /// state before `scroll_to_item` can work. On the first frame, layout hasn't happened
582    /// yet so the scroll is silently dropped. This method waits until child bounds are
583    /// available (meaning layout has completed at least once), then issues the scroll.
584    ///
585    /// ### Arguments
586    /// - `cx`: The application context
587    fn process_pending_tab_scroll(&mut self, cx: &mut Context<Self>) {
588        if let Some(index) = self.pending_tab_scroll {
589            if self.tab_scroll_handle.bounds_for_item(0).is_some() {
590                self.tab_scroll_handle.scroll_to_item(index);
591                self.pending_tab_scroll = None;
592            } else {
593                cx.notify();
594            }
595        }
596    }

598    /// Handle a right-click in the editor area to show a custom context menu.
599    ///
600    /// Called during the capture phase so propagation can be stopped before
601    /// the editor's built-in context menu fires.
602    ///
603    /// ### Arguments
604    /// - `event`: The mouse-down event
605    /// - `window`: The window context
606    /// - `cx`: The application context
607    fn on_editor_right_click(
608        &mut self,
609        event: &MouseDownEvent,
610        window: &mut Window,
611        cx: &mut Context<Self>,
612    ) {
613        if event.button != MouseButton::Right {
614            return;
615        }
616        cx.stop_propagation();
617
618        let Some(active_index) = self.active_tab_index else {
619            return;
620        };
621        let Some(Tab::Editor(editor_tab)) = self.tabs.get(active_index) else {
622            return;
623        };
624        let editor_focus = editor_tab.content.focus_handle(cx);
625        let has_file = editor_tab.file_path().is_some();
626        let position = event.position;
627
628        let menu = PopupMenu::build(window, cx, {
629            let editor_focus = editor_focus.clone();
630            move |mut menu, _window, _cx| {
631                menu = menu.action_context(editor_focus);
632                if has_file {
633                    menu = menu
634                        .menu(
635                            crate::fulgur::ui::components_utils::reveal_in_file_manager_label(),
636                            Box::new(ui::tabs::tab_bar::ShowInFileManager(active_index)),
637                        )
638                        .separator();
639                }
640                menu.menu("Cut", Box::new(gpui_component::input::Cut))
641                    .menu("Copy", Box::new(gpui_component::input::Copy))
642                    .menu("Paste", Box::new(gpui_component::input::Paste))
643                    .separator()
644                    .menu("Select All", Box::new(gpui_component::input::SelectAll))
645            }
646        });
647
648        let subscription = cx.subscribe_in(
649            &menu,
650            window,
651            |this: &mut Self, _, _: &DismissEvent, _, cx| {
652                this.editor_context_menu = None;
653                this._editor_context_menu_subscription = None;
654                cx.notify();
655            },
656        );
657
658        self.editor_context_menu = Some((position, menu));
659        self._editor_context_menu_subscription = Some(subscription);
660        cx.notify();
661    }

663    /// Render the content area (editor or settings)
664    ///
665    /// ### Arguments
666    /// - `active_tab_index`: The index of the active tab (if any)
667    /// - `window`: The window context
668    /// - `cx`: The application context
669    ///
670    /// ### Returns
671    /// - `AnyElement`: The rendered content area element (wrapped in `AnyElement`)
672    fn render_content_area(
673        &mut self,
674        active_tab_index: Option<usize>,
675        window: &mut Window,
676        cx: &mut Context<Self>,
677    ) -> AnyElement {
678        enum ActiveTabRenderData {
679            Editor {
680                tab_id: usize,
681                language: SupportedLanguage,
682                show_markdown_preview: bool,
683                content: Entity<InputState>,
684            },
685            Settings,
686            MarkdownPreview {
687                source_tab_id: usize,
688                content: Entity<InputState>,
689            },
690        }
691
692        let active_tab = active_tab_index.and_then(|active_index| {
693            self.tabs.get(active_index).map(|tab| match tab {
694                Tab::Editor(editor_tab) => ActiveTabRenderData::Editor {
695                    tab_id: editor_tab.id,
696                    language: editor_tab.language,
697                    show_markdown_preview: editor_tab.show_markdown_preview,
698                    content: editor_tab.content.clone(),
699                },
700                Tab::Settings(_) => ActiveTabRenderData::Settings,
701                Tab::MarkdownPreview(preview_tab) => ActiveTabRenderData::MarkdownPreview {
702                    source_tab_id: preview_tab.source_tab_id,
703                    content: preview_tab.content.clone(),
704                },
705            })
706        });
707
708        if let Some(tab) = active_tab {
709            match tab {
710                ActiveTabRenderData::Editor {
711                    tab_id,
712                    language,
713                    show_markdown_preview,
714                    content,
715                } => {
716                    let editor_input = Input::new(&content)
717                        .bordered(false)
718                        .p_0()
719                        .h_full()
720                        .font_family(self.settings.editor_settings.font_family.clone())
721                        .text_size(px(self.settings.editor_settings.font_size))
722                        .focus_bordered(false);
723                    let capture_right_click =
724                        cx.listener(|this, event: &MouseDownEvent, window, cx| {
725                            this.on_editor_right_click(event, window, cx);
726                        });
727                    if language == SupportedLanguage::Markdown
728                        && show_markdown_preview
729                        && self.settings.editor_settings.markdown_settings.preview_mode
730                            == crate::fulgur::settings::MarkdownPreviewMode::Panel
731                    {
732                        let preview_text = self.markdown_preview_text_for(tab_id, &content, cx);
733                        return v_flex()
734                            .w_full()
735                            .flex_1()
736                            .child(
737                                h_resizable("markdown-preview-container")
738                                    .child(
739                                        resizable_panel().child(
740                                            div()
741                                                .id("markdown-editor")
742                                                .size_full()
743                                                .capture_any_mouse_down(capture_right_click)
744                                                .child(editor_input),
745                                        ),
746                                    )
747                                    .child(
748                                        resizable_panel().child(
749                                            TextView::markdown("markdown-preview", preview_text)
750                                                .flex_none()
751                                                .py_0()
752                                                .px_2()
753                                                .scrollable(true)
754                                                .selectable(true)
755                                                .bg(cx.theme().muted),
756                                        ),
757                                    ),
758                            )
759                            .into_any_element();
760                    }
761                    return v_flex()
762                        .w_full()
763                        .flex_1()
764                        .capture_any_mouse_down(capture_right_click)
765                        .child(editor_input)
766                        .into_any_element();
767                }
768                ActiveTabRenderData::Settings => {
769                    return v_flex()
770                        .id("settings-tab-scrollable")
771                        .w_full()
772                        .flex_1()
773                        .overflow_y_scrollbar()
774                        .child(self.render_settings(window, cx))
775                        .into_any_element();
776                }
777                ActiveTabRenderData::MarkdownPreview {
778                    source_tab_id,
779                    content,
780                } => {
781                    let preview_text = self.markdown_preview_text_for(source_tab_id, &content, cx);
782                    return v_flex()
783                        .w_full()
784                        .flex_1()
785                        .child(
786                            TextView::markdown("markdown-preview-tab", preview_text)
787                                .py_2()
788                                .px_4()
789                                .scrollable(true)
790                                .selectable(true),
791                        )
792                        .into_any_element();
793                }
794            }
795        }
796        v_flex().w_full().flex_1().into_any_element()
797    }

799    /// Remove markdown preview cache entries for tabs that no longer exist.
800    ///
801    /// ### Arguments
802    /// - `cx`: The application context
803    fn prune_markdown_preview_cache(&mut self, cx: &mut Context<Self>) {
804        let source_ids: HashSet<usize> = self
805            .tabs
806            .iter()
807            .filter_map(|tab| match tab {
808                Tab::MarkdownPreview(preview_tab) => Some(preview_tab.source_tab_id),
809                Tab::Editor(editor_tab)
810                    if Self::should_track_markdown_preview_source(
811                        editor_tab.language,
812                        editor_tab.show_markdown_preview,
813                    ) =>
814                {
815                    Some(editor_tab.id)
816                }
817                _ => None,
818            })
819            .collect();
820
821        let before_cache = self.markdown_preview_cache.len();
822        let before_subs = self.markdown_preview_subscriptions.len();
823        self.markdown_preview_cache
824            .retain(|tab_id, _| source_ids.contains(tab_id));
825        self.markdown_preview_subscriptions
826            .retain(|tab_id, _| source_ids.contains(tab_id));
827        self.markdown_preview_to_refresh
828            .retain(|tab_id| source_ids.contains(tab_id));
829        if self.markdown_preview_cache.len() != before_cache
830            || self.markdown_preview_subscriptions.len() != before_subs
831        {
832            cx.notify();
833        }
834    }

836    /// Get cached markdown text for a source tab, refreshing lazily on demand.
837    ///
838    /// ### Arguments
839    /// - `source_tab_id`: Source editor tab id for this preview
840    /// - `content`: Source editor content entity
841    /// - `cx`: The application context
842    ///
843    /// ### Returns
844    /// - `SharedString`: Cached markdown source text for rendering
845    fn markdown_preview_text_for(
846        &mut self,
847        source_tab_id: usize,
848        content: &Entity<InputState>,
849        cx: &mut Context<Self>,
850    ) -> SharedString {
851        if let std::collections::hash_map::Entry::Vacant(entry) =
852            self.markdown_preview_subscriptions.entry(source_tab_id)
853        {
854            let subscription =
855                cx.subscribe(content, move |this: &mut Self, _, ev: &InputEvent, cx| {
856                    if !matches!(ev, InputEvent::Change) {
857                        return;
858                    }
859                    this.markdown_preview_to_refresh.insert(source_tab_id);
860                    cx.notify();
861                });
862            entry.insert(subscription);
863        }
864        let needs_refresh = self.markdown_preview_to_refresh.remove(&source_tab_id)
865            || !self.markdown_preview_cache.contains_key(&source_tab_id);
866        if needs_refresh {
867            self.markdown_preview_cache
868                .insert(source_tab_id, content.read(cx).value());
869        }
870        self.markdown_preview_cache
871            .get(&source_tab_id)
872            .cloned()
873            .unwrap_or_default()
874    }

895    /// Drain pending remote file results and open loaded content in new tabs.
896    ///
897    /// Called every render pass. When SSH background threads deliver results
898    /// (success or error), this method consumes them and either opens new tabs with
899    /// loaded content or shows error notifications.
900    ///
901    /// ### Arguments
902    /// - `window`: The window context
903    /// - `cx`: The application context
904    fn process_pending_remote_files(&mut self, window: &mut Window, cx: &mut Context<Self>) {
905        let outcomes = std::mem::take(&mut *self.pending_remote_open.lock());
906        if outcomes.is_empty() {
907            return;
908        }
909        for outcome in outcomes {
910            let target_tab_id = outcome.target_tab_id;
911            if let Some(tab_id) = target_tab_id {
912                if let Some(request_id) = outcome.target_request_id
913                    && self.latest_remote_open_request_by_tab.get(&tab_id).copied()
914                        != Some(request_id)
915                {
916                    // A newer request for this tab is already in flight; ignore stale completion.
917                    continue;
918                }
919                if let Some(request_id) = outcome.target_request_id
920                    && self.latest_remote_open_request_by_tab.get(&tab_id).copied()
921                        == Some(request_id)
922                {
923                    self.latest_remote_open_request_by_tab.remove(&tab_id);
924                }
925                self.inflight_remote_restore.remove(&tab_id);
926            }
927
928            match outcome.result {
929                Ok(RemoteOpenResult::File(remote_file)) => {
930                    if let Some(tab_id) = target_tab_id {
931                        self.pending_remote_restore.remove(&tab_id);
932                        self.apply_remote_reload_to_existing_tab(tab_id, remote_file, window, cx);
933                    } else {
934                        self.last_failed_remote_open_url = None;
935                        let recent_remote_url =
936                            crate::fulgur::sync::ssh::url::format_remote_url(&remote_file.spec);
937                        log::debug!(
938                            "Remote file loaded: {}:{}",
939                            remote_file.spec.host,
940                            remote_file.spec.path
941                        );
942                        let editor_tab = editor_tab::EditorTab::from_remote_loaded(
943                            self.next_tab_id,
944                            remote_file,
945                            window,
946                            cx,
947                            &self.settings.editor_settings,
948                        );
949                        let idx = self.tabs.len();
950                        self.tabs.push(Tab::Editor(editor_tab));
951                        self.active_tab_index = Some(idx);
952                        self.pending_tab_scroll = Some(idx);
953                        self.next_tab_id += 1;
954                        self.focus_active_tab(window, cx);
955                        if let Err(e) = self.settings.add_file(PathBuf::from(recent_remote_url)) {
956                            log::error!("Failed to add remote file to recent files: {e}");
957                        }
958                        let update_link = self
959                            .shared_state(cx)
960                            .update_info
961                            .lock()
962                            .as_ref()
963                            .map(|info| info.download_url.clone());
964                        let menus =
965                            build_menus(&self.settings.get_recent_files(), update_link.as_deref());
966                        self.update_menus(menus, cx);
967                        if let Err(e) = self.save_state(cx, window) {
968                            log::error!("Failed to save app state after opening remote file: {e}");
969                            self.pending_notification = Some((
970                                NotificationType::Warning,
971                                format!("Remote file opened but failed to save state: {e}").into(),
972                            ));
973                        }
974                        cx.notify();
975                    }
976                }
977                Ok(RemoteOpenResult::Browse(browse)) => {
978                    if let Some(tab_id) = target_tab_id {
979                        self.pending_remote_restore.insert(tab_id);
980                        self.pending_notification = Some((
981                            NotificationType::Error,
982                            "Restored remote tab path is no longer a file".into(),
983                        ));
984                    } else {
985                        self.show_remote_path_browser_dialog(window, cx, &browse);
986                    }
987                }
988                Err(msg) => {
989                    if let Some(tab_id) = target_tab_id {
990                        self.pending_remote_restore.insert(tab_id);
991                    }
992                    self.pending_notification = Some((NotificationType::Error, msg.into()));
993                }
994            }
995        }
996    }

998    /// Apply fresh remote contents to an already-restored tab after lazy reconnect.
999    ///
1000    /// ### Arguments
1001    /// - `tab_id`: Stable editor tab id to update
1002    /// - `remote_file`: Loaded remote payload from SSH worker
1003    /// - `window`: The window context
1004    /// - `cx`: The application context
1005    fn apply_remote_reload_to_existing_tab(
1006        &mut self,
1007        tab_id: usize,
1008        remote_file: RemoteFileResult,
1009        window: &mut Window,
1010        cx: &mut Context<Self>,
1011    ) {
1012        let Some(editor_tab) = self.tabs.iter_mut().find_map(|tab| match tab {
1013            Tab::Editor(editor_tab) if editor_tab.id == tab_id => Some(editor_tab),
1014            _ => None,
1015        }) else {
1016            return;
1017        };
1018
1019        editor_tab.content.update(cx, |input_state, cx| {
1020            input_state.set_value(&remote_file.content, window, cx);
1021        });
1022        editor_tab.location =
1023            crate::fulgur::editor_tab::TabLocation::Remote(remote_file.spec.clone());
1024        editor_tab.encoding = remote_file.encoding;
1025        editor_tab.set_original_content_from_str(&remote_file.content);
1026        editor_tab.modified = false;
1027        editor_tab.update_file_tooltip_cache(remote_file.file_size);
1028        let filename = remote_file
1029            .spec
1030            .path
1031            .rsplit('/')
1032            .next()
1033            .unwrap_or(&remote_file.spec.path)
1034            .to_string();
1035        editor_tab.title = filename.into();
1036        let language = crate::fulgur::languages::supported_languages::language_from_content(
1037            editor_tab.title.as_ref(),
1038            &remote_file.content,
1039        );
1040        editor_tab.force_language(window, cx, language, &self.settings.editor_settings);
1041        cx.notify();
1042    }
