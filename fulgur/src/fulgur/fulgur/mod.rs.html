211                window_id,
212                focus_handle: cx.focus_handle(),
213                title_bar,
214                tabs: vec![],
215                active_tab_index: None,
216                next_tab_id: 0,
217                search_state: SearchState::new(search_input, replace_input, search_subscription),
218                color_picker_bar_state: ColorPickerBarState::new(window, cx),
219                jump_to_line_input,
220                pending_jump: None,
221                jump_to_line_dialog_open: false,
222                settings,
223                settings_changed: false,
224                local_settings_version: 0,
225                rendered_tabs: HashSet::new(),
226                tabs_pending_update: HashSet::new(),
227                editor_modified_subscriptions: HashMap::new(),
228                markdown_preview_cache: HashMap::new(),
229                markdown_preview_subscriptions: HashMap::new(),
230                tab_scroll_handle: ScrollHandle::new(),
231                pending_tab_scroll: None,
232                file_watch_state: FileWatchState::new(),
233                sse_state: SseState::new(),
234                pending_notification: None,
235                save_failed_once: false,
236                pending_share_sheet: false,
237                cached_window_bounds: None,
238                _font_select_subscription: None,
239                editor_context_menu: None,
240                _editor_context_menu_subscription: None,
241                drag_ghost: None,
242                pending_tab_transfer: None,
243                pending_tab_removal: None,
244                pending_transfer_scroll: None,
245                #[cfg(any(target_os = "macos", target_os = "windows"))]
246                local_window_menu_fingerprint: 0,
247                #[cfg(target_os = "macos")]
248                last_dock_menu_revision: 0,
249                #[cfg(target_os = "macos")]
250                last_dock_menu_hash: 0,
251                #[cfg(target_os = "windows")]
252                last_jump_list_revision: 0,
253                #[cfg(target_os = "windows")]
254                last_jump_list_hash: 0,
255            }
256        });
257        let (sse_tx, sse_rx) = std::sync::mpsc::channel();
258        let sse_shutdown_flag = Arc::new(AtomicBool::new(false));
259        entity.update(cx, |this, cx| {
260            this.sse_state.sse_events = Some(sse_rx);
261            this.sse_state.sse_event_tx = Some(sse_tx);
262            this.sse_state.sse_shutdown_flag = Some(sse_shutdown_flag);
263            let shared = cx.global::<shared_state::SharedAppState>();
264            if let Some(error_msg) = shared.sync_error.lock().as_ref() {
265                this.pending_notification =
266                    Some((NotificationType::Error, error_msg.clone().into()));
267            }
268            if window_index == usize::MAX {
269                let initial_tab = Tab::Editor(EditorTab::new(
270                    0,
271                    crate::fulgur::ui::components_utils::UNTITLED,
272                    window,
273                    cx,
274                    &this.settings.editor_settings,
275                ));
276                this.tabs.push(initial_tab);
277                this.active_tab_index = Some(0);
278                this.next_tab_id = 1;
279            } else if window_index < usize::MAX - 1 {
280                // usize::MAX - 1 means new window receiving a tab transfer: skip initial tab
281                this.load_state(window, cx, window_index);
282                this.pending_tab_scroll = this.active_tab_index;
283            }
284            if this.settings.editor_settings.watch_files {
285                this.start_file_watcher();
286            }
287        });
288        sync::synchronization::begin_synchronization(&entity, cx);
289        entity
290    }

411    fn assemble_ui_tree(
412        &self,
413        app_content: impl IntoElement,
414        window: &mut Window,
415        cx: &mut Context<Self>,
416    ) -> impl IntoElement {
417        // Create root layout: TitleBar OUTSIDE of focus-tracked content
418        // This is critical for Windows hit-testing to work!
419        let root_content = v_flex()
420            .size_full()
421            .child(self.title_bar.clone())
422            .child(app_content);
423        let mut root = div()
424            .size_full()
425            .child(root_content)
426            .children(Root::render_sheet_layer(window, cx))
427            .children(Root::render_notification_layer(window, cx))
428            .children(Root::render_dialog_layer(window, cx));
429        if let Some((position, menu)) = self
430            .editor_context_menu
431            .as_ref()
432            .map(|(pos, menu)| (*pos, menu.clone()))
433        {
434            root = root.child(
435                gpui::deferred(
436                    gpui::anchored()
437                        .position(position)
438                        .snap_to_window_with_margin(px(8.))
439                        .anchor(Corner::TopLeft)
440                        .child(
441                            div()
442                                .font_family(cx.theme().font_family.clone())
443                                .cursor_default()
444                                .child(menu),
445                        ),
446                )
447                .with_priority(1),
448            );
449        }
450        root
451    }

467impl Render for Fulgur {
468    /// Render the Fulgur instance
469    ///
470    /// ### Arguments
471    /// - `window`: The window to render the Fulgur instance in
472    /// - `cx`: The application context
473    ///
474    /// ### Returns
475    /// - `impl IntoElement`: The rendered Fulgur instance
476    fn render(&mut self, window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
477        self.process_window_state_updates(window, cx);
478        self.process_update_notifications(window, cx);
479        self.synchronize_settings_from_other_windows(cx);
480        self.process_pending_files_from_macos(window, cx);
481        #[cfg(target_os = "windows")]
482        self.process_pending_ipc_commands(window, cx);
483        self.process_shared_files_from_sync(window, cx);
484        self.process_file_watch_events(window, cx);
485        self.process_sse_events(window, cx);
486        self.process_pending_share_sheet(window, cx);
487        if self.tabs.is_empty() {
488            self.active_tab_index = None;
489        }
490        self.update_search_if_needed(window, cx);
491        self.propagate_settings_to_tabs(window, cx);
492        self.track_newly_rendered_tabs(cx);
493        self.handle_pending_transfer_scroll(window, cx);
494        self.handle_pending_tab_transfer(window, cx);
495        self.handle_pending_tab_removal(window, cx);
496        self.handle_pending_jump_to_line(window, cx);
497        if !self.jump_to_line_dialog_open {
498            window.close_dialog(cx);
499            self.jump_to_line_dialog_open = true;
500        }
501        self.update_modified_status(cx);
502        self.prune_markdown_preview_cache(cx);
503        self.process_pending_tab_scroll(cx);
504        let app_content = self.build_app_content_with_actions(self.active_tab_index, window, cx);
505        self.assemble_ui_tree(app_content, window, cx)
506    }

509impl Fulgur {
510    /// Process a deferred scroll-to-tab request
511    ///
512    /// GPUI's ScrollHandle needs one render cycle to populate child bounds and overflow
513    /// state before `scroll_to_item` can work. On the first frame, layout hasn't happened
514    /// yet so the scroll is silently dropped. This method waits until child bounds are
515    /// available (meaning layout has completed at least once), then issues the scroll.
516    ///
517    /// ### Arguments
518    /// - `cx`: The application context
519    fn process_pending_tab_scroll(&mut self, cx: &mut Context<Self>) {
520        if let Some(index) = self.pending_tab_scroll {
521            if self.tab_scroll_handle.bounds_for_item(0).is_some() {
522                self.tab_scroll_handle.scroll_to_item(index);
523                self.pending_tab_scroll = None;
524            } else {
525                cx.notify();
526            }
527        }
528    }
529
530    /// Handle a right-click in the editor area to show a custom context menu.
531    ///
532    /// Called during the capture phase so propagation can be stopped before
533    /// the editor's built-in context menu fires.
534    ///
535    /// ### Arguments
536    /// - `event`: The mouse-down event
537    /// - `window`: The window context
538    /// - `cx`: The application context
539    fn on_editor_right_click(
540        &mut self,
541        event: &MouseDownEvent,
542        window: &mut Window,
543        cx: &mut Context<Self>,
544    ) {
545        if event.button != MouseButton::Right {
546            return;
547        }
548        cx.stop_propagation();
549
550        let Some(active_index) = self.active_tab_index else {
551            return;
552        };
553        let Some(Tab::Editor(editor_tab)) = self.tabs.get(active_index) else {
554            return;
555        };
556        let editor_focus = editor_tab.content.focus_handle(cx);
557        let has_file = editor_tab.file_path.is_some();
558        let position = event.position;
559
560        let menu = PopupMenu::build(window, cx, {
561            let editor_focus = editor_focus.clone();
562            move |mut menu, _window, _cx| {
563                menu = menu.action_context(editor_focus);
564                if has_file {
565                    menu = menu
566                        .menu(
567                            crate::fulgur::ui::components_utils::reveal_in_file_manager_label(),
568                            Box::new(tab_bar::ShowInFileManager(active_index)),
569                        )
570                        .separator();
571                }
572                menu.menu("Cut", Box::new(gpui_component::input::Cut))
573                    .menu("Copy", Box::new(gpui_component::input::Copy))
574                    .menu("Paste", Box::new(gpui_component::input::Paste))
575                    .separator()
576                    .menu("Select All", Box::new(gpui_component::input::SelectAll))
577            }
578        });
579
580        let subscription = cx.subscribe_in(
581            &menu,
582            window,
583            |this: &mut Self, _, _: &DismissEvent, _, cx| {
584                this.editor_context_menu = None;
585                this._editor_context_menu_subscription = None;
586                cx.notify();
587            },
588        );
589
590        self.editor_context_menu = Some((position, menu));
591        self._editor_context_menu_subscription = Some(subscription);
592        cx.notify();
593    }

595    /// Render the content area (editor or settings)
596    ///
597    /// ### Arguments
598    /// - `active_tab_index`: The index of the active tab (if any)
599    /// - `window`: The window context
600    /// - `cx`: The application context
601    ///
602    /// ### Returns
603    /// - `AnyElement`: The rendered content area element (wrapped in AnyElement)
604    fn render_content_area(
605        &mut self,
606        active_tab_index: Option<usize>,
607        window: &mut Window,
608        cx: &mut Context<Self>,
609    ) -> AnyElement {
610        enum ActiveTabRenderData {
611            Editor {
612                tab_id: usize,
613                language: SupportedLanguage,
614                show_markdown_preview: bool,
615                content: Entity<InputState>,
616            },
617            Settings,
618            MarkdownPreview {
619                source_tab_id: usize,
620                content: Entity<InputState>,
621            },
622        }
623
624        let active_tab = active_tab_index.and_then(|active_index| {
625            self.tabs.get(active_index).map(|tab| match tab {
626                Tab::Editor(editor_tab) => ActiveTabRenderData::Editor {
627                    tab_id: editor_tab.id,
628                    language: editor_tab.language,
629                    show_markdown_preview: editor_tab.show_markdown_preview,
630                    content: editor_tab.content.clone(),
631                },
632                Tab::Settings(_) => ActiveTabRenderData::Settings,
633                Tab::MarkdownPreview(preview_tab) => ActiveTabRenderData::MarkdownPreview {
634                    source_tab_id: preview_tab.source_tab_id,
635                    content: preview_tab.content.clone(),
636                },
637            })
638        });
639
640        if let Some(tab) = active_tab {
641            match tab {
642                ActiveTabRenderData::Editor {
643                    tab_id,
644                    language,
645                    show_markdown_preview,
646                    content,
647                } => {
648                    let editor_input = Input::new(&content)
649                        .bordered(false)
650                        .p_0()
651                        .h_full()
652                        .font_family(self.settings.editor_settings.font_family.clone())
653                        .text_size(px(self.settings.editor_settings.font_size))
654                        .focus_bordered(false);
655                    let capture_right_click =
656                        cx.listener(|this, event: &MouseDownEvent, window, cx| {
657                            this.on_editor_right_click(event, window, cx);
658                        });
659                    if language == SupportedLanguage::Markdown
660                        && show_markdown_preview
661                        && self.settings.editor_settings.markdown_settings.preview_mode
662                            == crate::fulgur::settings::MarkdownPreviewMode::Panel
663                    {
664                        let preview_text = self.markdown_preview_text_for(tab_id, &content, cx);
665                        return v_flex()
666                            .w_full()
667                            .flex_1()
668                            .child(
669                                h_resizable("markdown-preview-container")
670                                    .child(
671                                        resizable_panel().child(
672                                            div()
673                                                .id("markdown-editor")
674                                                .size_full()
675                                                .capture_any_mouse_down(capture_right_click)
676                                                .child(editor_input),
677                                        ),
678                                    )
679                                    .child(
680                                        resizable_panel().child(
681                                            TextView::markdown("markdown-preview", preview_text)
682                                                .flex_none()
683                                                .py_0()
684                                                .px_2()
685                                                .scrollable(true)
686                                                .selectable(true)
687                                                .bg(cx.theme().muted),
688                                        ),
689                                    ),
690                            )
691                            .into_any_element();
692                    }
693                    return v_flex()
694                        .w_full()
695                        .flex_1()
696                        .capture_any_mouse_down(capture_right_click)
697                        .child(editor_input)
698                        .into_any_element();
699                }
700                ActiveTabRenderData::Settings => {
701                    return v_flex()
702                        .id("settings-tab-scrollable")
703                        .w_full()
704                        .flex_1()
705                        .overflow_y_scrollbar()
706                        .child(self.render_settings(window, cx))
707                        .into_any_element();
708                }
709                ActiveTabRenderData::MarkdownPreview {
710                    source_tab_id,
711                    content,
712                } => {
713                    let preview_text = self.markdown_preview_text_for(source_tab_id, &content, cx);
714                    return v_flex()
715                        .w_full()
716                        .flex_1()
717                        .child(
718                            TextView::markdown("markdown-preview-tab", preview_text)
719                                .py_2()
720                                .px_4()
721                                .scrollable(true)
722                                .selectable(true),
723                        )
724                        .into_any_element();
725                }
726            }
727        }
728        v_flex().w_full().flex_1().into_any_element()
729    }

731    /// Remove markdown preview cache entries for tabs that no longer exist.
732    ///
733    /// ### Arguments
734    /// - `cx`: The application context
735    fn prune_markdown_preview_cache(&mut self, cx: &mut Context<Self>) {
736        let source_ids: HashSet<usize> = self
737            .tabs
738            .iter()
739            .filter_map(|tab| match tab {
740                Tab::MarkdownPreview(preview_tab) => Some(preview_tab.source_tab_id),
741                Tab::Editor(editor_tab)
742                    if Self::should_track_markdown_preview_source(
743                        editor_tab.language,
744                        editor_tab.show_markdown_preview,
745                    ) =>
746                {
747                    Some(editor_tab.id)
748                }
749                _ => None,
750            })
751            .collect();
752
753        let before_cache = self.markdown_preview_cache.len();
754        let before_subs = self.markdown_preview_subscriptions.len();
755        self.markdown_preview_cache
756            .retain(|tab_id, _| source_ids.contains(tab_id));
757        self.markdown_preview_subscriptions
758            .retain(|tab_id, _| source_ids.contains(tab_id));
759        if self.markdown_preview_cache.len() != before_cache
760            || self.markdown_preview_subscriptions.len() != before_subs
761        {
762            cx.notify();
763        }
764    }

766    /// Get cached markdown text for a source tab, updating the cache lazily.
767    ///
768    /// ### Arguments
769    /// - `source_tab_id`: Source editor tab id for this preview
770    /// - `content`: Source editor content entity
771    /// - `cx`: The application context
772    ///
773    /// ### Returns
774    /// - `SharedString`: Cached markdown source text for rendering
775    fn markdown_preview_text_for(
776        &mut self,
777        source_tab_id: usize,
778        content: &Entity<InputState>,
779        cx: &mut Context<Self>,
780    ) -> SharedString {
781        if let std::collections::hash_map::Entry::Vacant(entry) =
782            self.markdown_preview_cache.entry(source_tab_id)
783        {
784            entry.insert(content.read(cx).value().clone());
785        }
786        if let std::collections::hash_map::Entry::Vacant(entry) =
787            self.markdown_preview_subscriptions.entry(source_tab_id)
788        {
789            let subscription =
790                cx.subscribe(content, move |this: &mut Self, _, ev: &InputEvent, cx| {
791                    if !matches!(ev, InputEvent::Change) {
792                        return;
793                    }
794                    if let Some(source_tab) = this.tabs.iter().find(|tab| tab.id() == source_tab_id)
795                        && let Tab::Editor(editor_tab) = source_tab
796                    {
797                        this.markdown_preview_cache
798                            .insert(source_tab_id, editor_tab.content.read(cx).value().clone());
799                        cx.notify();
800                    }
801                });
802            entry.insert(subscription);
803        }
804        self.markdown_preview_cache
805            .get(&source_tab_id)
806            .cloned()
807            .unwrap_or_default()
808    }

810    /// Set the title of the title bar.
811    ///
812    /// Looks up the current window name from the global `WindowManager` and passes it to
813    /// `CustomTitleBar::set_title` so the suffix is automatically included when multiple
814    /// windows are open.
815    ///
816    /// ### Arguments
817    /// - `title`: The title to set (if None, the default title is used)
818    /// - `cx`: The application context
819    fn set_title(&self, title: Option<String>, cx: &mut Context<Self>) {
820        let window_name = cx
821            .global::<window_manager::WindowManager>()
822            .get_window_name(self.window_id)
823            .map(|s| s.to_string());
824        self.title_bar.update(cx, |this, _cx| {
825            this.set_title(title, window_name.as_deref());
826        });
827    }
