28/// Sync-related state for a single profile, shared across all windows.
29pub struct SyncState {
30    /// Sync server connection status for this profile.
31    pub connection_status: Arc<Mutex<SynchronizationStatus>>,
32    /// Timestamp when the connection attempt started (for delayed spinner display).
33    pub connecting_since: Arc<Mutex<Option<std::time::Instant>>>,
34    /// Device name reported by the server for this profile.
35    pub device_name: Arc<Mutex<Option<String>>>,
36    /// Pending shared files arrived from this profile's server.
37    pub pending_shared_files: Arc<Mutex<Vec<SharedFileResponse>>>,
38    /// JWT token state manager with condition variable for efficient token
39    /// refresh coordination, scoped to this profile.
40    pub token_state: Arc<crate::fulgur::sync::access_token::TokenStateManager>,
41    /// Last heartbeat time received from this profile's SSE stream.
42    pub last_heartbeat: Arc<Mutex<Option<std::time::Instant>>>,
43    /// Pending notification from background sync operations (checked in render loop).
44    pub pending_notification: Arc<Mutex<Option<(NotificationType, SharedString)>>>,
45    /// Pending devices list from background fetch (checked in render loop to open share sheet).
46    pub pending_devices: Arc<Mutex<Option<PendingDevicesResult>>>,
47    /// Maximum file size for sharing (bytes), as reported by this profile's server.
48    pub max_file_size_bytes: Arc<AtomicU64>,
49}

51impl SyncState {
52    /// Create a new per-profile sync state with the given connection status.
53    ///
54    /// ### Arguments

56    ///
57    /// ### Returns
58    /// - `Self`: The new sync state.

66            last_heartbeat: Arc::new(Mutex::new(None)),
67            pending_notification: Arc::new(Mutex::new(None)),
68            pending_devices: Arc::new(Mutex::new(None)),
69            max_file_size_bytes: Arc::new(AtomicU64::new(u64::MAX)),
70        }
71    }

74/// State that is shared across all windows. This includes settings, themes, and sync-related state.
75pub struct SharedAppState {
76    /// Settings (shared across all windows)
77    pub settings: Arc<Mutex<Settings>>,
78    /// Settings version counter, incremented whenever settings change. All windows check this to detect when they need to reload settings.
79    pub settings_version: Arc<AtomicU64>,
80    /// Available themes
81    pub themes: Arc<Mutex<Option<Themes>>>,
82    /// Per-profile sync states keyed by profile id.
83    pub sync_states: Arc<RwLock<HashMap<ProfileId, Arc<SyncState>>>>,
84    /// Global flag to ensure sync bootstrap runs only once across all windows.
85    pub sync_initialized: Arc<AtomicBool>,
86    /// Synchronization initialization error (if key generation failed)
87    pub sync_error: Arc<Mutex<Option<String>>>,
88    /// Update info if available
89    pub update_info: Arc<Mutex<Option<UpdateInfo>>>,
90    /// Files from macOS "Open with" events (already `Arc<Mutex>`)
91    pub pending_files_from_macos: Arc<Mutex<Vec<PathBuf>>>,
92    /// Pending IPC commands from Windows jump list ("new-tab", "new-window")
93    #[cfg(target_os = "windows")]
94    pub pending_ipc_commands: Arc<Mutex<Vec<String>>>,
95    /// Shared HTTP agent for connection pooling across all requests
96    pub http_agent: Arc<ureq::Agent>,
97    /// Session-scoped SSH password cache keyed by `(host, port, user)`.
98    ///
99    /// The cache is memory-only and dropped on app exit.
100    pub ssh_session_cache: Arc<Mutex<SshCredentialCache>>,
101    /// Process-wide pool of authenticated SSH sessions used to amortize TCP +
102    /// SSH handshakes across successive remote operations.
103    pub ssh_session_pool: Arc<SshSessionPool>,
104    /// Dedicated background writer for `WindowsState` persistence.
105    pub state_writer: Arc<StateWriter>,
106}

110impl SharedAppState {
111    /// Create a new shared app state by orchestrating the initialization of shared application state
112    /// by loading themes, and determining the initial synchronization status.
113    ///
114    /// ### Arguments
115    /// - `settings`: Already-loaded application settings
116    /// - `pending_files_from_macos`: Arc to the pending files queue from macOS open events
117    ///
118    /// ### Returns
119    /// - `Self`: The new shared app state
120    pub fn new(settings: Settings, pending_files_from_macos: Arc<Mutex<Vec<PathBuf>>>) -> Self {
121        let (settings, sync_error) = Self::validate_settings(settings);
122        let themes = Self::load_themes();
123        let synchronization_status = Self::determine_initial_sync_status(&settings);
124        let sync_states = Self::seed_sync_states(&settings, synchronization_status);
125
126        Self {
127            settings: Arc::new(Mutex::new(settings)),
128            settings_version: Arc::new(AtomicU64::new(0)),
129            themes: Arc::new(Mutex::new(themes)),
130            sync_states: Arc::new(RwLock::new(sync_states)),
131            sync_initialized: Arc::new(AtomicBool::new(false)),
132            sync_error: Arc::new(Mutex::new(sync_error)),
133            update_info: Arc::new(Mutex::new(None)),
134            pending_files_from_macos,
135            #[cfg(target_os = "windows")]
136            pending_ipc_commands: Arc::new(Mutex::new(Vec::new())),
137            http_agent: Arc::new(ureq::Agent::new_with_config(
138                ureq::config::Config::builder()
139                    .timeout_connect(Some(std::time::Duration::from_secs(5)))
140                    .timeout_global(Some(std::time::Duration::from_secs(10)))
141                    .build(),
142            )),
143            ssh_session_cache: Arc::new(Mutex::new(SshCredentialCache::new())),
144            ssh_session_pool: Arc::new(SshSessionPool::new()),
145            state_writer: Arc::new(StateWriter::new()),
146        }
147    }
148
149    /// Validate encryption keys against pre-loaded settings.
150    ///
151    /// If synchronization is activated but keys cannot be validated, disables
152    /// synchronization and returns the error message for user notification.
153    ///
154    /// ### Arguments
155    /// - `settings`: Application settings to validate
156    ///
157    /// ### Returns
158    /// - `(Settings, Option<String>)`: The validated settings and an optional error message
159    ///   if key validation failed
160    fn validate_settings(mut settings: Settings) -> (Settings, Option<String>) {
161        if let Err(e) = migrate_legacy_keychain_entries_if_present(&settings) {
162            //TODO: remove in 0.10.0
163            log::warn!("Legacy keychain migration failed: {e}");
164        }
165        let sync_error = if settings
166            .app_settings
167            .synchronization_settings
168            .is_synchronization_activated
169        {
170            match check_private_public_keys(&mut settings) {
171                Ok(()) => None,
172                Err(e) => {
173                    let error_msg = format!(
174                        "Failed to initialize encryption keys. Synchronization has been disabled. Error: {e}"
175                    );
176                    log::error!("{error_msg}");
177                    settings
178                        .app_settings
179                        .synchronization_settings
180                        .is_synchronization_activated = false;
181                    Some(error_msg)
182                }
183            }
184        } else {
185            None
186        };
187        (settings, sync_error)
188    }

190    /// Load themes from disk
191    ///
192    /// ### Returns
193    /// - `Some(Themes)`: The loaded themes
194    /// - `None`: If themes cannot be loaded
195    fn load_themes() -> Option<Themes> {
196        Themes::load().ok()
197    }

199    /// Determine the initial synchronization status based on settings
200    ///
201    /// ### Arguments
202    /// - `settings`: The application settings
203    ///
204    /// ### Returns
205    /// - `SynchronizationStatus`: Connecting if sync is activated, `NotActivated` otherwise
206    fn determine_initial_sync_status(settings: &Settings) -> SynchronizationStatus {
207        if settings
208            .app_settings
209            .synchronization_settings
210            .is_synchronization_activated
211        {
212            SynchronizationStatus::Connecting
213        } else {
214            SynchronizationStatus::NotActivated
215        }
216    }
217
218    /// Build the initial per-profile `SyncState` map from settings.
219    ///
220    /// ### Arguments
221    /// - `settings`: The application settings.
222    /// - `default_status`: The status assigned to profiles flagged as active
223    ///   when the master switch is on; ignored otherwise.
224    ///
225    /// ### Returns
226    /// - `HashMap<ProfileId, Arc<SyncState>>`: Map keyed by profile id.
227    fn seed_sync_states(
228        settings: &Settings,
229        default_status: SynchronizationStatus,
230    ) -> HashMap<ProfileId, Arc<SyncState>> {
231        let master_on = settings
232            .app_settings
233            .synchronization_settings
234            .is_synchronization_activated;
235        settings
236            .app_settings
237            .synchronization_settings
238            .profiles
239            .iter()
240            .map(|profile| {
241                let status = if master_on && profile.is_active {
242                    default_status
243                } else {
244                    SynchronizationStatus::NotActivated
245                };
246                (profile.id.clone(), Arc::new(SyncState::new(status)))
247            })
248            .collect()
249    }
250
251    /// Get the `SyncState` for a specific profile, creating it on demand.
252    ///
253    /// ### Arguments
254    /// - `profile_id`: The profile id.
255    ///
256    /// ### Returns
257    /// - `Arc<SyncState>`: The shared sync state for the profile. A fresh
258    ///   `NotActivated` state is inserted when the profile is unknown.
259    pub fn sync_state_for(&self, profile_id: &str) -> Arc<SyncState> {
260        if let Some(existing) = self.sync_states.read().get(profile_id) {
261            return Arc::clone(existing);
262        }
263        let mut map = self.sync_states.write();
264        Arc::clone(
265            map.entry(profile_id.to_string())
266                .or_insert_with(|| Arc::new(SyncState::new(SynchronizationStatus::NotActivated))),
267        )
268    }
269
270    /// Get the `SyncState` for the first configured profile.
271    ///
272    /// ### Returns
273    /// - `Arc<SyncState>`: The first profile's sync state, or the
274    ///   empty-id-keyed fallback state when there are no profiles.
275    pub fn primary_sync_state(&self) -> Arc<SyncState> {
276        let primary_id = self
277            .settings
278            .lock()
279            .app_settings
280            .synchronization_settings
281            .profiles
282            .first()
283            .map(|p| p.id.clone())
284            .unwrap_or_default();
285        self.sync_state_for(&primary_id)
286    }
287
288    /// Remove the `SyncState` entry for a profile and return it.
289    ///
290    /// ### Arguments
291    /// - `profile_id`: The profile id whose state should be dropped.
292    ///
293    /// ### Returns
294    /// - `Some(Arc<SyncState>)`: The removed state if it existed.
295    /// - `None`: When no entry exists for the profile.
296    pub fn remove_sync_state(&self, profile_id: &str) -> Option<Arc<SyncState>> {
297        self.sync_states.write().remove(profile_id)
298    }
