21/// Sync-related state that is shared across all windows
22pub struct SyncState {
23    /// Sync server connection status
24    pub connection_status: Arc<Mutex<SynchronizationStatus>>,
25    /// Timestamp when the connection attempt started (for delayed spinner display)
26    pub connecting_since: Arc<Mutex<Option<std::time::Instant>>>,
27    /// Device name from server
28    pub device_name: Arc<Mutex<Option<String>>>,
29    /// Pending shared files from sync server
30    pub pending_shared_files: Arc<Mutex<Vec<SharedFileResponse>>>,
31    /// JWT token state manager with condition variable for efficient token refresh coordination
32    pub token_state: Arc<crate::fulgur::sync::access_token::TokenStateManager>,
33    /// Last heartbeat time for sync connection
34    pub last_heartbeat: Arc<Mutex<Option<std::time::Instant>>>,
35    /// Flag to track if sync has been initialized (to prevent multiple initializations)
36    pub initialized: Arc<AtomicBool>,
37    /// Pending notification from background sync operations (checked in render loop)
38    pub pending_notification: Arc<Mutex<Option<(NotificationType, SharedString)>>>,
39    /// Pending devices list from background fetch (checked in render loop to open share sheet)
40    pub pending_devices: Arc<Mutex<Option<PendingDevicesResult>>>,
41}

66/// State that is shared across all windows. This includes settings, themes, and sync-related state.
67pub struct SharedAppState {
68    /// Settings (shared across all windows)
69    pub settings: Arc<Mutex<Settings>>,
70    /// Settings version counter, incremented whenever settings change. All windows check this to detect when they need to reload settings.
71    pub settings_version: Arc<AtomicU64>,
72    /// Available themes
73    pub themes: Arc<Mutex<Option<Themes>>>,
74    /// Sync-related state (connection status, device name, pending files, token state, heartbeat, initialization flag)
75    pub sync_state: SyncState,
76    /// Synchronization initialization error (if key generation failed)
77    pub sync_error: Arc<Mutex<Option<String>>>,
78    /// Update info if available
79    pub update_info: Arc<Mutex<Option<UpdateInfo>>>,
80    /// Files from macOS "Open with" events (already `Arc<Mutex>`)
81    pub pending_files_from_macos: Arc<Mutex<Vec<PathBuf>>>,
82    /// Shared HTTP agent for connection pooling across all requests
83    pub http_agent: Arc<ureq::Agent>,
84}

86impl gpui::Global for SharedAppState {}

88impl SharedAppState {
89    /// Create a new shared app state by orchestrating the initialization of shared application state
90    /// by loading themes, and determining the initial synchronization status.
91    ///
92    /// ### Arguments
93    /// - `settings`: Already-loaded application settings
94    /// - `pending_files_from_macos`: Arc to the pending files queue from macOS open events
95    ///
96    /// ### Returns
97    /// - `Self`: The new shared app state
98    pub fn new(settings: Settings, pending_files_from_macos: Arc<Mutex<Vec<PathBuf>>>) -> Self {
99        let (settings, sync_error) = Self::validate_settings(settings);
100        let themes = Self::load_themes();
101        let synchronization_status = Self::determine_initial_sync_status(&settings);
102
103        Self {
104            settings: Arc::new(Mutex::new(settings)),
105            settings_version: Arc::new(AtomicU64::new(0)),
106            themes: Arc::new(Mutex::new(themes)),
107            sync_state: SyncState::new(synchronization_status),
108            sync_error: Arc::new(Mutex::new(sync_error)),
109            update_info: Arc::new(Mutex::new(None)),
110            pending_files_from_macos,
111            http_agent: Arc::new(ureq::Agent::new_with_config(
112                ureq::config::Config::builder()
113                    .timeout_connect(Some(std::time::Duration::from_secs(5)))
114                    .timeout_global(Some(std::time::Duration::from_secs(10)))
115                    .build(),
116            )),
117        }
118    }
119
120    /// Validate encryption keys against pre-loaded settings.
121    ///
122    /// If synchronization is activated but keys cannot be validated, disables
123    /// synchronization and returns the error message for user notification.
124    ///
125    /// ### Arguments
126    /// - `settings`: Application settings to validate
127    ///
128    /// ### Returns
129    /// - `(Settings, Option<String>)`: The validated settings and an optional error message
130    ///   if key validation failed
131    fn validate_settings(mut settings: Settings) -> (Settings, Option<String>) {
132        let sync_error = if settings
133            .app_settings
134            .synchronization_settings
135            .is_synchronization_activated
136        {
137            match check_private_public_keys(&mut settings) {
138                Ok(_) => None,
139                Err(e) => {
140                    let error_msg = format!(
141                        "Failed to initialize encryption keys. Synchronization has been disabled. Error: {}",
142                        e
143                    );
144                    log::error!("{}", error_msg);
145                    settings
146                        .app_settings
147                        .synchronization_settings
148                        .is_synchronization_activated = false;
149                    Some(error_msg)
150                }
151            }
152        } else {
153            None
154        };
155        (settings, sync_error)
156    }
157
158    /// Load themes from disk
159    ///
160    /// ### Returns
161    /// - `Some(Themes)`: The loaded themes
162    /// - `None`: If themes cannot be loaded
163    fn load_themes() -> Option<Themes> {
164        Themes::load().ok()
165    }
166
167    /// Determine the initial synchronization status based on settings
168    ///
169    /// ### Arguments
170    /// - `settings`: The application settings
171    ///
172    /// ### Returns
173    /// - `SynchronizationStatus`: Connecting if sync is activated, NotActivated otherwise
174    fn determine_initial_sync_status(settings: &Settings) -> SynchronizationStatus {
175        if settings
176            .app_settings
177            .synchronization_settings
178            .is_synchronization_activated
179        {
180            SynchronizationStatus::Connecting
181        } else {
182            SynchronizationStatus::NotActivated
183        }
184    }
185}
