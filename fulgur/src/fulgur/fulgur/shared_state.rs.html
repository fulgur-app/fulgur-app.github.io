36/// Sync-related state for a single profile, shared across all windows.
37pub struct SyncState {
38    /// Sync server connection status for this profile.
39    pub connection_status: Arc<Mutex<SynchronizationStatus>>,
40    /// Timestamp when the connection attempt started (for delayed spinner display).
41    pub connecting_since: Arc<Mutex<Option<std::time::Instant>>>,
42    /// Device name reported by the server for this profile.
43    pub device_name: Arc<Mutex<Option<String>>>,
44    /// Pending shared files arrived from this profile's server (still encrypted).
45    pub pending_shared_files: Arc<Mutex<Vec<SharedFileResponse>>>,
46    /// Share IDs fetched via the v2 read/ack flow (`GET /api/v2/shares/:id`) that still need acknowledging
47    /// (`POST /api/v2/shares/:id/successful`) once decryption succeeds.
48    pub pending_ack_share_ids: Arc<Mutex<HashSet<String>>>,
49    /// Shares already decrypted off the UI thread, awaiting tab creation in the render loop.
50    pub pending_decrypted_files: Arc<Mutex<Vec<crate::fulgur::sync::share::DecryptedShare>>>,
51    /// Guard ensuring at most one background decryption worker runs per profile.
52    pub decrypt_in_flight: Arc<AtomicBool>,
53    /// JWT token state manager with condition variable for efficient token
54    /// refresh coordination, scoped to this profile.
55    pub token_state: Arc<crate::fulgur::sync::access_token::TokenStateManager>,
56    /// Last heartbeat time received from this profile's SSE stream.
57    pub last_heartbeat: Arc<Mutex<Option<std::time::Instant>>>,
58    /// Sender for user-facing notifications produced by background sync operations,
59    /// delivered by the app-scope consumer task (see `spawn_notification_consumer`).
60    pub notification_tx: UnboundedSender<AppNotification>,
61    /// Last emitted receive-error signature for shared-file processing (error deduplication).
62    pub last_share_receive_error_signature: Arc<Mutex<Option<String>>>,
63    /// Pending devices list from background fetch (checked in render loop to open share sheet).
64    pub pending_devices: Arc<Mutex<Option<PendingDevicesResult>>>,
65    /// Maximum file size for sharing (bytes), as reported by this profile's server.
66    pub max_file_size_bytes: Arc<AtomicU64>,
67    /// Raw `x-fulgurant-version` value advertised by this profile's server, if any.
68    /// `None` means the server did not advertise a version (Fulgurant before 0.7.0).
69    pub server_version: Arc<Mutex<Option<String>>>,
70    /// Minimum Fulgur version this profile's server requires for all its features.
71    /// `None` means the server did not advertise one (legacy v1 server).
72    pub server_min_fulgur_version: Arc<Mutex<Option<String>>>,
73    /// SSE channel and worker lifecycle state for this profile.
74    pub sse: Arc<Mutex<SseState>>,
75}

122/// State that is shared across all windows. This includes settings, themes, and sync-related state.
123pub struct SharedAppState {
124    /// Settings (shared across all windows). Mutate via `cx.update_global` only.
125    pub settings: Settings,
126    /// Available themes. Mutate via `cx.update_global` only.
127    pub themes: Option<Themes>,
128    /// Per-profile sync states keyed by profile id.
129    pub sync_states: Arc<RwLock<HashMap<ProfileId, Arc<SyncState>>>>,
130    /// Global flag to ensure sync bootstrap runs only once across all windows.
131    pub sync_initialized: Arc<AtomicBool>,
132    /// Synchronization initialization error (if key generation failed)
133    pub sync_error: Arc<Mutex<Option<String>>>,
134    /// Update info if available
135    pub update_info: Arc<Mutex<Option<UpdateInfo>>>,
136    /// Files from macOS "Open with" events (already `Arc<Mutex>`)
137    pub pending_files_from_macos: Arc<Mutex<Vec<PathBuf>>>,
138    /// Pending IPC commands from Windows jump list ("new-tab", "new-window")
139    #[cfg(target_os = "windows")]
140    pub pending_ipc_commands: Arc<Mutex<Vec<String>>>,
141    /// Shared HTTP agent for connection pooling across all short-lived REST
142    /// requests (token, ping, share fetch). Carries a 10s global timeout.
143    pub http_agent: Arc<ureq::Agent>,
144    /// Dedicated HTTP agent for the long-lived SSE stream only.
145    pub sse_http_agent: Arc<ureq::Agent>,
146    /// Session-scoped SSH password cache keyed by `(host, port, user)`.
147    ///
148    /// The cache is memory-only and dropped on app exit.
149    pub ssh_session_cache: Arc<Mutex<SshCredentialCache>>,
150    /// Process-wide pool of authenticated SSH sessions used to amortize TCP +
151    /// SSH handshakes across successive remote operations.
152    pub ssh_session_pool: Arc<SshSessionPool>,
153    /// Dedicated background writer for `WindowsState` persistence.
154    pub state_writer: Arc<StateWriter>,
155    /// In-memory snapshot of `WindowsState` taken once at startup, used to
156    /// restore every window without re-reading `state.json` per window.
157    ///
158    /// Holding the snapshot read-only also closes the consistency window: a
159    /// `save_state` landing between a window's spawn and its restore can no
160    /// longer change what that window sees.
161    pub restore_state: Arc<Mutex<Option<WindowsState>>>,
162    /// Sender for user-facing notifications produced anywhere in the app.
163    pub notification_tx: UnboundedSender<AppNotification>,
164    /// Receiver side of the notification channel, taken exactly once by `spawn_notification_consumer`.
165    notification_rx: Mutex<Option<UnboundedReceiver<AppNotification>>>,
166}

171    /// Create a new shared app state by orchestrating the initialization of shared application state
172    /// by loading themes, and determining the initial synchronization status.
173    ///
174    /// ### Arguments
175    /// - `settings`: Already-loaded application settings
176    /// - `pending_files_from_macos`: Arc to the pending files queue from macOS open events
177    /// - `restore_state`: Startup snapshot of `WindowsState` shared with every window for restoration
178    ///
179    /// ### Returns
180    /// - `Self`: The new shared app state
181    pub fn new(
182        settings: Settings,
183        pending_files_from_macos: Arc<Mutex<Vec<PathBuf>>>,
184        restore_state: Option<WindowsState>,
185    ) -> Self {
186        let (settings, sync_error) = Self::validate_settings(settings);
187        let themes = Self::load_themes();
188        let synchronization_status = Self::determine_initial_sync_status(&settings);
189        let (notification_tx, notification_rx) = unbounded();
190        let sync_states =
191            Self::seed_sync_states(&settings, synchronization_status, &notification_tx);
192
193        Self {
194            settings,
195            themes,
196            sync_states: Arc::new(RwLock::new(sync_states)),
197            sync_initialized: Arc::new(AtomicBool::new(false)),
198            sync_error: Arc::new(Mutex::new(sync_error)),
199            update_info: Arc::new(Mutex::new(None)),
200            pending_files_from_macos,
201            #[cfg(target_os = "windows")]
202            pending_ipc_commands: Arc::new(Mutex::new(Vec::new())),
203            http_agent: Arc::new(ureq::Agent::new_with_config(
204                ureq::config::Config::builder()
205                    .timeout_connect(Some(std::time::Duration::from_secs(5)))
206                    .timeout_global(Some(std::time::Duration::from_secs(10)))
207                    .build(),
208            )),
209            sse_http_agent: Arc::new(ureq::Agent::new_with_config(
210                ureq::config::Config::builder()
211                    .timeout_connect(Some(std::time::Duration::from_secs(5)))
212                    .timeout_global(Some(std::time::Duration::from_secs(90)))
213                    .build(),
214            )),
215            ssh_session_cache: Arc::new(Mutex::new(SshCredentialCache::new())),
216            ssh_session_pool: Arc::new(SshSessionPool::new()),
217            state_writer: Arc::new(StateWriter::new()),
218            restore_state: Arc::new(Mutex::new(restore_state)),
219            notification_tx,
220            notification_rx: Mutex::new(Some(notification_rx)),
221        }
222    }

234    /// Validate encryption keys against pre-loaded settings.
235    ///
236    /// If synchronization is activated but keys cannot be validated, disables
237    /// synchronization and returns the error message for user notification.
238    ///
239    /// ### Arguments
240    /// - `settings`: Application settings to validate
241    ///
242    /// ### Returns
243    /// - `(Settings, Option<String>)`: The validated settings and an optional error message
244    ///   if key validation failed
245    fn validate_settings(mut settings: Settings) -> (Settings, Option<String>) {
246        if let Err(e) = migrate_legacy_keychain_entries_if_present(&settings) {
247            //TODO: remove in 0.10.0
248            log::warn!("Legacy keychain migration failed: {e}");
249        }
250        let sync_error = if settings
251            .app_settings
252            .synchronization_settings
253            .is_synchronization_activated
254        {
255            match check_private_public_keys(&mut settings) {
256                Ok(()) => None,
257                Err(e) => {
258                    let error_msg = format!(
259                        "Failed to initialize encryption keys. Synchronization has been disabled. Error: {e}"
260                    );
261                    log::error!("{error_msg}");
262                    settings
263                        .app_settings
264                        .synchronization_settings
265                        .is_synchronization_activated = false;
266                    Some(error_msg)
267                }
268            }
269        } else {
270            None
271        };
272        (settings, sync_error)
273    }

284    /// Determine the initial synchronization status based on settings
285    ///
286    /// ### Arguments
287    /// - `settings`: The application settings
288    ///
289    /// ### Returns
290    /// - `SynchronizationStatus`: Connecting if sync is activated, `NotActivated` otherwise
291    fn determine_initial_sync_status(settings: &Settings) -> SynchronizationStatus {
292        if settings
293            .app_settings
294            .synchronization_settings
295            .is_synchronization_activated
296        {
297            SynchronizationStatus::Connecting
298        } else {
299            SynchronizationStatus::NotActivated
300        }

303    /// Build the initial per-profile `SyncState` map from settings.
304    ///
305    /// ### Arguments
306    /// - `settings`: The application settings.
307    /// - `default_status`: The status assigned to profiles flagged as active
308    ///   when the master switch is on; ignored otherwise.
309    /// - `notification_tx`: The app-wide notification sender cloned into each state.
310    ///
311    /// ### Returns
312    /// - `HashMap<ProfileId, Arc<SyncState>>`: Map keyed by profile id.
313    fn seed_sync_states(
314        settings: &Settings,
315        default_status: SynchronizationStatus,
316        notification_tx: &UnboundedSender<AppNotification>,
317    ) -> HashMap<ProfileId, Arc<SyncState>> {
318        let master_on = settings
319            .app_settings
320            .synchronization_settings
321            .is_synchronization_activated;
322        settings
323            .app_settings
324            .synchronization_settings
325            .profiles
326            .iter()
327            .map(|profile| {
328                let status = if master_on && profile.is_active {
329                    default_status
330                } else {
331                    SynchronizationStatus::NotActivated
332                };
333                (
334                    profile.id.clone(),
335                    Arc::new(SyncState::new(status, notification_tx.clone())),
336                )
337            })
338            .collect()
339    }

341    /// Get the `SyncState` for a specific profile, creating it on demand.
342    ///
343    /// ### Arguments
344    /// - `profile_id`: The profile id.
345    ///
346    /// ### Returns
347    /// - `Arc<SyncState>`: The shared sync state for the profile. A fresh
348    ///   `NotActivated` state is inserted when the profile is unknown.
349    #[must_use]
350    pub fn sync_state_for(&self, profile_id: &str) -> Arc<SyncState> {
351        if let Some(existing) = self.sync_states.read().get(profile_id) {
352            return Arc::clone(existing);
353        }
354        let mut map = self.sync_states.write();
355        Arc::clone(map.entry(profile_id.to_string()).or_insert_with(|| {
356            Arc::new(SyncState::new(
357                SynchronizationStatus::NotActivated,
358                self.notification_tx.clone(),
359            ))
360        }))
361    }

363    /// Get the `SyncState` for the first configured profile.
364    ///
365    /// ### Returns
366    /// - `Arc<SyncState>`: The first profile's sync state, or the
367    ///   empty-id-keyed fallback state when there are no profiles.
368    #[must_use]
369    pub fn primary_sync_state(&self) -> Arc<SyncState> {
370        let primary_id = self
371            .settings
372            .app_settings
373            .synchronization_settings
374            .profiles
375            .first()
376            .map(|p| p.id.clone())
377            .unwrap_or_default();
378        self.sync_state_for(&primary_id)
379    }

381    /// Remove the `SyncState` entry for a profile and return it.
382    ///
383    /// ### Arguments
384    /// - `profile_id`: The profile id whose state should be dropped.
385    ///
386    /// ### Returns
387    /// - `Some(Arc<SyncState>)`: The removed state if it existed.
388    /// - `None`: When no entry exists for the profile.

390    pub fn remove_sync_state(&self, profile_id: &str) -> Option<Arc<SyncState>> {
391        self.sync_states.write().remove(profile_id)
392    }

395/// Maximum delivery attempts before a notification with no window to show it in is dropped.
396const NOTIFICATION_DELIVERY_MAX_ATTEMPTS: usize = 60;
397
398/// Delay between notification delivery attempts while no window is available.
399const NOTIFICATION_DELIVERY_RETRY_DELAY: Duration = Duration::from_millis(500);
400
401/// Spawn the app-scope task that delivers queued notifications to the focused window.
402///
403/// ### Arguments
404/// - `cx`: The application context.
405pub fn spawn_notification_consumer(cx: &mut App) {
406    let Some(mut rx) = cx.global::<SharedAppState>().notification_rx.lock().take() else {
407        return;
408    };
409    cx.spawn(async move |cx| {
410        while let Some(notification) = rx.next().await {
411            deliver_notification(notification, cx).await;
412        }
413    })
414    .detach();
415}
416
417/// Deliver one notification to the focused window, retrying while no window exists.
418///
419/// ### Arguments
420/// - `notification`: The notification to deliver.
421/// - `cx`: The async application context.
422async fn deliver_notification(notification: AppNotification, cx: &mut AsyncApp) {
423    for attempt in 0..NOTIFICATION_DELIVERY_MAX_ATTEMPTS {
424        if attempt > 0 {
425            cx.background_executor()
426                .timer(NOTIFICATION_DELIVERY_RETRY_DELAY)
427                .await;
428        }
429        if cx.update(|cx| push_notification_to_focused_window(notification.clone(), cx)) {
430            return;
431        }
432    }
433    log::warn!(
434        "No window available to display notification, dropping: {}",
435        notification.1
436    );
437}

439/// Push a notification to the focused window, falling back to any open window.
440///
441/// ### Arguments
442/// - `notification`: The notification to display.
443/// - `cx`: The application context.
444///
445/// ### Returns
446/// - `true`: The notification was pushed to a window.
447/// - `false`: No window is currently available.
448fn push_notification_to_focused_window(notification: AppNotification, cx: &mut App) -> bool {
449    let focused_window_id = cx
450        .try_global::<crate::fulgur::window_manager::WindowManager>()
451        .and_then(super::window_manager::WindowManager::get_last_focused);
452    let windows = cx.windows();
453    let handle = focused_window_id
454        .and_then(|id| windows.iter().find(|w| w.window_id() == id).copied())
455        .or_else(|| windows.first().copied());
456    let Some(handle) = handle else {
457        return false;
458    };
459    handle
460        .update(cx, |_, window, cx| {
461            window.push_notification(notification, cx);
462        })
463        .is_ok()
