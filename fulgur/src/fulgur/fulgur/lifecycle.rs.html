36    /// Create a new Fulgur instance
37    ///
38    /// ### Arguments
39    /// - `window`: The window to create the Fulgur instance in
40    /// - `cx`: The application context
41    /// - `window_id`: The window ID for this instance, obtained from `window.window_handle().window_id()`
42    /// - `window_index`: Index of this window in saved state (0 = first window, etc.). Use `usize::MAX` for new empty windows.
43    ///
44    /// ### Returns
45    /// - `Entity<Self>`: The new Fulgur instance
46    pub fn new(
47        window: &mut Window,
48        cx: &mut App,
49        window_id: WindowId,
50        window_index: usize,
51    ) -> Entity<Self> {
52        let title_bar = CustomTitleBar::new(window, cx);
53        let shared = cx.global::<shared_state::SharedAppState>();
54        let settings = shared.settings.clone();
55        let jump_to_line_input =
56            cx.new(|cx| InputState::new(window, cx).placeholder("Jump to line or line:character"));
57        let entity = cx.new(|cx| {
58            let weak_fulgur = cx.weak_entity();
59            let search_bar = cx.new(|cx| SearchBar::new(weak_fulgur.clone(), window, cx));
60            let search_bar_subscription = cx.subscribe_in(
61                &search_bar,
62                window,
63                |this: &mut Self, _, event: &SearchBarEvent, window, cx| {
64                    this.on_search_bar_event(*event, window, cx);
65                },
66            );
67
68            let status_bar = cx.new(|_| StatusBar::new(weak_fulgur));
69            let status_bar_subscription = cx.subscribe_in(
70                &status_bar,
71                window,
72                |this: &mut Self, _, event: &StatusBarEvent, window, cx| {
73                    this.on_status_bar_event(*event, window, cx);
74                },
75            );
76
77            let shared_state_observation =
78                cx.observe_global::<shared_state::SharedAppState>(|this: &mut Self, cx| {
79                    let shared = cx.global::<shared_state::SharedAppState>();
80                    if this.settings != shared.settings {
81                        this.settings = shared.settings.clone();
82                        this.settings_changed = true;
83                    }
84                    cx.notify();
85                });
86            Self {
87                window_id,
88                focus_handle: cx.focus_handle(),
89                title_bar,
90                tabs: vec![],
91                active_tab_id: None,
92                next_tab_id: TabId(0),
93                search_bar,
94                _search_bar_subscription: search_bar_subscription,
95                color_picker_bar_state: ColorPickerBarState::new(window, cx),
96                jump_to_line_input,
97                pending_jump: None,
98                settings,
99                settings_changed: false,
100                _shared_state_observation: shared_state_observation,
101                rendered_tabs: HashSet::new(),
102                tabs_pending_update: HashSet::new(),
103                editor_modified_subscriptions: HashMap::new(),
104                markdown_preview_cache: HashMap::new(),
105                markdown_preview_to_refresh: HashSet::new(),
106                markdown_preview_subscriptions: HashMap::new(),
107                log_tail_state: HashMap::new(),
108                log_tail_cancel: HashMap::new(),
109                tab_scroll_handle: ScrollHandle::new(),
110                pending_tab_scroll: None,
111                file_watch_state: FileWatchState::new(),
112                save_failed_once: false,
113                share_sheet_state: None,
114                cached_window_bounds: None,
115                font_select_subscription: None,
116                editor_context_menu: None,
117                editor_context_menu_subscription: None,
118                drag_ghost: None,
119                status_bar,
120                _status_bar_subscription: status_bar_subscription,
121                cached_tab_filename_counts: HashMap::new(),
122                tab_filename_fp: u64::MAX, // sentinel: differs from fingerprint of any real tab list
123                pending_tab_transfer: None,
124                pending_tab_removal: None,
125                pending_transfer_scroll: None,
126                pending_remote_open: Arc::new(parking_lot::Mutex::new(Vec::new())),
127                next_remote_request_id: 1,
128                latest_remote_open_request_by_tab: HashMap::new(),
129                latest_remote_save_request_by_tab: HashMap::new(),
130                last_failed_remote_open_url: None,
131                pending_remote_restore: HashSet::new(),
132                inflight_remote_restore: HashSet::new(),
133                pending_initial_active_tab: None,
134                has_rendered_once: false,
135                #[cfg(any(target_os = "macos", target_os = "windows"))]
136                local_window_menu_fingerprint: 0,
137                #[cfg(target_os = "macos")]
138                last_dock_menu_revision: 0,
139                #[cfg(target_os = "macos")]
140                last_dock_menu_hash: 0,
141                #[cfg(target_os = "windows")]
142                last_jump_list_revision: 0,
143                #[cfg(target_os = "windows")]
144                last_jump_list_hash: 0,
145            }
146        });
147        entity.update(cx, |this, cx| {
148            let shared = cx.global::<shared_state::SharedAppState>();
149            if let Some(error_msg) = shared.sync_error.lock().as_ref() {
150                shared.notify((NotificationType::Error, error_msg.clone().into()));
151            }
152            if window_index == usize::MAX {
153                let initial_tab = Tab::Editor(editor_tab::EditorTab::new(
154                    TabId(0),
155                    crate::fulgur::ui::components_utils::UNTITLED,
156                    window,
157                    cx,
158                    &this.settings.editor_settings,
159                ));
160                this.active_tab_id = Some(initial_tab.id());
161                this.tabs.push(initial_tab);
162                this.next_tab_id = TabId(1);
163            } else if window_index < usize::MAX - 1 {
164                // usize::MAX - 1 means new window receiving a tab transfer: skip initial tab
165                this.load_state(window, cx, window_index);
166                this.pending_tab_scroll = this.active_tab_id;
167                this.pending_initial_active_tab = this.active_tab_id;
168            }
169            if this.settings.editor_settings.watch_files {
170                this.start_file_watcher(cx);
171            }
172        });
173        sync::synchronization::begin_synchronization(&entity, cx);
174        entity
175    }

177    /// Initialize the Fulgur instance
178    ///
179    /// ### Arguments
180    /// - `cx`: The application context
181    /// - `settings`: The application settings (already loaded and resolved, including first-run overrides)
182    pub fn init(cx: &mut App, settings: &mut Settings) {
183        let recent_files = settings.get_recent_files();
184        languages::supported_languages::register_external_languages();
185        themes::init(settings, cx, move |cx| {
186            cx.bind_keys(build_default_key_bindings());
187            let menus = build_menus(&recent_files, None);
188            cx.set_menus(menus);
189            #[cfg(not(target_os = "macos"))]
190            if let Some(owned_menus) = cx.get_menus() {
191                gpui_component::GlobalState::global_mut(cx).set_app_menus(owned_menus);
192            }
193        });
194    }

196    /// Set the title of the title bar.
197    ///
198    /// Looks up the current window name from the global `WindowManager` and passes it to
199    /// `CustomTitleBar::set_title` so the suffix is automatically included when multiple
200    /// windows are open.
201    ///
202    /// ### Arguments
203    /// - `title`: The title to set (if None, the default title is used)
204    /// - `cx`: The application context
205    pub(super) fn set_title(&self, title: Option<String>, cx: &mut Context<Self>) {
206        let window_name = cx
207            .global::<window_manager::WindowManager>()
208            .get_window_name(self.window_id)
209            .map(std::string::ToString::to_string);
210        self.title_bar.update(cx, |this, _cx| {
211            this.set_title(title, window_name.as_deref());
212        });
213    }
