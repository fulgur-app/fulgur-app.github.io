36    /// Create a new Fulgur instance
37    ///
38    /// ### Arguments
39    /// - `window`: The window to create the Fulgur instance in
40    /// - `cx`: The application context
41    /// - `window_id`: The window ID for this instance, obtained from `window.window_handle().window_id()`
42    /// - `window_index`: Index of this window in saved state (0 = first window, etc.). Use `usize::MAX` for new empty windows.
43    ///
44    /// ### Returns
45    /// - `Entity<Self>`: The new Fulgur instance
46    pub fn new(
47        window: &mut Window,
48        cx: &mut App,
49        window_id: WindowId,
50        window_index: usize,
51    ) -> Entity<Self> {
52        let title_bar = CustomTitleBar::new(window, cx);
53        let shared = cx.global::<shared_state::SharedAppState>();
54        let settings = shared.settings.clone();
55        let search_input = cx.new(|cx| InputState::new(window, cx).placeholder("Search"));
56        let replace_input = cx.new(|cx| InputState::new(window, cx).placeholder("Replace"));
57        let jump_to_line_input =
58            cx.new(|cx| InputState::new(window, cx).placeholder("Jump to line or line:character"));
59        let entity = cx.new(|cx| {
60            let search_subscription =
61                cx.subscribe(&search_input, |this: &mut Self, _, ev: &InputEvent, cx| {
62                    if let InputEvent::Change = ev
63                        && this.search_state.show_search
64                    {
65                        cx.notify();
66                    }
67                });
68
69            let shared_state_observation =
70                cx.observe_global::<shared_state::SharedAppState>(|this: &mut Self, cx| {
71                    let shared = cx.global::<shared_state::SharedAppState>();
72                    if this.settings != shared.settings {
73                        this.settings = shared.settings.clone();
74                        this.settings_changed = true;
75                    }
76                    cx.notify();
77                });
78            Self {
79                window_id,
80                focus_handle: cx.focus_handle(),
81                title_bar,
82                tabs: vec![],
83                active_tab_index: None,
84                next_tab_id: 0,
85                search_state: SearchState::new(search_input, replace_input, search_subscription),
86                color_picker_bar_state: ColorPickerBarState::new(window, cx),
87                jump_to_line_input,
88                pending_jump: None,
89                settings,
90                settings_changed: false,
91                _shared_state_observation: shared_state_observation,
92                rendered_tabs: HashSet::new(),
93                tabs_pending_update: HashSet::new(),
94                editor_modified_subscriptions: HashMap::new(),
95                markdown_preview_cache: HashMap::new(),
96                markdown_preview_to_refresh: HashSet::new(),
97                markdown_preview_subscriptions: HashMap::new(),
98                log_tail_state: HashMap::new(),
99                log_tail_cancel: HashMap::new(),
100                tab_scroll_handle: ScrollHandle::new(),
101                pending_tab_scroll: None,
102                file_watch_state: FileWatchState::new(),
103                pending_notification: None,
104                save_failed_once: false,
105                share_sheet_state: None,
106                cached_window_bounds: None,
107                font_select_subscription: None,
108                editor_context_menu: None,
109                editor_context_menu_subscription: None,
110                drag_ghost: None,
111                status_bar_cache: StatusBarCache::default(),
112                cached_tab_filename_counts: HashMap::new(),
113                tab_filename_fp: u64::MAX, // sentinel: differs from fingerprint of any real tab list
114                pending_tab_transfer: None,
115                pending_tab_removal: None,
116                pending_transfer_scroll: None,
117                pending_remote_open: Arc::new(parking_lot::Mutex::new(Vec::new())),
118                next_remote_request_id: 1,
119                latest_remote_open_request_by_tab: HashMap::new(),
120                latest_remote_save_request_by_tab: HashMap::new(),
121                last_failed_remote_open_url: None,
122                pending_remote_restore: HashSet::new(),
123                inflight_remote_restore: HashSet::new(),
124                pending_initial_active_tab: None,
125                has_rendered_once: false,
126                #[cfg(any(target_os = "macos", target_os = "windows"))]
127                local_window_menu_fingerprint: 0,
128                #[cfg(target_os = "macos")]
129                last_dock_menu_revision: 0,
130                #[cfg(target_os = "macos")]
131                last_dock_menu_hash: 0,
132                #[cfg(target_os = "windows")]
133                last_jump_list_revision: 0,
134                #[cfg(target_os = "windows")]
135                last_jump_list_hash: 0,
136            }
137        });
138        entity.update(cx, |this, cx| {
139            let shared = cx.global::<shared_state::SharedAppState>();
140            if let Some(error_msg) = shared.sync_error.lock().as_ref() {
141                this.pending_notification =
142                    Some((NotificationType::Error, error_msg.clone().into()));
143            }
144            if window_index == usize::MAX {
145                let initial_tab = Tab::Editor(editor_tab::EditorTab::new(
146                    0,
147                    crate::fulgur::ui::components_utils::UNTITLED,
148                    window,
149                    cx,
150                    &this.settings.editor_settings,
151                ));
152                this.tabs.push(initial_tab);
153                this.active_tab_index = Some(0);
154                this.next_tab_id = 1;
155            } else if window_index < usize::MAX - 1 {
156                // usize::MAX - 1 means new window receiving a tab transfer: skip initial tab
157                this.load_state(window, cx, window_index);
158                this.pending_tab_scroll = this.active_tab_index;
159                this.pending_initial_active_tab = this.active_tab_index;
160            }
161            if this.settings.editor_settings.watch_files {
162                this.start_file_watcher();
163            }
164        });
165        sync::synchronization::begin_synchronization(&entity, cx);
166        entity
167    }

169    /// Initialize the Fulgur instance
170    ///
171    /// ### Arguments
172    /// - `cx`: The application context
173    /// - `settings`: The application settings (already loaded and resolved, including first-run overrides)
174    pub fn init(cx: &mut App, settings: &mut Settings) {
175        let recent_files = settings.get_recent_files();
176        languages::supported_languages::register_external_languages();
177        themes::init(settings, cx, move |cx| {
178            cx.bind_keys(build_default_key_bindings());
179            let menus = build_menus(&recent_files, None);
180            cx.set_menus(menus);
181            #[cfg(not(target_os = "macos"))]
182            if let Some(owned_menus) = cx.get_menus() {
183                gpui_component::GlobalState::global_mut(cx).set_app_menus(owned_menus);
184            }
185        });
186    }

188    /// Set the title of the title bar.
189    ///
190    /// Looks up the current window name from the global `WindowManager` and passes it to
191    /// `CustomTitleBar::set_title` so the suffix is automatically included when multiple
192    /// windows are open.
193    ///
194    /// ### Arguments
195    /// - `title`: The title to set (if None, the default title is used)
196    /// - `cx`: The application context
197    pub(super) fn set_title(&self, title: Option<String>, cx: &mut Context<Self>) {
198        let window_name = cx
199            .global::<window_manager::WindowManager>()
200            .get_window_name(self.window_id)
201            .map(std::string::ToString::to_string);
202        self.title_bar.update(cx, |this, _cx| {
203            this.set_title(title, window_name.as_deref());
204        });
205    }
