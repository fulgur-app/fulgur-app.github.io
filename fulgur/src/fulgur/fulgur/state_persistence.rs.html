7/// Persisted SSH/SFTP tab location metadata.
8///
9/// This representation intentionally excludes any credential material.
10#[derive(Serialize, Deserialize, Debug, Clone, PartialEq, Eq)]
11pub struct SerializedRemoteSpec {
12    /// Remote hostname
13    pub host: String,
14    /// SSH port
15    pub port: u16,
16    /// Remote username. Empty means "prompt user".
17    pub user: String,
18    /// Remote file path
19    pub path: String,
20}
21
22impl SerializedRemoteSpec {
23    /// Build a persisted remote spec from a runtime `RemoteSpec`.
24    ///
25    /// ### Arguments
26    /// - `spec`: Runtime remote spec to persist.
27    ///
28    /// ### Returns
29    /// - `SerializedRemoteSpec`: Persistable remote spec with no password field.
30    pub fn from_remote_spec(spec: &RemoteSpec) -> Self {
31        Self {
32            host: spec.host.clone(),
33            port: spec.port,
34            user: spec.user.clone().unwrap_or_default(),
35            path: spec.path.clone(),
36        }
37    }
38
39    /// Convert persisted remote metadata back into a runtime `RemoteSpec`.
40    ///
41    /// ### Returns
42    /// - `RemoteSpec`: Runtime remote spec with `password_in_url` cleared.
43    pub fn to_remote_spec(&self) -> RemoteSpec {
44        RemoteSpec {
45            host: self.host.clone(),
46            port: self.port,
47            user: (!self.user.trim().is_empty()).then_some(self.user.clone()),
48            path: self.path.clone(),
49            password_in_url: None,
50        }
51    }
52}

54/// Persisted state of a single editor tab
55///
56/// Tab IDs are not persisted as they are assigned at runtime based on position.
57#[derive(Serialize, Deserialize, Debug, Clone)]
58pub struct TabState {
59    /// Display title shown in the tab bar (usually the filename)
60    pub title: String,
61    /// Path to the file on disk, if the tab has an associated file. `None` for unsaved/new tabs.
62    pub file_path: Option<PathBuf>,
63    /// The text content of the tab, stored for unsaved tabs or when the file may have been modified since last save
64    pub content: Option<String>,
65    /// ISO 8601 timestamp of when the content was last saved to disk. Used to detect if the file has been modified externally.
66    pub last_saved: Option<String>,
67    /// Serialized remote location metadata for SSH/SFTP tabs.
68    #[serde(default)]
69    pub remote: Option<SerializedRemoteSpec>,
70}

72/// Serialized window bounds that can be saved to JSON
73#[derive(Serialize, Deserialize, Debug, Clone)]
74pub struct SerializedWindowBounds {
75    /// Window state: "Windowed", "Maximized", or "Fullscreen"
76    pub state: String,
77    /// X position of window origin in pixels
78    pub x: f32,
79    /// Y position of window origin in pixels
80    pub y: f32,
81    /// Window width in pixels
82    pub width: f32,
83    /// Window height in pixels
84    pub height: f32,
85    /// Display ID (monitor) where the window was located
86    #[serde(default)]
87    pub display_id: Option<u32>,
88}

90impl Default for SerializedWindowBounds {
91    /// Default values for serialized window bounds
92    ///
93    /// ### Returns
94    /// - `SerializedWindowBounds`: The default serialized window bounds
95    fn default() -> Self {
96        Self {
97            state: "Windowed".to_string(),
98            x: 100.0,
99            y: 100.0,
100            width: 1200.0,
101            height: 800.0,
102            display_id: None,
103        }
104    }

107impl SerializedWindowBounds {
108    /// Convert GPUI WindowBounds to SerializedWindowBounds
109    ///
110    /// ### Arguments
111    /// - `bounds`: The GPUI WindowBounds to convert
112    /// - `display_id`: Optional display ID (monitor) for the window
113    ///
114    /// ### Returns
115    /// - `SerializedWindowBounds`: The serialized window bounds
116    pub fn from_gpui_bounds(bounds: gpui::WindowBounds, display_id: Option<u32>) -> Self {
117        use gpui::WindowBounds;
118        match bounds {
119            WindowBounds::Windowed(rect) => Self {
120                state: "Windowed".to_string(),
121                x: rect.origin.x.into(),
122                y: rect.origin.y.into(),
123                width: rect.size.width.into(),
124                height: rect.size.height.into(),
125                display_id,
126            },
127            WindowBounds::Maximized(rect) => Self {
128                state: "Maximized".to_string(),
129                x: rect.origin.x.into(),
130                y: rect.origin.y.into(),
131                width: rect.size.width.into(),
132                height: rect.size.height.into(),
133                display_id,
134            },
135            WindowBounds::Fullscreen(rect) => Self {
136                state: "Fullscreen".to_string(),
137                x: rect.origin.x.into(),
138                y: rect.origin.y.into(),
139                width: rect.size.width.into(),
140                height: rect.size.height.into(),
141                display_id,
142            },
143        }
144    }
145
146    /// Convert SerializedWindowBounds to GPUI WindowBounds
147    ///
148    /// ### Returns
149    /// - `gpui::WindowBounds`: The GPUI window bounds
150    pub fn to_gpui_bounds(&self) -> gpui::WindowBounds {
151        use gpui::{Bounds, WindowBounds, point, px, size};
152        let bounds = Bounds {
153            origin: point(px(self.x), px(self.y)),
154            size: size(px(self.width), px(self.height)),
155        };
156        match self.state.as_str() {
157            "Maximized" => WindowBounds::Maximized(bounds),
158            "Fullscreen" => WindowBounds::Fullscreen(bounds),
159            _ => WindowBounds::Windowed(bounds), // Default to Windowed for unknown states
160        }
161    }
162}

178/// Top-level container for all persisted application state
179///
180/// This struct is serialized to `state.json` and contains the complete
181/// state of all windows. On startup, each window in this list is restored
182/// with its tabs, positions, and content.
183///
184/// File location:
185/// - Windows: `%APPDATA%\Fulgur\state.json`
186/// - macOS/Linux: `~/.fulgur/state.json`
187#[derive(Serialize, Deserialize, Debug)]
188pub struct WindowsState {
189    /// All application windows to be restored
190    pub windows: Vec<WindowState>,
191}
192
193impl WindowsState {
194    /// Get the path to the state file
195    ///
196    /// ### Returns
197    /// - `Ok(PathBuf)`: The path to the state file
198    /// - `Err(anyhow::Error)`: If the state file path could not be determined
199    pub(crate) fn state_file_path() -> anyhow::Result<PathBuf> {
200        let mut path = crate::fulgur::utils::paths::config_dir()?;
201        path.push("state.json");
202        Ok(path)
203    }
204
205    /// Save the app state to a specific path
206    ///
207    /// ### Description
208    /// Core implementation for saving window state. Uses atomic file writes to
209    /// prevent corruption when multiple windows write simultaneously.
210    ///
211    /// ### Arguments
212    /// - `path`: The path to save the state to
213    ///
214    /// ### Returns
215    /// - `Ok(())`: If the app state was saved successfully
216    /// - `Err(anyhow::Error)`: If the app state could not be saved
217    pub fn save_to_path(&self, path: &Path) -> anyhow::Result<()> {
218        let json = serde_json::to_string_pretty(self)?;
219        if path.exists() {
220            let backup = crate::fulgur::utils::atomic_write::backup_path_for(path);
221            if let Err(e) = fs::copy(path, &backup) {
222                log::warn!("Failed to back up state to '{}': {}", backup.display(), e);
223            }
224        }
225        atomic_write_file(path, json.as_bytes())
226    }
227
228    /// Load the windows state from a specific path
229    ///
230    /// ### Description
231    /// Core implementation for loading window state.
232    ///
233    /// ### Arguments
234    /// - `path`: The path to load the state from
235    ///
236    /// ### Returns
237    /// - `Ok(WindowsState)`: The loaded windows state
238    /// - `Err(anyhow::Error)`: If the windows state could not be loaded
239    pub fn load_from_path(path: &PathBuf) -> anyhow::Result<Self> {
240        let json = fs::read_to_string(path)
241            .map_err(|e| anyhow::anyhow!("Failed to read state file: {}", e))?;
242        match serde_json::from_str::<WindowsState>(&json) {
243            Ok(state) => Ok(state),
244            Err(primary_err) => {
245                let backup = crate::fulgur::utils::atomic_write::backup_path_for(path);
246                log::warn!(
247                    "State file is corrupted ({}), attempting recovery from '{}'",
248                    primary_err,
249                    backup.display()
250                );
251                let bak_json = fs::read_to_string(&backup)
252                    .map_err(|_| anyhow::anyhow!("Failed to parse state: {}", primary_err))?;
253                let state = serde_json::from_str::<WindowsState>(&bak_json).map_err(|bak_err| {
254                    anyhow::anyhow!(
255                        "State and backup are both corrupted: primary={}, backup={}",
256                        primary_err,
257                        bak_err
258                    )
259                })?;
260                log::warn!("State recovered from backup '{}'", backup.display());
261                Ok(state)
262            }
263        }
264    }
265
266    /// Save the app state to the default state file location
267    ///
268    /// ### Returns
269    /// - `Ok(())`: If the app state was saved successfully
270    /// - `Err(anyhow::Error)`: If the app state could not be saved
271    pub fn save(&self) -> anyhow::Result<()> {
272        let path = Self::state_file_path()?;
273        self.save_to_path(&path)
274    }
275
276    /// Load the windows state from the default state file location
277    ///
278    /// ### Returns
279    /// - `Ok(WindowsState)`: The loaded windows state
280    /// - `Err(anyhow::Error)`: If the windows state could not be loaded
281    pub fn load() -> anyhow::Result<Self> {
282        let path = Self::state_file_path()?;
283        Self::load_from_path(&path)
284    }
285}

287/// Get the last modified time of a file as ISO 8601 string
288///
289/// ### Arguments
290/// - `path`: The path to the file
291///
292/// ### Returns
293/// - `Some(String)`: The last modified time of the file
294/// - `None`: If the file could not be found or the last modified time could not be determined
295pub fn get_file_modified_time(path: &PathBuf) -> Option<String> {
296    let metadata = fs::metadata(path).ok()?;
297    let modified = metadata.modified().ok()?;
298    let datetime = time::OffsetDateTime::from(modified);
299    datetime
300        .format(&time::format_description::well_known::Rfc3339)
301        .ok()
302}
303
304/// Compare two ISO 8601 timestamps
305///
306/// ### Arguments
307/// - `file_time`: The time of the file
308/// - `saved_time`: The time of the saved file
309///
310/// ### Returns
311/// - `True`: If the file is newer than the saved file, `False` otherwise
312pub fn is_file_newer(file_time: &str, saved_time: &str) -> bool {
313    let file_dt =
314        time::OffsetDateTime::parse(file_time, &time::format_description::well_known::Rfc3339).ok();
315    let saved_dt =
316        time::OffsetDateTime::parse(saved_time, &time::format_description::well_known::Rfc3339)
317            .ok();
318    match (file_dt, saved_dt) {
319        (Some(file), Some(saved)) => file > saved,
320        _ => false,
321    }
322}

762    fn test_serialized_remote_spec_roundtrip_omits_password() {
763        let spec = crate::fulgur::sync::ssh::url::RemoteSpec {
764            host: "example.com".to_string(),
765            port: 22,
766            user: Some("alice".to_string()),
767            path: "/tmp/test.txt".to_string(),
768            password_in_url: Some(zeroize::Zeroizing::new("secret".to_string())),
769        };
770
771        let serialized = SerializedRemoteSpec::from_remote_spec(&spec);
772        assert_eq!(serialized.host, "example.com");
773        assert_eq!(serialized.user, "alice");
774
775        let json = serde_json::to_string(&serialized).expect("serialize serialized remote spec");
776        assert!(
777            !json.contains("password"),
778            "serialized remote spec must not include a password field"
779        );
780
781        let restored = serialized.to_remote_spec();
782        assert_eq!(restored.host, "example.com");
783        assert_eq!(restored.port, 22);
784        assert_eq!(restored.user.as_deref(), Some("alice"));
785        assert_eq!(restored.path, "/tmp/test.txt");
786        assert!(restored.password_in_url.is_none());
787    }
