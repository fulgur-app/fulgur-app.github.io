324/// Get the Language enum from a filename or file extension.
325/// Exact filename matches (e.g. `Dockerfile`, `Makefile`) are checked first,
326/// then falls back to the file extension.
327///
328/// ### Arguments
329/// - `filename`: The file name
330///
331/// ### Returns
332/// - `SupportedLanguage`: The detected language
333#[must_use]
334pub fn language_from_filename(filename: &str) -> SupportedLanguage {
335    let lower = filename.to_lowercase();
336    let exact = match lower.as_str() {
337        "dockerfile" => Some(SupportedLanguage::Dockerfile),
338        "makefile" | "gnumakefile" => Some(SupportedLanguage::Make),
339        "gemfile" | "rakefile" | "guardfile" | "podfile" => Some(SupportedLanguage::Ruby),
340        _ => None,
341    };
342    if let Some(lang) = exact {
343        return lang;
344    }
345
346    let extension = std::path::Path::new(filename)
347        .extension()
348        .and_then(|ext| ext.to_str())
349        .unwrap_or("");
350
351    if extension == "txt" {
352        return SupportedLanguage::Plain;
353    }
354
355    // Overriding gpui-component's default language mapping.
356    if let Some(lang) = match extension {
357        "scss" => Some(SupportedLanguage::Scss),
358        "csv" | "tsv" => Some(SupportedLanguage::Csv),
359        _ => None,
360    } {
361        return lang;
362    }
363
364    let mut language = from_language(Language::from_str(extension));
365    if language == SupportedLanguage::Plain {
366        language = match extension {
367            "ada" | "ads" | "adb" => SupportedLanguage::Ada,
368            "asm" | "s" | "S" | "nasm" | "masm" => SupportedLanguage::Asm,
369            "clojure" | "clj" | "cljs" => SupportedLanguage::Clojure,
370            "d" | "di" => SupportedLanguage::D,
371            "dart" => SupportedLanguage::Dart,
372            "dockerfile" => SupportedLanguage::Dockerfile,
373            "erl" | "hrl" | "escript" => SupportedLanguage::Erlang,
374            "f" | "f77" | "f90" | "f95" | "f03" | "f08" | "for" | "ftn" | "pf" => {
375                SupportedLanguage::Fortran
376            }
377            "fs" | "fsi" | "fsx" | "fsscript" => SupportedLanguage::FSharp,
378            "hs" | "lhs" => SupportedLanguage::Haskell,
379            "ini" | "cfg" | "conf" | "config" => SupportedLanguage::Ini,
380            "jinja" | "jinja2" | "j2" => SupportedLanguage::Jinja2,
381            "jl" => SupportedLanguage::Julia,
382            "lock" => SupportedLanguage::Toml,
383            "groovy" | "gvy" | "gy" | "gsh" => SupportedLanguage::Groovy,
384            "mjs" => SupportedLanguage::JavaScript,
385            "pas" | "pp" | "dpr" | "dpk" | "lpr" => SupportedLanguage::Pascal,
386            "perl" | "pl" | "pm" | "plx" => SupportedLanguage::Perl,
387            "powershell" | "ps1" | "psm1" | "psd1" => SupportedLanguage::Powershell,
388            "pro" | "prolog" => SupportedLanguage::Prolog,
389            "m" | "M" | "mm" => SupportedLanguage::ObjectiveC,
390            "ml" | "mli" => SupportedLanguage::Ocaml,
391            "r" | "rmd" => SupportedLanguage::R,
392            "svg" => SupportedLanguage::Html,
393            "tsx" | "jsx" => SupportedLanguage::React,
394            "vue" => SupportedLanguage::Vue,
395            "xml" => SupportedLanguage::Xml,
396            _ => SupportedLanguage::Plain,
397        };
398    }
399    language
400}

402/// Detect whether a `.m` file contains Objective-C or MATLAB source by scanning
403/// the first 50 lines for distinctive markers.
404///
405/// Returns `ObjectiveC` if any Objective-C marker is encountered first, `Matlab` if any
406/// MATLAB marker is encountered first, and `ObjectiveC` as the default when no signal is found.
407fn detect_m_file_language(content: &str) -> SupportedLanguage {
408    for line in content.lines().take(50) {
409        let trimmed = line.trim();
410        // Objective-C markers
411        if trimmed.starts_with("#import")
412            || trimmed.starts_with("#include")
413            || trimmed.starts_with("@interface")
414            || trimmed.starts_with("@implementation")
415            || trimmed.starts_with("@protocol")
416            || trimmed.starts_with("@property")
417            || trimmed.starts_with("- (")
418            || trimmed.starts_with("+ (")
419        {
420            return SupportedLanguage::ObjectiveC;
421        }
422        // MATLAB markers
423        if trimmed.starts_with('%')
424            || trimmed.starts_with("function ")
425            || trimmed == "function"
426            || trimmed.starts_with("classdef ")
427        {
428            return SupportedLanguage::Matlab;
429        }
430    }
431    SupportedLanguage::ObjectiveC
432}

434/// Get the language from a filename and file content.
435///
436/// Behaves like [`language_from_filename`] but additionally uses content-based heuristics
437/// for extensions that are shared between multiple languages. Currently handles:
438/// - `.m`: disambiguates between Objective-C and MATLAB by scanning the first 50 lines.
439///
440/// ### Arguments
441/// - `filename`: The file name
442/// - `content`: The file content
443///
444/// ### Returns
445/// - `SupportedLanguage`: The detected language
446#[must_use]
447pub fn language_from_content(filename: &str, content: &str) -> SupportedLanguage {
448    let base = language_from_filename(filename);
449    if base == SupportedLanguage::ObjectiveC {
450        let ext = std::path::Path::new(filename)
451            .extension()
452            .and_then(|e| e.to_str())
453            .unwrap_or("");
454        if ext == "m" {
455            return detect_m_file_language(content);
456        }
457    }
458    base
459}

538impl Fulgur {
539    /// Get the current language of the active tab
540    ///
541    /// ### Arguments
542    /// - `cx`: The application context
543    ///
544    /// ### Returns
545    /// - `SupportedLanguage`: The active tab's language
546    #[must_use]
547    pub fn get_current_language(&self, cx: &gpui::App) -> SupportedLanguage {
548        self.active_tab(cx)
549            .and_then(crate::fulgur::tab::Tab::as_editor)
550            .map_or(SupportedLanguage::Plain, |editor_tab| editor_tab.language)
551    }
552
553    /// Check if the current active tab's language is a Markdown language
554    ///
555    /// ### Arguments
556    /// - `cx`: The application context
557    ///
558    /// ### Returns
559    /// - `True` if the active tab's language is a Markdown language, `False` otherwise
560    #[must_use]
561    pub fn is_markdown(&self, cx: &gpui::App) -> bool {
562        let current_language = self.get_current_language(cx);
563        current_language == SupportedLanguage::Markdown
564            || current_language == SupportedLanguage::MarkdownInline
565    }
566}
