1use gpui_component::highlighter::{LanguageConfig, LanguageRegistry};
2
3/// Add Assembly language syntax highlighting support.
4pub fn add_asm_support() {
5    LanguageRegistry::singleton().register(
6        "asm",
7        &LanguageConfig::new(
8            "asm",
9            arborium_asm::language().into(),
10            vec![],
11            ASM_HIGHLIGHTS_QUERY,
12            "",
13            "",
14        ),
15    );
16}
17
18/// Highlights query for Assembly, adapted from the `arborium_asm` bundled query.
19///
20/// Key mapping choices vs. the upstream tree-sitter-asm highlights.scm:
21/// - `@variable.builtin` is used for registers - this capture is defined in all
22///   bundled themes, giving registers a distinct color (typically blue).
23/// - `@function.builtin` is used for instruction mnemonics (e.g. `mov`, `push`) and
24///   assembler directives (e.g. `.section`, `.global`) - both use the `kind` field.
25/// - `@label` is used for labels (e.g. `_start:`, `main:`) - this capture exists in
26///   all bundled themes.
27/// - `@number.float` is kept separate from `@number` so themes can color them differently.
28/// - `@spell` from the upstream query is omitted - it is neovim-specific and has no
29///   equivalent in gpui-component.
30const ASM_HIGHLIGHTS_QUERY: &str = r#"
31; Labels (e.g. _start:, .loop:, main:)
32(label
33  [(ident) (word)] @label)
34
35; Registers (e.g. %rax, eax, rsp, x0)
36(reg) @variable.builtin
37
38; Instruction mnemonics (e.g. mov, push, call, ret)
39(instruction
40  kind: (_) @function.builtin)
41
42; Assembler directives (e.g. .section, .global, .byte)
43(meta
44  kind: (_) @function.builtin)
45
46; Named constants (const name = value)
47(const
48  name: (word) @constant)
49
50; Comments
51[
52  (line_comment)
53  (block_comment)
54] @comment
55
56; Integer literals (decimal, hex, binary, octal)
57(int) @number
58
59; Floating-point literals
60(float) @number.float
61
62; String literals
63(string) @string
64
65; Size specifiers and pointer keywords
66[
67  "byte"
68  "word"
69  "dword"
70  "qword"
71  "ptr"
72  "rel"
73  "label"
74  "const"
75] @keyword
76
77; Arithmetic and bitwise operators
78[
79  "+"
80  "-"
81  "*"
82  "/"
83  "%"
84  "|"
85  "^"
86  "&"
87] @operator
88
89; Brackets and parentheses
90[
91  "("
92  ")"
93  "["
94  "]"
95] @punctuation.bracket
96
97; Commas and colons
98[
99  ","
100  ":"
101] @punctuation.delimiter
102"#;
