1use gpui_component::highlighter::{LanguageConfig, LanguageRegistry};

3/// Add Jinja2 template syntax highlighting support.
4pub fn add_jinja2_support() {
5    LanguageRegistry::singleton().register(
6        "jinja2",
7        &LanguageConfig::new(
8            "jinja2",
9            arborium_jinja2::language().into(),
10            vec![],
11            JINJA2_HIGHLIGHTS_QUERY,
12            "",
13            "",
14        ),
15    );
16}
17
18/// Highlights query for Jinja2, remapped from the arborium_jinja2 nvim-treesitter naming
19/// conventions to the gpui-component recognized names.
20///
21/// Key points:
22/// - `keyword` is a named node holding the template tag keyword (if, for, block, etc.),
23///   captured directly rather than via anonymous string literals.
24/// - `expression_begin/end` and `statement_begin/end` are named nodes wrapping `{{`, `}}`,
25///   `{%`, `%}` delimiters (with optional whitespace-control `-`).
26/// - `identifier` is the catch-all for variable names, filter names, and tag arguments.
27/// - Delimiters use `@tag` because it is defined in all bundled themes; `@punctuation.bracket`
28///   is only defined in Catppuccin and would be invisible elsewhere.
29///
30/// HTML injection is intentionally omitted. Injecting HTML into the full `source_file`
31/// causes the HTML parser to mis-parse Jinja2 comparison operators (`< N`) as HTML tags,
32/// bleeding `@attribute` highlights over all subsequent Jinja2 content. Since the Jinja2
33/// grammar exposes no named nodes for the raw template text between tags, there is no
34/// way to restrict the injection to non-Jinja2 regions.
35const JINJA2_HIGHLIGHTS_QUERY: &str = r#"
36; Comments: {# ... #}
37(comment) @comment
38
39; String literals within expressions and statements
40(string) @string
41
42; Template keywords: if, elif, else, endif, for, endfor, block, endblock, etc.
43(keyword) @keyword
44
45; Operators: |, ==, !=, <, >, and, or, not, is, in, ~, +, -, *, /
46(operator) @operator
47
48; Template delimiters: {{ }}, {% %}, with optional whitespace control (-/+)
49; Using @tag because it is present in all bundled themes.
50(expression_begin) @tag
51(expression_end) @tag
52(statement_begin) @tag
53(statement_end) @tag
54
55; Identifiers: variables, filter names, macro names, block names
56(identifier) @variable
57"#;
