16/// Initialize the themes
17///
18/// ### Arguments
19/// - `settings`: The application settings containing theme preferences
20/// - `cx`: The application context
21/// - `on_themes_loaded`: The callback to call when the themes are loaded
22pub fn init(settings: &Settings, cx: &mut App, on_themes_loaded: impl Fn(&mut App) + 'static) {
23    let theme_name = settings.app_settings.theme.clone();
24    let scrollbar_show = settings.app_settings.scrollbar_show;
25    let themes_directory = match themes_directory_path() {
26        Ok(path) => {
27            if let Err(e) = extract_bundled_themes(&path) {
28                log::error!("Failed to extract bundled themes: {e}");
29            }
30            path
31        }
32        Err(e) => {
33            log::error!("Failed to get themes directory: {e}");
34            return;
35        }
36    };
37    if let Err(err) = ThemeRegistry::watch_dir(themes_directory, cx, move |cx| {
38        if let Some(theme) = ThemeRegistry::global(cx).themes().get(&theme_name).cloned() {
39            Theme::global_mut(cx).apply_config(&theme);
40        }
41        on_themes_loaded(cx);
42    }) {
43        log::error!("Failed to watch themes directory: {err}");
44    }
45    if let Some(scrollbar_show) = scrollbar_show {
46        Theme::global_mut(cx).scrollbar_show = scrollbar_show;
47    }
48    cx.refresh_windows();
49    cx.on_action(|switch: &SwitchThemeMode, cx| {
50        let mode = switch.0;
51        Theme::change(mode, None, cx);
52        cx.refresh_windows();
53    });
54}

56/// Extract bundled themes to the themes directory if they don't exist
57///
58/// ### Arguments
59/// - `themes_dir`: The themes directory path
60///
61/// ### Returns
62/// - `Ok(())`: Result indicating success
63/// - `Err(anyhow::Error)`: Result indicating failure
64fn extract_bundled_themes(themes_dir: &PathBuf) -> anyhow::Result<()> {
65    fs::create_dir_all(themes_dir)?;
66    for file in BundledThemes::iter() {
67        let file_path = themes_dir.join(file.as_ref());
68        if !file_path.exists()
69            && let Some(content) = BundledThemes::get(&file)
70        {
71            fs::write(&file_path, content.data.as_ref())?;
72        }
73    }
74    Ok(())
75}

77/// Get the path to the themes directory

86/// Reload themes and update the Fulgur instance: this function initializes the theme registry, reloads themes from disk, updates the Fulgur entity's themes field, and refreshes the window.
87///
88/// ### Arguments
89/// - `settings`: The application settings
90/// - `entity`: The Fulgur entity to update
91/// - `cx`: The application context
92pub fn reload_themes_and_update(settings: &Settings, entity: &Entity<Fulgur>, cx: &mut App) {
93    let entity_clone = entity.clone();
94    init(settings, cx, move |cx| {
95        let themes = match Themes::load() {
96            Ok(themes) => Some(themes),
97            Err(e) => {
98                log::error!("Failed to load themes: {e}");
99                None
100            }
101        };
102        if let Some(themes) = themes {
103            entity_clone.update(cx, |_, cx| {
104                let shared = Fulgur::shared_state(cx);
105                *shared.themes.lock() = Some(themes);
106            });
107        }
108    });
109}

126    /// ### Arguments

132        let timestamp = SystemTime::now()
133            .duration_since(UNIX_EPOCH)

136        std::env::temp_dir().join(format!("{}_{}_{}", prefix, std::process::id(), timestamp))

202    /// Wait until a condition becomes true while pumping the GPUI async runtime.
203    ///
204    /// ### Arguments
205    /// - `cx`: The GPUI test context to drive.
206    /// - `attempts`: Maximum polling attempts before timing out.
207    /// - `delay`: Sleep duration between attempts.
208    /// - `predicate`: The condition to evaluate after each pump.
209    ///
210    /// ### Returns
211    /// - `true`: The predicate returned true within the allowed attempts.
212    /// - `false`: The predicate never became true before timeout.
213    fn wait_until(
214        cx: &mut TestAppContext,
215        attempts: usize,
216        delay: Duration,
217        mut predicate: impl FnMut(&mut TestAppContext) -> bool,
218    ) -> bool {
219        for _ in 0..attempts {
220            cx.run_until_parked();
221            if predicate(cx) {
222                return true;
223            }
224            thread::sleep(delay);
225        }
226        false
227    }

268            fulgur_slot
269                .into_inner()
270                .expect("failed to capture test Fulgur entity"),
