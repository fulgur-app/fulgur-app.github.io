13/// Build the menus for the Fulgur instance
14///
15/// ### Arguments
16/// - `recent_files`: The list of recent files to display
17/// - `update_link`: The optional link to the update
18///
19/// ### Returns
20/// - `Vec<Menu>`: The menus for the Fulgur instance
21pub fn build_menus(recent_files: &[PathBuf], update_link: Option<&str>) -> Vec<Menu> {
22    let recent_files_items = if recent_files.is_empty() {
23        vec![MenuItem::action("No recent files", NoneAction)]
24    } else {
25        let mut items: Vec<MenuItem> = recent_files
26            .iter()
27            .map(|file| MenuItem::action(file.display().to_string(), OpenRecentFile(file.clone())))
28            .collect();
29        items.push(MenuItem::Separator);
30        items.push(MenuItem::action("Clear recent files", ClearRecentFiles));
31        items
32    };
33    vec![
34        Menu {
35            name: "Fulgur".into(),
36            disabled: false,
37            items: vec![
38                MenuItem::action("About Fulgur", About),
39                if update_link.is_some() {
40                    MenuItem::action("Update available", CheckForUpdates)
41                } else {
42                    MenuItem::action("Check for updates", CheckForUpdates)
43                },
44                MenuItem::Separator,
45                MenuItem::action("Settings", SettingsTab),
46                MenuItem::action("Select theme", SelectTheme),
47                MenuItem::action("Get more themes...", GetTheme),
48                MenuItem::Separator,
49                MenuItem::action("Close Window", CloseWindow),
50                MenuItem::action("Quit", Quit),
51            ],
52        },
53        Menu {
54            name: "File".into(),
55            disabled: false,
56            items: vec![
57                MenuItem::action("New", NewFile),
58                MenuItem::action("New Window", NewWindow),
59                MenuItem::action("Open...", OpenFile),
60                MenuItem::action("Open from path...", OpenPath),
61                MenuItem::action("Open remote file...", OpenRemote),
62                MenuItem::Submenu(Menu {
63                    name: "Recent Files".into(),
64                    disabled: false,
65                    items: recent_files_items,
66                }),
67                MenuItem::separator(),
68                MenuItem::action("Save", SaveFile),
69                MenuItem::action("Save as...", SaveFileAs),
70                MenuItem::separator(),
71                MenuItem::action("Print...", PrintFile),
72                MenuItem::separator(),
73                MenuItem::action("Close file", CloseFile),
74                MenuItem::action("Close all files", CloseAllFiles),
75            ],
76        },
77        Menu {
78            name: "Edit".into(),
79            disabled: false,
80            items: vec![
81                MenuItem::action("Undo", gpui_component::input::Undo),
82                MenuItem::action("Redo", gpui_component::input::Redo),
83                MenuItem::separator(),
84                MenuItem::action("Cut", gpui_component::input::Cut),
85                MenuItem::action("Copy", gpui_component::input::Copy),
86                MenuItem::action("Paste", gpui_component::input::Paste),
87                MenuItem::separator(),
88                MenuItem::action("Find & Replace", FindInFile),
89            ],
90        },
91        Menu {
92            name: "View".into(),
93            disabled: false,
94            items: vec![
95                MenuItem::action("Color picker", ToggleColorPicker),
96                MenuItem::separator(),
97            ],
98        },
99        Menu {
100            name: "Go".into(),
101            disabled: false,
102            items: vec![
103                MenuItem::action("Next Tab", NextTab),
104                MenuItem::action("Previous Tab", PreviousTab),
105                MenuItem::Separator,
106                MenuItem::action("Jump to line", JumpToLine),
107            ],
108        },
109    ]
110}

112impl Fulgur {
113    /// Set the application menus and sync them to the `AppMenuBar` on non-macOS platforms.
114    ///
115    /// ### Arguments
116    /// - `menus`: The menus to set
117    /// - `cx`: The application context
118    pub fn update_menus(&mut self, menus: Vec<Menu>, cx: &mut Context<Self>) {
119        cx.set_menus(menus);
120        #[cfg(not(target_os = "macos"))]
121        {
122            if let Some(owned_menus) = cx.get_menus() {
123                GlobalState::global_mut(cx).set_app_menus(owned_menus);
124            }
125            self.title_bar.update(
126                cx,
127                crate::fulgur::ui::bars::titlebar::CustomTitleBar::reload_app_menu_bar,
128            );
129        }
130    }
