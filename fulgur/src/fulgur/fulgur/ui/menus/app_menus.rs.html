14/// Build the menus for the Fulgur instance
15///
16/// ### Arguments
17/// - `recent_files`: The list of recent files to display
18/// - `update_link`: The optional link to the update
19///
20/// ### Returns
21/// - `Vec<Menu>`: The menus for the Fulgur instance
22pub fn build_menus(recent_files: &[PathBuf], update_link: Option<&str>) -> Vec<Menu> {
23    let recent_files_items = if recent_files.is_empty() {
24        vec![MenuItem::action("No recent files", NoneAction)]
25    } else {
26        let mut items: Vec<MenuItem> = recent_files
27            .iter()
28            .map(|file| MenuItem::action(file.display().to_string(), OpenRecentFile(file.clone())))
29            .collect();
30        items.push(MenuItem::Separator);
31        items.push(MenuItem::action("Clear recent files", ClearRecentFiles));
32        items
33    };
34    vec![
35        Menu {
36            name: "Fulgur".into(),
37            disabled: false,
38            items: vec![
39                MenuItem::action("About Fulgur", About),
40                if update_link.is_some() {
41                    MenuItem::action("Update available", CheckForUpdates)
42                } else {
43                    MenuItem::action("Check for updates", CheckForUpdates)
44                },
45                MenuItem::Separator,
46                MenuItem::action("Settings", SettingsTab),
47                MenuItem::action("Select theme", SelectTheme),
48                MenuItem::action("Get more themes...", GetTheme),
49                MenuItem::Separator,
50                MenuItem::action("Close Window", CloseWindow),
51                MenuItem::action("Quit", Quit),
52            ],
53        },
54        Menu {
55            name: "File".into(),
56            disabled: false,
57            items: vec![
58                MenuItem::action("New", NewFile),
59                MenuItem::action("New Window", NewWindow),
60                MenuItem::action("Open...", OpenFile),
61                MenuItem::action("Open from path...", OpenPath),
62                MenuItem::action("Open remote file...", OpenRemote),
63                MenuItem::Submenu(Menu {
64                    name: "Recent Files".into(),
65                    disabled: false,
66                    items: recent_files_items,
67                }),
68                MenuItem::separator(),
69                MenuItem::action("Save", SaveFile),
70                MenuItem::action("Save as...", SaveFileAs),
71                MenuItem::separator(),
72                MenuItem::action("Print...", PrintFile),
73                MenuItem::separator(),
74                MenuItem::action("Close file", CloseFile),
75                MenuItem::action("Close all files", CloseAllFiles),
76            ],
77        },
78        Menu {
79            name: "Edit".into(),
80            disabled: false,
81            items: vec![
82                MenuItem::action("Undo", gpui_kit::component::input::Undo),
83                MenuItem::action("Redo", gpui_kit::component::input::Redo),
84                MenuItem::separator(),
85                MenuItem::action("Cut", gpui_kit::component::input::Cut),
86                MenuItem::action("Copy", gpui_kit::component::input::Copy),
87                MenuItem::action("Paste", gpui_kit::component::input::Paste),
88                MenuItem::separator(),
89                MenuItem::action("Add Cursor Above", AddCursorAbove),
90                MenuItem::action("Add Cursor Below", AddCursorBelow),
91                MenuItem::separator(),
92                MenuItem::action("Find & Replace", FindInFile),
93            ],
94        },
95        Menu {
96            name: "View".into(),
97            disabled: false,
98            items: vec![
99                MenuItem::action("Command palette", ToggleCommandPalette),
100                MenuItem::separator(),
101                MenuItem::action("Color picker", ToggleColorPicker),
102                MenuItem::separator(),
103            ],
104        },
105        Menu {
106            name: "Go".into(),
107            disabled: false,
108            items: vec![
109                MenuItem::action("Next Tab", NextTab),
110                MenuItem::action("Previous Tab", PreviousTab),
111                MenuItem::Separator,
112                MenuItem::action("Jump to line", JumpToLine),
113            ],
114        },
115    ]
116}

118impl Fulgur {
119    /// Set the application menus and sync them to the `AppMenuBar` on non-macOS platforms.
120    ///
121    /// ### Arguments
122    /// - `menus`: The menus to set
123    /// - `cx`: The application context
124    pub fn update_menus(&mut self, menus: Vec<Menu>, cx: &mut Context<Self>) {
125        cx.set_menus(menus);
126        #[cfg(not(target_os = "macos"))]
127        {
128            if let Some(owned_menus) = cx.get_menus() {
129                GlobalState::global_mut(cx).set_app_menus(owned_menus);
130            }
131            self.title_bar.update(
132                cx,
133                crate::fulgur::ui::bars::titlebar::CustomTitleBar::reload_app_menu_bar,
134            );
135        }
136    }
137}

143    use gpui_kit::base::input::{AddCursorAbove, AddCursorBelow};
144    use gpui_kit::{Action, MenuItem};

146    fn edit_menu_exposes(action: &dyn Action) -> bool {
147        build_menus(&[], None)
148            .iter()
149            .filter(|menu| menu.name == "Edit")
150            .flat_map(|menu| menu.items.iter())
151            .any(|item| match item {
152                MenuItem::Action { action: bound, .. } => bound.partial_eq(action),
153                _ => false,
154            })
155    }
156
157    #[test]
158    fn test_edit_menu_exposes_multi_cursor_actions() {
159        assert!(edit_menu_exposes(&AddCursorAbove));
160        assert!(edit_menu_exposes(&AddCursorBelow));
161    }
