34/// A platform keybinding dispatch specification used to build runtime keybindings.
35#[derive(Clone, Copy, Debug, PartialEq, Eq)]
36struct KeybindingDispatchSpec {
37    keystroke: &'static str,
38    action: KeybindingDispatchAction,
39}
40
41impl KeybindingDispatchSpec {
42    /// Create a keybinding dispatch specification.
43    ///
44    /// ### Parameters:
45    /// - `keystroke`: The key combination string consumed by GPUI.
46    /// - `action`: The action to dispatch for this keybinding.
47    ///
48    /// ### Returns:
49    /// - `Self`: The new keybinding dispatch specification.
50    const fn new(keystroke: &'static str, action: KeybindingDispatchAction) -> Self {
51        Self { keystroke, action }
52    }
53
54    /// Convert this dispatch specification into a GPUI keybinding instance.
55    ///
56    /// ### Returns:
57    /// - `KeyBinding`: The runtime keybinding bound to the configured action, scoped
58    ///   to this action's key context.
59    fn into_key_binding(self) -> KeyBinding {
60        let context = self.action.key_context();
61        match self.action {
62            KeybindingDispatchAction::OpenFile => {
63                KeyBinding::new(self.keystroke, OpenFile, context)
64            }
65            KeybindingDispatchAction::NewFile => KeyBinding::new(self.keystroke, NewFile, context),
66            KeybindingDispatchAction::OpenPath => {
67                KeyBinding::new(self.keystroke, OpenPath, context)
68            }
69            KeybindingDispatchAction::OpenRemote => {
70                KeyBinding::new(self.keystroke, OpenRemote, context)
71            }
72            KeybindingDispatchAction::NewWindow => {
73                KeyBinding::new(self.keystroke, NewWindow, context)
74            }
75            KeybindingDispatchAction::CloseFile => {
76                KeyBinding::new(self.keystroke, CloseFile, context)
77            }
78            KeybindingDispatchAction::CloseAllFiles => {
79                KeyBinding::new(self.keystroke, CloseAllFiles, context)
80            }
81            KeybindingDispatchAction::Quit => KeyBinding::new(self.keystroke, Quit, context),
82            KeybindingDispatchAction::SaveFile => {
83                KeyBinding::new(self.keystroke, SaveFile, context)
84            }
85            KeybindingDispatchAction::SaveFileAs => {
86                KeyBinding::new(self.keystroke, SaveFileAs, context)
87            }
88            KeybindingDispatchAction::FindInFile => {
89                KeyBinding::new(self.keystroke, FindInFile, context)
90            }
91            KeybindingDispatchAction::NextTab => KeyBinding::new(self.keystroke, NextTab, context),
92            KeybindingDispatchAction::PreviousTab => {
93                KeyBinding::new(self.keystroke, PreviousTab, context)
94            }
95            KeybindingDispatchAction::JumpToLine => {
96                KeyBinding::new(self.keystroke, JumpToLine, context)
97            }
98            KeybindingDispatchAction::PrintFile => {
99                KeyBinding::new(self.keystroke, PrintFile, context)
100            }
101            KeybindingDispatchAction::ToggleColorPicker => {
102                KeyBinding::new(self.keystroke, ToggleColorPicker, context)
103            }
104        }
105    }
106}
107
108impl KeybindingDispatchAction {
109    /// Get the key context under which this action's binding is active.
110    ///
111    /// ### Returns
112    /// - `Some(&'static str)`: The key context the binding is scoped to
113    /// - `None`: The binding is global
114    const fn key_context(self) -> Option<&'static str> {
115        match self {
116            Self::OpenFile
117            | Self::NewFile
118            | Self::OpenPath
119            | Self::OpenRemote
120            | Self::NewWindow
121            | Self::Quit => None,
122            Self::CloseFile
123            | Self::CloseAllFiles
124            | Self::SaveFile
125            | Self::SaveFileAs
126            | Self::FindInFile
127            | Self::NextTab
128            | Self::PreviousTab
129            | Self::JumpToLine
130            | Self::PrintFile
131            | Self::ToggleColorPicker => Some(SCOPED_BINDING_PREDICATE),
132        }
133    }
134}

205/// Build the default runtime keybindings for the application.
206///
207/// ### Returns:
208/// - `Vec<KeyBinding>`: The platform-specific list of GPUI keybindings.
209pub fn build_default_key_bindings() -> Vec<KeyBinding> {
210    default_keybinding_dispatch_specs()
211        .into_iter()
212        .map(KeybindingDispatchSpec::into_key_binding)
213        .collect()
214}

226        keystroke: &str,

302    #[test]
303    fn test_scoped_predicate_matches_editor_input_depth_but_not_modal_inputs() {
304        let scoped =
305            gpui::KeyBindingContextPredicate::parse(super::SCOPED_BINDING_PREDICATE).unwrap();
306        let input = gpui::KeyBindingContextPredicate::parse("Input").unwrap();
307
308        let editor_stack = [
309            gpui::KeyContext::parse("Fulgur").unwrap(),
310            gpui::KeyContext::parse("Input").unwrap(),
311        ];
312        assert_eq!(
313            scoped.depth_of(&editor_stack),
314            input.depth_of(&editor_stack),
315            "scoped bindings must match at the same depth as gpui-component's \
316             Input bindings, so Fulgur's later registration wins the tie"
317        );
318
319        let dialog_stack = [
320            gpui::KeyContext::parse("Dialog").unwrap(),
321            gpui::KeyContext::parse("Input").unwrap(),
322        ];
323        assert_eq!(
324            scoped.depth_of(&dialog_stack),
325            None,
326            "scoped bindings must not fire inside modal inputs"
327        );
328
329        let content_stack = [gpui::KeyContext::parse("Fulgur").unwrap()];
330        assert!(
331            scoped.depth_of(&content_stack).is_some(),
332            "scoped bindings must fire when the app content itself is focused"
333        );
334    }
