1//! Log view mode: a live "tail -f" surface for log-like files.
2//!
3//! A log-view tab keeps its editable `content` buffer untouched and instead
4//! renders a dedicated read-only `log_content` buffer that is refreshed by a
5//! per-active-tab polling task. The poll reads only newly appended bytes,
6//! appends them to the display, and trims to the last `LOG_LINE_CAP` lines
7//! (unless the user loaded the full file).
8//!
9//! Following is an explicit, user-controlled toggle (`log_follow`): when on,
10//! every refresh snaps to the bottom; when off, the view is frozen and new
11//! text is buffered until the user re-enables follow. The mode toggle and the
12//! `Follow` / `Load full` controls live in the status bar; this module owns the
13//! state machine and tailing logic only (it renders no UI of its own).

25/// Maximum number of trailing lines kept in the log view before trimming.
26pub const LOG_LINE_CAP: usize = 10_000;
27
28/// Per-tab tail bookkeeping, held centrally in `Fulgur` and keyed by tab id.
29pub struct LogTailState {
30    /// Byte offset in the file up to which content has already been consumed.
31    pub byte_offset: u64,
32    /// Whether the line cap has dropped older lines from the display.
33    pub dropped_lines: bool,
34    /// Newly appended text accumulated while follow is paused (frozen view).
35    pub pending: String,
36}
37
38impl LogTailState {
39    /// Create a fresh tail state seeded at the given byte offset.
40    ///
41    /// ### Arguments
42    /// - `byte_offset`: The initial file offset already consumed by the seed
43    /// - `dropped_lines`: Whether the seed already exceeded the line cap
44    ///
45    /// ### Returns
46    /// - `LogTailState`: The initialized state
47    fn new(byte_offset: u64, dropped_lines: bool) -> Self {
48        Self {
49            byte_offset,
50            dropped_lines,
51            pending: String::new(),
52        }
53    }
54}
55
56impl Fulgur {
57    /// Borrow an editor tab by id.
58    ///
59    /// ### Arguments
60    /// - `tab_id`: The tab id to look up
61    ///
62    /// ### Returns
63    /// - `Some(&EditorTab)`: The matching editor tab
64    /// - `None`: If no editor tab has that id
65    fn editor_tab(&self, tab_id: TabId) -> Option<&crate::fulgur::editor_tab::EditorTab> {
66        self.tabs
67            .iter()
68            .find_map(|tab| tab.as_editor().filter(|editor| editor.id == tab_id))
69    }
70
71    /// Mutably borrow an editor tab by id.
72    ///
73    /// ### Arguments
74    /// - `tab_id`: The tab id to look up
75    ///
76    /// ### Returns
77    /// - `Some(&mut EditorTab)`: The matching editor tab
78    /// - `None`: If no editor tab has that id
79    fn editor_tab_mut(
80        &mut self,
81        tab_id: TabId,
82    ) -> Option<&mut crate::fulgur::editor_tab::EditorTab> {
83        self.tabs
84            .iter_mut()
85            .find_map(|tab| tab.as_editor_mut().filter(|editor| editor.id == tab_id))
86    }
