9/// Return the zero-based index of the last line in a buffer.
10///
11/// ### Arguments
12/// - `text`: The buffer text
13///
14/// ### Returns
15/// - `u32`: The last line index, clamped to `u32::MAX`
16fn last_line_index(text: &str) -> u32 {
17    u32::try_from(text.matches('\n').count()).unwrap_or(u32::MAX)
18}

20/// Replace a log buffer's content and snap the view to the last line.
21///
22/// `set_value` resets the scroll to the top, so it is always followed by
23/// `set_cursor_position` on the last line to scroll back to the bottom.
24///
25/// ### Arguments
26/// - `log_content`: The display input state to update
27/// - `display`: The full text to show
28/// - `window`: The active window
29/// - `cx`: The application context
30pub(super) fn write_log_to_bottom(
31    log_content: &Entity<EditorState>,
32    display: &str,
33    window: &mut Window,
34    cx: &mut Context<Fulgur>,
35) {
36    let last_line = last_line_index(display);
37    log_content.update(cx, |state, cx| {
38        state.set_value(display, window, cx);
39        state.set_cursor_position(
40            Position {
41                line: last_line,
42                character: 0,
43            },
44            window,
45            cx,
46        );
47    });
48}

50/// Append text to a log buffer with incremental edits and snap to the bottom.
51///
52/// ### Arguments
53/// - `log_content`: The display input state to update
54/// - `text`: The newly appended text (may be empty; only the snap happens)
55/// - `log_full`: Whether the line cap is lifted for this tab
56/// - `window`: The active window
57/// - `cx`: The application context
58///
59/// ### Returns
60/// - `bool`: Whether the line cap dropped lines from the front on this write
61pub(super) fn append_log_to_bottom(
62    log_content: &Entity<EditorState>,
63    text: &str,
64    log_full: bool,
65    window: &mut Window,
66    cx: &mut Context<Fulgur>,
67) -> bool {
68    log_content.update(cx, |state, cx| {
69        if !text.is_empty() {
70            let end = state.text().len();
71            state.set_selected_range(end..end, cx);
72            state.insert(text, window, cx);
73        }
74        let mut dropped = false;
75        if !log_full {
76            let newline_count = state.text().lines_len().saturating_sub(1);
77            if newline_count > LOG_LINE_CAP {
78                let cut = state.text().line_start_offset(newline_count - LOG_LINE_CAP);
79                state.set_selected_range(0..cut, cx);
80                state.replace("", window, cx);
81                dropped = true;
82            }
83        }
84        let last_line =
85            u32::try_from(state.text().lines_len().saturating_sub(1)).unwrap_or(u32::MAX);
86        state.set_cursor_position(
87            Position {
88                line: last_line,
89                character: 0,
90            },
91            window,
92            cx,
93        );
94        dropped
95    })
96}

98/// Build the read-only display `EditorState` for a log view buffer.
99///
100/// ### Arguments
101/// - `window`: The window the input is created in
102/// - `cx`: The input state context
103/// - `content`: The initial buffer content
104/// - `soft_wrap`: Whether soft wrapping is enabled
105///
106/// ### Returns
107/// - `EditorState`: A multi-line editor seeded with the content
108pub(super) fn make_log_input_state(
109    window: &mut Window,
110    cx: &mut Context<EditorState>,
111    content: &str,
112    soft_wrap: bool,
113) -> EditorState {
114    EditorState::new(window, cx)
115        .language("log")
116        .line_number(true)
117        .soft_wrap(soft_wrap)
118        .default_value(content.to_string())
119}
