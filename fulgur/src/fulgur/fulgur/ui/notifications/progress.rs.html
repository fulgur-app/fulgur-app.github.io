16/// Grace period before a long-running operation indicator becomes visible.
17const PROGRESS_NOTIFICATION_SHOW_AFTER: Duration = Duration::from_millis(300);
18
19/// Polling interval used to detect operation completion and dismiss the
20/// notification.
21const PROGRESS_NOTIFICATION_POLL_INTERVAL: Duration = Duration::from_millis(100);
22
23/// Stable name fragment used to construct unique notification ids.
24const PROGRESS_NOTIFICATION_KEY: &str = "fulgur-progress";
25
26/// Marker type used so progress notifications share a `TypeId` namespace
27/// distinct from app status toasts (success/info/error).
28struct ProgressKind;
29
30/// Monotonic id used as the `ElementId` integer for each notification so
31/// concurrent indicators do not replace each other.
32static NEXT_PROGRESS_ID: AtomicU64 = AtomicU64::new(1);
33
34/// Closure invoked once when the user clicks the Cancel button.
35pub type CancelCallback = Box<dyn FnOnce(&mut Window, &mut App) + 'static>;
36
37/// RAII handle to a long-running operation indicator.
38pub struct ProgressNotification {
39    completed: Arc<AtomicBool>,
40    cancelled: Arc<AtomicBool>,
41}
42
43impl ProgressNotification {
44    /// Cancel flag the worker thread can poll before publishing results.
45    ///
46    /// ### Returns
47    /// - `Arc<AtomicBool>`: Becomes `true` after the user clicks Cancel.
48    pub fn cancel_flag(&self) -> Arc<AtomicBool> {
49        Arc::clone(&self.cancelled)
50    }
51}
52
53impl Drop for ProgressNotification {
54    fn drop(&mut self) {
55        self.completed.store(true, Ordering::Release);
56    }
57}
58
59/// Start a long-running operation indicator.
60///
61/// ### Arguments
62/// - `window`: Target window for the notification.
63/// - `cx`: Application context.
64/// - `label`: Description shown next to the spinner.
65/// - `on_cancel`: Optional callback invoked once when the user clicks Cancel.
66///   Should remove any in-flight request entry so the eventual worker
67///   completion is discarded by the caller's freshness check.
68///
69/// ### Returns
70/// - `ProgressNotification`: RAII handle whose drop dismisses the
71///   notification.
72pub fn start_progress(
73    window: &mut Window,
74    cx: &mut App,
75    label: SharedString,
76    on_cancel: Option<CancelCallback>,
77) -> ProgressNotification {
78    let completed = Arc::new(AtomicBool::new(false));
79    let cancelled = Arc::new(AtomicBool::new(false));
80    let progress_id = NEXT_PROGRESS_ID.fetch_add(1, Ordering::Relaxed);
81
82    let completed_for_task = Arc::clone(&completed);
83    let cancelled_for_task = Arc::clone(&cancelled);
84    let cancel_slot: Rc<RefCell<Option<CancelCallback>>> = Rc::new(RefCell::new(on_cancel));
85
86    window
87        .spawn(cx, async move |async_cx| {
88            async_cx
89                .background_executor()
90                .timer(PROGRESS_NOTIFICATION_SHOW_AFTER)
91                .await;
92            if completed_for_task.load(Ordering::Acquire) {
93                return;
94            }
95
96            let element_id =
97                ElementId::NamedInteger(SharedString::from(PROGRESS_NOTIFICATION_KEY), progress_id);
98            let has_cancel = cancel_slot.borrow().is_some();
99            let label_for_content = label;
100            let cancelled_for_action = Arc::clone(&cancelled_for_task);
101            let completed_for_action = Arc::clone(&completed_for_task);
102            let cancel_slot_for_action = Rc::clone(&cancel_slot);
103
104            let entity: Option<WeakEntity<Notification>> = match async_cx.update(|window, cx| {
105                let mut note = Notification::new()
106                    .autohide(false)
107                    .id1::<ProgressKind>(element_id)
108                    .content({
109                        let label = label_for_content;
110                        move |_, _, _| {
111                            h_flex()
112                                .gap_2()
113                                .items_center()
114                                .child(Spinner::new().icon(CustomIcon::LoaderCircle).small())
115                                .child(div().text_sm().child(label.clone()))
116                                .into_any_element()
117                        }
118                    });
119
120                if has_cancel {
121                    let cancelled = cancelled_for_action;
122                    let completed = completed_for_action;
123                    let cancel_slot = cancel_slot_for_action;
124                    note = note.action(move |_, _, cx| {
125                        let cancelled = Arc::clone(&cancelled);
126                        let completed = Arc::clone(&completed);
127                        let cancel_slot = Rc::clone(&cancel_slot);
128                        Button::new("progress-cancel")
129                            .label("Cancel")
130                            .on_click(cx.listener(move |this, _, window, cx| {
131                                cancelled.store(true, Ordering::Release);
132                                completed.store(true, Ordering::Release);
133                                let maybe_cb = cancel_slot.borrow_mut().take();
134                                if let Some(cb) = maybe_cb {
135                                    cb(window, cx);
136                                }
137                                this.dismiss(window, cx);
138                            }))
139                    });
140                }
141
142                window.push_notification(note, cx);
143                window.notifications(cx).last().map(gpui::Entity::downgrade)
144            }) {
145                Ok(entity) => entity,
146                Err(_) => return,
147            };
148
149            loop {
150                async_cx
151                    .background_executor()
152                    .timer(PROGRESS_NOTIFICATION_POLL_INTERVAL)
153                    .await;
154                if completed_for_task.load(Ordering::Acquire) {
155                    let _ = async_cx.update(|window, cx| {
156                        if let Some(entity) = entity.as_ref().and_then(gpui::WeakEntity::upgrade) {
157                            entity.update(cx, |note, cx| note.dismiss(window, cx));
158                        }
159                    });
160                    return;
161                }
162            }
163        })
164        .detach();
165
166    ProgressNotification {
167        completed,
168        cancelled,
169    }
170}
