17/// The palette dialog's width, wide enough for a command label and its keybinding hint.
18const PALETTE_WIDTH: f32 = 560.;
19
20/// The tallest the command list grows before it scrolls.
21const PALETTE_MAX_LIST_HEIGHT: f32 = 420.;
22
23impl CommandPalette {
24    /// Open the dialog hosting the palette.
25    ///
26    /// ### Arguments
27    /// - `window`: The window to open the dialog in
28    /// - `cx`: The application context
29    pub(super) fn open_dialog(window: &mut Window, cx: &mut Context<Self>) {
30        let palette = cx.entity();
31        let focus_on_mount = Rc::new(Cell::new(true));
32        window.open_dialog(cx, move |dialog, _, _| {
33            let content_palette = palette.clone();
34            let close_palette = palette.clone();
35            let focus_on_mount = focus_on_mount.clone();
36            dialog
37                .close_button(false)
38                .w(px(PALETTE_WIDTH))
39                .p_0()
40                .on_close(move |_, _, cx| {
41                    close_palette.update(cx, |this, cx| {
42                        this.is_open = false;
43                        cx.notify();
44                    });
45                })
46                .content(move |content, window, cx| {
47                    let palette = content_palette.clone();
48                    if focus_on_mount.replace(false) {
49                        let palette = palette.clone();
50                        window.defer(cx, move |window, cx| {
51                            let focus_handle = palette.read(cx).state.read(cx).focus_handle(cx);
52                            window.focus(&focus_handle, cx);
53                        });
54                    }
55                    content.child(build_command(&palette, cx))
56                })
57        });
58    }
59}

61/// Build the palette element for the layout the palette entity currently holds.
62///
63/// ### Arguments
64/// - `palette`: The palette entity to read the layout and interaction state from
65/// - `cx`: The application context
66///
67/// ### Returns
68/// - `Command`: The configured palette element
69fn build_command(palette: &gpui_kit::Entity<CommandPalette>, cx: &mut App) -> Command {
70    let (state, layout) = {
71        let this = palette.read(cx);
72        (this.state.clone(), this.layout.clone())
73    };
74    let confirm_palette = palette.clone();
75
76    layout.into_iter().fold(
77        Command::new(&state)
78            .bordered(false)
79            .placeholder("Type a command...")
80            .max_h(px(PALETTE_MAX_LIST_HEIGHT))
81            .on_confirm(move |index, window, cx| {
82                confirm_palette.update(cx, |this, cx| {
83                    this.confirm(index, window, cx);
84                });
85            })
86            .footer(render_footer),
87        |command, (group, commands)| command.group(build_group(group, &commands)),
88    )
89}

91/// Build one titled section of the palette.
92///
93/// ### Arguments
94/// - `group`: The group being rendered, providing the heading
95/// - `commands`: The commands the group offers, in order
96///
97/// ### Returns
98/// - `CommandGroup`: The section, with one item per command
99fn build_group(group: PaletteGroup, commands: &[PaletteCommand]) -> CommandGroup {
100    commands.iter().fold(
101        CommandGroup::new().label(group.label()),
102        |section, command| section.item(build_item(*command)),
103    )
104}

106/// Build the palette row for a single command.
107///
108/// ### Arguments
109/// - `command`: The command to render
110///
111/// ### Returns
112/// - `CommandItem`: The row, searchable on its label and keywords
113fn build_item(command: PaletteCommand) -> CommandItem {
114    let hint = command.keybinding_hint();
115    CommandItem::new()
116        .label(command.label())
117        .keywords(command.keywords().iter().copied())
118        .child(move |window: &mut Window, cx: &mut App| {
119            render_row(command, hint.as_ref(), window, cx)
120        })
121}

123/// Render a command's icon, label and keybinding hint.
124///
125/// ### Arguments
126/// - `command`: The command the row stands for
127/// - `hint`: The keybinding hint to display, when the command has one
128/// - `window`: The window the keybinding is resolved against
129/// - `cx`: The application context
130///
131/// ### Returns
132/// - `impl IntoElement`: The row element
133fn render_row(
134    command: PaletteCommand,
135    hint: Option<&KeybindingHint>,
136    window: &Window,
137    cx: &App,
138) -> impl IntoElement + use<> {
139    let binding = hint
140        .and_then(|hint| Kbd::binding_for_action(hint.action.as_ref(), Some(hint.context), window));
141
142    h_flex()
143        .w_full()
144        .gap_2()
145        .items_center()
146        .child(
147            Icon::from(command.icon())
148                .size_4()
149                .text_color(cx.theme().muted_foreground),
150        )
151        .child(command.label())
152        .when_some(binding, |this, binding| this.child(binding.ml_auto()))
153}

155/// Render the palette's key hint footer.
156///
157/// ### Arguments
158/// - `_state`: The palette interaction state, unused by this footer
159/// - `_window`: The window the palette is rendered in
160/// - `cx`: The application context
161///
162/// ### Returns
163/// - `impl IntoElement`: The footer element
164fn render_footer(
165    _state: &gpui_kit::component::command::CommandState,
166    _window: &mut Window,
167    cx: &mut App,
168) -> impl IntoElement + use<> {
169    h_flex()
170        .gap_3()
171        .px_3()
172        .py_2()
173        .border_t_1()
174        .border_color(cx.theme().border)
175        .text_xs()
176        .text_color(cx.theme().muted_foreground)
177        .child("Up/Down to navigate")
178        .child("Enter to run")
179        .child("Escape to close")
180}
