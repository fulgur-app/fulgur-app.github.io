23/// Convert an HSLA color to OkLCH (Lightness, Chroma, Hue) components.
24///
25/// ### Arguments
26/// - `color`: The HSLA color to convert
27///
28/// ### Returns
29/// `(f32, f32, f32)` where L is 0..1, C is chroma, and H is hue in degrees 0..360.
30fn hsla_to_oklch(color: Hsla) -> (f32, f32, f32) {
31    let rgb = color.to_rgb();
32    let to_linear = |c: f32| -> f32 {
33        if c <= 0.04045 {
34            c / 12.92
35        } else {
36            ((c + 0.055) / 1.055).powf(2.4)
37        }
38    };
39    let lr = to_linear(rgb.r);
40    let lg = to_linear(rgb.g);
41    let lb = to_linear(rgb.b);
42    #[allow(clippy::excessive_precision)]
43    const LMS: [[f32; 3]; 3] = [
44        [0.4122214708, 0.5363325363, 0.0514459929],
45        [0.2119034982, 0.6806995451, 0.1073969566],
46        [0.0883024619, 0.2817188376, 0.6299787005],
47    ];
48    let l = LMS[0][0] * lr + LMS[0][1] * lg + LMS[0][2] * lb;
49    let m = LMS[1][0] * lr + LMS[1][1] * lg + LMS[1][2] * lb;
50    let s = LMS[2][0] * lr + LMS[2][1] * lg + LMS[2][2] * lb;
51    let l_ = l.cbrt();
52    let m_ = m.cbrt();
53    let s_ = s.cbrt();
54    #[allow(clippy::excessive_precision)]
55    const OKLAB: [[f32; 3]; 3] = [
56        [0.2104542553, 0.7936177850, -0.0040720468],
57        [1.9779984951, -2.4285922050, 0.4505937099],
58        [0.0259040371, 0.7827717662, -0.8086757660],
59    ];
60    let ok_l = OKLAB[0][0] * l_ + OKLAB[0][1] * m_ + OKLAB[0][2] * s_;
61    let ok_a = OKLAB[1][0] * l_ + OKLAB[1][1] * m_ + OKLAB[1][2] * s_;
62    let ok_b = OKLAB[2][0] * l_ + OKLAB[2][1] * m_ + OKLAB[2][2] * s_;
63    let chroma = (ok_a * ok_a + ok_b * ok_b).sqrt();
64    let hue = ok_b.atan2(ok_a).to_degrees();
65    let hue = if hue < 0.0 { hue + 360.0 } else { hue };
66    (ok_l, chroma, hue)
67}

69/// Convert OkLCH (Lightness, Chroma, Hue) components back to an HSLA color.
70///
71/// ### Arguments
72/// - `l`: Lightness in 0..1
73/// - `c`: Chroma (non-negative)
74/// - `h`: Hue in degrees 0..360
75///
76/// ### Returns
77/// `Hsla`: The resulting color
78fn oklch_to_hsla(l: f32, c: f32, h: f32) -> Hsla {
79    let h_rad = h.to_radians();
80    let ok_a = c * h_rad.cos();
81    let ok_b = c * h_rad.sin();
82    #[allow(clippy::excessive_precision)]
83    const INV_OKLAB: [[f32; 3]; 3] = [
84        [1.0000000000, 0.3963377774, 0.2158037573],
85        [1.0000000000, -0.1055613458, -0.0638541728],
86        [1.0000000000, -0.0894841775, -1.2914855480],
87    ];
88    let l_ = INV_OKLAB[0][0] * l + INV_OKLAB[0][1] * ok_a + INV_OKLAB[0][2] * ok_b;
89    let m_ = INV_OKLAB[1][0] * l + INV_OKLAB[1][1] * ok_a + INV_OKLAB[1][2] * ok_b;
90    let s_ = INV_OKLAB[2][0] * l + INV_OKLAB[2][1] * ok_a + INV_OKLAB[2][2] * ok_b;
91    let lms_l = l_ * l_ * l_;
92    let lms_m = m_ * m_ * m_;
93    let lms_s = s_ * s_ * s_;
94    #[allow(clippy::excessive_precision)]
95    const INV_LMS: [[f32; 3]; 3] = [
96        [4.0767416621, -3.3077115913, 0.2309699292],
97        [-1.2684380046, 2.6097574011, -0.3413193965],
98        [-0.0041960863, -0.7034186147, 1.7076147010],
99    ];
100    let lin_r = INV_LMS[0][0] * lms_l + INV_LMS[0][1] * lms_m + INV_LMS[0][2] * lms_s;
101    let lin_g = INV_LMS[1][0] * lms_l + INV_LMS[1][1] * lms_m + INV_LMS[1][2] * lms_s;
102    let lin_b = INV_LMS[2][0] * lms_l + INV_LMS[2][1] * lms_m + INV_LMS[2][2] * lms_s;
103    let from_linear = |c: f32| -> f32 {
104        let c = c.clamp(0.0, 1.0);
105        if c <= 0.0031308 {
106            c * 12.92
107        } else {
108            1.055 * c.powf(1.0 / 2.4) - 0.055
109        }
110    };
111    gpui::Rgba {
112        r: from_linear(lin_r),
113        g: from_linear(lin_g),
114        b: from_linear(lin_b),
115        a: 1.0,
116    }
117    .into()
118}

120/// Format an HSLA color as an OkLCH CSS string.
121///
122/// ### Arguments
123/// - `color`: The HSLA color to format
124///
125/// ### Returns
126/// `String`: The OkLCH CSS string (e.g. "oklch(0.63 0.26 29.2)")
127fn format_oklch(color: Hsla) -> String {
128    let (l, c, h) = hsla_to_oklch(color);
129    format!("oklch({:.2} {:.4} {:.1})", l, c, h)
130}
131
132/// Format an HSLA color as an HSLA CSS string.
133///
134/// ### Arguments
135/// - `color`: The HSLA color to format
136///
137/// ### Returns
138/// `String`: The HSLA CSS string (e.g. "hsla(210, 50%, 60%, 1.00)")
139fn format_hsla(color: Hsla) -> String {
140    format!(
141        "hsla({:.0}, {:.0}%, {:.0}%, {:.2})",
142        color.h * 360.0,
143        color.s * 100.0,
144        color.l * 100.0,
145        color.a,
146    )
147}

149/// Parse a hex color string into an HSLA color.
150///
151/// ### Arguments
152/// - `s`: A hex string, with or without the leading `#` (e.g. "#FF0000" or "FF0000")
153///
154/// ### Returns
155/// - `Option<Hsla>`: The parsed color
156/// - `None`: If the string is not a valid HSLA string
157fn parse_hex(s: &str) -> Option<Hsla> {
158    gpui_component::Colorize::parse_hex(s).ok()
159}

161/// Parse an OkLCH CSS string into an HSLA color.
162///
163/// ### Arguments
164/// - `s`: An OkLCH string in the form "oklch(L C H)" (e.g. "oklch(0.63 0.26 29.2)")
165///
166/// ### Returns
167/// - `Some<Hsla>`: The parsed color
168/// - `None`: If the string is not a valid HSLA string
169fn parse_oklch(s: &str) -> Option<Hsla> {
170    let inner = s.trim().strip_prefix("oklch(")?.strip_suffix(')')?;
171    let mut parts = inner.split_whitespace();
172    let l: f32 = parts.next()?.parse().ok()?;
173    let c: f32 = parts.next()?.parse().ok()?;
174    let h: f32 = parts.next()?.parse().ok()?;
175    Some(oklch_to_hsla(l, c, h))
176}
177
178/// Parse an HSLA CSS string into an HSLA color.
179///
180/// ### Arguments
181/// - `s`: An HSLA string in the form "hsla(H, S%, L%, A)" (e.g. "hsla(210, 50%, 60%, 1.00)")
182///
183/// ### Returns
184/// - `Option<Hsla>`: The parsed color
185/// - `None`: If the string is not a valid HSLA string
186fn parse_hsla(s: &str) -> Option<Hsla> {
187    let inner = s.trim().strip_prefix("hsla(")?.strip_suffix(')')?;
188    let parts: Vec<&str> = inner.split(',').collect();
189    if parts.len() != 4 {
190        return None;
191    }
192    let h: f32 = parts[0].trim().parse().ok()?;
193    let s: f32 = parts[1].trim().trim_end_matches('%').parse().ok()?;
194    let l: f32 = parts[2].trim().trim_end_matches('%').parse().ok()?;
195    let a: f32 = parts[3].trim().parse().ok()?;
196    Some(hsla(h / 360.0, s / 100.0, l / 100.0, a))
197}

219impl ColorPickerBarState {
220    /// Create a new ColorPickerBarState.
221    ///
222    /// ### Arguments
223    /// - `window`: The window context
224    /// - `cx`: The application context
225    ///
226    /// ### Returns
227    /// - `Self`: A new ColorPickerBarState initialized to white
228    pub fn new(window: &mut Window, cx: &mut Context<Fulgur>) -> Self {
229        let color_picker_state =
230            cx.new(|cx| ColorPickerState::new(window, cx).default_value(gpui::white()));
231        let initial = gpui::white();
232        let initial_hex = gpui_component::Colorize::to_hex(&initial).to_string();
233        let initial_oklch = format_oklch(initial);
234        let initial_hsla = format_hsla(initial);
235        let hex_input = cx.new(|cx| InputState::new(window, cx).default_value(initial_hex.clone()));
236        let oklch_input =
237            cx.new(|cx| InputState::new(window, cx).default_value(initial_oklch.clone()));
238        let hsla_input =
239            cx.new(|cx| InputState::new(window, cx).default_value(initial_hsla.clone()));
240        let _color_picker_subscription = cx.subscribe_in(
241            &color_picker_state,
242            window,
243            |this: &mut Fulgur, _, event: &ColorPickerEvent, window, cx| {
244                if let ColorPickerEvent::Change(Some(color)) = event {
245                    let color = *color;
246                    let hex = gpui_component::Colorize::to_hex(&color);
247                    let oklch = format_oklch(color);
248                    let hsla_str = format_hsla(color);
249                    this.color_picker_bar_state.cached_hex = hex.to_string();
250                    this.color_picker_bar_state.cached_oklch = oklch.clone();
251                    this.color_picker_bar_state.cached_hsla = hsla_str.clone();
252                    this.color_picker_bar_state
253                        .hex_input
254                        .update(cx, |state, cx| {
255                            state.set_value(hex, window, cx);
256                        });
257                    this.color_picker_bar_state
258                        .oklch_input
259                        .update(cx, |state, cx| {
260                            state.set_value(oklch, window, cx);
261                        });
262                    this.color_picker_bar_state
263                        .hsla_input
264                        .update(cx, |state, cx| {
265                            state.set_value(hsla_str, window, cx);
266                        });
267                }
268                cx.notify();
269            },
270        );
271        // On Change: update the color picker preview only (no cross-input sync, which
272        // would loop since InputState::set_value always emits InputEvent::Change).
273        // On PressEnter / Blur: parse and sync all three inputs + the color picker.
274        let _hex_input_subscription = cx.subscribe_in(
275            &hex_input,
276            window,
277            |this: &mut Fulgur, _, event: &InputEvent, window, cx| {
278                let value = this.color_picker_bar_state.hex_input.read(cx).value();
279                let Some(color) = parse_hex(&value) else {
280                    return;
281                };
282                match event {
283                    InputEvent::Change => {
284                        this.color_picker_bar_state
285                            .color_picker_state
286                            .update(cx, |state, cx| {
287                                state.set_value(color, window, cx);
288                            });
289                    }
290                    InputEvent::PressEnter { .. } | InputEvent::Blur => {
291                        this.color_picker_bar_state
292                            .color_picker_state
293                            .update(cx, |state, cx| {
294                                state.set_value(color, window, cx);
295                            });
296                        this.color_picker_bar_state
297                            .oklch_input
298                            .update(cx, |state, cx| {
299                                state.set_value(format_oklch(color), window, cx);
300                            });
301                        this.color_picker_bar_state
302                            .hsla_input
303                            .update(cx, |state, cx| {
304                                state.set_value(format_hsla(color), window, cx);
305                            });
306                    }
307                    _ => {}
308                }
309            },
310        );
311        let _oklch_input_subscription = cx.subscribe_in(
312            &oklch_input,
313            window,
314            |this: &mut Fulgur, _, event: &InputEvent, window, cx| {
315                let value = this.color_picker_bar_state.oklch_input.read(cx).value();
316                let Some(color) = parse_oklch(&value) else {
317                    return;
318                };
319                match event {
320                    InputEvent::Change => {
321                        this.color_picker_bar_state
322                            .color_picker_state
323                            .update(cx, |state, cx| {
324                                state.set_value(color, window, cx);
325                            });
326                    }
327                    InputEvent::PressEnter { .. } | InputEvent::Blur => {
328                        this.color_picker_bar_state
329                            .color_picker_state
330                            .update(cx, |state, cx| {
331                                state.set_value(color, window, cx);
332                            });
333                        this.color_picker_bar_state
334                            .hex_input
335                            .update(cx, |state, cx| {
336                                state.set_value(
337                                    gpui_component::Colorize::to_hex(&color),
338                                    window,
339                                    cx,
340                                );
341                            });
342                        this.color_picker_bar_state
343                            .hsla_input
344                            .update(cx, |state, cx| {
345                                state.set_value(format_hsla(color), window, cx);
346                            });
347                    }
348                    _ => {}
349                }
350            },
351        );
352        let _hsla_input_subscription = cx.subscribe_in(
353            &hsla_input,
354            window,
355            |this: &mut Fulgur, _, event: &InputEvent, window, cx| {
356                let value = this.color_picker_bar_state.hsla_input.read(cx).value();
357                let Some(color) = parse_hsla(&value) else {
358                    return;
359                };
360                match event {
361                    InputEvent::Change => {
362                        this.color_picker_bar_state
363                            .color_picker_state
364                            .update(cx, |state, cx| {
365                                state.set_value(color, window, cx);
366                            });
367                    }
368                    InputEvent::PressEnter { .. } | InputEvent::Blur => {
369                        this.color_picker_bar_state
370                            .color_picker_state
371                            .update(cx, |state, cx| {
372                                state.set_value(color, window, cx);
373                            });
374                        this.color_picker_bar_state
375                            .hex_input
376                            .update(cx, |state, cx| {
377                                state.set_value(
378                                    gpui_component::Colorize::to_hex(&color),
379                                    window,
380                                    cx,
381                                );
382                            });
383                        this.color_picker_bar_state
384                            .oklch_input
385                            .update(cx, |state, cx| {
386                                state.set_value(format_oklch(color), window, cx);
387                            });
388                    }
389                    _ => {}
390                }

393        Self {
394            color_picker_state,

400            cached_oklch: initial_oklch,
401            cached_hsla: initial_hsla,
402            _color_picker_subscription,
403            _hex_input_subscription,
404            _oklch_input_subscription,

406        }

411    /// Insert a value at the cursor position in the active editor tab,
412    /// replacing the current selection if any.
413    ///
414    /// ### Arguments
415    /// - `value`: The string to insert
416    /// - `window`: The window context
417    /// - `cx`: The application context
418    pub fn insert_color_value(
419        &mut self,
420        value: String,
421        window: &mut Window,
422        cx: &mut Context<Self>,
423    ) {
424        if let Some(index) = self.active_tab_index
425            && let Some(crate::fulgur::tab::Tab::Editor(editor_tab)) = self.tabs.get_mut(index)
426        {
427            editor_tab.content.update(cx, |input_state, cx| {
428                let selection = input_state.selected_text_range(true, window, cx);
429                if selection.is_some() {
430                    input_state.replace(value, window, cx);
431                } else {
432                    input_state.insert(value, window, cx);
433                }
434                cx.notify();
435            });
436        }
437    }
