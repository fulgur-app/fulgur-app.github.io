18/// Convert an HSLA color to OkLCH (Lightness, Chroma, Hue) components.
19///
20/// ### Arguments
21/// - `color`: The HSLA color to convert
22///
23/// ### Returns
24/// `(f32, f32, f32)` where L is 0..1, C is chroma, and H is hue in degrees 0..360.
25fn hsla_to_oklch(color: Hsla) -> (f32, f32, f32) {
26    let rgb = color.to_rgb();
27    let to_linear = |c: f32| -> f32 {
28        if c <= 0.04045 {
29            c / 12.92
30        } else {
31            ((c + 0.055) / 1.055).powf(2.4)
32        }
33    };
34    let lr = to_linear(rgb.r);
35    let lg = to_linear(rgb.g);
36    let lb = to_linear(rgb.b);
37    #[allow(clippy::excessive_precision)]
38    const LMS: [[f32; 3]; 3] = [
39        [0.4122214708, 0.5363325363, 0.0514459929],
40        [0.2119034982, 0.6806995451, 0.1073969566],
41        [0.0883024619, 0.2817188376, 0.6299787005],
42    ];
43    let l = LMS[0][0] * lr + LMS[0][1] * lg + LMS[0][2] * lb;
44    let m = LMS[1][0] * lr + LMS[1][1] * lg + LMS[1][2] * lb;
45    let s = LMS[2][0] * lr + LMS[2][1] * lg + LMS[2][2] * lb;
46    let l_ = l.cbrt();
47    let m_ = m.cbrt();
48    let s_ = s.cbrt();
49    #[allow(clippy::excessive_precision)]
50    const OKLAB: [[f32; 3]; 3] = [
51        [0.2104542553, 0.7936177850, -0.0040720468],
52        [1.9779984951, -2.4285922050, 0.4505937099],
53        [0.0259040371, 0.7827717662, -0.8086757660],
54    ];
55    let ok_l = OKLAB[0][0] * l_ + OKLAB[0][1] * m_ + OKLAB[0][2] * s_;
56    let ok_a = OKLAB[1][0] * l_ + OKLAB[1][1] * m_ + OKLAB[1][2] * s_;
57    let ok_b = OKLAB[2][0] * l_ + OKLAB[2][1] * m_ + OKLAB[2][2] * s_;
58    let chroma = (ok_a * ok_a + ok_b * ok_b).sqrt();
59    let hue = ok_b.atan2(ok_a).to_degrees();
60    let hue = if hue < 0.0 { hue + 360.0 } else { hue };
61    (ok_l, chroma, hue)
62}

64/// Convert OkLCH (Lightness, Chroma, Hue) components back to an HSLA color.
65///
66/// ### Arguments
67/// - `l`: Lightness in 0..1
68/// - `c`: Chroma (non-negative)
69/// - `h`: Hue in degrees 0..360
70///
71/// ### Returns
72/// `Hsla`: The resulting color
73fn oklch_to_hsla(l: f32, c: f32, h: f32) -> Hsla {
74    let h_rad = h.to_radians();
75    let ok_a = c * h_rad.cos();
76    let ok_b = c * h_rad.sin();
77    #[allow(clippy::excessive_precision)]
78    const INV_OKLAB: [[f32; 3]; 3] = [
79        [1.0000000000, 0.3963377774, 0.2158037573],
80        [1.0000000000, -0.1055613458, -0.0638541728],
81        [1.0000000000, -0.0894841775, -1.2914855480],
82    ];
83    let l_ = INV_OKLAB[0][0] * l + INV_OKLAB[0][1] * ok_a + INV_OKLAB[0][2] * ok_b;
84    let m_ = INV_OKLAB[1][0] * l + INV_OKLAB[1][1] * ok_a + INV_OKLAB[1][2] * ok_b;
85    let s_ = INV_OKLAB[2][0] * l + INV_OKLAB[2][1] * ok_a + INV_OKLAB[2][2] * ok_b;
86    let lms_l = l_ * l_ * l_;
87    let lms_m = m_ * m_ * m_;
88    let lms_s = s_ * s_ * s_;
89    #[allow(clippy::excessive_precision)]
90    const INV_LMS: [[f32; 3]; 3] = [
91        [4.0767416621, -3.3077115913, 0.2309699292],
92        [-1.2684380046, 2.6097574011, -0.3413193965],
93        [-0.0041960863, -0.7034186147, 1.7076147010],
94    ];
95    let lin_r = INV_LMS[0][0] * lms_l + INV_LMS[0][1] * lms_m + INV_LMS[0][2] * lms_s;
96    let lin_g = INV_LMS[1][0] * lms_l + INV_LMS[1][1] * lms_m + INV_LMS[1][2] * lms_s;
97    let lin_b = INV_LMS[2][0] * lms_l + INV_LMS[2][1] * lms_m + INV_LMS[2][2] * lms_s;
98    let from_linear = |c: f32| -> f32 {
99        let c = c.clamp(0.0, 1.0);
100        if c <= 0.0031308 {
101            c * 12.92
102        } else {
103            1.055 * c.powf(1.0 / 2.4) - 0.055
104        }
105    };
106    Rgba {
107        r: from_linear(lin_r),
108        g: from_linear(lin_g),
109        b: from_linear(lin_b),
110        a: 1.0,
111    }
112    .into()
113}

115/// Format an HSLA color as an OkLCH CSS string.
116///
117/// ### Arguments
118/// - `color`: The HSLA color to format
119///
120/// ### Returns
121/// `String`: The OkLCH CSS string (e.g. "oklch(0.63 0.26 29.2)")
122fn format_oklch(color: Hsla) -> String {
123    let (l, c, h) = hsla_to_oklch(color);
124    format!("oklch({:.2} {:.4} {:.1})", l, c, h)
125}
126
127/// Format an HSLA color as an HSLA CSS string.
128///
129/// ### Arguments
130/// - `color`: The HSLA color to format
131///
132/// ### Returns
133/// `String`: The HSLA CSS string (e.g. "hsla(210, 50%, 60%, 1.00)")
134fn format_hsla(color: Hsla) -> String {
135    format!(
136        "hsla({:.0}, {:.0}%, {:.0}%, {:.2})",
137        color.h * 360.0,
138        color.s * 100.0,
139        color.l * 100.0,
140        color.a,
141    )
142}

144/// Parse a hex color string into an HSLA color.
145///
146/// ### Arguments
147/// - `s`: A hex string, with or without the leading `#` (e.g. "#FF0000" or "FF0000")
148///
149/// ### Returns
150/// - `Option<Hsla>`: The parsed color
151/// - `None`: If the string is not a valid HSLA string
152fn parse_hex(s: &str) -> Option<Hsla> {
153    gpui_component::Colorize::parse_hex(s).ok()
154}
155
156/// Parse an OkLCH CSS string into an HSLA color.
157///
158/// ### Arguments
159/// - `s`: An OkLCH string in the form "oklch(L C H)" (e.g. "oklch(0.63 0.26 29.2)")
160///
161/// ### Returns
162/// - `Some<Hsla>`: The parsed color
163/// - `None`: If the string is not a valid HSLA string
164fn parse_oklch(s: &str) -> Option<Hsla> {
165    let inner = s.trim().strip_prefix("oklch(")?.strip_suffix(')')?;
166    let mut parts = inner.split_whitespace();
167    let l: f32 = parts.next()?.parse().ok()?;
168    let c: f32 = parts.next()?.parse().ok()?;
169    let h: f32 = parts.next()?.parse().ok()?;
170    Some(oklch_to_hsla(l, c, h))
171}
172
173/// Parse an HSLA CSS string into an HSLA color.
174///
175/// ### Arguments
176/// - `s`: An HSLA string in the form "hsla(H, S%, L%, A)" (e.g. "hsla(210, 50%, 60%, 1.00)")
177///
178/// ### Returns
179/// - `Option<Hsla>`: The parsed color
180/// - `None`: If the string is not a valid HSLA string
181fn parse_hsla(s: &str) -> Option<Hsla> {
182    let inner = s.trim().strip_prefix("hsla(")?.strip_suffix(')')?;
183    let parts: Vec<&str> = inner.split(',').collect();
184    if parts.len() != 4 {
185        return None;
186    }
187    let h: f32 = parts[0].trim().parse().ok()?;
188    let s: f32 = parts[1].trim().trim_end_matches('%').parse().ok()?;
189    let l: f32 = parts[2].trim().trim_end_matches('%').parse().ok()?;
190    let a: f32 = parts[3].trim().parse().ok()?;
191    Some(hsla(h / 360.0, s / 100.0, l / 100.0, a))
192}

194/// State for the color picker bar.
195///
196/// Groups the ColorPickerState entity, input states for each color format,
197/// and the subscriptions for change events.
198pub struct ColorPickerBarState {
199    pub color_picker_state: Entity<ColorPickerState>,
200    pub show_color_picker: bool,
201    pub hex_input: Entity<InputState>,
202    pub oklch_input: Entity<InputState>,
203    pub hsla_input: Entity<InputState>,
204    _color_picker_subscription: Subscription,
205    _hex_input_subscription: Subscription,
206    _oklch_input_subscription: Subscription,
207    _hsla_input_subscription: Subscription,
208}

210impl ColorPickerBarState {
211    /// Create a new ColorPickerBarState.
212    ///
213    /// ### Arguments
214    /// - `window`: The window context
215    /// - `cx`: The application context
216    ///
217    /// ### Returns
218    /// - `Self`: A new ColorPickerBarState initialized to white
219    pub fn new(window: &mut Window, cx: &mut Context<Fulgur>) -> Self {
220        let color_picker_state =
221            cx.new(|cx| ColorPickerState::new(window, cx).default_value(gpui::white()));
222        let initial = gpui::white();
223        let hex_input = cx.new(|cx| {
224            InputState::new(window, cx).default_value(gpui_component::Colorize::to_hex(&initial))
225        });
226        let oklch_input =
227            cx.new(|cx| InputState::new(window, cx).default_value(format_oklch(initial)));
228        let hsla_input =
229            cx.new(|cx| InputState::new(window, cx).default_value(format_hsla(initial)));
230        let _color_picker_subscription = cx.subscribe_in(
231            &color_picker_state,
232            window,
233            |this: &mut Fulgur, _, event: &ColorPickerEvent, window, cx| {
234                if let ColorPickerEvent::Change(Some(color)) = event {
235                    let color = *color;
236                    let hex = gpui_component::Colorize::to_hex(&color);
237                    let oklch = format_oklch(color);
238                    let hsla_str = format_hsla(color);
239                    this.color_picker_bar_state
240                        .hex_input
241                        .update(cx, |state, cx| {
242                            state.set_value(hex, window, cx);
243                        });
244                    this.color_picker_bar_state
245                        .oklch_input
246                        .update(cx, |state, cx| {
247                            state.set_value(oklch, window, cx);
248                        });
249                    this.color_picker_bar_state
250                        .hsla_input
251                        .update(cx, |state, cx| {
252                            state.set_value(hsla_str, window, cx);
253                        });
254                }
255                cx.notify();
256            },
257        );
258        // On Change: update the color picker preview only (no cross-input sync, which
259        // would loop since InputState::set_value always emits InputEvent::Change).
260        // On PressEnter / Blur: parse and sync all three inputs + the color picker.
261        let _hex_input_subscription = cx.subscribe_in(
262            &hex_input,
263            window,
264            |this: &mut Fulgur, _, event: &InputEvent, window, cx| {
265                let value = this.color_picker_bar_state.hex_input.read(cx).value();
266                let Some(color) = parse_hex(&value) else {
267                    return;
268                };
269                match event {
270                    InputEvent::Change => {
271                        this.color_picker_bar_state
272                            .color_picker_state
273                            .update(cx, |state, cx| {
274                                state.set_value(color, window, cx);
275                            });
276                    }
277                    InputEvent::PressEnter { .. } | InputEvent::Blur => {
278                        this.color_picker_bar_state
279                            .color_picker_state
280                            .update(cx, |state, cx| {
281                                state.set_value(color, window, cx);
282                            });
283                        this.color_picker_bar_state
284                            .oklch_input
285                            .update(cx, |state, cx| {
286                                state.set_value(format_oklch(color), window, cx);
287                            });
288                        this.color_picker_bar_state
289                            .hsla_input
290                            .update(cx, |state, cx| {
291                                state.set_value(format_hsla(color), window, cx);
292                            });
293                    }
294                    _ => {}
295                }
296            },
297        );
298        let _oklch_input_subscription = cx.subscribe_in(
299            &oklch_input,
300            window,
301            |this: &mut Fulgur, _, event: &InputEvent, window, cx| {
302                let value = this.color_picker_bar_state.oklch_input.read(cx).value();
303                let Some(color) = parse_oklch(&value) else {
304                    return;
305                };
306                match event {
307                    InputEvent::Change => {
308                        this.color_picker_bar_state
309                            .color_picker_state
310                            .update(cx, |state, cx| {
311                                state.set_value(color, window, cx);
312                            });
313                    }
314                    InputEvent::PressEnter { .. } | InputEvent::Blur => {
315                        this.color_picker_bar_state
316                            .color_picker_state
317                            .update(cx, |state, cx| {
318                                state.set_value(color, window, cx);
319                            });
320                        this.color_picker_bar_state
321                            .hex_input
322                            .update(cx, |state, cx| {
323                                state.set_value(
324                                    gpui_component::Colorize::to_hex(&color),
325                                    window,
326                                    cx,
327                                );
328                            });
329                        this.color_picker_bar_state
330                            .hsla_input
331                            .update(cx, |state, cx| {
332                                state.set_value(format_hsla(color), window, cx);
333                            });
334                    }
335                    _ => {}
336                }
337            },
338        );
339        let _hsla_input_subscription = cx.subscribe_in(
340            &hsla_input,
341            window,
342            |this: &mut Fulgur, _, event: &InputEvent, window, cx| {
343                let value = this.color_picker_bar_state.hsla_input.read(cx).value();
344                let Some(color) = parse_hsla(&value) else {
345                    return;
346                };
347                match event {
348                    InputEvent::Change => {
349                        this.color_picker_bar_state
350                            .color_picker_state
351                            .update(cx, |state, cx| {
352                                state.set_value(color, window, cx);
353                            });
354                    }
355                    InputEvent::PressEnter { .. } | InputEvent::Blur => {
356                        this.color_picker_bar_state
357                            .color_picker_state
358                            .update(cx, |state, cx| {
359                                state.set_value(color, window, cx);
360                            });
361                        this.color_picker_bar_state
362                            .hex_input
363                            .update(cx, |state, cx| {
364                                state.set_value(
365                                    gpui_component::Colorize::to_hex(&color),
366                                    window,
367                                    cx,
368                                );
369                            });
370                        this.color_picker_bar_state
371                            .oklch_input
372                            .update(cx, |state, cx| {
373                                state.set_value(format_oklch(color), window, cx);
374                            });
375                    }
376                    _ => {}
377                }
378            },
379        );
380        Self {
381            color_picker_state,
382            show_color_picker: false,
383            hex_input,
384            oklch_input,
385            hsla_input,
386            _color_picker_subscription,
387            _hex_input_subscription,
388            _oklch_input_subscription,
389            _hsla_input_subscription,
390        }
391    }

395    /// Insert a value at the cursor position in the active editor tab,
396    /// replacing the current selection if any.
397    ///
398    /// ### Arguments
399    /// - `value`: The string to insert
400    /// - `window`: The window context
401    /// - `cx`: The application context
402    pub fn insert_color_value(
403        &mut self,
404        value: String,
405        window: &mut Window,
406        cx: &mut Context<Self>,
407    ) {
408        if let Some(index) = self.active_tab_index
409            && let Some(crate::fulgur::tab::Tab::Editor(editor_tab)) = self.tabs.get_mut(index)
410        {
411            editor_tab.content.update(cx, |input_state, cx| {
412                let selection = input_state.selected_text_range(true, window, cx);
413                if selection.is_some() {
414                    input_state.replace(value, window, cx);
415                } else {
416                    input_state.insert(value, window, cx);
417                }
418                cx.notify();
419            });
420        }
421    }

717        let (l, c, h) = hsla_to_oklch(original);
718        let recovered = oklch_to_hsla(l, c, h);

720        let rec_rgb = recovered.to_rgb();
721        assert!(
722            (orig_rgb.r - rec_rgb.r).abs() < 0.01,
723            "red channel mismatch"
724        );
725        assert!(
726            (orig_rgb.g - rec_rgb.g).abs() < 0.01,
727            "green channel mismatch"
728        );
729        assert!(
730            (orig_rgb.b - rec_rgb.b).abs() < 0.01,
731            "blue channel mismatch"
732        );
