11/// Render the window control cluster sitting at the end of the unified row
12///
13/// ### Arguments
14/// - `window`: The window being rendered, queried for the controls it supports
15/// - `cx`: The title bar context
16///
17/// ### Returns
18/// - `AnyElement`: The rendered cluster, empty when the window manager supports nothing
19pub(super) fn render_window_controls(
20    window: &mut Window,
21    cx: &mut Context<CustomTitleBar>,
22) -> AnyElement {
23    // A tiling compositor may honor neither minimize nor maximize; close is always ours.
24    let supported = window.window_controls();
25    let mut controls = h_flex()
26        .id("unified-window-controls")
27        .flex_shrink_0()
28        .h_full()
29        .items_center()
30        .border_b_1()
31        .border_color(cx.theme().border);
32    if supported.minimize {
33        controls = controls.child(render_control(Control::Minimize, cx));
34    }
35    if supported.maximize {
36        let control = if window.is_maximized() {
37            Control::Restore
38        } else {
39            Control::Maximize
40        };
41        controls = controls.child(render_control(control, cx));
42    }
43    controls
44        .child(render_control(Control::Close, cx))
45        .into_any_element()
46}

48/// One of the window control buttons
49#[derive(Clone, Copy)]
50enum Control {
51    Minimize,
52    Maximize,
53    Restore,
54    Close,
55}
56
57impl Control {
58    /// The stable element identifier of this button
59    ///
60    /// ### Returns
61    /// - `&'static str`: The element id
62    fn id(self) -> &'static str {
63        match self {
64            Self::Minimize => "window-minimize",
65            Self::Maximize => "window-maximize",
66            Self::Restore => "window-restore",
67            Self::Close => "window-close",
68        }
69    }
70
71    /// The icon drawn inside this button
72    ///
73    /// ### Returns
74    /// - `CustomIcon`: The icon of the button
75    fn icon(self) -> CustomIcon {
76        match self {
77            Self::Minimize => CustomIcon::WindowMinimize,
78            Self::Maximize => CustomIcon::WindowMaximize,
79            Self::Restore => CustomIcon::WindowRestore,
80            Self::Close => CustomIcon::WindowClose,
81        }
82    }
83
84    /// The area the platform window has to associate with this button
85    ///
86    /// ### Returns
87    /// - `WindowControlArea`: The matching control area
88    fn area(self) -> gpui::WindowControlArea {
89        match self {
90            Self::Minimize => gpui::WindowControlArea::Min,
91            Self::Maximize | Self::Restore => gpui::WindowControlArea::Max,
92            Self::Close => gpui::WindowControlArea::Close,
93        }
94    }
95
96    /// The hover and pressed colors of this button
97    ///
98    /// ### Arguments
99    /// - `cx`: The application context, read for the active theme
100    ///
101    /// ### Returns
102    /// - `(Hsla, Hsla, Hsla)`: The hovered background, the pressed background and the text color
103    fn colors(self, cx: &App) -> (Hsla, Hsla, Hsla) {
104        if matches!(self, Self::Close) {
105            (
106                cx.theme().danger,
107                cx.theme().danger_active,
108                cx.theme().danger_foreground,
109            )
110        } else {
111            (
112                cx.theme().secondary_hover,
113                cx.theme().secondary_active,
114                cx.theme().secondary_foreground,
115            )
116        }
117    }
118}
119
120/// Render a single window control button
121///
122/// ### Arguments
123/// - `control`: The button to render
124/// - `cx`: The title bar context
125///
126/// ### Returns
127/// - `AnyElement`: The rendered button
128fn render_control(control: Control, cx: &mut Context<CustomTitleBar>) -> AnyElement {
129    let (hovered_bg, pressed_bg, emphasis) = control.colors(cx);
130    let button = div()
131        .id(control.id())
132        .flex()
133        .w(TITLE_BAR_HEIGHT)
134        .h_full()
135        .flex_shrink_0()
136        .justify_center()
137        .items_center()
138        .text_color(cx.theme().foreground)
139        .hover(|style| style.bg(hovered_bg).text_color(emphasis))
140        .active(|style| style.bg(pressed_bg).text_color(emphasis))
141        .child(control.icon().icon().small());
142    #[cfg(target_os = "windows")]
143    {
144        button
145            .window_control_area(control.area())
146            .into_any_element()
147    }
148    #[cfg(not(target_os = "windows"))]
149    {
150        let _ = control.area();
151        button
152            .on_mouse_down(gpui::MouseButton::Left, |_, window, cx| {
153                window.prevent_default();
154                cx.stop_propagation();
155            })
156            .on_click(move |_, window, cx| {
157                cx.stop_propagation();
158                match control {
159                    Control::Minimize => window.minimize_window(),
160                    Control::Maximize | Control::Restore => window.zoom_window(),
161                    Control::Close => window.remove_window(),
162                }
163            })
164            .into_any_element()
165    }
166}
