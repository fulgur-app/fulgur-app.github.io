57    /// Compute the status bar label strings from the active tab's cursor, language, and encoding
58    ///
59    /// ### Arguments
60    /// - `fulgur`: The owning window to derive the labels from
61    /// - `cx`: The application context
62    ///
63    /// ### Returns
64    /// - `StatusBarLabels`: The line/column, language, and encoding labels
65    pub(super) fn compute_labels(fulgur: &Fulgur, cx: &App) -> StatusBarLabels {
66        let active_tab_index = fulgur.active_tab_index();
67        let (cursor_pos, language, encoding) = match active_tab_index {
68            Some(index) => {
69                if let Some(editor_tab) = fulgur.tabs[index].as_editor() {
70                    let cursor = editor_tab.content.read(cx).cursor_position();
71                    let enc = editor_tab.encoding.clone();
72                    (cursor, Some(editor_tab.language), enc)
73                } else {
74                    (
75                        Position::default(),
76                        Some(SupportedLanguage::Plain),
77                        EMPTY.to_string(),
78                    )
79                }
80            }
81            None => (Position::default(), None, String::new()),
82        };
83
84        let language_label = match &language {
85            Some(lang) => pretty_name(lang),
86            None => EMPTY.to_string(),
87        };
88        let encoding_label = match active_tab_index {
89            Some(_) => encoding,
90            None => UTF_8.to_string(),
91        };
92
93        StatusBarLabels {
94            line_col: format!(
95                "Ln {}, Col {}",
96                cursor_pos.line + 1,
97                cursor_pos.character + 1
98            ),
99            language_label,
100            encoding_label,
101        }
102    }

104    /// Aggregate sync button state across all active profiles.
105    ///
106    /// Priority order: Connected beats Connecting beats Disconnected. The spinner is
107    /// shown once the earliest connecting-since timestamp has exceeded
108    /// `CONNECTING_SPINNER_DELAY`.
109    ///
110    /// ### Arguments
111    /// - `profiles`: The configured server profiles to aggregate over
112    /// - `cx`: The application context
113    ///
114    /// ### Returns
115    /// - `(SyncButtonState, bool)`: The aggregated state and whether to show the spinner.
116    pub(super) fn sync_button_state(
117        profiles: &[ServerProfile],
118        cx: &App,
119    ) -> (SyncButtonState, bool) {
120        let shared = Fulgur::shared_state(cx);
121        let sync_states = shared.sync_states.read();
122
123        let mut any_connected = false;
124        let mut any_connecting = false;
125        let mut earliest_connecting_since: Option<Instant> = None;
126
127        for profile in profiles.iter().filter(|p| p.is_active) {
128            let Some(state) = sync_states.get(&profile.id) else {
129                continue;
130            };
131            match *state.connection_status.lock() {
132                SynchronizationStatus::Connected => any_connected = true,
133                SynchronizationStatus::Connecting => {
134                    any_connecting = true;
135                    if let Some(since) = *state.connecting_since.lock() {
136                        earliest_connecting_since = Some(match earliest_connecting_since {
137                            None => since,
138                            Some(existing) if since < existing => since,
139                            Some(existing) => existing,
140                        });
141                    }
142                }
143                _ => {}
144            }
145        }
146
147        if any_connected {
148            (SyncButtonState::Connected, false)
149        } else if any_connecting {
150            let show = earliest_connecting_since
151                .is_some_and(|since| since.elapsed() >= CONNECTING_SPINNER_DELAY);
152            (SyncButtonState::Connecting, show)
153        } else {
154            (SyncButtonState::Disconnected, false)
155        }
156    }

158    /// Collect per-profile tooltip data for all active profiles.
159    ///
160    /// ### Arguments
161    /// - `profiles`: The configured server profiles to collect tooltip data for
162    /// - `cx`: The application context
163    ///
164    /// ### Returns
165    /// - `Vec<(String, String)>`: Profile name and its human-readable status label.
166    pub(super) fn sync_profiles_tooltip_data(
167        profiles: &[ServerProfile],
168        cx: &App,
169    ) -> Vec<(String, String)> {
170        let shared = Fulgur::shared_state(cx);
171        let sync_states = shared.sync_states.read();
172
173        profiles
174            .iter()
175            .filter(|p| p.is_active)
176            .map(|profile| {
177                let state = sync_states.get(&profile.id);
178                let label = state.map_or("Inactive", |s| s.connection_status.lock().label());
179                let device_name = state.and_then(|s| s.device_name.lock().clone());
180                let name = match device_name {
181                    Some(device) if !device.is_empty() => {
182                        format!("{} @ {}", device, profile.name)
183                    }
184                    _ => profile.name.clone(),
185                };
186                (name, label.to_string())
187            })
188            .collect()
189    }
190}

192impl Fulgur {
193    /// Dispatch a status bar event to the matching window-level handler
194    ///
195    /// ### Arguments
196    /// - `event`: The status bar event to handle
197    /// - `window`: The window context
198    /// - `cx`: The application context
199    pub(crate) fn on_status_bar_event(
200        &mut self,
201        event: StatusBarEvent,
202        window: &mut Window,
203        cx: &mut Context<Self>,
204    ) {
205        match event {
206            StatusBarEvent::JumpToLine => self.show_jump_to_line_dialog(window, cx),
207            StatusBarEvent::SelectLanguage => self.render_select_language_sheet(window, cx),
208            StatusBarEvent::ToggleMarkdownPreview => {
209                if self.settings.editor_settings.markdown_settings.preview_mode
210                    == MarkdownPreviewMode::DedicatedTab
211                {
212                    self.open_markdown_preview_tab(window, cx);
213                } else {
214                    if let Some(active_editor_tab) = self.get_active_editor_tab_mut() {
215                        active_editor_tab.show_markdown_preview =
216                            !active_editor_tab.show_markdown_preview;
217                    }
218                    cx.notify();
219                }
220            }
221            StatusBarEvent::ToggleMarkdownToolbar => {
222                if let Some(active_editor_tab) = self.get_active_editor_tab_mut() {
223                    active_editor_tab.show_markdown_toolbar =
224                        !active_editor_tab.show_markdown_toolbar;
225                }
226                cx.notify();
227            }
228            StatusBarEvent::ToggleCsvView => self.toggle_csv_view_mode(window, cx),
229            StatusBarEvent::ToggleLogView => self.toggle_log_view(window, cx),
230            StatusBarEvent::ToggleLogFollow => self.toggle_log_follow(window, cx),
231            StatusBarEvent::LoadFullLog => self.load_full_log(window, cx),
232            StatusBarEvent::OpenShareSheet => self.open_share_file_sheet(window, cx),
233            StatusBarEvent::ToggleColorPicker => self.toggle_color_picker(window, cx),
234        }
235    }
236}
