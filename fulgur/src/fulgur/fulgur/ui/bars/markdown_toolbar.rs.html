13/// Create a markdown bar button
14///
15/// ### Arguments
16/// - `id`: The ID of the button
17/// - `tooltip`: The tooltip of the button
18/// - `icon`: The icon of the button
19/// - `border_color`: The color of the border
20///
21/// ### Returns
22/// - `Button`: A markdown bar button
23pub fn markdown_bar_button_factory(
24    id: &'static str,
25    tooltip: &'static str,
26    icon: CustomIcon,
27    border_color: Hsla,
28) -> Button {
29    button_factory(id, tooltip, icon, border_color)
30        .h(MARKDOWN_BAR_BUTTON_SIZE)
31        .w(MARKDOWN_BAR_BUTTON_SIZE)
32}

34/// The markdown formatting toolbar, rendered as its own entity
35pub(crate) struct MarkdownToolbar {
36    fulgur: WeakEntity<Fulgur>,
37}
38
39impl MarkdownToolbar {
40    /// Create a new markdown toolbar view
41    ///
42    /// ### Arguments
43    /// - `fulgur`: Weak handle to the owning window entity the bar reads the active editor from
44    ///
45    /// ### Returns
46    /// - `MarkdownToolbar`: The new markdown toolbar view
47    pub(crate) fn new(fulgur: WeakEntity<Fulgur>) -> Self {
48        Self { fulgur }
49    }
50
51    /// Get the active editor's content entity from the owning window
52    ///
53    /// ### Arguments
54    /// - `cx`: The application context
55    ///
56    /// ### Returns
57    /// - `Some(Entity<InputState>)`: The active editor tab's content
58    /// - `None`: If the window is gone or the active tab is not an editor
59    fn active_editor_content(&self, cx: &App) -> Option<Entity<InputState>> {
60        let fulgur = self.fulgur.upgrade()?;
61        fulgur
62            .read(cx)
63            .get_active_editor_tab(cx)
64            .map(|editor_tab| editor_tab.content.clone())
65    }
66
67    /// Surround the active editor's selection with a prefix and suffix, or insert them at the cursor
68    ///
69    /// ### Arguments
70    /// - `prefix`: The prefix to insert or surround with
71    /// - `suffix`: The suffix to insert or surround with
72    /// - `window`: The window context
73    /// - `cx`: The application context
74    pub(crate) fn insert_or_surround(
75        &mut self,
76        prefix: &str,
77        suffix: &str,
78        window: &mut Window,
79        cx: &mut Context<Self>,
80    ) {
81        if let Some(content) = self.active_editor_content(cx) {
82            content.update(cx, |input_state, cx| {
83                let selection = input_state.selected_text_range(true, window, cx);
84                if let Some(selection) = selection {
85                    let selected_text = input_state
86                        .text()
87                        .slice(selection.range.start..selection.range.end)
88                        .to_string();
89                    let surrounded_text = format!("{prefix}{selected_text}{suffix}");
90                    input_state.replace(surrounded_text, window, cx);
91                } else {
92                    let inserted_text = format!("{}{}{}", prefix, " ", suffix);
93                    input_state.insert(inserted_text, window, cx);
94                }
95                cx.notify();
96            });
97        }
98    }
99}

101impl Fulgur {
102    /// Whether the markdown toolbar should be mounted for the active tab
103    ///
104    /// ### Arguments
105    /// - `cx`: The application context
106    ///
107    /// ### Returns
108    /// - `bool`: True if the active tab is markdown and its toolbar is enabled
109    pub(crate) fn markdown_toolbar_visible(&self, cx: &gpui::App) -> bool {
110        self.is_markdown(cx)
111            && self
112                .get_active_editor_tab(cx)
113                .is_some_and(|editor_tab| editor_tab.show_markdown_toolbar)
114    }
115}

386    #[cfg(feature = "gpui-test-support")]
387    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
388        cx.update(|cx| {
389            gpui_component::init(cx);
390            let mut settings = Settings::new();
391            settings.editor_settings.watch_files = false;
392            let pending_files: Arc<Mutex<Vec<PathBuf>>> = Arc::new(Mutex::new(Vec::new()));
393            cx.set_global(SharedAppState::new(settings, pending_files, None));
394            cx.set_global(WindowManager::new());
395        });
396
397        let fulgur_slot: RefCell<Option<Entity<Fulgur>>> = RefCell::new(None);
398        let window = cx
399            .update(|cx| {
400                cx.open_window(WindowOptions::default(), |window, cx| {
401                    let window_id = window.window_handle().window_id();
402                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
403                    *fulgur_slot.borrow_mut() = Some(fulgur.clone());
404                    cx.new(|cx| Root::new(fulgur, window, cx))
405                })
406            })
407            .expect("failed to open test window");
408
409        let visual_cx = VisualTestContext::from_window(window.into(), cx);
410        visual_cx.run_until_parked();
411        let fulgur = fulgur_slot
412            .into_inner()
413            .expect("failed to capture Fulgur entity");
414        (fulgur, visual_cx)
415    }

427    #[cfg(feature = "gpui-test-support")]
428    #[gpui::test]
429    fn test_insert_or_surround_wraps_selected_text(cx: &mut TestAppContext) {
430        let (fulgur, toolbar, mut visual_cx) = setup_toolbar(cx);
431
432        visual_cx.update(|window, cx| {
433            fulgur.update(cx, |this, cx| {
434                this.update_active_editor_tab(cx, |editor, cx| {
435                    editor.content.update(cx, |content, cx| {
436                        content.set_value("hello", window, cx);
437                    });
438                })
439                .expect("expected active editor tab");
440
441                this.focus_active_tab(window, cx);
442                let focus = this
443                    .get_active_editor_tab(cx)
444                    .expect("expected active editor tab")
445                    .content
446                    .read(cx)
447                    .focus_handle(cx);
448                focus.dispatch_action(&SelectAll, window, cx);
449                let selected = this
450                    .get_active_editor_tab(cx)
451                    .expect("expected active editor tab")
452                    .content
453                    .read(cx)
454                    .selected_value()
455                    .to_string();
456                assert_eq!(selected, "hello");
457            });
458
459            toolbar.update(cx, |bar, cx| {
460                bar.insert_or_surround("**", "**", window, cx);
461            });
462
463            let text = fulgur
464                .read(cx)
465                .get_active_editor_tab(cx)
466                .expect("expected active editor tab")
467                .content
468                .read(cx)
469                .text()
470                .to_string();
471            assert_eq!(text, "**hello**");
472        });
473    }
