11/// The three color text-input formats handled by the color picker bar.
12#[derive(Clone, Copy, PartialEq, Eq)]
13enum ColorInput {
14    Hex,
15    Oklch,
16    Hsla,
17}
18
19impl ColorInput {
20    /// All formats, in display order.
21    const ALL: [ColorInput; 3] = [ColorInput::Hex, ColorInput::Oklch, ColorInput::Hsla];
22
23    /// Borrow the input state entity backing this format.
24    ///
25    /// ### Arguments
26    /// - `state`: The color picker bar state holding the inputs
27    ///
28    /// ### Returns
29    /// - `&Entity<InputState>`: The input entity for this format
30    fn input(self, state: &ColorPickerBarState) -> &Entity<InputState> {
31        match self {
32            ColorInput::Hex => &state.hex_input,
33            ColorInput::Oklch => &state.oklch_input,
34            ColorInput::Hsla => &state.hsla_input,
35        }
36    }
37
38    /// Format a color into the textual representation for this format.
39    ///
40    /// ### Arguments
41    /// - `color`: The color to format
42    ///
43    /// ### Returns
44    /// - `String`: The formatted color string
45    fn format(self, color: Hsla) -> String {
46        match self {
47            ColorInput::Hex => gpui_component::Colorize::to_hex(&color),
48            ColorInput::Oklch => format_oklch(color),
49            ColorInput::Hsla => format_hsla(color),
50        }
51    }
52
53    /// Parse a textual color in this format into a color.
54    ///
55    /// ### Arguments
56    /// - `value`: The string to parse
57    ///
58    /// ### Returns
59    /// - `Some(Hsla)`: The parsed color
60    /// - `None`: If the string is not valid for this format
61    fn parse(self, value: &str) -> Option<Hsla> {
62        match self {
63            ColorInput::Hex => parse_hex(value),
64            ColorInput::Oklch => parse_oklch(value),
65            ColorInput::Hsla => parse_hsla(value),
66        }
67    }
68}

70/// State for the color picker bar.
71pub struct ColorPickerBarState {
72    pub color_picker_state: Entity<ColorPickerState>,
73    pub show_color_picker: bool,
74    pub hex_input: Entity<InputState>,
75    pub oklch_input: Entity<InputState>,
76    pub hsla_input: Entity<InputState>,
77    pub(super) cached_hex: String,
78    pub(super) cached_oklch: String,
79    pub(super) cached_hsla: String,
80    #[allow(dead_code, reason = "RAII guard: keeps the subscription alive")]
81    color_picker_subscription: Subscription,
82    #[allow(dead_code, reason = "RAII guard: keeps the subscription alive")]
83    hex_input_subscription: Subscription,
84    #[allow(dead_code, reason = "RAII guard: keeps the subscription alive")]
85    oklch_input_subscription: Subscription,
86    #[allow(dead_code, reason = "RAII guard: keeps the subscription alive")]
87    hsla_input_subscription: Subscription,
88}
89
90impl ColorPickerBarState {
91    /// Create a new `ColorPickerBarState`.
92    ///
93    /// ### Arguments
94    /// - `window`: The window context
95    /// - `cx`: The application context
96    ///
97    /// ### Returns
98    /// - `Self`: A new `ColorPickerBarState` initialized to white
99    pub fn new(window: &mut Window, cx: &mut Context<Fulgur>) -> Self {
100        let color_picker_state =
101            cx.new(|cx| ColorPickerState::new(window, cx).default_value(gpui::white()));
102        let initial = gpui::white();
103        let initial_hex = gpui_component::Colorize::to_hex(&initial).clone();
104        let initial_oklch = format_oklch(initial);
105        let initial_hsla = format_hsla(initial);
106        let hex_input = cx.new(|cx| InputState::new(window, cx).default_value(initial_hex.clone()));
107        let oklch_input =
108            cx.new(|cx| InputState::new(window, cx).default_value(initial_oklch.clone()));
109        let hsla_input =
110            cx.new(|cx| InputState::new(window, cx).default_value(initial_hsla.clone()));
111        let color_picker_subscription = cx.subscribe_in(
112            &color_picker_state,
113            window,
114            |this: &mut Fulgur, _, event: &ColorPickerEvent, window, cx| {
115                if let ColorPickerEvent::Change(Some(color)) = event {
116                    let color = *color;
117                    let hex = gpui_component::Colorize::to_hex(&color);
118                    let oklch = format_oklch(color);
119                    let hsla_str = format_hsla(color);
120                    this.color_picker_bar_state.cached_hex.clone_from(&hex);
121                    this.color_picker_bar_state.cached_oklch.clone_from(&oklch);
122                    this.color_picker_bar_state
123                        .cached_hsla
124                        .clone_from(&hsla_str);
125                    this.color_picker_bar_state
126                        .hex_input
127                        .update(cx, |state, cx| {
128                            state.set_value(hex, window, cx);
129                        });
130                    this.color_picker_bar_state
131                        .oklch_input
132                        .update(cx, |state, cx| {
133                            state.set_value(oklch, window, cx);
134                        });
135                    this.color_picker_bar_state
136                        .hsla_input
137                        .update(cx, |state, cx| {
138                            state.set_value(hsla_str, window, cx);
139                        });
140                }
141                cx.notify();
142            },
143        );
144        // On Change: update the color picker preview only.
145        // On PressEnter / Blur: parse and sync all three inputs + the color picker.
146        let hex_input_subscription = cx.subscribe_in(
147            &hex_input,
148            window,
149            |this: &mut Fulgur, _, event: &InputEvent, window, cx| {
150                Self::handle_input_event(this, ColorInput::Hex, event, window, cx);
151            },
152        );
153        let oklch_input_subscription = cx.subscribe_in(
154            &oklch_input,
155            window,
156            |this: &mut Fulgur, _, event: &InputEvent, window, cx| {
157                Self::handle_input_event(this, ColorInput::Oklch, event, window, cx);
158            },
159        );
160        let hsla_input_subscription = cx.subscribe_in(
161            &hsla_input,
162            window,
163            |this: &mut Fulgur, _, event: &InputEvent, window, cx| {
164                Self::handle_input_event(this, ColorInput::Hsla, event, window, cx);
165            },
166        );
167        Self {
168            color_picker_state,
169            show_color_picker: false,
170            hex_input,
171            oklch_input,
172            hsla_input,
173            cached_hex: initial_hex,
174            cached_oklch: initial_oklch,
175            cached_hsla: initial_hsla,
176            color_picker_subscription,
177            hex_input_subscription,
178            oklch_input_subscription,
179            hsla_input_subscription,
180        }
181    }
182
183    /// Handle a change event from one of the color text inputs.
184    ///
185    /// ### Arguments
186    /// - `this`: The owning `Fulgur` instance
187    /// - `source`: The input format that emitted the event
188    /// - `event`: The input event to handle
189    /// - `window`: The window context
190    /// - `cx`: The application context
191    fn handle_input_event(
192        this: &mut Fulgur,
193        source: ColorInput,
194        event: &InputEvent,
195        window: &mut Window,
196        cx: &mut Context<Fulgur>,
197    ) {
198        let value = source.input(&this.color_picker_bar_state).read(cx).value();
199        let Some(color) = source.parse(&value) else {
200            return;
201        };
202        match event {
203            InputEvent::Change => Self::apply_input_color(this, source, color, false, window, cx),
204            InputEvent::PressEnter { .. } | InputEvent::Blur => {
205                Self::apply_input_color(this, source, color, true, window, cx);
206            }
207            InputEvent::Focus => {}
208        }
209    }
210
211    /// Apply a color parsed from one of the text inputs.
212    ///
213    /// ### Arguments
214    /// - `this`: The owning `Fulgur` instance
215    /// - `source`: The input format the color originated from
216    /// - `color`: The parsed color to apply
217    /// - `sync_siblings`: Whether to rewrite the other two inputs
218    /// - `window`: The window context
219    /// - `cx`: The application context
220    fn apply_input_color(
221        this: &mut Fulgur,
222        source: ColorInput,
223        color: Hsla,
224        sync_siblings: bool,
225        window: &mut Window,
226        cx: &mut Context<Fulgur>,
227    ) {
228        this.color_picker_bar_state
229            .color_picker_state
230            .update(cx, |state, cx| {
231                state.set_value(color, window, cx);
232            });
233        if !sync_siblings {
234            return;
235        }
236        for target in ColorInput::ALL {
237            if target == source {
238                continue;
239            }
240            let formatted = target.format(color);
241            target
242                .input(&this.color_picker_bar_state)
243                .update(cx, |state, cx| {
244                    state.set_value(formatted, window, cx);
245                });
246        }
247    }
248}
