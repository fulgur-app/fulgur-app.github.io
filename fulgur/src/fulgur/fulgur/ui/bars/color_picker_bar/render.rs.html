21    /// Insert a value at the cursor position in the active editor tab, replacing the current selection if any.
22    ///
23    /// ### Arguments
24    /// - `value`: The string to insert
25    /// - `window`: The window context
26    /// - `cx`: The application context
27    pub fn insert_color_value(
28        &mut self,
29        value: String,
30        window: &mut Window,
31        cx: &mut Context<Self>,
32    ) {
33        if let Some(index) = self.active_tab_index()
34            && let Some(crate::fulgur::tab::Tab::Editor(editor_tab)) = self.tabs.get_mut(index)
35        {
36            editor_tab.content.update(cx, |input_state, cx| {
37                let selection = input_state.selected_text_range(true, window, cx);
38                if selection.is_some() {
39                    input_state.replace(value, window, cx);
40                } else {
41                    input_state.insert(value, window, cx);
42                }
43                cx.notify();
44            });
45        }
46    }

48    /// Toggle the color picker bar visibility.
49    ///
50    /// ### Arguments
51    /// - `_window`: The window context
52    /// - `cx`: The application context
53    pub fn toggle_color_picker(&mut self, _window: &mut Window, cx: &mut Context<Self>) {
54        self.color_picker_bar_state.show_color_picker =
55            !self.color_picker_bar_state.show_color_picker;
56        cx.notify();
57    }

59    /// Render the color picker bar.
60    ///
61    /// ### Arguments
62    /// - `cx`: The application context
63    ///
64    /// ### Returns
65    /// - `Some(Div)`: The rendered color picker bar
66    /// - `None`: If the color picker bar is hidden
67    pub fn render_color_picker_bar(&self, cx: &mut Context<Self>) -> Option<Div> {
68        if !self.color_picker_bar_state.show_color_picker {
69            return None;
70        }
71        let hex_value = &self.color_picker_bar_state.cached_hex;
72        let oklch_value = &self.color_picker_bar_state.cached_oklch;
73        let hsla_value = &self.color_picker_bar_state.cached_hsla;
74        Some(
75            div()
76                .flex()
77                .items_center()
78                .justify_between()
79                .bg(cx.theme().tab_bar)
80                .p_0()
81                .m_0()
82                .w_full()
83                .h(SEARCH_BAR_HEIGHT)
84                .border_t_1()
85                .border_color(cx.theme().border)
86                .child(
87                    div()
88                        .id("color-picker-scroll-container")
89                        .flex()
90                        .flex_1()
91                        .overflow_x_scroll()
92                        .child(
93                            div()
94                                .flex()
95                                .flex_1()
96                                .items_center()
97                                .h(SEARCH_BAR_HEIGHT)
98                                .child(self.render_color_picker_section(cx))
99                                .child(Self::render_color_value_section(
100                                    "Hex",
101                                    hex_value,
102                                    &self.color_picker_bar_state.hex_input,
103                                    cx,
104                                ))
105                                .child(Self::render_color_value_section(
106                                    "OkLCH",
107                                    oklch_value,
108                                    &self.color_picker_bar_state.oklch_input,
109                                    cx,
110                                ))
111                                .child(Self::render_color_value_section(
112                                    "HSLA",
113                                    hsla_value,
114                                    &self.color_picker_bar_state.hsla_input,
115                                    cx,
116                                ))
117                                .child(self.render_highlight_toggle_button(cx)),
118                        ),
119                )
120                .child(Self::render_color_picker_close_button(cx)),
121        )
122    }

142    /// Render a color value section with an editable input and a clipboard copy button.
143    ///
144    /// ### Arguments
145    /// - `label`: The label for the color format (e.g. "Hex", "`OkLCH`", "HSLA")
146    /// - `value`: The current formatted color value string (used for the clipboard button)
147    /// - `input_state`: The input state entity for this field
148    /// - `cx`: The application context
149    ///
150    /// ### Returns
151    /// - `Div`: The rendered color value section
152    fn render_color_value_section(
153        label: &'static str,
154        value: &str,
155        input_state: &Entity<InputState>,
156        cx: &mut Context<Self>,
157    ) -> Div {
158        div()
159            .flex()
160            .items_center()
161            .flex_1()
162            .min_w(gpui::px(330.0))
163            .h(SEARCH_BAR_HEIGHT)
164            .border_l_1()
165            .border_color(cx.theme().border)
166            .child(
167                div()
168                    .text_xs()
169                    .px_2()
170                    .text_color(cx.theme().muted_foreground)
171                    .child(label),
172            )
173            .child(
174                Input::new(input_state)
175                    .appearance(false)
176                    .bordered(false)
177                    .px_0(),
178            )
179            .child(
180                InsertButton::new(SharedString::from(format!("color-insert-{label}"))).on_click(
181                    cx.listener({
182                        let value = value.to_string();
183                        move |this, _, window, cx| {
184                            this.insert_color_value(value.clone(), window, cx);
185                        }
186                    }),
187                ),
188            )
189            .child(
190                CopyButton::new(SharedString::from(format!("color-copy-{label}")))
191                    .value(SharedString::from(value.to_string())),
192            )
193    }

281    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
282        cx.update(|cx| {
283            gpui_component::init(cx);
284            let mut settings = Settings::new();
285            settings.editor_settings.watch_files = false;
286            let pending_files: Arc<Mutex<Vec<PathBuf>>> = Arc::new(Mutex::new(Vec::new()));
287            cx.set_global(SharedAppState::new(settings, pending_files, None));
288            cx.set_global(WindowManager::new());
289        });
290
291        let fulgur_slot: RefCell<Option<Entity<Fulgur>>> = RefCell::new(None);
292        let window = cx
293            .update(|cx| {
294                cx.open_window(WindowOptions::default(), |window, cx| {
295                    let window_id = window.window_handle().window_id();
296                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
297                    *fulgur_slot.borrow_mut() = Some(fulgur);
298                    cx.new(|_| EmptyView)
299                })
300            })
301            .expect("failed to open test window");
302
303        let visual_cx = VisualTestContext::from_window(window.into(), cx);
304        visual_cx.run_until_parked();
305        let fulgur = fulgur_slot
306            .into_inner()
307            .expect("failed to capture Fulgur entity");
308        (fulgur, visual_cx)
309    }

349        let (fulgur, mut visual_cx) = setup_fulgur(cx);
350        visual_cx.update(|window, cx| {

353                    .get_active_editor_tab_mut()
354                    .expect("expected active editor tab");
355                editor.content.update(cx, |content, cx| {
356                    content.set_value("color: ;", window, cx);
357                    content.set_cursor_position(
358                        Position {
359                            line: 0,
360                            character: 7,
361                        },
362                        window,
363                        cx,
364                    );
365                });
366                this.insert_color_value("#FF0000".to_string(), window, cx);
367                let text = this
368                    .get_active_editor_tab()
369                    .expect("expected active editor tab")
370                    .content
371                    .read(cx)
372                    .text()

374                assert_eq!(text, "color: #FF0000;");
