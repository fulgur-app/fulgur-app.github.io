18impl Render for ColorPickerBar {
19    /// Render the color picker bar
20    ///
21    /// ### Arguments
22    /// - `_window`: The window to render the color picker bar in
23    /// - `cx`: The application context
24    ///
25    /// ### Returns
26    /// - `impl IntoElement`: The rendered color picker bar, or an empty element when hidden
27    fn render(&mut self, _window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
28        if !self.show_color_picker {
29            return div().into_any_element();
30        }
31        div()
32            .flex()
33            .items_center()
34            .justify_between()
35            .bg(cx.theme().tab_bar)
36            .p_0()
37            .m_0()
38            .w_full()
39            .h(SEARCH_BAR_HEIGHT)
40            .border_t_1()
41            .border_color(cx.theme().border)
42            .child(
43                div()
44                    .id("color-picker-scroll-container")
45                    .flex()
46                    .flex_1()
47                    .overflow_x_scroll()
48                    .child(
49                        div()
50                            .flex()
51                            .flex_1()
52                            .items_center()
53                            .h(SEARCH_BAR_HEIGHT)
54                            .child(self.render_color_picker_section(cx))
55                            .child(Self::render_color_value_section(
56                                "Hex",
57                                &self.cached_hex,
58                                &self.hex_input,
59                                cx,
60                            ))
61                            .child(Self::render_color_value_section(
62                                "OkLCH",
63                                &self.cached_oklch,
64                                &self.oklch_input,
65                                cx,
66                            ))
67                            .child(Self::render_color_value_section(
68                                "HSLA",
69                                &self.cached_hsla,
70                                &self.hsla_input,
71                                cx,
72                            ))
73                            .child(self.render_highlight_toggle_button(cx)),
74                    ),
75            )
76            .child(Self::render_close_button(cx))
77            .into_any_element()
78    }
79}

81impl ColorPickerBar {
82    /// Render the color picker section (left part with the color picker widget).
83    ///
84    /// ### Returns
85    /// - `impl IntoElement`: The rendered color picker section
86    fn render_color_picker_section(&self, _cx: &mut Context<Self>) -> impl IntoElement {
87        h_flex()
88            .items_center()
89            .flex_shrink_0()
90            .px_2()
91            .gap_2()
92            .h(SEARCH_BAR_HEIGHT)
93            .child(
94                gpui_component::color_picker::ColorPicker::new(&self.color_picker_state)
95                    .anchor(Anchor::BottomLeft),
96            )
97    }
98
99    /// Render a color value section with an editable input and a clipboard copy button.
100    ///
101    /// ### Arguments
102    /// - `label`: The label for the color format (e.g. "Hex", "`OkLCH`", "HSLA")
103    /// - `value`: The current formatted color value string (used for the insert and clipboard buttons)
104    /// - `input_state`: The input state entity for this field
105    /// - `cx`: The application context
106    ///
107    /// ### Returns
108    /// - `Div`: The rendered color value section
109    fn render_color_value_section(
110        label: &'static str,
111        value: &str,
112        input_state: &Entity<InputState>,
113        cx: &mut Context<Self>,
114    ) -> Div {
115        div()
116            .flex()
117            .items_center()
118            .flex_1()
119            .min_w(gpui::px(330.0))
120            .h(SEARCH_BAR_HEIGHT)
121            .border_l_1()
122            .border_color(cx.theme().border)
123            .child(
124                div()
125                    .text_xs()
126                    .px_2()
127                    .text_color(cx.theme().muted_foreground)
128                    .child(label),
129            )
130            .child(
131                Input::new(input_state)
132                    .appearance(false)
133                    .bordered(false)
134                    .px_0(),
135            )
136            .child(
137                InsertButton::new(SharedString::from(format!("color-insert-{label}"))).on_click(
138                    cx.listener({
139                        let value = value.to_string();
140                        move |this, _, window, cx| {
141                            this.insert_color_value(value.clone(), window, cx);
142                        }
143                    }),
144                ),
145            )
146            .child(
147                CopyButton::new(SharedString::from(format!("color-copy-{label}")))
148                    .value(SharedString::from(value.to_string())),
149            )
150    }
151
152    /// Render the highlight colors toggle button for the color picker bar.
153    ///
154    /// ### Arguments
155    /// - `cx`: The application context
156    ///
157    /// ### Returns
158    /// - `Div`: The rendered toggle button
159    fn render_highlight_toggle_button(&self, cx: &mut Context<Self>) -> Div {
160        let highlight_colors = self
161            .fulgur
162            .upgrade()
163            .is_some_and(|fulgur| fulgur.read(cx).settings.editor_settings.highlight_colors);
164        div()
165            .flex()
166            .items_center()
167            .p_0()
168            .m_0()
169            .border_l_1()
170            .border_color(cx.theme().border)
171            .child(
172                search_bar_toggle_button_factory(
173                    "highlight-colors-toggle",
174                    "Toggle color highlighting",
175                    CustomIcon::Highlighter,
176                    cx.theme().border,
177                    cx.theme().tab_bar,
178                    cx.theme().accent,
179                    highlight_colors,
180                )
181                .on_click(cx.listener(|_, _, _window, cx| {
182                    cx.emit(ColorPickerBarEvent::ToggleHighlightColors);
183                })),
184            )
185    }
186
187    /// Render the close button for the color picker bar.
188    ///
189    /// ### Arguments
190    /// - `cx`: The application context
191    ///
192    /// ### Returns
193    /// - `Div`: The rendered close button
194    fn render_close_button(cx: &mut Context<Self>) -> Div {
195        div()
196            .flex()
197            .items_center()
198            .p_0()
199            .m_0()
200            .border_l_1()
201            .border_color(cx.theme().border)
202            .child(
203                search_bar_button_factory(
204                    "close-color-picker-button",
205                    "Close",
206                    CustomIcon::Close,
207                    cx.theme().border,
208                )
209                .on_click(cx.listener(|this, _, _window, cx| {
210                    this.close(cx);
211                })),
212            )
213    }
214}

238    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
239        cx.update(|cx| {
240            gpui_component::init(cx);
241            let mut settings = Settings::new();
242            settings.editor_settings.watch_files = false;
243            let pending_files: Arc<Mutex<Vec<PathBuf>>> = Arc::new(Mutex::new(Vec::new()));
244            cx.set_global(SharedAppState::new(settings, pending_files, None));
245            cx.set_global(WindowManager::new());
246        });
247
248        let fulgur_slot: RefCell<Option<Entity<Fulgur>>> = RefCell::new(None);
249        let window = cx
250            .update(|cx| {
251                cx.open_window(WindowOptions::default(), |window, cx| {
252                    let window_id = window.window_handle().window_id();
253                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
254                    *fulgur_slot.borrow_mut() = Some(fulgur);
255                    cx.new(|_| EmptyView)
256                })
257            })
258            .expect("failed to open test window");
259
260        let visual_cx = VisualTestContext::from_window(window.into(), cx);
261        visual_cx.run_until_parked();
262        let fulgur = fulgur_slot
263            .into_inner()
264            .expect("failed to capture Fulgur entity");
265        (fulgur, visual_cx)
266    }
