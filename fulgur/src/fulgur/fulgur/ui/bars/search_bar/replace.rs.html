7impl SearchBar {
8    /// Force a fresh search, bypassing the query/option dedup cache
9    ///
10    /// ### Arguments
11    /// - `content`: The active editor tab's content, if any
12    /// - `window`: The window context
13    /// - `cx`: The search bar context
14    fn force_perform_search(
15        &mut self,
16        content: Option<Entity<EditorState>>,
17        window: &mut Window,
18        cx: &mut Context<Self>,
19    ) {
20        self.last_search_query.clear();
21        self.search_matches.clear();
22        self.perform_search(content, window, cx);
23    }
24
25    /// Replace the current search match
26    ///
27    /// ### Arguments
28    /// - `content`: The active editor tab's content, if any
29    /// - `window`: The window context
30    /// - `cx`: The search bar context
31    pub(super) fn replace_current(
32        &mut self,
33        content: Option<Entity<EditorState>>,
34        window: &mut Window,
35        cx: &mut Context<Self>,
36    ) {
37        // Recompute matches against the current buffer before slicing: the cached
38        // offsets may be stale if the document was edited since the last search.
39        self.force_perform_search(content.clone(), window, cx);
40        if let Some(match_index) = self.current_match_index
41            && let Some(search_match) = self.search_matches.get(match_index).cloned()
42            && let Some(content_entity) = content
43        {
44            let replace_text = self.replace_input.read(cx).text().to_string();
45            let text = content_entity.read(cx).text().to_string();
46            // Defensive guard against stale offsets: bail out instead of slicing
47            // out of bounds or on a non-char-boundary if the buffer changed.
48            if search_match.end > text.len()
49                || search_match.start > search_match.end
50                || !text.is_char_boundary(search_match.start)
51                || !text.is_char_boundary(search_match.end)
52            {
53                cx.notify();
54                return;
55            }
56            let mut new_text = String::new();
57            new_text.push_str(&text[..search_match.start]);
58            new_text.push_str(&replace_text);
59            new_text.push_str(&text[search_match.end..]);
60            content_entity.update(cx, |content, cx| {
61                content.set_value(&new_text, window, cx);
62            });
63            self.search_matches.clear();
64            self.perform_search(Some(content_entity.clone()), window, cx);
65            if !self.search_matches.is_empty() {
66                if match_index < self.search_matches.len() {
67                    self.current_match_index = Some(match_index);
68                } else {
69                    self.current_match_index = Some(0);
70                }
71                self.highlight_current_match(&content_entity, window, cx);
72            }
73        }
74        cx.notify();
75    }
76
77    /// Replace all search matches
78    ///
79    /// ### Arguments
80    /// - `content`: The active editor tab's content, if any
81    /// - `window`: The window context
82    /// - `cx`: The search bar context
83    pub(super) fn replace_all(
84        &mut self,
85        content: Option<Entity<EditorState>>,
86        window: &mut Window,
87        cx: &mut Context<Self>,
88    ) {
89        self.force_perform_search(content.clone(), window, cx);
90        if self.search_matches.is_empty() {
91            return;
92        }
93        if let Some(content_entity) = content {
94            let replace_text = self.replace_input.read(cx).text().to_string();
95            let search_query = self.search_input.read(cx).text().to_string();
96            let match_case = self.match_case;
97            let match_whole_word = self.match_whole_word;
98            let text = content_entity.read(cx).text().to_string();
99            let new_text = if match_case && !match_whole_word {
100                text.replace(&search_query, &replace_text)
101            } else {
102                apply_replacements(&self.search_matches, &text, &replace_text)
103            };
104            content_entity.update(cx, |content, cx| {
105                content.set_value(&new_text, window, cx);
106            });
107            self.search_matches.clear();
108            self.current_match_index = None;
109        }
110        cx.notify();
111    }
