9impl Render for SearchBar {
10    /// Render the search bar
11    ///
12    /// ### Arguments
13    /// - `_window`: The window to render the search bar in
14    /// - `cx`: The application context
15    ///
16    /// ### Returns
17    /// - `impl IntoElement`: The rendered search bar, or an empty element when hidden
18    fn render(&mut self, _window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
19        if !self.show_search {
20            return div().into_any_element();
21        }
22        div()
23            .flex()
24            .justify_between()
25            .items_center()
26            .bg(cx.theme().tab_bar)
27            .p_0()
28            .m_0()
29            .w_full()
30            .h(SEARCH_BAR_HEIGHT)
31            .border_t_1()
32            .border_color(cx.theme().border)
33            .child(self.render_search_input_section(cx))
34            .child(self.render_search_navigation_section(cx))
35            .child(self.render_replace_section(cx))
36            .child(Self::render_search_close_button(cx))
37            .into_any_element()
38    }
39}
