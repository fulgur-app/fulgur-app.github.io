9/// The pair of decoration collections painting the search matches of one editor
10///
11/// `all` carries every match except the current one, `current` carries the
12/// current one alone, so the two never overlap and their styles never compete.
13/// The weak handle lets dead editors be pruned from the cache.
14pub(super) struct MatchDecorations {
15    pub(super) editor: WeakEntity<EditorState>,
16    pub(super) all: TextDecorationCollection,
17    pub(super) current: TextDecorationCollection,
18}

42/// Typed events emitted by the search bar toward the owning `Fulgur` window
43#[derive(Clone, Copy, Debug, PartialEq, Eq)]
44pub(crate) enum SearchBarEvent {
45    Closed,
46}

48impl EventEmitter<SearchBarEvent> for SearchBar {}

50impl SearchBar {
51    /// Create a new search bar view owning its search and replace inputs
52    ///
53    /// Subscribes to its own search input so the search re-runs whenever the
54    /// query changes while the bar is visible.
55    ///
56    /// ### Arguments
57    /// - `fulgur`: Weak handle to the owning window entity the bar reads the active editor from
58    /// - `window`: The window context
59    /// - `cx`: The search bar context
60    ///
61    /// ### Returns
62    /// - `SearchBar`: The new search bar view
63    pub(crate) fn new(
64        fulgur: WeakEntity<Fulgur>,
65        window: &mut Window,
66        cx: &mut Context<Self>,
67    ) -> Self {
68        let search_input = cx.new(|cx| InputState::new(window, cx).placeholder("Search"));
69        let replace_input = cx.new(|cx| InputState::new(window, cx).placeholder("Replace"));
70        let search_input_subscription = cx.subscribe_in(
71            &search_input,
72            window,
73            |this: &mut Self, _, ev: &InputEvent, window, cx| {
74                if let InputEvent::Change = ev
75                    && this.show_search
76                {
77                    this.on_query_changed(window, cx);
78                }
79            },
80        );
81        Self {
82            fulgur,
83            show_search: false,
84            search_input,
85            replace_input,
86            match_case: false,
87            match_whole_word: false,
88            search_matches: Vec::new(),
89            current_match_index: None,
90            last_search_query: String::new(),
91            last_search_match_case: false,
92            last_search_match_whole_word: false,
93            search_text_scratch: String::new(),
94            search_newline_offsets_scratch: Vec::new(),
95            search_lowercase_text_scratch: String::new(),
96            search_lowercase_offsets_scratch: Vec::new(),
97            match_decorations: HashMap::new(),
98            _search_input_subscription: search_input_subscription,
99        }
100    }
101
102    /// Whether the search bar is currently shown
103    ///
104    /// ### Returns
105    /// - `bool`: True if the bar is visible
106    pub(crate) fn is_visible(&self) -> bool {
107        self.show_search
108    }
109
110    /// Get the active editor's content entity from the owning window
111    ///
112    /// ### Arguments
113    /// - `cx`: The application context
114    ///
115    /// ### Returns
116    /// - `Some(Entity<EditorState>)`: The active editor tab's content
117    /// - `None`: If the window is gone or the active tab is not an editor
118    pub(super) fn active_editor_content(&self, cx: &App) -> Option<Entity<EditorState>> {
119        let fulgur = self.fulgur.upgrade()?;
120        fulgur
121            .read(cx)
122            .get_active_editor_tab(cx)
123            .map(|editor_tab| editor_tab.content.clone())
124    }
125}

127impl Fulgur {
128    /// Dispatch a search bar event to the matching window-level handler
129    ///
130    /// ### Arguments
131    /// - `event`: The search bar event to handle
132    /// - `window`: The window context
133    /// - `cx`: The application context
134    pub(crate) fn on_search_bar_event(
135        &mut self,
136        event: SearchBarEvent,
137        window: &mut Window,
138        cx: &mut Context<Self>,
139    ) {
140        match event {
141            SearchBarEvent::Closed => {
142                self.focus_active_tab(window, cx);
143                cx.notify();
144            }
145        }
146    }
