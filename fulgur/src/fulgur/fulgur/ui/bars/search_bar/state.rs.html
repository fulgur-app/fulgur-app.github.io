27/// Typed events emitted by the search bar toward the owning `Fulgur` window
28#[derive(Clone, Copy, Debug, PartialEq, Eq)]
29pub(crate) enum SearchBarEvent {
30    Closed,
31}
32
33impl EventEmitter<SearchBarEvent> for SearchBar {}

35impl SearchBar {
36    /// Create a new search bar view owning its search and replace inputs
37    ///
38    /// Subscribes to its own search input so the search re-runs whenever the
39    /// query changes while the bar is visible.
40    ///
41    /// ### Arguments
42    /// - `fulgur`: Weak handle to the owning window entity the bar reads the active editor from
43    /// - `window`: The window context
44    /// - `cx`: The search bar context
45    ///
46    /// ### Returns
47    /// - `SearchBar`: The new search bar view
48    pub(crate) fn new(
49        fulgur: WeakEntity<Fulgur>,
50        window: &mut Window,
51        cx: &mut Context<Self>,
52    ) -> Self {
53        let search_input = cx.new(|cx| InputState::new(window, cx).placeholder("Search"));
54        let replace_input = cx.new(|cx| InputState::new(window, cx).placeholder("Replace"));
55        let search_input_subscription = cx.subscribe_in(
56            &search_input,
57            window,
58            |this: &mut Self, _, ev: &InputEvent, window, cx| {
59                if let InputEvent::Change = ev
60                    && this.show_search
61                {
62                    this.on_query_changed(window, cx);
63                }
64            },
65        );
66        Self {
67            fulgur,
68            show_search: false,
69            search_input,
70            replace_input,
71            match_case: false,
72            match_whole_word: false,
73            search_matches: Vec::new(),
74            current_match_index: None,
75            last_search_query: String::new(),
76            last_search_match_case: false,
77            last_search_match_whole_word: false,
78            search_text_scratch: String::new(),
79            search_newline_offsets_scratch: Vec::new(),
80            search_lowercase_text_scratch: String::new(),
81            search_lowercase_offsets_scratch: Vec::new(),
82            _search_input_subscription: search_input_subscription,
83        }
84    }
85
86    /// Whether the search bar is currently shown
87    ///
88    /// ### Returns
89    /// - `bool`: True if the bar is visible
90    pub(crate) fn is_visible(&self) -> bool {
91        self.show_search
92    }
93
94    /// Get the active editor's content entity from the owning window
95    ///
96    /// ### Arguments
97    /// - `cx`: The application context
98    ///
99    /// ### Returns
100    /// - `Some(Entity<EditorState>)`: The active editor tab's content
101    /// - `None`: If the window is gone or the active tab is not an editor
102    pub(super) fn active_editor_content(&self, cx: &App) -> Option<Entity<EditorState>> {
103        let fulgur = self.fulgur.upgrade()?;
104        fulgur
105            .read(cx)
106            .get_active_editor_tab(cx)
107            .map(|editor_tab| editor_tab.content.clone())
108    }
109}
110
111impl Fulgur {
112    /// Dispatch a search bar event to the matching window-level handler
113    ///
114    /// ### Arguments
115    /// - `event`: The search bar event to handle
116    /// - `window`: The window context
117    /// - `cx`: The application context
118    pub(crate) fn on_search_bar_event(
119        &mut self,
120        event: SearchBarEvent,
121        window: &mut Window,
122        cx: &mut Context<Self>,
123    ) {
124        match event {
125            SearchBarEvent::Closed => {
126                self.focus_active_tab(window, cx);
127                cx.notify();
128            }
129        }
130    }
