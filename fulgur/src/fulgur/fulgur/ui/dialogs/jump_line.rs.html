7    /// Show the jump to line dialog
8    ///
9    /// ### Arguments
10    /// - `window`: The window to show the dialog in
11    /// - `cx`: The application context
12    pub fn show_jump_to_line_dialog(&mut self, window: &mut Window, cx: &mut Context<Self>) {
13        self.jump_to_line_input.update(cx, |input_state, cx| {
14            input_state.set_value("", window, cx);
15            cx.notify();
16        });
17        let jump_to_line_input = self.jump_to_line_input.clone();
18        let entity = cx.entity().clone();
19        window.open_alert_dialog(cx, move |modal, window, cx| {
20            let focus_handle = jump_to_line_input.read(cx).focus_handle(cx);
21            window.focus(&focus_handle, cx);
22            let jump_to_line_input_clone = jump_to_line_input.clone();
23            let entity_for_ok = entity.clone();
24            let entity_for_cancel = entity.clone();
25            modal
26                .title(div().text_size(px(16.)).child("Jump to line..."))
27                .keyboard(true)
28                .button_props(
29                    DialogButtonProps::default()
30                        .show_cancel(true)
31                        .cancel_text("Cancel")
32                        .cancel_variant(ButtonVariant::Secondary)
33                        .ok_text("Jump")
34                        .ok_variant(ButtonVariant::Primary),
35                )
36                .close_button(false)
37                .child(Input::new(&jump_to_line_input))
38                .on_ok(move |_, _window, cx| {
39                    let text = jump_to_line_input_clone.read(cx).value();
40                    let jump_result = editor_tab::extract_line_number(&text);
41                    let is_ok = jump_result.is_ok();
42                    entity_for_ok.update(cx, |this, cx| {
43                        if let Ok(jump) = jump_result {
44                            this.pending_jump = Some(jump);
45                            cx.notify();
46                        } else {
47                            this.pending_jump = None;
48                        }
49                    });
50
51                    is_ok
52                })
53                .on_cancel(move |_, _, cx| {
54                    entity_for_cancel.update(cx, |_this, cx| {
55                        cx.notify();
56                    });
57                    true
58                })
59        });
60    }

62    /// Handle pending jump-to-line action
63    ///
64    /// ### Arguments
65    /// - `window`: The window containing the editor
66    /// - `cx`: The application context
67    pub fn handle_pending_jump_to_line(&mut self, window: &mut Window, cx: &mut Context<Self>) {
68        if let Some(jump) = self.pending_jump.take() {
69            self.update_active_editor_tab(cx, |editor_tab, cx| {
70                editor_tab.jump_to_line(window, cx, jump);
71            });
72        }
73    }

87    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
88        cx.update(gpui_component::init);
89        cx.update(|cx| {
90            cx.set_global(SharedAppState::new(
91                Settings::new(),
92                Arc::new(Mutex::new(Vec::new())),
93                None,
94                None,
95            ));
96            cx.set_global(WindowManager::new());
97        });
98        let fulgur_slot: Rc<RefCell<Option<Entity<Fulgur>>>> = Rc::new(RefCell::new(None));
99        let slot = Rc::clone(&fulgur_slot);
100        let window = cx
101            .update(|cx| {
102                cx.open_window(WindowOptions::default(), |window, cx| {
103                    let window_id = window.window_handle().window_id();
104                    let fulgur = Fulgur::new(window, cx, window_id, WindowInit::Empty);
105                    *slot.borrow_mut() = Some(fulgur.clone());
106                    cx.new(|cx| gpui_component::Root::new(fulgur, window, cx))
107                })
108            })
109            .expect("failed to open test window");
110        let fulgur = fulgur_slot
111            .borrow_mut()
112            .take()
113            .expect("expected fulgur entity");
114        let visual_cx = VisualTestContext::from_window(window.into(), cx);
115        (fulgur, visual_cx)
116    }

120    #[gpui::test]
121    fn test_handle_pending_jump_consumes_pending_jump(cx: &mut TestAppContext) {
122        let (fulgur, mut visual_cx) = setup_fulgur(cx);
123        visual_cx.update(|window, cx| {
124            fulgur.update(cx, |this, cx| {
125                this.pending_jump = Some(Jump {
126                    line: 0,
127                    character: None,
128                });
129                this.handle_pending_jump_to_line(window, cx);
130            });
131        });
132        let pending = fulgur.read_with(&visual_cx, |this, _| this.pending_jump.is_some());
133        assert!(!pending, "pending_jump should be consumed after handling");
134    }

136    #[gpui::test]
137    fn test_handle_pending_jump_applies_cursor_line(cx: &mut TestAppContext) {
138        let (fulgur, mut visual_cx) = setup_fulgur(cx);
139        // Set multi-line content in the active editor tab
140        visual_cx.update(|window, cx| {
141            fulgur.update(cx, |this, cx| {
142                this.tabs
143                    .first()
144                    .expect("expected at least one tab")
145                    .clone()
146                    .update(cx, |tab, cx| {
147                        if let Some(editor_tab) = tab.as_editor_mut() {
148                            editor_tab.content.update(cx, |input, cx| {
149                                input.set_value("line one\nline two\nline three", window, cx);
150                            });
151                        }
152                    });
153            });
154        });
155        // Jump to line 2 (0-indexed: line = 1)
156        visual_cx.update(|window, cx| {
157            fulgur.update(cx, |this, cx| {
158                this.pending_jump = Some(Jump {
159                    line: 1,
160                    character: None,
161                });
162                this.handle_pending_jump_to_line(window, cx);
163            });
164        });
165        let cursor_line = fulgur.read_with(&visual_cx, |this, cx| {
166            this.tabs
167                .first()
168                .and_then(|t| t.read(cx).as_editor())
169                .map(|e| e.content.read(cx).cursor_position().line)
170        });
171        assert_eq!(cursor_line, Some(1), "cursor should be on line index 1");
172    }
