8impl Fulgur {
9    /// Show the jump to line dialog
10    ///
11    /// ### Arguments
12    /// - `window`: The window to show the dialog in
13    /// - `cx`: The application context
14    pub fn show_jump_to_line_dialog(&mut self, window: &mut Window, cx: &mut Context<Self>) {
15        self.jump_to_line_input.update(cx, |input_state, cx| {
16            input_state.set_value("", window, cx);
17            cx.notify();
18        });
19        let jump_to_line_input = self.jump_to_line_input.clone();
20        let entity = cx.entity().clone();
21        self.jump_to_line_dialog_open = true;
22        window.open_alert_dialog(cx.deref_mut(), move |modal, window, cx| {
23            let focus_handle = jump_to_line_input.read(cx).focus_handle(cx);
24            window.focus(&focus_handle, cx);
25            let jump_to_line_input_clone = jump_to_line_input.clone();
26            let entity_for_ok = entity.clone();
27            let entity_for_cancel = entity.clone();
28            modal
29                .title(div().text_size(px(16.)).child("Jump to line..."))
30                .keyboard(true)
31                .button_props(
32                    DialogButtonProps::default()
33                        .show_cancel(true)
34                        .cancel_text("Cancel")
35                        .cancel_variant(ButtonVariant::Secondary)
36                        .ok_text("Jump")
37                        .ok_variant(ButtonVariant::Primary),
38                )
39                .overlay_closable(true)
40                .close_button(false)
41                .child(Input::new(&jump_to_line_input))
42                .on_ok(move |_, _window, cx| {
43                    let text = jump_to_line_input_clone.read(cx).value();
44                    let jump_result = editor_tab::extract_line_number(text);
45                    let is_ok = jump_result.is_ok();
46                    entity_for_ok.update(cx, |this, cx| {
47                        if let Ok(jump) = jump_result {
48                            this.pending_jump = Some(jump);
49                            this.jump_to_line_dialog_open = false;
50                            cx.notify();
51                        } else {
52                            this.pending_jump = None;
53                        }
54                    });
55
56                    is_ok
57                })
58                .on_cancel(move |_, _, cx| {
59                    entity_for_cancel.update(cx, |this, cx| {
60                        this.jump_to_line_dialog_open = false;
61                        cx.notify();
62                    });
63                    true
64                })
65        });
66    }

82    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
83        cx.update(gpui_component::init);
84        cx.update(|cx| {
85            cx.set_global(SharedAppState::new(
86                Settings::new(),
87                Arc::new(Mutex::new(Vec::new())),
88            ));
89            cx.set_global(WindowManager::new());
90        });
91        let fulgur_slot: Rc<RefCell<Option<Entity<Fulgur>>>> = Rc::new(RefCell::new(None));
92        let slot = Rc::clone(&fulgur_slot);
93        let window = cx
94            .update(|cx| {
95                cx.open_window(Default::default(), |window, cx| {
96                    let window_id = window.window_handle().window_id();
97                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
98                    *slot.borrow_mut() = Some(fulgur.clone());
99                    cx.new(|cx| gpui_component::Root::new(fulgur, window, cx))
100                })
101            })
102            .expect("failed to open test window");
103        let fulgur = fulgur_slot
104            .borrow_mut()
105            .take()
106            .expect("expected fulgur entity");
107        let visual_cx = VisualTestContext::from_window(window.into(), cx);
108        (fulgur, visual_cx)
109    }

115        let (fulgur, mut visual_cx) = setup_fulgur(cx);

124        let after = fulgur.read_with(&visual_cx, |this, _| this.jump_to_line_dialog_open);

149    #[gpui::test]
150    fn test_handle_pending_jump_applies_cursor_line(cx: &mut TestAppContext) {
151        let (fulgur, mut visual_cx) = setup_fulgur(cx);
152        // Set multi-line content in the active editor tab
153        visual_cx.update(|window, cx| {
154            fulgur.update(cx, |this, cx| {
155                if let Some(Tab::Editor(editor_tab)) = this.tabs.first_mut() {
156                    editor_tab.content.update(cx, |input, cx| {
157                        input.set_value("line one\nline two\nline three", window, cx);
158                    });
159                }
160            });
161        });
162        // Jump to line 2 (0-indexed: line = 1)
163        visual_cx.update(|window, cx| {
164            fulgur.update(cx, |this, cx| {
165                this.pending_jump = Some(Jump {
166                    line: 1,
167                    character: None,
168                });
169                this.handle_pending_jump_to_line(window, cx);
170            });
171        });
172        let cursor_line = fulgur.read_with(&visual_cx, |this, cx| {
173            this.tabs
174                .first()
175                .and_then(|t| t.as_editor())
176                .map(|e| e.content.read(cx).cursor_position().line)
177        });
178        assert_eq!(cursor_line, Some(1), "cursor should be on line index 1");
179    }
