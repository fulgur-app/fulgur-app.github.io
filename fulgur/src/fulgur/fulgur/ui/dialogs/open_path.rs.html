11/// Validate a raw path string entered by the user for the open-file dialog.
12///
13/// ### Arguments
14/// - `path_str`: The raw path string to validate (may include leading/trailing whitespace)
15///
16/// ### Returns
17/// - `Ok(PathBuf)`: a canonicalised `PathBuf` when the string names an existing file
18/// - `Err(SharedString)`: a human-readable error message for any other case
19fn validate_open_path(path_str: &str) -> Result<PathBuf, SharedString> {
20    let trimmed = path_str.trim();
21    if trimmed.is_empty() {
22        return Err(SharedString::from("Please enter a file path"));
23    }
24    let path = PathBuf::from(trimmed);
25    if !path.exists() {
26        return Err(SharedString::from(format!(
27            "Path does not exist: {trimmed}"
28        )));
29    }
30    if !path.is_file() {
31        return Err(SharedString::from(format!("Path is not a file: {trimmed}")));
32    }
33    Ok(path)
34}

36impl Fulgur {
37    pub fn show_open_from_path_dialog(&self, window: &mut Window, cx: &mut Context<Self>) {
38        let entity = cx.entity().clone();
39        let path_browser = cx.new(|cx| PathBrowser::new(window, cx));
40        let input = path_browser.read(cx).input().clone();
41        let input_clone = input.clone();
42        window.open_alert_dialog(cx, move |modal, window, cx| {
43            let focus_handle = input.read(cx).focus_handle(cx);
44            window.focus(&focus_handle, cx);
45            let entity_ok = entity.clone();
46            let input_ok = input_clone.clone();
47            let path_browser = path_browser.clone();
48            modal
49                .title(div().text_size(px(16.)).child("Open file from path..."))
50                .keyboard(true)
51                .button_props(
52                    DialogButtonProps::default()
53                        .show_cancel(true)
54                        .cancel_text("Cancel")
55                        .cancel_variant(ButtonVariant::Secondary)
56                        .ok_text("Open")
57                        .ok_variant(ButtonVariant::Primary),
58                )
59                .close_button(false)
60                .child(path_browser)
61                .on_ok(move |_, window: &mut Window, cx| {
62                    let path_str = input_ok.read(cx).value().to_string();
63                    match validate_open_path(&path_str) {
64                        Ok(path) => {
65                            entity_ok.update(cx, |this, cx| {
66                                this.do_open_file(window, cx, path);
67                            });
68                            true
69                        }
70                        Err(msg) => {
71                            window.push_notification((NotificationType::Error, msg), cx);
72                            false
73                        }
74                    }
75                })
76                .on_cancel(|_, _, _| true)
77        });
78    }

87    fn make_temp_file(dir: &TempDir, name: &str, content: &str) -> std::path::PathBuf {
88        let path = dir.path().join(name);
89        std::fs::write(&path, content).expect("failed to write temp file");
90        path
91    }

153    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
154        cx.update(gpui_component::init);
155        cx.update(|cx| {
156            cx.set_global(SharedAppState::new(
157                Settings::new(),
158                Arc::new(Mutex::new(Vec::new())),
159                None,
160                None,
161            ));
162            cx.set_global(WindowManager::new());
163        });
164        let fulgur_slot: Rc<RefCell<Option<Entity<Fulgur>>>> = Rc::new(RefCell::new(None));
165        let slot = Rc::clone(&fulgur_slot);
166        let window = cx
167            .update(|cx| {
168                cx.open_window(WindowOptions::default(), |window, cx| {
169                    let window_id = window.window_handle().window_id();
170                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
171                    *slot.borrow_mut() = Some(fulgur.clone());
172                    cx.new(|cx| gpui_component::Root::new(fulgur, window, cx))
173                })
174            })
175            .expect("failed to open test window");
176        let fulgur = fulgur_slot
177            .borrow_mut()
178            .take()
179            .expect("expected fulgur entity");
180        let visual_cx = VisualTestContext::from_window(window.into(), cx);
181        (fulgur, visual_cx)
182    }
