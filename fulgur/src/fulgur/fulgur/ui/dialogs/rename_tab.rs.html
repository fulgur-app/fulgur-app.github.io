8impl Fulgur {
9    /// Show the rename tab dialog for a renameable tab
10    ///
11    /// ### Arguments
12    /// - `tab_id`: The identifier of the tab to rename
13    /// - `window`: The window to show the dialog in
14    /// - `cx`: The application context
15    pub fn show_rename_tab_dialog(
16        &mut self,
17        tab_id: TabId,
18        window: &mut Window,
19        cx: &mut Context<Self>,
20    ) {
21        let Some(current_title) = self.tab_entity_of(tab_id, cx).and_then(|tab| {
22            tab.read(cx)
23                .as_editor()
24                .filter(|editor_tab| editor_tab.is_renameable())
25                .map(|editor_tab| editor_tab.title.to_string())
26        }) else {
27            return;
28        };
29        let initial_value = if current_title.starts_with(UNTITLED) {
30            String::new()
31        } else {
32            current_title
33        };
34        self.rename_tab_input.update(cx, |input_state, cx| {
35            input_state.set_value(initial_value, window, cx);
36            cx.notify();
37        });
38        let rename_tab_input = self.rename_tab_input.clone();
39        let entity = cx.entity().clone();
40        window.open_alert_dialog(cx, move |modal, window, cx| {
41            let focus_handle = rename_tab_input.read(cx).focus_handle(cx);
42            window.focus(&focus_handle, cx);
43            let rename_tab_input_clone = rename_tab_input.clone();
44            let entity_for_ok = entity.clone();
45            modal
46                .title(div().text_size(px(16.)).child("Rename tab..."))
47                .keyboard(true)
48                .button_props(
49                    DialogButtonProps::default()
50                        .show_cancel(true)
51                        .cancel_text("Cancel")
52                        .cancel_variant(ButtonVariant::Secondary)
53                        .ok_text("Rename")
54                        .ok_variant(ButtonVariant::Primary),
55                )
56                .close_button(false)
57                .child(Input::new(&rename_tab_input))
58                .on_ok(move |_, window, cx| {
59                    let name = rename_tab_input_clone.read(cx).value().to_string();
60                    entity_for_ok.update(cx, |this, cx| this.rename_tab(tab_id, &name, window, cx))
61                })
62        });
63    }

65    /// Whether a tab can be renamed by the user
66    ///
67    /// ### Arguments
68    /// - `tab_id`: The identifier of the tab to inspect
69    /// - `cx`: The application context
70    ///
71    /// ### Returns
72    /// - `bool`: `true` when the tab is an editor tab with no associated file
73    pub fn is_tab_renameable(&self, tab_id: TabId, cx: &gpui::App) -> bool {
74        self.tab_entity_of(tab_id, cx).is_some_and(|tab| {
75            tab.read(cx)
76                .as_editor()
77                .is_some_and(EditorTab::is_renameable)
78        })
79    }
80}

97    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
98        cx.update(gpui_component::init);
99        cx.update(|cx| {
100            cx.set_global(SharedAppState::new(
101                Settings::new(),
102                Arc::new(Mutex::new(Vec::new())),
103                None,
104                None,
105            ));
106            cx.set_global(WindowManager::new());
107        });
108        let fulgur_slot: Rc<RefCell<Option<Entity<Fulgur>>>> = Rc::new(RefCell::new(None));
109        let slot = Rc::clone(&fulgur_slot);
110        let window = cx
111            .update(|cx| {
112                cx.open_window(WindowOptions::default(), |window, cx| {
113                    let window_id = window.window_handle().window_id();
114                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
115                    *slot.borrow_mut() = Some(fulgur.clone());
116                    cx.new(|cx| gpui_component::Root::new(fulgur, window, cx))
117                })
118            })
119            .expect("failed to open test window");
120        let fulgur = fulgur_slot
121            .borrow_mut()
122            .take()
123            .expect("expected fulgur entity");
124        let visual_cx = VisualTestContext::from_window(window.into(), cx);
125        (fulgur, visual_cx)
126    }

230    fn test_rename_tab_refuses_file_backed_tab(cx: &mut TestAppContext) {
231        let (fulgur, mut visual_cx) = setup_fulgur(cx);
232        let tab_id = first_tab_id(&fulgur, &mut visual_cx);
233        visual_cx.update(|_, cx| {
234            fulgur.update(cx, |this, cx| {
235                this.tabs
236                    .first()
237                    .expect("expected at least one tab")
238                    .clone()
239                    .update(cx, |tab, _cx| {
240                        if let Some(editor_tab) = tab.as_editor_mut() {
241                            editor_tab.location = TabLocation::Local("/tmp/notes.txt".into());
242                        }
243                    });
244            });
245        });
246        let renamed = visual_cx.update(|window, cx| {
247            fulgur.update(cx, |this, cx| {
248                this.rename_tab(tab_id, "Something else", window, cx)
249            })
250        });
251        assert!(!renamed, "a file-backed tab must not be renameable");
252        let is_renameable =
253            fulgur.read_with(&visual_cx, |this, cx| this.is_tab_renameable(tab_id, cx));
254        assert!(!is_renameable);
255    }
