2use std::rc::Rc;

8use gpui_component::{
9    WindowExt, button::ButtonVariant, dialog::DialogButtonProps, input::Input,
10    notification::NotificationType, v_flex,
11};
12use zeroize::Zeroizing;

14use crate::fulgur::Fulgur;

16impl Fulgur {
17    /// Shows a modal with an optional username field (when `user` is `None`) and a masked
18    /// password field. Calls `on_confirm` with the resolved username and password on submit.
19    ///
20    /// ### Arguments
21    /// - `window`: The window to show the dialog in
22    /// - `cx`: The application context
23    /// - `host`: Remote hostname displayed in the dialog title
24    /// - `port`: SSH port (appended to title only when not 22)
25    /// - `user`: Pre-filled username, or `None` to show an editable username field
26    /// - `on_confirm`: Callback invoked with `(username, password, window, cx)` on OK
27    pub fn show_ssh_password_dialog(
28        &self,
29        window: &mut Window,
30        cx: &mut Context<Self>,
31        host: &str,
32        port: u16,
33        user: Option<String>,
34        on_confirm: impl Fn(String, Zeroizing<String>, &mut Window, &mut App) + 'static,
35    ) {
36        let title: SharedString = if port == 22 {
37            format!("SSH login for {host}").into()
38        } else {
39            format!("SSH login for {host}:{port}").into()
40        };
41        let show_user_field = user.is_none();
42        let prefilled_user = user.unwrap_or_default();
43
44        let user_input =
45            cx.new(|cx| gpui_component::input::InputState::new(window, cx).placeholder("Username"));
46        let password_input = cx.new(|cx| {
47            gpui_component::input::InputState::new(window, cx)
48                .placeholder("Password")
49                .masked(true)
50        });
51
52        let user_input_ok = user_input.clone();
53        let password_input_ok = password_input.clone();
54        let prefilled_ok = prefilled_user.clone();
55        let on_confirm = Arc::new(on_confirm);
56        let has_initialized_focus = Rc::new(Cell::new(false));
57
58        window.open_alert_dialog(cx, move |modal, window, cx| {
59            if !has_initialized_focus.get() {
60                let focus_handle = if show_user_field {
61                    user_input.read(cx).focus_handle(cx)
62                } else {
63                    password_input.read(cx).focus_handle(cx)
64                };
65                window.focus(&focus_handle, cx);
66                has_initialized_focus.set(true);
67            }
68
69            let user_input_inner = user_input_ok.clone();
70            let password_input_inner = password_input_ok.clone();
71            let prefilled_inner = prefilled_ok.clone();
72            let on_confirm_inner = Arc::clone(&on_confirm);
73
74            let m = modal
75                .title(div().text_size(px(16.)).child(title.clone()))
76                .keyboard(true)
77                .button_props(
78                    DialogButtonProps::default()
79                        .show_cancel(true)
80                        .cancel_text("Cancel")
81                        .cancel_variant(ButtonVariant::Secondary)
82                        .ok_text("Connect")
83                        .ok_variant(ButtonVariant::Primary),
84                )
85                .close_button(false);
86
87            let form = if show_user_field {
88                v_flex()
89                    .w_full()
90                    .gap_2()
91                    .child(Input::new(&user_input))
92                    .child(Input::new(&password_input))
93            } else {
94                v_flex().w_full().gap_2().child(Input::new(&password_input))
95            };
96
97            m.child(form)
98                .on_ok(move |_, window: &mut Window, cx| {
99                    let username = if show_user_field {
100                        user_input_inner.read(cx).value().to_string()
101                    } else {
102                        prefilled_inner.clone()
103                    };
104                    if username.trim().is_empty() {
105                        window.push_notification(
106                            (
107                                NotificationType::Error,
108                                SharedString::from("Username is required"),
109                            ),
110                            cx,
111                        );
112                        return false;
113                    }
114                    let password =
115                        Zeroizing::new(password_input_inner.read(cx).value().to_string());
116                    if password.is_empty() {
117                        window.push_notification(
118                            (
119                                NotificationType::Error,
120                                SharedString::from("Password is required"),
121                            ),
122                            cx,
123                        );
124                        return false;
125                    }
126                    on_confirm_inner(username, password, window, cx);
127                    true
128                })
129                .on_cancel(|_, _, _| true)
130        });
131    }
132}
