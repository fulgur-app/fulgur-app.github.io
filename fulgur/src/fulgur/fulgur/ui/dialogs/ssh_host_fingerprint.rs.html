1use gpui::{Context, ParentElement, SharedString, Styled, Window, div, px};
2use gpui_component::{WindowExt, button::ButtonVariant, dialog::DialogButtonProps};
3
4use crate::fulgur::{
5    Fulgur,
6    sync::ssh::session::{HostKeyDecision, HostKeyRequest},
7};
8
9impl Fulgur {
10    /// Show the SSH host key fingerprint dialog for TOFU verification.
11    ///
12    /// ### Arguments
13    /// - `window`: The window to show the dialog in
14    /// - `cx`: The application context
15    /// - `request`: The host key verification request from the SSH background thread
16    pub fn show_ssh_host_fingerprint_dialog(
17        &self,
18        window: &mut Window,
19        cx: &mut Context<Self>,
20        request: HostKeyRequest,
21    ) {
22        let host_label: SharedString = if request.port == 22 {
23            format!("Host: {}", request.host).into()
24        } else {
25            format!("Host: {}:{}", request.host, request.port).into()
26        };
27        let fingerprint_label: SharedString = request.fingerprint.into();
28        let tx = request.decision_tx;
29
30        window.open_alert_dialog(cx, move |modal, _, _| {
31            let tx_ok = tx.clone();
32            let tx_cancel = tx.clone();
33
34            modal
35                .title(div().text_size(px(16.)).child("Unknown SSH host"))
36                .keyboard(true)
37                .button_props(
38                    DialogButtonProps::default()
39                        .show_cancel(true)
40                        .cancel_text("Reject")
41                        .cancel_variant(ButtonVariant::Danger)
42                        .ok_text("Trust and connect")
43                        .ok_variant(ButtonVariant::Primary),
44                )
45                .close_button(false)
46                .child(div().text_sm().child(host_label.clone()))
47                .child(div().text_sm().mt_2().child("SHA-256 fingerprint:"))
48                .child(
49                    div()
50                        .text_xs()
51                        .font_family("monospace")
52                        .mt_1()
53                        .child(fingerprint_label.clone()),
54                )
55                .on_ok(move |_, _, _| {
56                    let _ = tx_ok.send(HostKeyDecision::Accept);
57                    true
58                })
59                .on_cancel(move |_, _, _| {
60                    let _ = tx_cancel.send(HostKeyDecision::Reject);
61                    true
62                })
63        });
64    }
65}
