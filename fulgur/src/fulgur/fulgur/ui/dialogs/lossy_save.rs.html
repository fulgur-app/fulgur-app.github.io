8impl Fulgur {
9    /// Show a confirmation dialog when saving would lose data through encoding.
10    ///
11    /// ### Arguments
12    /// - `tab_id`: Stable id of the tab being saved
13    /// - `encoding`: The current encoding label that cannot represent the text
14    /// - `window`: The window to show the dialog in
15    /// - `cx`: The application context
16    pub fn show_lossy_save_dialog(
17        &self,
18        tab_id: usize,
19        encoding: &str,
20        window: &mut Window,
21        cx: &mut Context<Self>,
22    ) {
23        let entity = cx.entity().clone();
24        let encoding = encoding.to_string();
25        window.open_alert_dialog(cx, move |modal, _, _| {
26            let entity_for_ok = entity.clone();
27            modal
28                .title(div().text_size(px(16.)).child("Cannot Save in Encoding"))
29                .keyboard(true)
30                .button_props(
31                    DialogButtonProps::default()
32                        .show_cancel(true)
33                        .cancel_text("Cancel")
34                        .cancel_variant(ButtonVariant::Secondary)
35                        .ok_text("Save as UTF-8")
36                        .ok_variant(ButtonVariant::Primary),
37                )
38                .overlay_closable(false)
39                .close_button(false)
40                .child(
41                    v_flex()
42                        .gap_2()
43                        .child(format!(
44                            "This file contains characters that cannot be represented in {encoding}."
45                        ))
46                        .child("Save it as UTF-8 instead? This changes the file's encoding."),
47                )
48                .on_ok(move |_, window, cx| {
49                    entity_for_ok.update(cx, |this, cx| {
50                        if let Some(Tab::Editor(editor_tab)) =
51                            this.tabs.iter_mut().find(|tab| tab.id() == tab_id)
52                        {
53                            editor_tab.encoding = UTF_8.to_string();
54                            editor_tab.lossy_decode = false;
55                        }
56                        this.save_file(window, cx);
57                    });
58                    true
59                })
60                .on_cancel(|_, _, _| true)
61        });
62    }
