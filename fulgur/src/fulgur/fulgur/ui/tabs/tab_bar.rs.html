69    /// Get the display title for a tab, including parent folder if there are duplicates
70    ///
71    /// ### Arguments
72    /// - `index`: The index of the tab
73    /// - `tab`: The tab to get the title for
74    ///
75    /// ### Returns
76    /// - `(String, Option<String>)`: A tuple of (filename, optional parent folder)
77    fn get_tab_display_title(&self, index: usize, tab: &Tab) -> (String, Option<String>) {
78        let base_title = tab.title();
79        if let Some(editor_tab) = tab.as_editor()
80            && let Some(ref path) = editor_tab.file_path
81        {
82            let filename = path.file_name().and_then(|n| n.to_str()).unwrap_or("");
83            let duplicate_count = self
84                .tabs
85                .iter()
86                .enumerate()
87                .filter(|(i, t)| {
88                    if *i == index {
89                        return false;
90                    }
91                    if let Some(editor) = t.as_editor()
92                        && let Some(ref other_path) = editor.file_path
93                        && let Some(other_filename) =
94                            other_path.file_name().and_then(|n| n.to_str())
95                    {
96                        return other_filename == filename;
97                    }
98                    false
99                })
100                .count();
101            if duplicate_count > 0
102                && let Some(parent) = path.parent()
103                && let Some(parent_name) = parent.file_name().and_then(|n| n.to_str())
104            {
105                return (filename.to_string(), Some(format!("../{}", parent_name)));
106            }
107
108            return (filename.to_string(), None);
109        }
110        (base_title.to_string(), None)
111    }

188    /// Handle show in file manager action from context menu
189    ///
190    /// Opens the file manager and selects the file associated with the given tab.
191    ///
192    /// On macOS, uses `open -R` to reveal and select the file in Finder.
193    /// On Windows, uses `explorer /select,` to select the file in Explorer.
194    /// On Linux, falls back to opening the parent directory, as there is no
195    /// universal "reveal file" command across desktop environments.
196    ///
197    /// ### Arguments
198    /// - `action`: The action carrying the tab index
199    /// - `_window`: The window context
200    /// - `_cx`: The application context
201    pub fn on_show_in_file_manager(
202        &mut self,
203        action: &ShowInFileManager,
204        _window: &mut Window,
205        _cx: &mut Context<Self>,
206    ) {
207        let Some(tab) = self.tabs.get(action.0) else {
208            return;
209        };
210        let Some(editor_tab) = tab.as_editor() else {
211            return;
212        };
213        let Some(ref file_path) = editor_tab.file_path else {
214            return;
215        };
216
217        let result = reveal_file_in_file_manager(file_path);
218        if let Err(e) = result {
219            log::error!("Failed to open file manager: {}", e);
220        }
221    }

262    /// Handle theme switching action
263    ///
264    /// Applies the selected theme, updates settings, refreshes windows, and rebuilds menus.
265    ///
266    /// ### Arguments
267    /// - `theme_name`: The name of the theme to switch to (as SharedString from action)
268    /// - `cx`: The application context
269    pub fn switch_to_theme(&mut self, theme_name: gpui::SharedString, cx: &mut Context<Self>) {
270        if let Some(theme_config) = ThemeRegistry::global(cx)
271            .themes()
272            .get(theme_name.as_ref())
273            .cloned()
274        {
275            Theme::global_mut(cx).apply_config(&theme_config);
276            self.settings.app_settings.theme = theme_name;
277            self.settings.app_settings.scrollbar_show = Some(cx.theme().scrollbar_show);
278            let _ = self.update_and_propagate_settings(cx);
279        }
280        cx.refresh_windows();
281        let menus =
282            crate::fulgur::ui::menus::build_menus(self.settings.recent_files.get_files(), None);
283        self.update_menus(menus, cx);
284    }

381    /// Render a ghost tab shown at the insertion point during a drag operation.
382    ///
383    /// The ghost tab previews where the dragged tab will land when dropped. It uses
384    /// a muted, semi-transparent style to distinguish it from real tabs.
385    ///
386    /// ### Arguments
387    /// - `dragged`: The dragged tab data (used for title and modified state)
388    /// - `cx`: The application context
389    ///
390    /// ### Returns
391    /// - `AnyElement`: The rendered ghost tab element
392    fn render_ghost_tab(
393        &self,
394        slot: usize,
395        dragged: &DraggedTab,
396        cx: &mut Context<Self>,
397    ) -> AnyElement {
398        let modified_indicator = if dragged.is_modified { " •" } else { "" };
399        div()
400            .id(("ghost-tab", slot))
401            .flex()
402            .items_center()
403            .h(TAB_BAR_HEIGHT)
404            .px_2()
405            .gap_2()
406            .border_r_1()
407            .border_b_0()
408            .border_color(cx.theme().border)
409            .bg(cx.theme().tab_active)
410            .opacity(0.45)
411            .child(
412                div()
413                    .pl_1()
414                    .text_sm()
415                    .text_color(cx.theme().tab_active_foreground)
416                    .child(format!("{}{}", dragged.title, modified_indicator)),
417            )
418            .on_drop(cx.listener(|this, dragged: &DraggedTab, window, cx| {
419                if let Some((slot, _)) = this.drag_ghost.take() {
420                    this.handle_tab_drop(dragged, slot, window, cx);
421                }
422            }))
423            .into_any_element()
424    }

426    /// Render all tabs, inserting a ghost tab at the current drag insertion point.
427    ///
428    /// During a drag operation, a ghost tab is rendered at the slot determined by
429    /// the most recent `on_drag_move` event. The ghost is suppressed for no-op
430    /// positions (where the tab would not actually move).
431    ///
432    /// ### Arguments
433    /// - `cx`: The application context
434    ///
435    /// ### Returns
436    /// - `Vec<AnyElement>`: Tab elements, with a ghost tab inserted at the drag target
437    fn render_tabs_with_slots(&self, cx: &mut Context<Self>) -> Vec<AnyElement> {
438        let ghost = if cx.has_active_drag() {
439            self.drag_ghost.as_ref().and_then(|(slot, dragged)| {
440                let from = dragged.tab_index;
441                let is_noop = *slot == from || *slot == from + 1;
442                if is_noop {
443                    None
444                } else {
445                    Some((*slot, dragged))
446                }
447            })
448        } else {
449            None
450        };
451        let capacity = self.tabs.len() + if ghost.is_some() { 1 } else { 0 };
452        let mut elements: Vec<AnyElement> = Vec::with_capacity(capacity);
453        if let Some((0, dragged)) = ghost {
454            elements.push(self.render_ghost_tab(0, dragged, cx));
455        }
456        for (index, tab) in self.tabs.iter().enumerate() {
457            elements.push(self.render_tab(index, tab, cx));
458            if let Some((slot, dragged)) = ghost
459                && slot == index + 1
460            {
461                elements.push(self.render_ghost_tab(slot, dragged, cx));
462            }
463        }
464        elements
465    }

467    /// Render a single tab
468    ///
469    /// ### Arguments
470    /// - `index`: The index of the tab
471    /// - `tab`: The tab to render
472    /// - `cx`: The application context
473    ///
474    /// ### Returns
475    /// - `AnyElement`: The rendered tab element
476    pub fn render_tab(&self, index: usize, tab: &Tab, cx: &mut Context<Self>) -> AnyElement {
477        let tab_id = tab.id();
478        let is_active = match self.active_tab_index {
479            Some(active_index) => index == active_index,
480            None => false,
481        };
482        let has_tabs_on_left = index > 0;
483        let has_tabs_on_right = index < self.tabs.len() - 1;
484        let total_tabs = self.tabs.len();
485        let file_path = tab.as_editor().and_then(|editor_tab| {
486            editor_tab
487                .file_path
488                .as_ref()
489                .and_then(|path| path.to_str().map(|s| s.to_string()))
490        });
491        let has_file_path = file_path.is_some();
492        let is_editor_tab = tab.as_editor().is_some();
493        let other_windows: Vec<(String, WeakEntity<Fulgur>)> = {
494            let manager = cx.global::<WindowManager>();
495            let current_window_id = self.window_id;
496            manager
497                .get_all_window_ids()
498                .into_iter()
499                .filter(|id| *id != current_window_id)
500                .filter_map(|id| {
501                    manager
502                        .get_window_name(id)
503                        .map(|name| name.to_string())
504                        .zip(manager.get_window(id))
505                })
506                .collect()
507        };
508        let source_entity = cx.entity().downgrade();
509        let cached_file_size = tab
510            .as_editor()
511            .and_then(|editor_tab| editor_tab.file_size_bytes)
512            .map(components_utils::format_file_size);
513        let cached_last_modified = tab
514            .as_editor()
515            .and_then(|editor_tab| editor_tab.file_last_modified)
516            .and_then(components_utils::format_system_time);
517        let mut tab_div = div()
518            .id(("tab", tab_id))
519            .flex()
520            .items_center()
521            .h(TAB_BAR_HEIGHT)
522            .px_2()
523            .gap_2()
524            .border_r_1()
525            .border_b_1()
526            .border_color(cx.theme().border)
527            .on_mouse_down(
528                MouseButton::Left,
529                cx.listener(move |this, _, window, cx: &mut Context<'_, Fulgur>| {
530                    if !is_active {
531                        this.set_active_tab(index, window, cx);
532                    }
533                }),
534            );
535        if is_active {
536            tab_div = tab_div.bg(cx.theme().tab_active).border_b_0();
537            self.set_title(Some(tab.title().to_string()), cx);
538        } else {
539            tab_div = tab_div
540                .bg(cx.theme().tab)
541                .hover(|this| this.bg(cx.theme().muted))
542                .cursor_pointer();
543        }
544        if let Some(path) = file_path {
545            tab_div = tab_div.tooltip(move |window, cx| {
546                let path_clone = path.clone();
547                let file_size = cached_file_size.clone();
548                let last_modified = cached_last_modified.clone();
549                Tooltip::element(move |_, cx| {
550                    let mut tooltip = v_flex().gap_1().py_2().px_1().child(
551                        h_flex()
552                            .gap_3()
553                            .child(CustomIcon::File.icon())
554                            .child(path_clone.clone())
555                            .text_sm()
556                            .font_semibold(),
557                    );
558                    let mut details = h_flex().gap_4().justify_between();
559                    if let Some(ref size) = file_size {
560                        details = details.child(
561                            div()
562                                .child(format!("Size: {}", size))
563                                .text_xs()
564                                .text_color(cx.theme().muted_foreground),
565                        );
566                    }
567                    if let Some(ref last_modified) = last_modified {
568                        details = details.child(
569                            div()
570                                .child(format!("Last Modified: {}", last_modified))
571                                .text_xs()
572                                .text_color(cx.theme().muted_foreground),
573                        );
574                    }
575                    tooltip = tooltip.child(details);
576                    tooltip
577                })
578                .build(window, cx)
579            });
580        }
581        let (filename, folder) = self.get_tab_display_title(index, tab);
582        let modified_indicator = if tab.is_modified() { " •" } else { "" };
583        let mut title_container = div().flex().items_center().gap_1().pl_1().child(
584            div()
585                .text_sm()
586                .text_color(if is_active {
587                    cx.theme().tab_active_foreground
588                } else {
589                    cx.theme().tab_foreground
590                })
591                .child(format!("{}{}", filename, modified_indicator)),
592        );
593        if let Some(folder_path) = folder {
594            title_container = title_container.child(
595                div()
596                    .text_xs()
597                    .italic()
598                    .text_color(if is_active {
599                        cx.theme().tab_active_foreground
600                    } else {
601                        cx.theme().tab_foreground
602                    })
603                    .child(folder_path),
604            );
605        }
606        let is_markdown_preview = tab.as_markdown_preview().is_some();
607        let title: SharedString = tab.title().to_string().into();
608        let is_modified = tab.is_modified();
609        let mut tab_with_content = tab_div
610            .child(title_container)
611            .child(
612                Button::new(("close-tab", tab_id))
613                    .icon(CustomIcon::Close)
614                    .ghost()
615                    .xsmall()
616                    .cursor_pointer()
617                    .on_click(cx.listener(move |this, _, window, cx| {
618                        cx.stop_propagation();
619                        this.close_tab(tab_id, window, cx);
620                    })),
621            )
622            .on_mouse_down(
623                MouseButton::Middle,
624                cx.listener(move |this, _, window, cx| {
625                    this.close_tab(tab_id, window, cx);
626                }),
627            );
628        if !is_markdown_preview {
629            let is_source = cx.has_active_drag()
630                && self
631                    .drag_ghost
632                    .as_ref()
633                    .map(|(_, d)| d.tab_index == index)
634                    .unwrap_or(false);
635            if is_source {
636                tab_with_content = tab_with_content.opacity(0.45);
637            }
638            tab_with_content = tab_with_content
639                .on_drag(
640                    DraggedTab {
641                        tab_index: index,
642                        title,
643                        is_modified,
644                    },
645                    |dragged, _, _, cx| cx.new(|_| dragged.clone()),
646                )
647                .on_drag_move::<DraggedTab>(cx.listener(
648                    move |this, event: &DragMoveEvent<DraggedTab>, _window, cx| {
649                        let cursor = event.event.position;
650                        let bounds = event.bounds;
651                        if cursor.x < bounds.origin.x
652                            || cursor.x > bounds.origin.x + bounds.size.width
653                            || cursor.y < bounds.origin.y
654                            || cursor.y > bounds.origin.y + bounds.size.height
655                        {
656                            return;
657                        }
658                        let slot = if cursor.x < bounds.origin.x + bounds.size.width * 0.5 {
659                            index
660                        } else {
661                            index + 1
662                        };
663                        let dragged = event.drag(cx).clone();
664                        this.drag_ghost = Some((slot, dragged));
665                        cx.notify();
666                    },
667                ))
668                .on_drop(cx.listener(|this, dragged: &DraggedTab, window, cx| {
669                    if let Some((slot, _)) = this.drag_ghost.take() {
670                        this.handle_tab_drop(dragged, slot, window, cx);
671                    }
672                }));
673        }
674        let tab_with_content = tab_with_content.context_menu(move |this, window, cx| {
675            let other_windows_clone = other_windows.clone();
676            let source_entity_clone = source_entity.clone();
677            let this = this
678                .menu_with_disabled(
679                    crate::fulgur::ui::components_utils::reveal_in_file_manager_label(),
680                    Box::new(ShowInFileManager(index)),
681                    !has_file_path,
682                )
683                .menu_with_disabled(
684                    "Duplicate Tab",
685                    Box::new(DuplicateTab(index)),
686                    !is_editor_tab,
687                );
688            let this = if is_editor_tab {
689                this.submenu("Send to...", window, cx, move |sub, _window, _cx| {
690                    let mut sub = sub;
691                    for (name, weak_tgt) in &other_windows_clone {
692                        let label = format!("Window {}", name);
693                        let src = source_entity_clone.clone();
694                        let tgt = weak_tgt.clone();
695                        sub = sub.item(PopupMenuItem::new(label).on_click(move |_, _, cx| {
696                            let Some(src_entity) = src.upgrade() else {
697                                return;
698                            };
699                            let Some(tgt_entity) = tgt.upgrade() else {
700                                return;
701                            };
702                            let data = src_entity.update(cx, |fulgur, cx| {
703                                fulgur.extract_tab_transfer_data(tab_id, cx)
704                            });
705                            let Some(data) = data else { return };
706                            tgt_entity.update(cx, |fulgur, cx| {
707                                fulgur.pending_tab_transfer = Some(data);
708                                cx.notify();
709                            });
710                            src_entity.update(cx, |fulgur, cx| {
711                                fulgur.pending_tab_removal = Some(tab_id);
712                                cx.notify();
713                            });
714                        }));
715                    }
716                    if !other_windows_clone.is_empty() {
717                        sub = sub.separator();
718                    }
719                    let src = source_entity_clone.clone();
720                    sub = sub.item(PopupMenuItem::new("New Window").on_click(move |_, _, cx| {
721                        let Some(src_entity) = src.upgrade() else {
722                            return;
723                        };
724                        let data = src_entity.update(cx, |fulgur, cx| {
725                            fulgur.extract_tab_transfer_data(tab_id, cx)
726                        });
727                        let Some(data) = data else { return };
728                        src_entity.update(cx, |fulgur, cx| {
729                            fulgur.pending_tab_removal = Some(tab_id);
730                            cx.notify();
731                        });
732                        src_entity.update(cx, |fulgur, cx| {
733                            fulgur.open_new_window_with_tab(data, cx);
734                        });
735                    }));
736                    sub
737                })
738            } else {
739                this.menu_with_disabled("Send to...", Box::new(SendTabToWindowNoOp), true)
740            };
741            this.separator()
742                .menu("Close Tab", Box::new(CloseTabAction(tab_id)))
743                .menu_with_disabled(
744                    "Close Tabs to the Left",
745                    Box::new(CloseTabsToLeft(index)),
746                    !has_tabs_on_left,
747                )
748                .menu_with_disabled(
749                    "Close Tabs to the Right",
750                    Box::new(CloseTabsToRight(index)),
751                    !has_tabs_on_right,
752                )
753                .separator()
754                .menu_with_disabled(
755                    "Close All Tabs",
756                    Box::new(CloseAllTabsAction),
757                    total_tabs == 0,
758                )
759                .menu_with_disabled(
760                    "Close All Other Tabs",
761                    Box::new(CloseAllOtherTabs(index)),
762                    total_tabs <= 1,
763                )
764        });
765        tab_with_content.into_any_element()
766    }

769/// Reveals a file in the platform's native file manager with the file selected.
770///
771/// - **macOS**: `open -R <path>` — reveals and selects the file in Finder
772/// - **Windows**: `explorer /select,<path>` — selects the file in Explorer
773/// - **Linux**: falls back to opening the parent directory via the `open` crate,
774///   as there is no universal "reveal" command across desktop environments
775///
776/// ### Arguments
777/// - `file_path`: The path of the file to reveal
778///
779/// ### Returns
780/// - `Ok(())` on success, `Err` with an error message on failure
781fn reveal_file_in_file_manager(file_path: &std::path::Path) -> Result<(), String> {
782    #[cfg(target_os = "macos")]
783    {
784        std::process::Command::new("open")
785            .arg("-R")
786            .arg(file_path)
787            .spawn()
788            .map(|_| ())
789            .map_err(|e| e.to_string())
790    }
791
792    #[cfg(target_os = "windows")]
793    {
794        std::process::Command::new("explorer")
795            .arg(format!("/select,{}", file_path.display()))
796            .spawn()
797            .map(|_| ())
798            .map_err(|e| e.to_string())
799    }
800
801    #[cfg(not(any(target_os = "macos", target_os = "windows")))]
802    {
803        let parent = file_path
804            .parent()
805            .ok_or_else(|| "File has no parent directory".to_string())?;
806        open::that(parent).map_err(|e| e.to_string())
807    }
808}
