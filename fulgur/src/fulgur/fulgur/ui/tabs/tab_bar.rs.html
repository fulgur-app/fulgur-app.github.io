68    /// Get the display title for a tab, including parent folder if there are duplicates
69    ///
70    /// ### Arguments
71    /// - `index`: The index of the tab
72    /// - `tab`: The tab to get the title for
73    ///
74    /// ### Returns
75    /// - `(String, Option<String>)`: A tuple of (filename, optional parent folder)
76    fn get_tab_display_title(&self, index: usize, tab: &Tab) -> (String, Option<String>) {
77        let base_title = tab.title();
78        if let Some(editor_tab) = tab.as_editor()
79            && let Some(ref path) = editor_tab.file_path
80        {
81            let filename = path.file_name().and_then(|n| n.to_str()).unwrap_or("");
82            let duplicate_count = self
83                .tabs
84                .iter()
85                .enumerate()
86                .filter(|(i, t)| {
87                    if *i == index {
88                        return false;
89                    }
90                    if let Some(editor) = t.as_editor()
91                        && let Some(ref other_path) = editor.file_path
92                        && let Some(other_filename) =
93                            other_path.file_name().and_then(|n| n.to_str())
94                    {
95                        return other_filename == filename;
96                    }
97                    false
98                })
99                .count();
100            if duplicate_count > 0
101                && let Some(parent) = path.parent()
102                && let Some(parent_name) = parent.file_name().and_then(|n| n.to_str())
103            {
104                return (filename.to_string(), Some(format!("../{}", parent_name)));
105            }
106
107            return (filename.to_string(), None);
108        }
109        (base_title.to_string(), None)
110    }

187    /// Handle show in file manager action from context menu
188    ///
189    /// Opens the file manager and selects the file associated with the given tab.
190    ///
191    /// On macOS, uses `open -R` to reveal and select the file in Finder.
192    /// On Windows, uses `explorer /select,` to select the file in Explorer.
193    /// On Linux, falls back to opening the parent directory, as there is no
194    /// universal "reveal file" command across desktop environments.
195    ///
196    /// ### Arguments
197    /// - `action`: The action carrying the tab index
198    /// - `_window`: The window context
199    /// - `_cx`: The application context
200    pub fn on_show_in_file_manager(
201        &mut self,
202        action: &ShowInFileManager,
203        _window: &mut Window,
204        _cx: &mut Context<Self>,
205    ) {
206        let Some(tab) = self.tabs.get(action.0) else {
207            return;
208        };
209        let Some(editor_tab) = tab.as_editor() else {
210            return;
211        };
212        let Some(ref file_path) = editor_tab.file_path else {
213            return;
214        };
215
216        let result = reveal_file_in_file_manager(file_path);
217        if let Err(e) = result {
218            log::error!("Failed to open file manager: {}", e);
219        }
220    }

261    /// Handle theme switching action
262    ///
263    /// Applies the selected theme, updates settings, refreshes windows, and rebuilds menus.
264    ///
265    /// ### Arguments
266    /// - `theme_name`: The name of the theme to switch to (as SharedString from action)
267    /// - `cx`: The application context
268    pub fn switch_to_theme(&mut self, theme_name: gpui::SharedString, cx: &mut Context<Self>) {
269        if let Some(theme_config) = ThemeRegistry::global(cx)
270            .themes()
271            .get(theme_name.as_ref())
272            .cloned()
273        {
274            Theme::global_mut(cx).apply_config(&theme_config);
275            self.settings.app_settings.theme = theme_name;
276            self.settings.app_settings.scrollbar_show = Some(cx.theme().scrollbar_show);
277            let _ = self.update_and_propagate_settings(cx);
278        }
279        cx.refresh_windows();
280        let menus =
281            crate::fulgur::ui::menus::build_menus(self.settings.recent_files.get_files(), None);
282        self.update_menus(menus, cx);
283    }

380    /// Render a ghost tab shown at the insertion point during a drag operation.
381    ///
382    /// The ghost tab previews where the dragged tab will land when dropped. It uses
383    /// a muted, semi-transparent style to distinguish it from real tabs.
384    ///
385    /// ### Arguments
386    /// - `dragged`: The dragged tab data (used for title and modified state)
387    /// - `cx`: The application context
388    ///
389    /// ### Returns
390    /// - `AnyElement`: The rendered ghost tab element
391    fn render_ghost_tab(
392        &self,
393        slot: usize,
394        dragged: &DraggedTab,
395        cx: &mut Context<Self>,
396    ) -> AnyElement {
397        let modified_indicator = if dragged.is_modified { " •" } else { "" };
398        div()
399            .id(("ghost-tab", slot))
400            .flex()
401            .items_center()
402            .h(TAB_BAR_HEIGHT)
403            .px_2()
404            .gap_2()
405            .border_r_1()
406            .border_b_0()
407            .border_color(cx.theme().border)
408            .bg(cx.theme().tab_active)
409            .opacity(0.45)
410            .child(
411                div()
412                    .pl_1()
413                    .text_sm()
414                    .text_color(cx.theme().tab_active_foreground)
415                    .child(format!("{}{}", dragged.title, modified_indicator)),
416            )
417            .on_drop(cx.listener(|this, dragged: &DraggedTab, window, cx| {
418                if let Some((slot, _)) = this.drag_ghost.take() {
419                    this.handle_tab_drop(dragged, slot, window, cx);
420                }
421            }))
422            .into_any_element()
423    }

425    /// Render all tabs, inserting a ghost tab at the current drag insertion point.
426    ///
427    /// During a drag operation, a ghost tab is rendered at the slot determined by
428    /// the most recent `on_drag_move` event. The ghost is suppressed for no-op
429    /// positions (where the tab would not actually move).
430    ///
431    /// ### Arguments
432    /// - `cx`: The application context
433    ///
434    /// ### Returns
435    /// - `Vec<AnyElement>`: Tab elements, with a ghost tab inserted at the drag target
436    fn render_tabs_with_slots(&self, cx: &mut Context<Self>) -> Vec<AnyElement> {
437        let ghost = if cx.has_active_drag() {
438            self.drag_ghost.as_ref().and_then(|(slot, dragged)| {
439                let from = dragged.tab_index;
440                let is_noop = *slot == from || *slot == from + 1;
441                if is_noop {
442                    None
443                } else {
444                    Some((*slot, dragged))
445                }
446            })
447        } else {
448            None
449        };
450        let capacity = self.tabs.len() + if ghost.is_some() { 1 } else { 0 };
451        let mut elements: Vec<AnyElement> = Vec::with_capacity(capacity);
452        if let Some((0, dragged)) = ghost {
453            elements.push(self.render_ghost_tab(0, dragged, cx));
454        }
455        for (index, tab) in self.tabs.iter().enumerate() {
456            elements.push(self.render_tab(index, tab, cx));
457            if let Some((slot, dragged)) = ghost
458                && slot == index + 1
459            {
460                elements.push(self.render_ghost_tab(slot, dragged, cx));
461            }
462        }
463        elements
464    }

466    /// Render a single tab
467    ///
468    /// ### Arguments
469    /// - `index`: The index of the tab
470    /// - `tab`: The tab to render
471    /// - `cx`: The application context
472    ///
473    /// ### Returns
474    /// - `AnyElement`: The rendered tab element
475    pub fn render_tab(&self, index: usize, tab: &Tab, cx: &mut Context<Self>) -> AnyElement {
476        let tab_id = tab.id();
477        let is_active = match self.active_tab_index {
478            Some(active_index) => index == active_index,
479            None => false,
480        };
481        let has_tabs_on_left = index > 0;
482        let has_tabs_on_right = index < self.tabs.len() - 1;
483        let total_tabs = self.tabs.len();
484        let file_path = tab.as_editor().and_then(|editor_tab| {
485            editor_tab
486                .file_path
487                .as_ref()
488                .and_then(|path| path.to_str().map(|s| s.to_string()))
489        });
490        let has_file_path = file_path.is_some();
491        let is_editor_tab = tab.as_editor().is_some();
492        let cached_file_size = tab
493            .as_editor()
494            .and_then(|editor_tab| editor_tab.file_size_bytes)
495            .map(components_utils::format_file_size);
496        let cached_last_modified = tab
497            .as_editor()
498            .and_then(|editor_tab| editor_tab.file_last_modified)
499            .and_then(components_utils::format_system_time);
500        let mut tab_div = div()
501            .id(("tab", tab_id))
502            .flex()
503            .items_center()
504            .h(TAB_BAR_HEIGHT)
505            .px_2()
506            .gap_2()
507            .border_r_1()
508            .border_b_1()
509            .border_color(cx.theme().border)
510            .on_mouse_down(
511                MouseButton::Left,
512                cx.listener(move |this, _, window, cx: &mut Context<'_, Fulgur>| {
513                    if !is_active {
514                        this.set_active_tab(index, window, cx);
515                    }
516                }),
517            );
518        if is_active {
519            tab_div = tab_div.bg(cx.theme().tab_active).border_b_0();
520            self.set_title(Some(tab.title().to_string()), cx);
521        } else {
522            tab_div = tab_div
523                .bg(cx.theme().tab)
524                .hover(|this| this.bg(cx.theme().muted))
525                .cursor_pointer();
526        }
527        if let Some(path) = file_path {
528            tab_div = tab_div.tooltip(move |window, cx| {
529                let path_clone = path.clone();
530                let file_size = cached_file_size.clone();
531                let last_modified = cached_last_modified.clone();
532                Tooltip::element(move |_, cx| {
533                    let mut tooltip = v_flex().gap_1().py_2().px_1().child(
534                        h_flex()
535                            .gap_3()
536                            .child(CustomIcon::File.icon())
537                            .child(path_clone.clone())
538                            .text_sm()
539                            .font_semibold(),
540                    );
541                    let mut details = h_flex().gap_4().justify_between();
542                    if let Some(ref size) = file_size {
543                        details = details.child(
544                            div()
545                                .child(format!("Size: {}", size))
546                                .text_xs()
547                                .text_color(cx.theme().muted_foreground),
548                        );
549                    }
550                    if let Some(ref last_modified) = last_modified {
551                        details = details.child(
552                            div()
553                                .child(format!("Last Modified: {}", last_modified))
554                                .text_xs()
555                                .text_color(cx.theme().muted_foreground),
556                        );
557                    }
558                    tooltip = tooltip.child(details);
559                    tooltip
560                })
561                .build(window, cx)
562            });
563        }
564        let (filename, folder) = self.get_tab_display_title(index, tab);
565        let modified_indicator = if tab.is_modified() { " •" } else { "" };
566        let mut title_container = div().flex().items_center().gap_1().pl_1().child(
567            div()
568                .text_sm()
569                .text_color(if is_active {
570                    cx.theme().tab_active_foreground
571                } else {
572                    cx.theme().tab_foreground
573                })
574                .child(format!("{}{}", filename, modified_indicator)),
575        );
576        if let Some(folder_path) = folder {
577            title_container = title_container.child(
578                div()
579                    .text_xs()
580                    .italic()
581                    .text_color(if is_active {
582                        cx.theme().tab_active_foreground
583                    } else {
584                        cx.theme().tab_foreground
585                    })
586                    .child(folder_path),
587            );
588        }
589        let is_markdown_preview = tab.as_markdown_preview().is_some();
590        let title: SharedString = tab.title().to_string().into();
591        let is_modified = tab.is_modified();
592        let mut tab_with_content = tab_div
593            .child(title_container)
594            .child(
595                Button::new(("close-tab", tab_id))
596                    .icon(CustomIcon::Close)
597                    .ghost()
598                    .xsmall()
599                    .cursor_pointer()
600                    .on_click(cx.listener(move |this, _, window, cx| {
601                        cx.stop_propagation();
602                        this.close_tab(tab_id, window, cx);
603                    })),
604            )
605            .on_mouse_down(
606                MouseButton::Middle,
607                cx.listener(move |this, _, window, cx| {
608                    this.close_tab(tab_id, window, cx);
609                }),
610            );
611        if !is_markdown_preview {
612            let is_source = cx.has_active_drag()
613                && self
614                    .drag_ghost
615                    .as_ref()
616                    .map(|(_, d)| d.tab_index == index)
617                    .unwrap_or(false);
618            if is_source {
619                tab_with_content = tab_with_content.opacity(0.45);
620            }
621            tab_with_content = tab_with_content
622                .on_drag(
623                    DraggedTab {
624                        tab_index: index,
625                        title,
626                        is_modified,
627                    },
628                    |dragged, _, _, cx| cx.new(|_| dragged.clone()),
629                )
630                .on_drag_move::<DraggedTab>(cx.listener(
631                    move |this, event: &DragMoveEvent<DraggedTab>, _window, cx| {
632                        let cursor = event.event.position;
633                        let bounds = event.bounds;
634                        if cursor.x < bounds.origin.x
635                            || cursor.x > bounds.origin.x + bounds.size.width
636                            || cursor.y < bounds.origin.y
637                            || cursor.y > bounds.origin.y + bounds.size.height
638                        {
639                            return;
640                        }
641                        let slot = if cursor.x < bounds.origin.x + bounds.size.width * 0.5 {
642                            index
643                        } else {
644                            index + 1
645                        };
646                        let dragged = event.drag(cx).clone();
647                        this.drag_ghost = Some((slot, dragged));
648                        cx.notify();
649                    },
650                ))
651                .on_drop(cx.listener(|this, dragged: &DraggedTab, window, cx| {
652                    if let Some((slot, _)) = this.drag_ghost.take() {
653                        this.handle_tab_drop(dragged, slot, window, cx);
654                    }
655                }));
656        }
657        let tab_with_content = tab_with_content.context_menu(move |this, _window, _cx| {
658            this.menu_with_disabled(
659                crate::fulgur::ui::components_utils::reveal_in_file_manager_label(),
660                Box::new(ShowInFileManager(index)),
661                !has_file_path,
662            )
663            .menu_with_disabled(
664                "Duplicate Tab",
665                Box::new(DuplicateTab(index)),
666                !is_editor_tab,
667            )
668            .separator()
669            .menu("Close Tab", Box::new(CloseTabAction(tab_id)))
670            .menu_with_disabled(
671                "Close Tabs to the Left",
672                Box::new(CloseTabsToLeft(index)),
673                !has_tabs_on_left,
674            )
675            .menu_with_disabled(
676                "Close Tabs to the Right",
677                Box::new(CloseTabsToRight(index)),
678                !has_tabs_on_right,
679            )
680            .separator()
681            .menu_with_disabled(
682                "Close All Tabs",
683                Box::new(CloseAllTabsAction),
684                total_tabs == 0,
685            )
686            .menu_with_disabled(
687                "Close All Other Tabs",
688                Box::new(CloseAllOtherTabs(index)),
689                total_tabs <= 1,
690            )
691        });
692        tab_with_content.into_any_element()
693    }

696/// Reveals a file in the platform's native file manager with the file selected.
697///
698/// - **macOS**: `open -R <path>` — reveals and selects the file in Finder
699/// - **Windows**: `explorer /select,<path>` — selects the file in Explorer
700/// - **Linux**: falls back to opening the parent directory via the `open` crate,
701///   as there is no universal "reveal" command across desktop environments
702///
703/// ### Arguments
704/// - `file_path`: The path of the file to reveal
705///
706/// ### Returns
707/// - `Ok(())` on success, `Err` with an error message on failure
708fn reveal_file_in_file_manager(file_path: &std::path::Path) -> Result<(), String> {
709    #[cfg(target_os = "macos")]
710    {
711        std::process::Command::new("open")
712            .arg("-R")
713            .arg(file_path)
714            .spawn()
715            .map(|_| ())
716            .map_err(|e| e.to_string())
717    }
718
719    #[cfg(target_os = "windows")]
720    {
721        std::process::Command::new("explorer")
722            .arg(format!("/select,{}", file_path.display()))
723            .spawn()
724            .map(|_| ())
725            .map_err(|e| e.to_string())
726    }
727
728    #[cfg(not(any(target_os = "macos", target_os = "windows")))]
729    {
730        let parent = file_path
731            .parent()
732            .ok_or_else(|| "File has no parent directory".to_string())?;
733        open::that(parent).map_err(|e| e.to_string())
734    }
735}
