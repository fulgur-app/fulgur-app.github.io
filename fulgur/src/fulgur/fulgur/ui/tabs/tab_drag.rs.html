1use gpui::*;
2use gpui_component::ActiveTheme;
3
4/// Data carried during a tab drag operation.
5///
6/// Also implements `Render` so it can serve as the ghost view shown under
7/// the cursor while dragging.
8#[derive(Clone)]
9pub struct DraggedTab {
10    /// Index of the tab in the source window at the time the drag started
11    pub tab_index: usize,
12    pub title: SharedString,
13    pub is_modified: bool,
14}
15
16impl Render for DraggedTab {
17    /// Renders the floating ghost tab shown under the cursor while dragging
18    fn render(&mut self, _window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
19        let modified_indicator = if self.is_modified { " •" } else { "" };
20        div()
21            .px_3()
22            .py_1()
23            .rounded_md()
24            .shadow_md()
25            .border_1()
26            .border_color(cx.theme().border)
27            .bg(cx.theme().tab_active)
28            .text_color(cx.theme().tab_active_foreground)
29            .text_sm()
30            .child(format!("{}{}", self.title, modified_indicator))
31    }
32}
