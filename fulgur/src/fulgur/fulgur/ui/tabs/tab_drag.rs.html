1use gpui::*;
2use gpui_component::ActiveTheme;
3
4/// Data carried during a tab drag operation.
5///
6/// Also implements `Render` so it can serve as the ghost view shown under
7/// the cursor while dragging.
8#[derive(Clone)]
9pub struct DraggedTab {
10    /// Index of the tab in the source window at the time the drag started
11    pub tab_index: usize,
12    pub title: SharedString,
13    pub is_modified: bool,
14}
15
16impl Render for DraggedTab {
17    /// Renders the floating ghost tab shown under the cursor while dragging
18    fn render(&mut self, _window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
19        let modified_indicator = if self.is_modified { " •" } else { "" };
20        div()
21            .px_3()
22            .py_1()
23            .rounded_md()
24            .shadow_md()
25            .border_1()
26            .border_color(cx.theme().border)
27            .bg(cx.theme().tab_active)
28            .text_color(cx.theme().tab_active_foreground)
29            .text_sm()
30            .child(format!("{}{}", self.title, modified_indicator))
31    }
32}

34#[cfg(test)]
35mod tests {
36    use super::DraggedTab;
37
38    fn make_tab(index: usize, title: &'static str, modified: bool) -> DraggedTab {
39        DraggedTab {
40            tab_index: index,
41            title: title.into(),
42            is_modified: modified,
43        }
44    }
45
46    // ========== DraggedTab struct tests ==========
47
48    #[test]
49    fn test_dragged_tab_clone_preserves_all_fields() {
50        let original = make_tab(3, "main.rs", true);
51        let cloned = original.clone();
52        assert_eq!(cloned.tab_index, 3);
53        assert_eq!(cloned.title.as_ref(), "main.rs");
54        assert!(cloned.is_modified);
55    }
56
57    // ========== ghost tab title formatting tests ==========
58
59    #[test]
60    fn test_ghost_title_unmodified_has_no_bullet() {
61        let tab = make_tab(0, "readme.md", false);
62        let indicator = if tab.is_modified { " •" } else { "" };
63        assert_eq!(format!("{}{}", tab.title, indicator), "readme.md");
64    }
65
66    #[test]
67    fn test_ghost_title_modified_appends_bullet() {
68        let tab = make_tab(0, "readme.md", true);
69        let indicator = if tab.is_modified { " •" } else { "" };
70        assert_eq!(format!("{}{}", tab.title, indicator), "readme.md •");
71    }
