1use crate::fulgur::sync::ssh::url::RemoteSpec;
2use std::path::PathBuf;
3
4/// The source location of an editor tab's content.
5#[derive(Debug, Clone)]
6pub enum TabLocation {
7    /// A file on the local filesystem.
8    Local(PathBuf),
9    /// A file on a remote SSH/SFTP host.
10    Remote(RemoteSpec),
11    /// An unsaved buffer with no associated file.
12    Untitled,
13}
14
15impl TabLocation {
16    /// Return the local filesystem path if this is a local file.
17    ///
18    /// ### Returns
19    /// - `Some(&PathBuf)`: The local path.
20    /// - `None`: If the location is remote or untitled.
21    pub fn local_path(&self) -> Option<&PathBuf> {
22        match self {
23            TabLocation::Local(path) => Some(path),
24            _ => None,
25        }
26    }
27
28    /// Return a human-readable path string for UI display.
29    ///
30    /// ### Returns
31    /// - `String`: The file path for local tabs, `user@host:path` for remote tabs, or an empty string for untitled tabs.
32    pub fn display_path(&self) -> String {
33        match self {
34            TabLocation::Local(path) => path.to_string_lossy().into_owned(),
35            TabLocation::Remote(spec) => {
36                let user = spec.user.as_deref().unwrap_or("?");
37                format!("{}@{}:{}", user, spec.host, spec.path)
38            }
39            TabLocation::Untitled => String::new(),
40        }
41    }
42
43    /// Return whether this location has no associated file.
44    ///
45    /// ### Returns
46    /// - `bool`: `true` if untitled, `false` otherwise.
47    pub fn is_untitled(&self) -> bool {
48        matches!(self, TabLocation::Untitled)
49    }
50}
