129    /// Create a new tab from a file
130    ///
131    /// ### Arguments
132    /// - `params`: The parameters for creating the tab
133    /// - `window`: The window to create the tab in
134    /// - `cx`: The application context
135    /// - `settings`: The settings for the input state
136    ///
137    /// ### Returns
138    /// - `EditorTab`: The new tab
139    pub fn from_file(
140        params: FromFileParams,
141        window: &mut Window,
142        cx: &mut App,
143        settings: &EditorSettings,
144    ) -> Self {
145        let content_len = params.contents.len();
146        let (original_content_hash, original_content_len) =
147            super::content_fingerprint_from_str(&params.contents);
148        let file_name = params
149            .path
150            .file_name()
151            .and_then(|n| n.to_str())
152            .unwrap_or(UNTITLED)
153            .to_string();
154
155        let language = language_from_content(&file_name, &params.contents);
156        let (csv_view_mode, csv_delimiter) = initial_csv_state(language, &params.contents);
157        let content = cx.new(|cx| {
158            super::make_input_state(
159                window,
160                cx,
161                language_registry_name(&language),
162                Some(params.contents.clone()),
163                settings,
164            )
165        });
166        let title = format!(
167            "{}{}",
168            file_name,
169            if params.is_modified { " •" } else { "" }
170        );
171        Self {
172            id: params.id,
173            title: title.into(),
174            content,
175            location: TabLocation::Local(params.path),
176            modified: params.is_modified,
177            original_content_hash,
178            original_content_len,
179            encoding: params.encoding,
180            lossy_decode: false,
181            language,
182            show_markdown_toolbar: settings.markdown_settings.show_markdown_toolbar,
183            show_markdown_preview: settings.markdown_settings.show_markdown_preview,
184            file_size_bytes: Some(content_len as u64),
185            file_last_modified: Some(SystemTime::now()),
186            csv_view_mode,
187            csv_delimiter,
188            csv_table: None,
189            csv_table_source_hash: 0,
190            log_view: false,
191            log_follow: true,
192            log_full: false,
193            log_content: None,
194        }
195    }

380    /// Recreate a tab in a new window from transferred state.
381    ///
382    /// Preserves text content, file path, unsaved-change state, language, encoding,
383    /// markdown panel visibility, and cursor position. Called by the target window
384    /// when processing a deferred `pending_tab_transfer`.
385    ///
386    /// ### Arguments
387    /// - `id`: The new tab ID, allocated by the receiving window
388    /// - `data`: All state captured from the source tab
389    /// - `window`: The target window context
390    /// - `cx`: The application context
391    /// - `settings`: The receiving window's editor settings
392    ///
393    /// ### Returns
394    /// - `EditorTab`: The newly created tab, ready to be pushed onto the tab list
395    pub fn from_transfer(
396        id: usize,
397        data: TabTransferData,
398        window: &mut Window,
399        cx: &mut App,
400        settings: &EditorSettings,
401    ) -> Self {
402        let cursor_position = data.cursor_position;
403        let content = cx.new(|cx| {
404            super::make_input_state(
405                window,
406                cx,
407                language_registry_name(&data.language),
408                Some(data.content),
409                settings,
410            )
411        });
412        content.update(cx, |state, cx| {
413            state.set_cursor_position(cursor_position, window, cx);
414        });
415        Self {
416            id,
417            title: data.title,
418            content,
419            location: data.location,
420            modified: data.modified,
421            original_content_hash: data.original_content_hash,
422            original_content_len: data.original_content_len,
423            encoding: data.encoding,
424            lossy_decode: data.lossy_decode,
425            language: data.language,
426            show_markdown_toolbar: data.show_markdown_toolbar,
427            show_markdown_preview: data.show_markdown_preview,
428            file_size_bytes: data.file_size_bytes,
429            file_last_modified: data.file_last_modified,
430            csv_view_mode: data.csv_view_mode,
431            csv_delimiter: data.csv_delimiter,
432            csv_table: None,
433            csv_table_source_hash: 0,
434            log_view: data.log_view,
435            log_follow: true,
436            log_full: false,
437            log_content: None,
438        }
439    }
