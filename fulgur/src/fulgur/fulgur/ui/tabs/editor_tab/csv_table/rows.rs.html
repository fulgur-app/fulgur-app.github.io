1//! Row insert and delete operations.
2
3use gpui::{Context, Window};
4use gpui_component::table::TableState;
5
6use super::CsvTableDelegate;
7
8impl CsvTableDelegate {
9    /// Insert an empty row at `index` into a plain rows model.
10    ///
11    /// ### Arguments
12    /// - `headers`: The column headers, used to size the new row
13    /// - `rows`: The data rows to mutate
14    /// - `index`: The row index to insert at, clamped to the row count
15    pub(super) fn insert_row_at_in(headers: &[String], rows: &mut Vec<Vec<String>>, index: usize) {
16        let index = index.min(rows.len());
17        rows.insert(index, vec![String::new(); headers.len()]);
18    }
19
20    /// Insert an empty row above the selected row (or at the end if none).
21    ///
22    /// ### Arguments
23    /// - `window`: The active window
24    /// - `cx`: The table state context
25    pub fn insert_row_above(&mut self, window: &mut Window, cx: &mut Context<TableState<Self>>) {
26        let selected_row = self.selected_row_index();
27        Self::insert_row_above_in(&self.headers, &mut self.rows, selected_row);
28        self.commit_and_refresh(window, cx);
29    }
30
31    /// Insert an empty row above the selected row into a plain model.
32    ///
33    /// ### Arguments
34    /// - `headers`: The column headers, used to size the new row
35    /// - `rows`: The data rows to mutate
36    /// - `selected_row`: The currently selected row, if any; when `None` the
37    ///   row is appended at the end
38    pub(super) fn insert_row_above_in(
39        headers: &[String],
40        rows: &mut Vec<Vec<String>>,
41        selected_row: Option<usize>,
42    ) {
43        let index = selected_row.unwrap_or(rows.len());
44        Self::insert_row_at_in(headers, rows, index);
45    }
46
47    /// Insert an empty row below the selected row (or at the end if none).
48    ///
49    /// ### Arguments
50    /// - `window`: The active window
51    /// - `cx`: The table state context
52    pub fn insert_row_below(&mut self, window: &mut Window, cx: &mut Context<TableState<Self>>) {
53        let selected_row = self.selected_row_index();
54        Self::insert_row_below_in(&self.headers, &mut self.rows, selected_row);
55        self.commit_and_refresh(window, cx);
56    }
57
58    /// Insert an empty row below the selected row into a plain model.
59    ///
60    /// ### Arguments
61    /// - `headers`: The column headers, used to size the new row
62    /// - `rows`: The data rows to mutate
63    /// - `selected_row`: The currently selected row, if any; when `None` the
64    ///   row is appended at the end
65    pub(super) fn insert_row_below_in(
66        headers: &[String],
67        rows: &mut Vec<Vec<String>>,
68        selected_row: Option<usize>,
69    ) {
70        let index = selected_row.map_or(rows.len(), |row| row + 1);
71        Self::insert_row_at_in(headers, rows, index);
72    }
73
74    /// Delete the selected row, or the last row if none is selected.
75    ///
76    /// ### Arguments
77    /// - `window`: The active window
78    /// - `cx`: The table state context
79    pub fn delete_row(&mut self, window: &mut Window, cx: &mut Context<TableState<Self>>) {
80        let selected_row = self.selected_row_index();
81        if Self::delete_row_in(&mut self.rows, selected_row) {
82            self.commit_and_refresh(window, cx);
83        }
84    }
85
86    /// Delete the selected row from a plain model, clamping out-of-bounds
87    /// selections to the last row.
88    ///
89    /// ### Arguments
90    /// - `rows`: The data rows to mutate
91    /// - `selected_row`: The currently selected row, if any; when `None` the
92    ///   last row is removed
93    ///
94    /// ### Returns
95    /// - `true`: A row was removed
96    /// - `false`: The buffer was empty and nothing changed
97    pub(super) fn delete_row_in(rows: &mut Vec<Vec<String>>, selected_row: Option<usize>) -> bool {
98        if rows.is_empty() {
99            return false;
100        }
101        let last = rows.len() - 1;
102        let index = selected_row.unwrap_or(last).min(last);
103        rows.remove(index);
104        true
105    }
106}
