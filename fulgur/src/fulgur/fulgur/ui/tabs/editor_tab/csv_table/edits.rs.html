1//! Single header/cell editing and the modal edit dialog.
2
3use gpui::{Context, Focusable, ParentElement, Styled, Window, div, px};
4use gpui_component::{
5    WindowExt, button::ButtonVariant, dialog::DialogButtonProps, input::Input, table::TableState,
6};
7
8use super::{CsvTableDelegate, EditTarget};
9
10impl CsvTableDelegate {
11    /// Read the current value targeted by an edit dialog.
12    ///
13    /// ### Arguments
14    /// - `target`: The header or cell to read (data indices)
15    ///
16    /// ### Returns
17    /// - `String`: The current value, or an empty string if out of bounds
18    fn value_at(&self, target: EditTarget) -> String {
19        Self::value_at_in(&self.headers, &self.rows, target)
20    }
21
22    /// Read the value at a header or cell from a plain headers/rows model.
23    ///
24    /// ### Arguments
25    /// - `headers`: The column headers
26    /// - `rows`: The data rows
27    /// - `target`: The header or cell to read (data indices)
28    ///
29    /// ### Returns
30    /// - `String`: The current value, or an empty string if out of bounds
31    pub(super) fn value_at_in(
32        headers: &[String],
33        rows: &[Vec<String>],
34        target: EditTarget,
35    ) -> String {
36        match target {
37            EditTarget::Header(col) => headers.get(col).cloned().unwrap_or_default(),
38            EditTarget::Cell(row, col) => rows
39                .get(row)
40                .and_then(|cells| cells.get(col))
41                .cloned()
42                .unwrap_or_default(),
43        }
44    }
45
46    /// Open the modal edit dialog for a header or cell.
47    ///
48    /// ### Arguments
49    /// - `target`: The header or cell being edited (data indices)
50    /// - `window`: The active window
51    /// - `cx`: The table state context
52    pub(super) fn open_edit_dialog(
53        &self,
54        target: EditTarget,
55        window: &mut Window,
56        cx: &mut Context<TableState<Self>>,
57    ) {
58        let current = self.value_at(target);
59        let input = self.dialog_input.clone();
60        input.update(cx, |state, cx| {
61            state.set_value(current, window, cx);
62            cx.notify();
63        });
64
65        let table = cx.entity();
66        let title = match target {
67            EditTarget::Header(_) => "Edit column header",
68            EditTarget::Cell(_, _) => "Edit cell",
69        };
70
71        window.open_alert_dialog(cx, move |dialog, window, cx| {
72            let focus_handle = input.read(cx).focus_handle(cx);
73            window.focus(&focus_handle, cx);
74            let input_for_ok = input.clone();
75            let table_for_ok = table.clone();
76            dialog
77                .title(div().text_size(px(16.)).child(title))
78                .keyboard(true)
79                .button_props(
80                    DialogButtonProps::default()
81                        .show_cancel(true)
82                        .cancel_text("Cancel")
83                        .cancel_variant(ButtonVariant::Secondary)
84                        .ok_text("OK")
85                        .ok_variant(ButtonVariant::Primary),
86                )
87                .overlay_closable(true)
88                .close_button(false)
89                .child(Input::new(&input))
90                .on_ok(move |_, window, cx| {
91                    let value = input_for_ok.read(cx).value().to_string();
92                    table_for_ok.update(cx, |state, cx| {
93                        state.delegate_mut().apply_edit(target, value, window, cx);
94                    });
95                    true
96                })
97        });
98    }
99
100    /// Write an edited value back into the model and the canonical buffer.
101    ///
102    /// ### Arguments
103    /// - `target`: The header or cell being edited (data indices)
104    /// - `value`: The new value
105    /// - `window`: The active window
106    /// - `cx`: The table state context
107    fn apply_edit(
108        &mut self,
109        target: EditTarget,
110        value: String,
111        window: &mut Window,
112        cx: &mut Context<TableState<Self>>,
113    ) {
114        Self::apply_edit_in(&mut self.headers, &mut self.rows, target, value);
115        self.commit_and_refresh(window, cx);
116    }
117
118    /// Write an edited value into a plain headers/rows model, ignoring
119    /// out-of-bounds targets.
120    ///
121    /// ### Arguments
122    /// - `headers`: The column headers
123    /// - `rows`: The data rows
124    /// - `target`: The header or cell being edited (data indices)
125    /// - `value`: The new value
126    pub(super) fn apply_edit_in(
127        headers: &mut [String],
128        rows: &mut [Vec<String>],
129        target: EditTarget,
130        value: String,
131    ) {
132        match target {
133            EditTarget::Header(col) => {
134                if let Some(header) = headers.get_mut(col) {
135                    *header = value;
136                }
137            }
138            EditTarget::Cell(row, col) => {
139                if let Some(cell) = rows.get_mut(row).and_then(|cells| cells.get_mut(col)) {
140                    *cell = value;
141                }
142            }
143        }
144    }
145}
