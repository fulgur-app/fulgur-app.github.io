17/// Fixed width of the synthetic row-number column.
18const ROW_NUMBER_COLUMN_WIDTH: f32 = 52.0;
19/// Approximate rendered width of a single character, used to size columns.
20const CELL_CHARACTER_WIDTH: f32 = 7.5;
21/// Horizontal padding added to a column's content width.
22const CELL_PADDING_WIDTH: f32 = 24.0;
23/// Minimum width a data column is allowed to shrink to.
24const MIN_COLUMN_WIDTH: f32 = 60.0;
25/// Maximum width a data column is allowed to grow to.
26const MAX_COLUMN_WIDTH: f32 = 500.0;
27
28/// Identifies which value a CSV edit dialog is targeting (data indices).
29#[derive(Debug, Clone, Copy)]
30enum EditTarget {
31    /// The header of the data column at the given index.
32    Header(usize),
33    /// The cell at the given `(row, data column)` index.
34    Cell(usize, usize),
35}
36
37/// A [`TableDelegate`] that renders and edits CSV data over a canonical buffer.
38pub struct CsvTableDelegate {
39    headers: Vec<String>,
40    rows: Vec<Vec<String>>,
41    columns: Vec<Column>,
42    delimiter: u8,
43    content: Entity<InputState>,
44    dialog_input: Entity<InputState>,
45}
46
47impl CsvTableDelegate {
48    /// Build a delegate from parsed CSV data.
49    ///
50    /// ### Arguments
51    /// - `data`: The parsed headers and rows
52    /// - `delimiter`: The delimiter to use when serializing back to text
53    /// - `content`: The canonical `InputState` buffer to write edits back into
54    /// - `dialog_input`: A reusable input entity for the edit dialog
55    ///
56    /// ### Returns
57    /// - `CsvTableDelegate`: The constructed delegate
58    pub fn new(
59        data: CsvData,
60        delimiter: u8,
61        content: Entity<InputState>,
62        dialog_input: Entity<InputState>,
63    ) -> Self {
64        let columns = Self::compute_columns(&data.headers, &data.rows);
65        Self {
66            headers: data.headers,
67            rows: data.rows,
68            columns,
69            delimiter,
70            content,
71            dialog_input,
72        }
73    }
74
75    /// Build the grid columns: a synthetic row-number column plus one column
76    /// per header sized to its widest value (clamped to `MAX_COLUMN_WIDTH`).
77    ///
78    /// ### Arguments
79    /// - `headers`: The header strings
80    /// - `rows`: The data rows, used to measure content widths
81    ///
82    /// ### Returns
83    /// - `Vec<Column>`: The row-number column followed by one column per header
84    fn compute_columns(headers: &[String], rows: &[Vec<String>]) -> Vec<Column> {
85        let mut columns = Vec::with_capacity(headers.len() + 1);
86        columns.push(
87            Column::new(
88                SharedString::from("csv-row-number"),
89                SharedString::default(),
90            )
91            .width(px(ROW_NUMBER_COLUMN_WIDTH))
92            .resizable(false)
93            .movable(false)
94            .selectable(false)
95            .text_center(),
96        );
97
98        for (index, header) in headers.iter().enumerate() {
99            let mut max_chars = header.chars().count();
100            for row in rows {
101                if let Some(cell) = row.get(index) {
102                    max_chars = max_chars.max(cell.chars().count());
103                }
104            }
105            #[allow(clippy::cast_precision_loss)]
106            let width = ((max_chars as f32) * CELL_CHARACTER_WIDTH + CELL_PADDING_WIDTH)
107                .clamp(MIN_COLUMN_WIDTH, MAX_COLUMN_WIDTH);
108            columns.push(
109                Column::new(
110                    SharedString::from(format!("csv-col-{index}")),
111                    SharedString::from(header.clone()),
112                )
113                .width(px(width)),
114            );
115        }
116        columns
117    }
118
119    /// Recompute the columns from the current headers and rows.
120    fn refresh_columns(&mut self) {
121        self.columns = Self::compute_columns(&self.headers, &self.rows);
122    }
123
124    /// Map a selected grid cell to a data column index, ignoring the synthetic
125    /// row-number column.
126    ///
127    /// ### Arguments
128    /// - `selected`: The selected `(row, grid column)`, if any
129    ///
130    /// ### Returns
131    /// - `Some(usize)`: The data column index when a data column is selected
132    /// - `None`: When nothing or only the row-number column is selected
133    fn selected_data_column(selected: Option<(usize, usize)>) -> Option<usize> {
134        selected.and_then(|(_, grid_col)| grid_col.checked_sub(1))
135    }

137    /// Serialize the current model back into the canonical text buffer.
138    ///
139    /// ### Arguments
140    /// - `window`: The active window
141    /// - `cx`: The table state context
142    fn commit_to_buffer(&self, window: &mut Window, cx: &mut Context<TableState<Self>>) {
143        let text = serialize_csv(&self.headers, &self.rows, self.delimiter);
144        self.content.update(cx, |state, cx| {
145            state.set_value(text, window, cx);
146            cx.emit(InputEvent::Change);
147        });
148    }
149
150    /// Refresh columns, write the model back to the buffer, and notify.
151    ///
152    /// ### Arguments
153    /// - `window`: The active window
154    /// - `cx`: The table state context
155    fn commit_and_refresh(&mut self, window: &mut Window, cx: &mut Context<TableState<Self>>) {
156        self.refresh_columns();
157        self.commit_to_buffer(window, cx);
158        cx.notify();
159    }

161    /// Read the current value targeted by an edit dialog.
162    ///
163    /// ### Arguments
164    /// - `target`: The header or cell to read (data indices)
165    ///
166    /// ### Returns
167    /// - `String`: The current value, or an empty string if out of bounds
168    fn value_at(&self, target: EditTarget) -> String {
169        match target {
170            EditTarget::Header(col) => self.headers.get(col).cloned().unwrap_or_default(),
171            EditTarget::Cell(row, col) => self
172                .rows
173                .get(row)
174                .and_then(|cells| cells.get(col))
175                .cloned()
176                .unwrap_or_default(),
177        }
178    }
179
180    /// Open the modal edit dialog for a header or cell.
181    ///
182    /// ### Arguments
183    /// - `target`: The header or cell being edited (data indices)
184    /// - `window`: The active window
185    /// - `cx`: The table state context
186    fn open_edit_dialog(
187        &self,
188        target: EditTarget,
189        window: &mut Window,
190        cx: &mut Context<TableState<Self>>,
191    ) {
192        let current = self.value_at(target);
193        let input = self.dialog_input.clone();
194        input.update(cx, |state, cx| {
195            state.set_value(current, window, cx);
196            cx.notify();
197        });
198
199        let table = cx.entity();
200        let title = match target {
201            EditTarget::Header(_) => "Edit column header",
202            EditTarget::Cell(_, _) => "Edit cell",
203        };
204
205        window.open_alert_dialog(cx, move |dialog, window, cx| {
206            let focus_handle = input.read(cx).focus_handle(cx);
207            window.focus(&focus_handle, cx);
208            let input_for_ok = input.clone();
209            let table_for_ok = table.clone();
210            dialog
211                .title(div().text_size(px(16.)).child(title))
212                .keyboard(true)
213                .button_props(
214                    DialogButtonProps::default()
215                        .show_cancel(true)
216                        .cancel_text("Cancel")
217                        .cancel_variant(ButtonVariant::Secondary)
218                        .ok_text("OK")
219                        .ok_variant(ButtonVariant::Primary),
220                )
221                .overlay_closable(true)
222                .close_button(false)
223                .child(Input::new(&input))
224                .on_ok(move |_, window, cx| {
225                    let value = input_for_ok.read(cx).value().to_string();
226                    table_for_ok.update(cx, |state, cx| {
227                        state.delegate_mut().apply_edit(target, value, window, cx);
228                    });
229                    true
230                })
231        });
232    }

234    /// Write an edited value back into the model and the canonical buffer.
235    ///
236    /// ### Arguments
237    /// - `target`: The header or cell being edited (data indices)
238    /// - `value`: The new value
239    /// - `window`: The active window
240    /// - `cx`: The table state context
241    fn apply_edit(
242        &mut self,
243        target: EditTarget,
244        value: String,
245        window: &mut Window,
246        cx: &mut Context<TableState<Self>>,
247    ) {
248        match target {
249            EditTarget::Header(col) => {
250                if let Some(header) = self.headers.get_mut(col) {
251                    *header = value;
252                }
253            }
254            EditTarget::Cell(row, col) => {
255                if let Some(cell) = self.rows.get_mut(row).and_then(|cells| cells.get_mut(col)) {
256                    *cell = value;
257                }
258            }
259        }
260        self.commit_and_refresh(window, cx);
261    }

263    /// Insert an empty row at `index`.
264    ///
265    /// ### Arguments
266    /// - `index`: The row index to insert at
267    /// - `window`: The active window
268    /// - `cx`: The table state context
269    fn insert_row_at(
270        &mut self,
271        index: usize,
272        window: &mut Window,
273        cx: &mut Context<TableState<Self>>,
274    ) {
275        let index = index.min(self.rows.len());
276        self.rows
277            .insert(index, vec![String::new(); self.headers.len()]);
278        self.commit_and_refresh(window, cx);
279    }

281    /// Insert an empty row above the selected row (or at the end if none).
282    ///
283    /// ### Arguments
284    /// - `selected`: The currently selected `(row, grid column)`, if any
285    /// - `window`: The active window
286    /// - `cx`: The table state context
287    pub fn insert_row_above(
288        &mut self,
289        selected: Option<(usize, usize)>,
290        window: &mut Window,
291        cx: &mut Context<TableState<Self>>,
292    ) {
293        let index = selected.map_or(self.rows.len(), |(row, _)| row);
294        self.insert_row_at(index, window, cx);
295    }

297    /// Insert an empty row below the selected row (or at the end if none).
298    ///
299    /// ### Arguments
300    /// - `selected`: The currently selected `(row, grid column)`, if any
301    /// - `window`: The active window
302    /// - `cx`: The table state context
303    pub fn insert_row_below(
304        &mut self,
305        selected: Option<(usize, usize)>,
306        window: &mut Window,
307        cx: &mut Context<TableState<Self>>,
308    ) {
309        let index = selected.map_or(self.rows.len(), |(row, _)| row + 1);
310        self.insert_row_at(index, window, cx);
311    }

313    /// Delete the selected row, or the last row if none is selected.
314    ///
315    /// ### Arguments
316    /// - `selected`: The currently selected `(row, grid column)`, if any
317    /// - `window`: The active window
318    /// - `cx`: The table state context
319    pub fn delete_row(
320        &mut self,
321        selected: Option<(usize, usize)>,
322        window: &mut Window,
323        cx: &mut Context<TableState<Self>>,
324    ) {
325        if self.rows.is_empty() {
326            return;
327        }
328        let index = selected
329            .map_or(self.rows.len() - 1, |(row, _)| row)
330            .min(self.rows.len() - 1);
331        self.rows.remove(index);
332        self.commit_and_refresh(window, cx);
333    }

335    /// Insert an empty data column at `index`.
336    ///
337    /// ### Arguments
338    /// - `index`: The column index to insert at
339    /// - `window`: The active window
340    /// - `cx`: The table state context
341    fn insert_column_at(
342        &mut self,
343        index: usize,
344        window: &mut Window,
345        cx: &mut Context<TableState<Self>>,
346    ) {
347        let index = index.min(self.headers.len());
348        self.headers.insert(index, String::new());
349        for row in &mut self.rows {
350            let position = index.min(row.len());
351            row.insert(position, String::new());
352        }
353        self.commit_and_refresh(window, cx);
354    }

356    /// Insert an empty column before the selected column (or at the end).
357    ///
358    /// ### Arguments
359    /// - `selected`: The currently selected `(row, grid column)`, if any
360    /// - `window`: The active window
361    /// - `cx`: The table state context
362    pub fn insert_column_before(
363        &mut self,
364        selected: Option<(usize, usize)>,
365        window: &mut Window,
366        cx: &mut Context<TableState<Self>>,
367    ) {
368        let index = Self::selected_data_column(selected).unwrap_or(self.headers.len());
369        self.insert_column_at(index, window, cx);
370    }

372    /// Insert an empty column after the selected column (or at the end).
373    ///
374    /// ### Arguments
375    /// - `selected`: The currently selected `(row, grid column)`, if any
376    /// - `window`: The active window
377    /// - `cx`: The table state context
378    pub fn insert_column_after(
379        &mut self,
380        selected: Option<(usize, usize)>,
381        window: &mut Window,
382        cx: &mut Context<TableState<Self>>,
383    ) {
384        let index = Self::selected_data_column(selected).map_or(self.headers.len(), |col| col + 1);
385        self.insert_column_at(index, window, cx);
386    }
387
388    /// Delete the selected column, or the last column if none is selected.
389    ///
390    /// ### Arguments
391    /// - `selected`: The currently selected `(row, grid column)`, if any
392    /// - `window`: The active window
393    /// - `cx`: The table state context
394    pub fn delete_column(
395        &mut self,
396        selected: Option<(usize, usize)>,
397        window: &mut Window,
398        cx: &mut Context<TableState<Self>>,
399    ) {
400        if self.headers.is_empty() {
401            return;
402        }
403        let index = Self::selected_data_column(selected)
404            .unwrap_or(self.headers.len() - 1)
405            .min(self.headers.len() - 1);
406        self.headers.remove(index);
407        for row in &mut self.rows {
408            if index < row.len() {
409                row.remove(index);
410            }
411        }
412        self.commit_and_refresh(window, cx);
413    }

416impl TableDelegate for CsvTableDelegate {
417    fn columns_count(&self, _: &App) -> usize {
418        self.headers.len() + 1
419    }
420
421    fn rows_count(&self, _: &App) -> usize {
422        self.rows.len()
423    }
424
425    fn column(&self, col_ix: usize, _: &App) -> Column {
426        self.columns.get(col_ix).cloned().unwrap_or_default()
427    }
428
429    fn render_th(
430        &mut self,
431        col_ix: usize,
432        _: &mut Window,
433        cx: &mut Context<TableState<Self>>,
434    ) -> impl IntoElement {
435        let label = if col_ix == 0 {
436            String::new()
437        } else {
438            self.headers.get(col_ix - 1).cloned().unwrap_or_default()
439        };
440        div()
441            .id(SharedString::from(format!("csv-th-{col_ix}")))
442            .size_full()
443            .child(label)
444            .when(col_ix != 0, |this| {
445                let data_col = col_ix - 1;
446                this.on_click(cx.listener(move |state, event: &ClickEvent, window, cx| {
447                    if event.click_count() >= 2 {
448                        state.delegate_mut().open_edit_dialog(
449                            EditTarget::Header(data_col),
450                            window,
451                            cx,
452                        );
453                    }
454                }))
455            })
456    }
457
458    fn render_td(
459        &mut self,
460        row_ix: usize,
461        col_ix: usize,
462        _: &mut Window,
463        cx: &mut Context<TableState<Self>>,
464    ) -> impl IntoElement {
465        let is_row_number = col_ix == 0;
466        let muted = cx.theme().muted_foreground;
467        let text = if is_row_number {
468            (row_ix + 1).to_string()
469        } else {
470            self.rows
471                .get(row_ix)
472                .and_then(|cells| cells.get(col_ix - 1))
473                .cloned()
474                .unwrap_or_default()
475        };
476        div()
477            .id(SharedString::from(format!("csv-td-{row_ix}-{col_ix}")))
478            .size_full()
479            .when(is_row_number, |this| {
480                this.flex().justify_center().text_color(muted)
481            })
482            .child(text)
483            .when(!is_row_number, |this| {
484                let data_col = col_ix - 1;
485                this.on_click(cx.listener(move |state, event: &ClickEvent, window, cx| {
486                    if event.click_count() >= 2 {
487                        state.delegate_mut().open_edit_dialog(
488                            EditTarget::Cell(row_ix, data_col),
489                            window,
490                            cx,
491                        );
492                    }
493                }))
494            })
495    }
496
497    fn move_column(
498        &mut self,
499        col_ix: usize,
500        to_ix: usize,
501        window: &mut Window,
502        cx: &mut Context<TableState<Self>>,
503    ) {
504        // The row-number column (grid index 0) is not movable.
505        let Some(from_data) = col_ix.checked_sub(1) else {
506            return;
507        };
508        if from_data >= self.headers.len() {
509            return;
510        }
511        let to_data = to_ix.saturating_sub(1).min(self.headers.len() - 1);
512        if from_data == to_data {
513            return;
514        }
515
516        let header = self.headers.remove(from_data);
517        self.headers.insert(to_data, header);
518        for row in &mut self.rows {
519            if from_data < row.len() {
520                let cell = row.remove(from_data);
521                row.insert(to_data.min(row.len()), cell);
522            }
523        }
524        self.commit_and_refresh(window, cx);
525    }
526
527    fn cell_text(&self, row_ix: usize, col_ix: usize, _: &App) -> String {
528        if col_ix == 0 {
529            return (row_ix + 1).to_string();
530        }
531        self.rows
532            .get(row_ix)
533            .and_then(|cells| cells.get(col_ix - 1))
534            .cloned()
535            .unwrap_or_default()
536    }
