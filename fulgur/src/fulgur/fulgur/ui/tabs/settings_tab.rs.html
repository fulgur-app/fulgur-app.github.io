35impl SettingsTab {
36    /// Create a new settings tab
37    ///
38    /// ### Arguments
39    /// - `id`: The ID of the settings tab
40    /// - `current_font`: The currently selected font family
41    /// - `window`: The window
42    /// - `cx`: The application context
43    ///
44    /// ### Returns
45    /// - `Self`: The settings tab
46    pub fn new(id: usize, current_font: &str, window: &mut Window, cx: &mut App) -> Self {
47        let font_family_select = Self::build_font_select(current_font, window, cx);
48        Self {
49            id,
50            title: SharedString::from("Settings"),
51            font_family_select,
52        }
53    }
54
55    /// Build the font family select entity populated with all system fonts.
56    ///
57    /// ### Arguments
58    /// - `current_font`: The currently selected font family used to set the initial selection
59    /// - `window`: The window
60    /// - `cx`: The application context
61    ///
62    /// ### Returns
63    /// - `Entity<SelectState<SearchableVec<SharedString>>>`: The font family select state
64    pub fn build_font_select(
65        current_font: &str,
66        window: &mut Window,
67        cx: &mut App,
68    ) -> Entity<SelectState<SearchableVec<SharedString>>> {
69        let mut system_fonts: Vec<SharedString> = cx
70            .text_system()
71            .all_font_names()
72            .into_iter()
73            .map(SharedString::from)
74            .collect();
75        system_fonts.sort();
76        system_fonts.dedup();
77        let selected_index = system_fonts
78            .iter()
79            .position(|f| f.as_ref() == current_font)
80            .map(|ix| IndexPath::default().row(ix));
81        cx.new(|cx| {
82            SelectState::new(SearchableVec::new(system_fonts), selected_index, window, cx)
83                .searchable(true)
84        })
85    }
86}

734            let theme_name = theme.name.clone();
735            let theme_author = theme.author.clone();
736            let theme_path = theme.path.clone();
737            let themes_info = theme
738                .themes
739                .iter()
740                .map(|t| format!("{} ({})", t.name, t.mode))
741                .collect::<Vec<String>>()
742                .join(", ");
743            let button_id = SharedString::from(format!("delete-theme-{}", theme_name));
744            user_theme_items.push(SettingItem::render({
745                let entity = entity.clone();
746                move |_options, _window, cx| {
747                    let theme_path = theme_path.clone();
748                    let entity_clone = entity.clone();
749                    h_flex()
750                        .w_full()
751                        .justify_between()
752                        .gap_3()
753                        .child(
754                            v_flex()
755                                .gap_1()
756                                .child(
757                                    div()
758                                        .font_semibold()
759                                        .child(format!("{} by {}", theme_name, theme_author)),
760                                )
761                                .child(
762                                    div()
763                                        .text_xs()
764                                        .text_color(cx.theme().muted_foreground)
765                                        .child(themes_info.clone()),
766                                ),
767                        )
768                        .child(
769                            Button::new(button_id.clone())
770                                .icon(CustomIcon::Close)
771                                .small()
772                                .cursor_pointer()
773                                .on_click(move |_, _window, cx| {
774                                    if let Err(e) = fs::remove_file(&theme_path) {
775                                        log::error!(
776                                            "Failed to delete theme file {}: {}",
777                                            theme_path.display(),
778                                            e
779                                        );
780                                    } else {
781                                        log::info!("Deleted theme file: {:?}", theme_path);
782                                    }
783                                    let entity_for_update = entity_clone.clone();
784                                    entity_clone.update(cx, |this, cx| {
785                                        themes::reload_themes_and_update(
786                                            &this.settings,
787                                            entity_for_update,
788                                            cx,
789                                        );
790                                    });
791                                }),
792                        )
793                        .into_any_element()
794                }
795            }));
796        }
797        for theme in &themes.default_themes {
798            let theme_name = theme.name.clone();
799            let theme_author = theme.author.clone();
800            let themes_info = theme
801                .themes
802                .iter()
803                .map(|t| format!("{} ({})", t.name, t.mode))
804                .collect::<Vec<String>>()
805                .join(", ");
806            default_theme_items.push(SettingItem::render(move |_options, _window, cx| {
807                v_flex()
808                    .w_full()
809                    .gap_1()
810                    .child(
811                        div()
812                            .font_semibold()
813                            .child(format!("{} by {} (Default)", theme_name, theme_author)),
814                    )
815                    .child(
816                        div()
817                            .text_xs()
818                            .text_color(cx.theme().muted_foreground)
819                            .child(themes_info.clone()),
820                    )
821                    .into_any_element()
822            }));
823        }
824        let mut groups = Vec::new();
825        if !user_theme_items.is_empty() {
826            groups.push(
827                SettingGroup::new()
828                    .title("User Themes")
829                    .items(user_theme_items),
830            );
831        }
832        if !default_theme_items.is_empty() {
833            groups.push(
834                SettingGroup::new()
835                    .title("Default Themes")
836                    .items(default_theme_items),
837            );
838        }
839        SettingPage::new("Themes").groups(groups)
840    }

842    /// Create settings pages using the Settings component
843    ///
844    /// ### Arguments
845    /// - `window`: The window
846    /// - `cx`: The context
847    ///
848    /// ### Returns
849    /// - `Vec<SettingPage>`: The settings pages
850    fn create_settings_pages(
851        &self,
852        _window: &mut Window,
853        cx: &mut Context<Self>,
854    ) -> Vec<SettingPage> {
855        let entity = cx.entity();
856        let font_family_select = self
857            .tabs
858            .iter()
859            .find_map(|t| {
860                if let Tab::Settings(s) = t {
861                    Some(s.font_family_select.clone())
862                } else {
863                    None
864                }
865            })
866            .expect("create_settings_pages called without a settings tab in self.tabs");
867        let mut pages = vec![
868            Self::create_editor_page(entity.clone(), font_family_select),
869            Self::create_application_page(entity.clone()),
870        ];
871        let themes = self.shared_state(cx).themes.lock().clone();
872        if let Some(ref themes) = themes {
873            pages.push(Self::create_themes_page(entity, themes));
874        }
875        pages
876    }

878    /// Render the settings
879    ///
880    /// ### Arguments
881    /// - `window`: The window
882    /// - `cx`: The context
883    ///
884    /// ### Returns
885    /// - `impl IntoElement`: The settings UI
886    pub fn render_settings(&self, window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
887        const MAX_WIDTH: f32 = 1400.0;
888        let show_side_borders = window.viewport_size().width >= px(MAX_WIDTH);
889        div()
890            .id("settings-scroll-container")
891            .size_full()
892            .overflow_x_scroll()
893            .child(
894                v_flex()
895                    //.w(px(980.0))
896                    .h_full()
897                    .mx_auto()
898                    .max_w(px(MAX_WIDTH))
899                    .min_w(px(980.0))
900                    .text_color(cx.theme().foreground)
901                    .text_size(px(12.0))
902                    .when(show_side_borders, |el: gpui::Div| {
903                        el.border_l_1().border_r_1().border_color(cx.theme().border)
904                    })
905                    .child(
906                        SettingsComponent::new("fulgur-settings")
907                            .with_size(Size::Medium)
908                            .with_group_variant(GroupBoxVariant::Outline)
909                            .pages(self.create_settings_pages(window, cx)),
910                    ),
911            )
912    }
