11    /// Open or close the Markdown preview tab.
12    ///
13    /// ### Arguments
14    /// - `window`: The window context
15    /// - `cx`: The application context
16    pub fn open_markdown_preview_tab(&mut self, window: &mut Window, cx: &mut Context<Self>) {
17        if self.settings.editor_settings.markdown_settings.preview_mode
18            != MarkdownPreviewMode::DedicatedTab
19        {
20            return;
21        }
22        let active_tab = self.active_tab_index.and_then(|index| self.tabs.get(index));
23        let editor_id = match active_tab {
24            Some(Tab::Editor(editor_tab)) => editor_tab.id,
25            Some(Tab::MarkdownPreview(preview_tab)) => preview_tab.source_tab_id,
26            _ => return,
27        };
28        if let Some(preview_id) = self
29            .tabs
30            .iter()
31            .find(|t| matches!(t, Tab::MarkdownPreview(p) if p.source_tab_id == editor_id))
32            .map(super::super::tab::Tab::id)
33        {
34            self.remove_tab_by_id(preview_id, window, cx);
35        } else {
36            let Some(editor_tab) = self.get_active_editor_tab() else {
37                return;
38            };
39            let title = SharedString::from(format!("Preview - {}", editor_tab.title));
40            let content = editor_tab.content.clone();
41            let editor_pos = self.active_tab_index.unwrap_or(0);
42            let view_state = cx.new(|cx| TextViewState::markdown("", cx));
43            let preview_tab = Tab::MarkdownPreview(MarkdownPreviewTab {
44                id: self.next_tab_id,
45                title,
46                source_tab_id: editor_id,
47                content,
48                view_state,
49            });
50            self.tabs.insert(editor_pos + 1, preview_tab);
51            self.next_tab_id += 1;
52            self.set_active_tab(editor_pos + 1, window, cx);
53        }
54    }

56    /// Insert Markdown preview tabs for all eligible editor tabs.
57    ///
58    /// ### Arguments
59    /// - `cx`: The application context, used to allocate per-preview view state
60    pub fn insert_preview_tabs_for_markdown(&mut self, cx: &mut Context<Self>) {
61        let settings = &self.settings.editor_settings.markdown_settings;
62        if settings.preview_mode != MarkdownPreviewMode::DedicatedTab
63            || !settings.show_markdown_preview
64        {
65            return;
66        }
67        let original_count = self.tabs.len();
68        let mut offset = 0;
69        for orig_idx in 0..original_count {
70            let actual_idx = orig_idx + offset;
71            let info = match self.tabs.get(actual_idx) {
72                Some(Tab::Editor(et))
73                    if et.language == SupportedLanguage::Markdown
74                        || et.language == SupportedLanguage::MarkdownInline =>
75                {
76                    Some((et.id, et.title.clone(), et.content.clone()))
77                }
78                _ => None,
79            };
80            if let Some((editor_id, title, content)) = info {
81                let view_state = cx.new(|cx| TextViewState::markdown("", cx));
82                let preview_tab = Tab::MarkdownPreview(MarkdownPreviewTab {
83                    id: self.next_tab_id,
84                    title: SharedString::from(format!("Preview - {title}")),
85                    source_tab_id: editor_id,
86                    content,
87                    view_state,
88                });
89                self.tabs.insert(actual_idx + 1, preview_tab);
90                self.next_tab_id += 1;
91                offset += 1;
92            }
93        }
94    }

135    /// Check whether an editor tab should be tracked as a markdown preview source.
136    ///
137    /// ### Arguments
138    /// - `language`: The editor tab language
139    /// - `show_markdown_preview`: Per-tab markdown preview toggle
140    ///
141    /// ### Returns
142    /// - `True` when this tab should keep markdown preview cache/subscriptions alive
143    fn should_track_markdown_preview_source(
144        language: SupportedLanguage,
145        show_markdown_preview: bool,
146    ) -> bool {
147        show_markdown_preview
148            && matches!(
149                language,
150                SupportedLanguage::Markdown | SupportedLanguage::MarkdownInline
151            )
152    }

154    /// Remove markdown preview cache entries for tabs that no longer exist.
155    ///
156    /// ### Arguments
157    /// - `cx`: The application context
158    pub(crate) fn prune_markdown_preview_cache(&mut self, cx: &mut Context<Self>) {
159        let source_ids: HashSet<usize> = self
160            .tabs
161            .iter()
162            .filter_map(|tab| match tab {
163                Tab::MarkdownPreview(preview_tab) => Some(preview_tab.source_tab_id),
164                Tab::Editor(editor_tab)
165                    if Self::should_track_markdown_preview_source(
166                        editor_tab.language,
167                        editor_tab.show_markdown_preview,
168                    ) =>
169                {
170                    Some(editor_tab.id)
171                }
172                _ => None,
173            })
174            .collect();
175
176        let before_cache = self.markdown_preview_cache.len();
177        let before_subs = self.markdown_preview_subscriptions.len();
178        self.markdown_preview_cache
179            .retain(|tab_id, _| source_ids.contains(tab_id));
180        self.markdown_preview_subscriptions
181            .retain(|tab_id, _| source_ids.contains(tab_id));
182        self.markdown_preview_to_refresh
183            .retain(|tab_id| source_ids.contains(tab_id));
184        if self.markdown_preview_cache.len() != before_cache
185            || self.markdown_preview_subscriptions.len() != before_subs
186        {
187            cx.notify();
188        }
189    }

191    /// Get cached markdown text for a source tab, refreshing lazily on demand.
192    ///
193    /// ### Arguments
194    /// - `source_tab_id`: Source editor tab id for this preview
195    /// - `content`: Source editor content entity
196    /// - `cx`: The application context
197    ///
198    /// ### Returns
199    /// - `SharedString`: Cached markdown source text for rendering
200    pub(crate) fn markdown_preview_text_for(
201        &mut self,
202        source_tab_id: usize,
203        content: &Entity<InputState>,
204        cx: &mut Context<Self>,
205    ) -> SharedString {
206        let current_entity_id = content.entity_id();
207        let entity_changed = self
208            .markdown_preview_subscriptions
209            .get(&source_tab_id)
210            .is_none_or(|(entity_id, _)| *entity_id != current_entity_id);
211        if entity_changed {
212            let subscription =
213                cx.subscribe(content, move |this: &mut Self, _, ev: &InputEvent, cx| {
214                    if !matches!(ev, InputEvent::Change) {
215                        return;
216                    }
217                    this.markdown_preview_to_refresh.insert(source_tab_id);
218                    cx.notify();
219                });
220            self.markdown_preview_subscriptions
221                .insert(source_tab_id, (current_entity_id, subscription));
222        }
223        let refresh_requested = self.markdown_preview_to_refresh.remove(&source_tab_id);
224        let needs_refresh = entity_changed
225            || refresh_requested
226            || !self.markdown_preview_cache.contains_key(&source_tab_id);
227        if needs_refresh {
228            self.markdown_preview_cache
229                .insert(source_tab_id, content.read(cx).value());
230        }
231        self.markdown_preview_cache
232            .get(&source_tab_id)
233            .cloned()
234            .unwrap_or_default()
235    }
