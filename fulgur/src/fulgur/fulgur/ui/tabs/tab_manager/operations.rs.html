5    /// Update the modified status of the tabs
6    ///
7    /// ### Arguments
8    /// - `cx`: The application context
9    pub fn update_modified_status(&mut self, cx: &mut Context<Self>) {
10        for tab in self.tabs.iter_mut() {
11            if let Tab::Editor(editor_tab) = tab {
12                editor_tab.check_modified(cx);
13            }
14        }
15    }

17    /// Reorder a tab from one index to another within this window.
18    ///
19    /// `to` is the logical insertion slot (0 = before all tabs, N = after all tabs).
20    /// No-op when the operation would leave the tab in its current position.
21    ///
22    /// ### Arguments
23    /// - `from`: The current index of the tab to move
24    /// - `to`: The insertion slot index (0..=tabs.len())
25    /// - `window`: The window context
26    /// - `cx`: The application context
27    pub fn reorder_tab(
28        &mut self,
29        from: usize,
30        to: usize,
31        window: &mut Window,
32        cx: &mut Context<Self>,
33    ) {
34        if from >= self.tabs.len() || to > self.tabs.len() {
35            return;
36        }
37        // Inserting at slot `to` or `to-1` when `to > from` is equivalent to no move.
38        if to == from || to == from + 1 {
39            return;
40        }
41        let tab = self.tabs.remove(from);
42        // After removing `from`, the effective insert position shifts down by 1 when to > from.
43        let insert_at = if to > from { to - 1 } else { to };
44        self.tabs.insert(insert_at, tab);
45        if let Some(active) = self.active_tab_index {
46            self.active_tab_index = Some(if from == active {
47                insert_at
48            } else if from < active && insert_at >= active {
49                active - 1
50            } else if from > active && insert_at <= active {
51                active + 1
52            } else {
53                active
54            });
55        }
56        if let Err(e) = self.save_state(cx, window) {
57            log::error!("Failed to save app state after reordering tab: {}", e);
58            self.pending_notification = Some((
59                gpui_component::notification::NotificationType::Warning,
60                format!("Tab reordered but failed to save state: {}", e).into(),
61            ));
62        }
63        cx.notify();
64    }

66    /// Handle a tab drop onto an insertion slot.
67    ///
68    /// Called by `on_drop` handlers on the slot divs in the tab bar.
69    ///
70    /// ### Arguments
71    /// - `dragged`: The drag payload
72    /// - `slot_index`: The insertion slot (0 = before first tab, N = after last tab)
73    /// - `window`: The window context
74    /// - `cx`: The application context
75    pub fn handle_tab_drop(
76        &mut self,
77        dragged: &crate::fulgur::ui::tabs::tab_drag::DraggedTab,
78        slot_index: usize,
79        window: &mut Window,
80        cx: &mut Context<Self>,
81    ) {
82        self.reorder_tab(dragged.tab_index, slot_index, window, cx);
83    }
