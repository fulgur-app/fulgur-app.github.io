49/// Create a tab bar button
50///
51/// ### Arguments
52/// - `id`: The ID of the button
53/// - `tooltip`: The tooltip of the button
54/// - `icon`: The icon of the button
55/// - `border_color`: The color of the border
56///
57/// ### Returns
58/// - `Button`: A tab bar button
59pub fn tab_bar_button_factory(
60    id: &'static str,
61    tooltip: &'static str,
62    icon: CustomIcon,
63    border_color: Hsla,
64) -> Button {
65    button_factory(id, tooltip, icon, border_color)
66        .border_b_1()
67        .h(TAB_BAR_BUTTON_SIZE)
68        .w(TAB_BAR_BUTTON_SIZE)
69}
