29    /// Compute a fingerprint that changes whenever the set of open file paths changes.
30    ///
31    /// ### Returns
32    /// - `u64`: The computed fingerprint value.
33    fn compute_tab_filename_fingerprint(&self) -> u64 {
34        let mut hash: u64 = self.tabs.len() as u64;
35        for tab in &self.tabs {
36            if let Some(editor) = tab.as_editor()
37                && let Some(path) = editor.file_path()
38            {
39                for byte in path.as_os_str().as_encoded_bytes() {
40                    hash = hash
41                        .wrapping_mul(0x0000_0100_0000_01b3)
42                        .wrapping_add(u64::from(*byte));
43                }
44            }
45            hash ^= tab.id() as u64;
46            hash = hash.rotate_left(17);
47        }
48        hash
49    }

51    /// Rebuild the cached tab filename frequency map only when the tab list has changed.
52    pub(crate) fn refresh_tab_filename_counts(&mut self) {
53        let fp = self.compute_tab_filename_fingerprint();
54        if fp == self.tab_filename_fp {
55            return;
56        }
57        self.tab_filename_fp = fp;
58        self.cached_tab_filename_counts = self.build_tab_filename_counts();
59    }

62    ///

64    /// - `tab`: The tab to inspect.

78    /// Build a per-filename tab count map for disambiguation.
79    ///
80    /// ### Returns
81    /// - `HashMap<String, usize>`: Map of filename to number of open tabs with that filename
82    pub(crate) fn build_tab_filename_counts(&self) -> HashMap<String, usize> {
83        let mut filename_counts = HashMap::new();
84        for tab in &self.tabs {
85            if let Some(editor_tab) = tab.as_editor()
86                && let Some(path) = editor_tab.file_path()
87                && let Some(filename) = path.file_name().and_then(|n| n.to_str())
88            {
89                *filename_counts.entry(filename.to_string()).or_insert(0) += 1;
90            }
91        }
92        filename_counts
93    }

95    /// Get the display title for a tab, including parent folder when duplicated.
96    ///
97    /// ### Arguments
98    /// - `tab`: The tab to get the title for
99    /// - `filename_counts`: Precomputed filename frequencies for all open tabs
100    ///
101    /// ### Returns
102    /// - `(String, Option<String>)`: A tuple of (filename, optional parent folder)
103    pub(crate) fn get_tab_display_title(
104        tab: &Tab,
105        filename_counts: &HashMap<String, usize>,
106    ) -> (String, Option<String>) {
107        let base_title = tab.title();
108        if let Some(editor_tab) = tab.as_editor()
109            && let Some(path) = editor_tab.file_path()
110        {
111            let filename = path.file_name().and_then(|n| n.to_str()).unwrap_or("");
112            let duplicate_count = filename_counts.get(filename).copied().unwrap_or(0);
113            if duplicate_count > 1
114                && let Some(parent) = path.parent()
115                && let Some(parent_name) = parent.file_name().and_then(|n| n.to_str())
116            {
117                return (filename.to_string(), Some(format!("../{parent_name}")));
118            }
119
120            return (filename.to_string(), None);
121        }
122        (base_title.to_string(), None)
123    }

192                            .border_color(cx.theme().border)
193                            .h(TAB_BAR_HEIGHT)
194                            .on_click(cx.listener(|this, event: &ClickEvent, window, cx| {
195                                if event.click_count() == 2 {
196                                    this.new_tab(window, cx);
197                                }
198                            }))
199                            .on_drag_move::<DraggedTab>(cx.listener(
200                                |this, event: &DragMoveEvent<DraggedTab>, _window, cx| {
201                                    let cursor = event.event.position;
202                                    let bounds = event.bounds;
203                                    if cursor.x < bounds.origin.x
204                                        || cursor.x > bounds.origin.x + bounds.size.width
205                                        || cursor.y < bounds.origin.y
206                                        || cursor.y > bounds.origin.y + bounds.size.height

208                                        return;
209                                    }
210                                    let slot = this.tabs.len();
211                                    let dragged = event.drag(cx).clone();
212                                    this.drag_ghost = Some((slot, dragged));
213                                    cx.notify();
214                                },
215                            ))
216                            .on_drop(cx.listener(|this, dragged: &DraggedTab, window, cx| {
217                                if let Some((slot, _)) = this.drag_ghost.take() {
218                                    this.handle_tab_drop(dragged, slot, window, cx);
219                                }
220                            })),
221                    ),
222            );
223        tab_bar
224    }

226    /// Render a ghost tab shown at the insertion point during a drag operation.
227    ///
228    /// The ghost tab previews where the dragged tab will land when dropped. It uses
229    /// a muted, semi-transparent style to distinguish it from real tabs.
230    ///
231    /// ### Arguments
232    /// - `slot`: The insertion slot index
233    /// - `dragged`: The dragged tab data (used for title and modified state)
234    /// - `cx`: The application context
235    ///
236    /// ### Returns
237    /// - `AnyElement`: The rendered ghost tab element
238    fn render_ghost_tab(slot: usize, dragged: &DraggedTab, cx: &mut Context<Self>) -> AnyElement {
239        use crate::fulgur::ui::components_utils::TAB_BAR_HEIGHT;
240
241        let modified_indicator = if dragged.is_modified { " •" } else { "" };
242        div()
243            .id(("ghost-tab", slot))
244            .flex()
245            .items_center()
246            .h(TAB_BAR_HEIGHT)
247            .px_2()
248            .gap_2()
249            .border_r_1()
250            .border_b_0()
251            .border_color(cx.theme().border)
252            .bg(cx.theme().tab_active)
253            .opacity(0.45)
254            .child(
255                div()
256                    .pl_1()
257                    .text_sm()
258                    .text_color(cx.theme().tab_active_foreground)
259                    .child(format!("{}{}", dragged.title, modified_indicator)),
260            )
261            .on_drop(cx.listener(|this, dragged: &DraggedTab, window, cx| {
262                if let Some((slot, _)) = this.drag_ghost.take() {
263                    this.handle_tab_drop(dragged, slot, window, cx);
264                }
265            }))
266            .into_any_element()
267    }

269    /// Render all tabs, inserting a ghost tab at the current drag insertion point.
270    ///
271    /// During a drag operation, a ghost tab is rendered at the slot determined by
272    /// the most recent `on_drag_move` event. The ghost is suppressed for no-op
273    /// positions (where the tab would not actually move).
274    ///
275    /// ### Arguments
276    /// - `cx`: The application context
277    ///
278    /// ### Returns
279    /// - `Vec<AnyElement>`: Tab elements, with a ghost tab inserted at the drag target
280    fn render_tabs_with_slots(&self, cx: &mut Context<Self>) -> Vec<AnyElement> {
281        let ghost = if cx.has_active_drag() {
282            self.drag_ghost.as_ref().and_then(|(slot, dragged)| {
283                let from = dragged.tab_index;
284                let is_noop = *slot == from || *slot == from + 1;
285                if is_noop {
286                    None
287                } else {
288                    Some((*slot, dragged))
289                }
290            })
291        } else {
292            None
293        };
294        let filename_counts = &self.cached_tab_filename_counts;
295        let capacity = self.tabs.len() + usize::from(ghost.is_some());
296        let mut elements: Vec<AnyElement> = Vec::with_capacity(capacity);
297        if let Some((0, dragged)) = ghost {
298            elements.push(Self::render_ghost_tab(0, dragged, cx));
299        }
300        for (index, tab) in self.tabs.iter().enumerate() {
301            elements.push(self.render_tab(index, tab, filename_counts, cx));
302            if let Some((slot, dragged)) = ghost
303                && slot == index + 1
304            {
305                elements.push(Self::render_ghost_tab(slot, dragged, cx));
306            }
307        }
308        elements
309    }

311    /// Render a single tab in the tab bar
312    ///
313    /// ### Arguments
314    /// - `index`: Position of the tab in `self.tabs`
315    /// - `tab`: The tab to render
316    /// - `filename_counts`: Precomputed map of filename to occurrence count, used to show a
317    ///   disambiguating folder segment when multiple open tabs share the same filename
318    /// - `cx`: The application context
319    ///
320    /// ### Returns
321    /// - `AnyElement`: The fully composed tab element ready to be inserted into the tab bar
322    pub fn render_tab(
323        &self,
324        index: usize,
325        tab: &Tab,
326        filename_counts: &HashMap<String, usize>,
327        cx: &mut Context<Self>,
328    ) -> AnyElement {
329        use crate::fulgur::ui::components_utils::TAB_BAR_HEIGHT;
330
331        let tab_id = tab.id();
332        let is_active = match self.active_tab_index {
333            Some(active_index) => index == active_index,
334            None => false,
335        };
336        let has_tabs_on_left = index > 0;
337        let has_tabs_on_right = index < self.tabs.len() - 1;
338        let total_tabs = self.tabs.len();
339        let file_path = tab.as_editor().and_then(|editor_tab| {
340            editor_tab
341                .file_path()
342                .and_then(|path| path.to_str().map(std::string::ToString::to_string))
343        });
344        let has_file_path = file_path.is_some();
345        let is_editor_tab = tab.as_editor().is_some();
346        let other_windows: Vec<(String, WeakEntity<Fulgur>)> = {
347            let manager = cx.global::<WindowManager>();
348            let current_window_id = self.window_id;
349            manager
350                .get_all_window_ids()
351                .into_iter()
352                .filter(|id| *id != current_window_id)
353                .filter_map(|id| {
354                    manager
355                        .get_window_name(id)
356                        .map(std::string::ToString::to_string)
357                        .zip(manager.get_window(id))
358                })
359                .collect()
360        };
361        let source_entity = cx.entity().downgrade();
362        let cached_file_size = tab
363            .as_editor()
364            .and_then(|editor_tab| editor_tab.file_size_bytes)
365            .map(components_utils::format_file_size);
366        let cached_last_modified = tab
367            .as_editor()
368            .and_then(|editor_tab| editor_tab.file_last_modified)
369            .and_then(components_utils::format_system_time);
370        let mut tab_div = div()
371            .id(("tab", tab_id))
372            .flex()
373            .items_center()
374            .h(TAB_BAR_HEIGHT)
375            .px_2()
376            .gap_2()
377            .border_r_1()
378            .border_b_1()
379            .border_color(cx.theme().border)
380            .on_mouse_down(
381                MouseButton::Left,
382                cx.listener(move |this, _, window, cx: &mut Context<'_, Fulgur>| {
383                    if !is_active {
384                        this.set_active_tab(index, window, cx);
385                    }
386                }),
387            );
388        if is_active {
389            tab_div = tab_div.bg(cx.theme().tab_active).border_b_0();
390            self.set_title(Some(tab.title().to_string()), cx);
391        } else {
392            tab_div = tab_div
393                .bg(cx.theme().tab)
394                .hover(|this| this.bg(cx.theme().muted))
395                .cursor_pointer();
396        }
397        if let Some(path) = file_path {
398            tab_div = tab_div.tooltip(move |window, cx| {
399                let path_clone = path.clone();
400                let file_size = cached_file_size.clone();
401                let last_modified = cached_last_modified.clone();
402                Tooltip::element(move |_, cx| {
403                    let mut tooltip = v_flex().gap_1().py_2().px_1().child(
404                        h_flex()
405                            .gap_3()
406                            .child(CustomIcon::File.icon())
407                            .child(path_clone.clone())
408                            .text_sm()
409                            .font_semibold(),
410                    );
411                    let mut details = h_flex().gap_4().justify_between();
412                    if let Some(ref size) = file_size {
413                        details = details.child(
414                            div()
415                                .child(format!("Size: {size}"))
416                                .text_xs()
417                                .text_color(cx.theme().muted_foreground),
418                        );
419                    }
420                    if let Some(ref last_modified) = last_modified {
421                        details = details.child(
422                            div()
423                                .child(format!("Last Modified: {last_modified}"))
424                                .text_xs()
425                                .text_color(cx.theme().muted_foreground),
426                        );
427                    }
428                    tooltip = tooltip.child(details);
429                    tooltip
430                })
431                .build(window, cx)
432            });
433        }
434        let (filename, folder) = Self::get_tab_display_title(tab, filename_counts);
435        let modified_indicator = if tab.is_modified() { " •" } else { "" };
436        let mut title_container = div().flex().items_center().gap_1().pl_1();
437        if let Some(remote_indicator) = Self::remote_tab_indicator_label(tab) {
438            title_container = title_container.child(
439                div()
440                    .text_sm()
441                    .font_semibold()
442                    .text_color(if is_active {
443                        cx.theme().tab_active_foreground.opacity(0.50)
444                    } else {
445                        cx.theme().primary.opacity(0.80)
446                    })
447                    .child(remote_indicator),
448            );
449        }
450        title_container = title_container.child(
451            div()
452                .text_sm()
453                .text_color(if is_active {
454                    cx.theme().tab_active_foreground
455                } else {
456                    cx.theme().tab_foreground
457                })
458                .child(format!("{filename}{modified_indicator}")),
459        );
460        if let Some(folder_path) = folder {
461            title_container = title_container.child(
462                div()
463                    .text_xs()
464                    .italic()
465                    .text_color(if is_active {
466                        cx.theme().tab_active_foreground
467                    } else {
468                        cx.theme().tab_foreground
469                    })
470                    .child(folder_path),
471            );
472        }
473        let is_markdown_preview = tab.as_markdown_preview().is_some();
474        let title: SharedString = tab.title().to_string().into();
475        let is_modified = tab.is_modified();
476        let mut tab_with_content = tab_div
477            .child(title_container)
478            .child(
479                Button::new(("close-tab", tab_id))
480                    .icon(CustomIcon::Close)
481                    .ghost()
482                    .xsmall()
483                    .cursor_pointer()
484                    .on_click(cx.listener(move |this, _, window, cx| {
485                        cx.stop_propagation();
486                        this.close_tab(tab_id, window, cx);
487                    })),
488            )
489            .on_mouse_down(
490                MouseButton::Middle,
491                cx.listener(move |this, _, window, cx| {
492                    this.close_tab(tab_id, window, cx);
493                }),
494            );
495        if !is_markdown_preview {
496            let is_source = cx.has_active_drag()
497                && self
498                    .drag_ghost
499                    .as_ref()
500                    .is_some_and(|(_, d)| d.tab_index == index);
501            if is_source {
502                tab_with_content = tab_with_content.opacity(0.45);
503            }
504            tab_with_content = tab_with_content
505                .on_drag(
506                    DraggedTab {
507                        tab_index: index,
508                        title,
509                        is_modified,
510                    },
511                    |dragged, _, _, cx| cx.new(|_| dragged.clone()),
512                )
513                .on_drag_move::<DraggedTab>(cx.listener(
514                    move |this, event: &DragMoveEvent<DraggedTab>, _window, cx| {
515                        let cursor = event.event.position;
516                        let bounds = event.bounds;
517                        if cursor.x < bounds.origin.x
518                            || cursor.x > bounds.origin.x + bounds.size.width
519                            || cursor.y < bounds.origin.y
520                            || cursor.y > bounds.origin.y + bounds.size.height
521                        {
522                            return;
523                        }
524                        let slot = if cursor.x < bounds.origin.x + bounds.size.width * 0.5 {
525                            index
526                        } else {
527                            index + 1
528                        };
529                        let dragged = event.drag(cx).clone();
530                        this.drag_ghost = Some((slot, dragged));
531                        cx.notify();
532                    },
533                ))
534                .on_drop(cx.listener(|this, dragged: &DraggedTab, window, cx| {
535                    if let Some((slot, _)) = this.drag_ghost.take() {
536                        this.handle_tab_drop(dragged, slot, window, cx);
537                    }
538                }));
539        }
540        let tab_with_content = tab_with_content.context_menu(move |this, window, cx| {
541            let other_windows_clone = other_windows.clone();
542            let source_entity_clone = source_entity.clone();
543            let this = this
544                .menu_with_disabled(
545                    crate::fulgur::ui::components_utils::reveal_in_file_manager_label(),
546                    Box::new(ShowInFileManager(index)),
547                    !has_file_path,
548                )
549                .menu_with_disabled(
550                    "Duplicate Tab",
551                    Box::new(DuplicateTab(index)),
552                    !is_editor_tab,
553                );
554            let this = if is_editor_tab {
555                this.submenu("Send to...", window, cx, move |sub, _window, _cx| {
556                    let mut sub = sub;
557                    for (name, weak_tgt) in &other_windows_clone {
558                        let label = format!("Window {name}");
559                        let src = source_entity_clone.clone();
560                        let tgt = weak_tgt.clone();
561                        sub = sub.item(PopupMenuItem::new(label).on_click(move |_, _, cx| {
562                            let Some(src_entity) = src.upgrade() else {
563                                return;
564                            };
565                            let Some(tgt_entity) = tgt.upgrade() else {
566                                return;
567                            };
568                            let data = src_entity.update(cx, |fulgur, cx| {
569                                fulgur.extract_tab_transfer_data(tab_id, cx)
570                            });
571                            let Some(data) = data else { return };
572                            tgt_entity.update(cx, |fulgur, cx| {
573                                fulgur.pending_tab_transfer = Some(data);
574                                cx.notify();
575                            });
576                            src_entity.update(cx, |fulgur, cx| {
577                                fulgur.pending_tab_removal = Some(tab_id);
578                                cx.notify();
579                            });
580                        }));
581                    }
582                    if !other_windows_clone.is_empty() {
583                        sub = sub.separator();
584                    }
585                    let src = source_entity_clone.clone();
586                    sub = sub.item(PopupMenuItem::new("New Window").on_click(move |_, _, cx| {
587                        let Some(src_entity) = src.upgrade() else {
588                            return;
589                        };
590                        let data = src_entity.update(cx, |fulgur, cx| {
591                            fulgur.extract_tab_transfer_data(tab_id, cx)
592                        });
593                        let Some(data) = data else { return };
594                        src_entity.update(cx, |fulgur, cx| {
595                            fulgur.pending_tab_removal = Some(tab_id);
596                            cx.notify();
597                        });
598                        src_entity.update(cx, |fulgur, cx| {
599                            fulgur.open_new_window_with_tab(data, cx);
600                        });
601                    }));
602                    sub
603                })
604            } else {
605                this.menu_with_disabled("Send to...", Box::new(SendTabToWindowNoOp), true)
606            };
607            this.separator()
608                .menu("Close Tab", Box::new(CloseTabAction(tab_id)))
609                .menu_with_disabled(
610                    "Close Tabs to the Left",
611                    Box::new(CloseTabsToLeft(index)),
612                    !has_tabs_on_left,
613                )
614                .menu_with_disabled(
615                    "Close Tabs to the Right",
616                    Box::new(CloseTabsToRight(index)),
617                    !has_tabs_on_right,
618                )
619                .separator()
620                .menu_with_disabled(
621                    "Close All Tabs",
622                    Box::new(CloseAllTabsAction),
623                    total_tabs == 0,
624                )
625                .menu_with_disabled(
626                    "Close All Other Tabs",
627                    Box::new(CloseAllOtherTabs(index)),
628                    total_tabs <= 1,
629                )
630        });
631        tab_with_content.into_any_element()
632    }
