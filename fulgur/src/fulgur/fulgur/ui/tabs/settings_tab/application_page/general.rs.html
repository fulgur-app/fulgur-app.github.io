9/// Render the tab color style chooser as a compact dropdown.
10///
11/// ### Arguments
12/// - `entity`: The Fulgur entity, read for the current style and updated on change
13/// - `cx`: The application context
14///
15/// ### Returns
16/// - `impl IntoElement`: The small dropdown button reflecting the current style
17pub(super) fn render_tab_color_style_select(
18    entity: &Entity<Fulgur>,
19    cx: &App,
20) -> impl IntoElement + use<> {
21    let current = entity.read(cx).settings.app_settings.tab_color_style;
22    let label = match current {
23        TabColorStyle::TextColor => "Colored Title",
24        TabColorStyle::Dot => "Colored Dot",
25    };
26    let entity = entity.clone();
27    Button::new("tab-color-style")
28        .label(label)
29        .dropdown_caret(true)
30        .outline()
31        .small()
32        .dropdown_menu_with_anchor(Anchor::TopRight, move |menu, _window, _cx| {
33            let entity_text = entity.clone();
34            let entity_dot = entity.clone();
35            menu.item(
36                PopupMenuItem::new("Colored Title")
37                    .checked(current == TabColorStyle::TextColor)
38                    .on_click(move |_, _, cx| {
39                        set_tab_color_style(&entity_text, TabColorStyle::TextColor, cx);
40                    }),
41            )
42            .item(
43                PopupMenuItem::new("Colored Dot")
44                    .checked(current == TabColorStyle::Dot)
45                    .on_click(move |_, _, cx| {
46                        set_tab_color_style(&entity_dot, TabColorStyle::Dot, cx);
47                    }),
48            )
49        })
50}

52/// Persist a new tab color style selection and propagate it to open windows.
53///
54/// ### Arguments
55/// - `entity`: The Fulgur entity to update
56/// - `style`: The newly selected tab color style
57/// - `cx`: The application context
58fn set_tab_color_style(entity: &Entity<Fulgur>, style: TabColorStyle, cx: &mut App) {
59    entity.update(cx, |this, cx| {
60        this.settings.app_settings.tab_color_style = style;
61        if let Err(e) = this.update_and_propagate_settings(cx) {
62            log::error!("Failed to save settings: {e}");
63        }
64    });
65}
