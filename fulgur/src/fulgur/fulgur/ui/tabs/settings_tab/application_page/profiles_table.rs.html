17/// Render the list of configured profiles as a table-like element.
18///
19/// ### Arguments
20/// - `entity`: The Fulgur entity.
21///
22/// ### Returns
23/// - `SettingItem`: The profiles table row.
24pub(super) fn render_profiles_table(entity: &Entity<Fulgur>) -> SettingItem {
25    let entity = entity.clone();
26    SettingItem::render(move |_options, _window, cx| {
27        let profiles = entity
28            .read(cx)
29            .settings
30            .app_settings
31            .synchronization_settings
32            .profiles
33            .clone();
34        let master_on = entity
35            .read(cx)
36            .settings
37            .app_settings
38            .synchronization_settings
39            .is_synchronization_activated;
40        if profiles.is_empty() {
41            return v_flex()
42                .w_full()
43                .gap_1()
44                .child(div().text_sm().child("Fulgurant instances"))
45                .child(table_header(cx))
46                .child(
47                    div()
48                        .w_full()
49                        .p_3()
50                        .text_sm()
51                        .text_color(cx.theme().muted_foreground)
52                        .child("No server configured. Click \"Add server\" below to add your first one."),
53                )
54                .into_any_element();
55        }
56        let entity_for_rows = entity.clone();
57        v_flex()
58            .w_full()
59            .gap_1()
60            .child(div().text_sm().child("Fulgurant instances"))
61            .child(table_header(cx))
62            .children(
63                profiles
64                    .iter()
65                    .map(|profile| {
66                        render_profile_row(entity_for_rows.clone(), profile, master_on, cx)
67                    })
68                    .collect::<Vec<_>>(),
69            )
70            .into_any_element()
71    })
72}

101/// Render a single profile row in the profiles table.
102///
103/// ### Arguments
104/// - `entity`: The Fulgur entity (used to open the edit sheet).
105/// - `profile`: The profile to render.
106/// - `master_on`: Whether the master switch is on.
107/// - `cx`: The application context.
108///
109/// ### Returns
110/// - `impl IntoElement`: The row element.
111fn render_profile_row(
112    entity: Entity<Fulgur>,
113    profile: &ServerProfile,
114    master_on: bool,
115    cx: &App,
116) -> impl IntoElement {
117    let display_url = profile
118        .server_url
119        .clone()
120        .unwrap_or_else(|| "-".to_string());
121    let version_label = get_profile_version_display(profile, master_on, cx);
122    let update_warning = get_profile_version_warning(profile, master_on, cx).map(|tooltip| {
123        let warn_id = SharedString::from(format!("profile-row-warn-{}", profile.id));
124        let tooltip = SharedString::from(tooltip);
125        div()
126            .id(warn_id)
127            .cursor_pointer()
128            .child(
129                Icon::new(CustomIcon::TriangleAlert)
130                    .with_size(gpui::px(18.0))
131                    .text_color(cx.theme().warning),
132            )
133            .tooltip(move |window, cx| Tooltip::new(tooltip.clone()).build(window, cx))
134    });
135    let pill = render_status_pill(profile, master_on, cx);
136    let row_id = SharedString::from(format!("profile-row-{}", profile.id));
137    let edit_id = SharedString::from(format!("profile-row-edit-{}", profile.id));
138    let activate_id = SharedString::from(format!("profile-row-activate-{}", profile.id));
139    let profile_id_for_edit = profile.id.clone();
140    let profile_id_for_activate = profile.id.clone();
141    let entity_for_activate = entity.clone();
142    let is_active = profile.is_active;
143    let profile_name = profile.name.clone();
144    h_flex()
145        .id(row_id)
146        .w_full()
147        .px_2()
148        .py_2()
149        .gap_2()
150        .items_center()
151        .border_b_1()
152        .border_color(cx.theme().border)
153        .child(
154            div()
155                .flex_1()
156                .text_sm()
157                .text_color(cx.theme().foreground)
158                .child(profile_name),
159        )
160        .child(
161            div()
162                .w(gpui::px(90.0))
163                .text_sm()
164                .text_color(cx.theme().muted_foreground)
165                .child(version_label),
166        )
167        .child(
168            div()
169                .flex_1()
170                .text_sm()
171                .text_color(cx.theme().muted_foreground)
172                .child(display_url),
173        )
174        .child(
175            div()
176                .w(gpui::px(20.0))
177                .flex()
178                .items_center()
179                .justify_center()
180                .children(update_warning),
181        )
182        .child(div().w(gpui::px(125.0)).child(pill).pr_4())
183        .child(
184            div()
185                .w(gpui::px(60.0))
186                .child(Switch::new(activate_id).checked(is_active).on_click(
187                    move |val: &bool, window, cx| {
188                        let id = profile_id_for_activate.clone();
189                        handle_profile_active_toggle(&entity_for_activate, &id, *val, window, cx);
190                    },
191                )),
192        )
193        .child(
194            div().w(gpui::px(80.0)).child(
195                Button::new(edit_id)
196                    .child("Edit")
197                    .small()
198                    .cursor_pointer()
199                    .on_click(move |_, window, cx| {
200                        let id = profile_id_for_edit.clone();
201                        entity.update(cx, |this, cx| {
202                            this.open_edit_profile_sheet(Some(&id), window, cx);
203                        });
204                    }),
205            ),
206        )
207}

209/// Toggle a profile's activation state and refresh its SSE connection.
210///
211/// ### Arguments
212/// - `entity`: The Fulgur entity.
213/// - `profile_id`: The id of the profile to toggle.
214/// - `value`: The new activation value.
215/// - `window`: The window (used to show connection progress).
216/// - `cx`: The application context.
217fn handle_profile_active_toggle(
218    entity: &Entity<Fulgur>,
219    profile_id: &str,
220    value: bool,
221    window: &mut Window,
222    cx: &mut App,
223) {
224    entity.update(cx, |this, cx| {
225        match this.update_profile(profile_id, |profile| profile.is_active = value, cx) {
226            Ok(true) => {
227                this.restart_sse_connection_for_with_progress(profile_id, window, cx);
228            }
229            Ok(false) => {
230                log::warn!("Profile '{profile_id}' could not be toggled (no longer in settings)");
231            }
232            Err(e) => {
233                log::error!("Failed to toggle profile '{profile_id}': {e}");
234            }
235        }
236    });
237}

239/// Render a status pill for a profile.
240///
241/// ### Arguments
242/// - `profile`: The profile to render the pill for.
243/// - `master_on`: Whether the master switch is on (overrides the per-profile
244///   status to Inactive when off).
245/// - `cx`: The application context.
246///
247/// ### Returns
248/// - `impl IntoElement`: The pill element.
249fn render_status_pill(profile: &ServerProfile, master_on: bool, cx: &App) -> impl IntoElement {
250    let status = get_profile_status(profile, master_on, cx);
251    let (bg, fg) = pill_colors(status, cx);
252    Label::new(status.label())
253        .rounded_lg()
254        .border_1()
255        .border_color(bg)
256        .text_sm()
257        .text_color(fg)
258        .text_center()
259}

261/// Resolve the foreground/background colors for a status pill.
262///
263/// ### Arguments
264/// - `status`: The profile's status enum value.
265///
266/// ### Returns
267/// - `(gpui::Hsla, gpui::Hsla)`: Background and foreground colors.
268fn pill_colors(status: SynchronizationStatus, cx: &App) -> (gpui::Hsla, gpui::Hsla) {
269    let theme = cx.theme();
270    match status {
271        SynchronizationStatus::Connected => (theme.success, theme.success),
272        SynchronizationStatus::Connecting => (theme.warning, theme.warning),
273        SynchronizationStatus::AuthenticationFailed
274        | SynchronizationStatus::ConnectionFailed
275        | SynchronizationStatus::NotActivated
276        | SynchronizationStatus::Other => (theme.danger, theme.danger),
277        SynchronizationStatus::Disconnected => (theme.info, theme.info),
278    }
