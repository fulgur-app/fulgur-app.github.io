26impl SettingsTab {
27    /// Create a new settings tab
28    ///
29    /// ### Arguments
30    /// - `id`: The ID of the settings tab
31    /// - `current_font`: The currently selected font family
32    /// - `window`: The window
33    /// - `cx`: The application context
34    ///
35    /// ### Returns
36    /// - `Self`: The settings tab
37    pub fn new(id: usize, current_font: &str, window: &mut Window, cx: &mut App) -> Self {
38        let font_family_select = Self::build_font_select(current_font, window, cx);
39        Self {
40            id,
41            title: SharedString::from("Settings"),
42            font_family_select,
43        }
44    }
45
46    /// Build the font family select entity populated with all system fonts.
47    ///
48    /// ### Arguments
49    /// - `current_font`: The currently selected font family used to set the initial selection
50    /// - `window`: The window
51    /// - `cx`: The application context
52    ///
53    /// ### Returns
54    /// - `Entity<SelectState<SearchableVec<SharedString>>>`: The font family select state
55    pub fn build_font_select(
56        current_font: &str,
57        window: &mut Window,
58        cx: &mut App,
59    ) -> Entity<SelectState<SearchableVec<SharedString>>> {
60        let mut system_fonts: Vec<SharedString> = cx
61            .text_system()
62            .all_font_names()
63            .into_iter()
64            .map(SharedString::from)
65            .collect();
66        system_fonts.sort();
67        system_fonts.dedup();
68        let selected_index = system_fonts
69            .iter()
70            .position(|f| f.as_ref() == current_font)
71            .map(|ix| IndexPath::default().row(ix));
72        cx.new(|cx| {
73            SelectState::new(SearchableVec::new(system_fonts), selected_index, window, cx)
74                .searchable(true)
75        })
76    }
77}

79impl Fulgur {
80    /// Create settings pages using the Settings component
81    ///
82    /// ### Arguments
83    /// - `window`: The window
84    /// - `cx`: The context
85    ///
86    /// ### Returns
87    /// - `Vec<SettingPage>`: The settings pages
88    fn create_settings_pages(
89        &self,
90        _window: &mut Window,
91        cx: &mut Context<Self>,
92    ) -> Vec<SettingPage> {
93        let entity = cx.entity();
94        let font_family_select = self
95            .tabs
96            .iter()
97            .find_map(|t| {
98                if let Tab::Settings(s) = t {
99                    Some(s.font_family_select.clone())
100                } else {
101                    None
102                }
103            })
104            .expect("create_settings_pages called without a settings tab in self.tabs");
105        let mut pages = vec![
106            editor_page::create_editor_page(&entity, font_family_select),
107            application_page::create_application_page(&entity),
108        ];
109        let themes = Fulgur::shared_state(cx).themes.clone();
110        if let Some(ref themes) = themes {
111            pages.push(Self::create_themes_page(&entity, themes));
112        }
113        pages
114    }

116    /// Render the settings
117    ///
118    /// ### Arguments
119    /// - `window`: The window
120    /// - `cx`: The context
121    ///
122    /// ### Returns
123    /// - `impl IntoElement`: The settings UI
124    pub fn render_settings(&self, window: &mut Window, cx: &mut Context<Self>) -> impl IntoElement {
125        const MAX_WIDTH: f32 = 1400.0;
126        let show_side_borders = window.viewport_size().width >= px(MAX_WIDTH);
127        div()
128            .id("settings-scroll-container")
129            .size_full()
130            .overflow_x_scroll()
131            .child(
132                v_flex()
133                    //.w(px(980.0))
134                    .h_full()
135                    .mx_auto()
136                    .max_w(px(MAX_WIDTH))
137                    .min_w(px(980.0))
138                    .text_color(cx.theme().foreground)
139                    .text_size(px(12.0))
140                    .when(show_side_borders, |el: gpui::Div| {
141                        el.border_l_1().border_r_1().border_color(cx.theme().border)
142                    })
143                    .child(
144                        SettingsComponent::new("fulgur-settings")
145                            .with_size(Size::Medium)
146                            .with_group_variant(GroupBoxVariant::Outline)
147                            .pages(self.create_settings_pages(window, cx)),
148                    ),
149            )
150    }

152    /// Clear the recent files
153    ///
154    /// ### Arguments
155    /// - `cx`: The context
156    pub fn clear_recent_files(&mut self, cx: &mut Context<Self>) {
157        self.settings.recent_files.clear();
158        if let Err(e) = self.update_and_propagate_settings(cx) {
159            log::error!("Failed to save settings: {e}");
160        }
161        let menus = crate::fulgur::ui::menus::build_menus(
162            self.settings.recent_files.get_files(),
163            Fulgur::shared_state(cx)
164                .update_info
165                .lock()
166                .as_ref()
167                .map(|info| info.download_url.as_str()),
168        );
169        self.update_menus(menus, cx);
170    }
