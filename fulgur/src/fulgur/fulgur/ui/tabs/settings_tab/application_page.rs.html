26/// Create the Application settings page
27///
28/// ### Arguments
29/// - `entity`: The Fulgur entity
30///
31/// ### Returns
32/// - `SettingPage`: The Application settings page
33pub fn create_application_page(entity: &Entity<Fulgur>) -> SettingPage {
34    let default_app_settings = AppSettings::new();
35
36    SettingPage::new("Application")
37        .default_open(true)
38        .groups(vec![
39            SettingGroup::new().title("General").items(vec![
40                SettingItem::new(
41                    "Confirm Exit",
42                    SettingField::switch(
43                        {
44                            let entity = entity.clone();
45                            move |cx: &App| entity.read(cx).settings.app_settings.confirm_exit
46                        },
47                        {
48                            let entity = entity.clone();
49                            move |val: bool, cx: &mut App| {
50                                entity.update(cx, |this, cx| {
51                                    this.settings.app_settings.confirm_exit = val;
52                                    if let Err(e) = this.update_and_propagate_settings(cx) {
53                                        log::error!("Failed to save settings: {e}");
54                                    }
55                                });
56                            }
57                        },
58                    )
59                    .default_value(default_app_settings.confirm_exit),
60                )
61                .description("Show confirmation dialog before exiting the application."),
62                SettingItem::new(
63                    "Debug mode",
64                    SettingField::switch(
65                        {
66                            let entity = entity.clone();
67                            move |cx: &App| entity.read(cx).settings.app_settings.debug_mode
68                        },
69                        {
70                            let entity = entity.clone();
71                            move |val: bool, cx: &mut App| {
72                                entity.update(cx, |this, cx| {
73                                    this.settings.app_settings.debug_mode = val;
74                                    if let Err(e) = this.update_and_propagate_settings(cx) {
75                                        log::error!("Failed to save settings: {e}");
76                                    }
77                                });
78                            }
79                        },
80                    )
81                    .default_value(default_app_settings.debug_mode),
82                )
83                .description("Enables debug mode, showing more info in the logs."),
84            ]),
85            SettingGroup::new().title("Synchronization").items(vec![
86                render_sync_error_banner(),
87                render_master_switch(entity),
88                render_profiles_table(entity),
89                render_add_server_button(entity),

91        ])
92}

172/// Apply a master switch toggle and propagate the side effects.
173///
174/// ### Arguments
175/// - `entity`: The Fulgur entity.
176/// - `value`: The new master switch value.
177/// - `cx`: The application context.
178fn handle_master_toggle(entity: &Entity<Fulgur>, value: bool, cx: &mut App) {
179    let active_profile_ids = entity.update(cx, |this, cx| {
180        this.settings
181            .app_settings
182            .synchronization_settings
183            .is_synchronization_activated = value;
184        if value && let Err(e) = crypto_helper::check_private_public_keys(&mut this.settings) {
185            log::error!("Failed to check private/public keys: {e}");
186        }
187        if let Err(e) = this.update_and_propagate_settings(cx) {
188            log::error!("Failed to save settings: {e}");
189        }
190        let ids: Vec<String> = this
191            .settings
192            .app_settings
193            .synchronization_settings
194            .profiles
195            .iter()
196            .filter(|p| p.is_active)
197            .map(|p| p.id.clone())
198            .collect();
199        for profile_id in &ids {
200            this.restart_sse_connection_for(profile_id, cx);
201        }
202        ids
203    });
204    log::debug!(
205        "Master switch toggled to {value}; refreshed SSE for {} active profile(s)",
206        active_profile_ids.len()
207    );
208}

294/// Render a single profile row in the profiles table.
295///
296/// ### Arguments
297/// - `entity`: The Fulgur entity (used to open the edit sheet).
298/// - `profile`: The profile to render.
299/// - `master_on`: Whether the master switch is on.
300/// - `cx`: The application context.
301///
302/// ### Returns
303/// - `impl IntoElement`: The row element.
304fn render_profile_row(
305    entity: Entity<Fulgur>,
306    profile: &ServerProfile,
307    master_on: bool,
308    cx: &App,
309) -> impl IntoElement {
310    let display_url = profile
311        .server_url
312        .clone()
313        .unwrap_or_else(|| "-".to_string());
314    let version_label = get_profile_version_display(profile, master_on, cx);
315    let update_warning = get_profile_version_warning(profile, master_on, cx).map(|tooltip| {
316        let warn_id = SharedString::from(format!("profile-row-warn-{}", profile.id));
317        let tooltip = SharedString::from(tooltip);
318        div()
319            .id(warn_id)
320            .cursor_pointer()
321            .child(
322                Icon::new(CustomIcon::TriangleAlert)
323                    .with_size(gpui::px(18.0))
324                    .text_color(cx.theme().warning),
325            )
326            .tooltip(move |window, cx| Tooltip::new(tooltip.clone()).build(window, cx))
327    });
328    let pill = render_status_pill(profile, master_on, cx);
329    let row_id = SharedString::from(format!("profile-row-{}", profile.id));
330    let edit_id = SharedString::from(format!("profile-row-edit-{}", profile.id));
331    let activate_id = SharedString::from(format!("profile-row-activate-{}", profile.id));
332    let profile_id_for_edit = profile.id.clone();
333    let profile_id_for_activate = profile.id.clone();
334    let entity_for_activate = entity.clone();

336    let profile_name = profile.name.clone();
337    h_flex()
338        .id(row_id)
339        .w_full()
340        .px_2()
341        .py_2()
342        .gap_2()
343        .items_center()
344        .border_b_1()
345        .border_color(cx.theme().border)
346        .child(
347            div()
348                .flex_1()
349                .text_sm()
350                .text_color(cx.theme().foreground)

354            div()
355                .w(gpui::px(90.0))
356                .text_sm()
357                .text_color(cx.theme().muted_foreground)

360        .child(
361            div()
362                .flex_1()
363                .text_sm()

378                .w(gpui::px(60.0))

402/// Toggle a profile's activation state and refresh its SSE connection.
403///
404/// ### Arguments
405/// - `entity`: The Fulgur entity.
406/// - `profile_id`: The id of the profile to toggle.
407/// - `value`: The new activation value.
408/// - `window`: The window (used to show connection progress).
409/// - `cx`: The application context.
410fn handle_profile_active_toggle(
411    entity: &Entity<Fulgur>,
412    profile_id: &str,
413    value: bool,
414    window: &mut Window,
415    cx: &mut App,
416) {
417    entity.update(cx, |this, cx| {
418        match this.update_profile(profile_id, |profile| profile.is_active = value, cx) {
419            Ok(true) => {
420                this.restart_sse_connection_for_with_progress(profile_id, window, cx);
421            }
422            Ok(false) => {
423                log::warn!("Profile '{profile_id}' could not be toggled (no longer in settings)");
424            }
425            Err(e) => {
426                log::error!("Failed to toggle profile '{profile_id}': {e}");
427            }
428        }
429    });
430}

454/// Get the status  for a profile.
455///
456/// ### Arguments
457/// - `profile`: The profile.
458/// - `master_on`: The master switch value.
459/// - `cx`: The application context.
460///
461/// ### Returns
462/// - `SynchronizationStatus`: The profile's status.
463fn get_profile_status(profile: &ServerProfile, master_on: bool, cx: &App) -> SynchronizationStatus {
464    if !master_on || !profile.is_active {
465        return SynchronizationStatus::NotActivated;
466    }
467    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
468    let sync_states = shared.sync_states.read();
469    sync_states
470        .get(&profile.id)
471        .map_or(SynchronizationStatus::NotActivated, |state| {
472            *state.connection_status.lock()
473        })
474}

476/// Resolve the version label to display for a profile.
477///
478/// ### Arguments
479/// - `profile`: The profile to resolve the version for.
480/// - `master_on`: Whether the master switch is on.
481/// - `cx`: The application context.
482///
483/// ### Returns
484/// - `String`: The version label to render.
485fn get_profile_version_display(profile: &ServerProfile, master_on: bool, cx: &App) -> String {
486    if !matches!(
487        get_profile_status(profile, master_on, cx),
488        SynchronizationStatus::Connected
489    ) {
490        return "-".to_string();
491    }
492    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
493    let sync_states = shared.sync_states.read();
494    let Some(state) = sync_states.get(&profile.id) else {
495        return "-".to_string();
496    };
497    let version = state.server_version.lock();
498    match version.as_deref() {
499        Some(raw) => format!("v{}", raw.trim().trim_start_matches(['v', 'V'])),
500        None => "< v0.7.0".to_string(),
501    }
502}

504/// Resolve an "update Fulgur" warning for a profile when its server requires a
505/// newer Fulgur version than the running build.
506///
507/// ### Arguments
508/// - `profile`: The profile to check.
509/// - `master_on`: Whether the master switch is on.
510/// - `cx`: The application context.
511///
512/// ### Returns
513/// - `Some(String)`: A tooltip describing the required Fulgur update.
514/// - `None`: The running Fulgur is compatible, or the profile is not connected.
515fn get_profile_fulgur_update_warning(
516    profile: &ServerProfile,
517    master_on: bool,
518    cx: &App,
519) -> Option<String> {
520    if !matches!(
521        get_profile_status(profile, master_on, cx),
522        SynchronizationStatus::Connected
523    ) {
524        return None;
525    }
526    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
527    let sync_states = shared.sync_states.read();
528    let required = sync_states
529        .get(&profile.id)?
530        .server_min_fulgur_version
531        .lock()
532        .clone()?;
533    let current = env!("CARGO_PKG_VERSION");
534    match compare_required_version(current, &required) {
535        VersionCompatibility::Compatible => None,
536        VersionCompatibility::UpdateRecommended | VersionCompatibility::UpdateRequired => {
537            Some(format!(
538                "This server needs Fulgur v{required} or newer (you have v{current}). Please update Fulgur."
539            ))
540        }
541    }
542}

544/// Resolve an "update Fulgurant" warning for a profile when the connected
545/// Fulgurant is older than the version this Fulgur build is best paired with.
546///
547/// ### Arguments
548/// - `profile`: The profile to check.
549/// - `master_on`: Whether the master switch is on.
550/// - `cx`: The application context.
551///
552/// ### Returns
553/// - `Some(String)`: A tooltip describing the recommended Fulgurant update.
554/// - `None`: Fulgurant is recent enough, or the profile is not connected.
555fn get_profile_fulgurant_update_warning(
556    profile: &ServerProfile,
557    master_on: bool,
558    cx: &App,
559) -> Option<String> {
560    if !matches!(
561        get_profile_status(profile, master_on, cx),
562        SynchronizationStatus::Connected
563    ) {
564        return None;
565    }
566    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
567    let sync_states = shared.sync_states.read();
568    let reported = sync_states.get(&profile.id)?.server_version.lock().clone();
569    let current = reported.as_deref().map_or_else(
570        || FULGURANT_VERSION_WITHOUT_HEADER.to_string(),
571        |raw| raw.trim().trim_start_matches(['v', 'V']).to_string(),
572    );
573    let required = RECOMMENDED_FULGURANT_VERSION;
574    match compare_required_version(&current, required) {
575        VersionCompatibility::Compatible => None,
576        VersionCompatibility::UpdateRecommended | VersionCompatibility::UpdateRequired => {
577            Some(match reported {
578                Some(_) => format!(
579                    "This server runs Fulgurant v{current}, but Fulgur works best with v{required} or newer. Please update Fulgurant."
580                ),
581                None => format!(
582                    "This server runs an old Fulgurant (pre-0.7.0). Fulgur works best with v{required} or newer. Please update Fulgurant."
583                ),
584            })
585        }
586    }
587}

589/// Resolve the combined version-mismatch warning for a profile, covering both
590/// "update Fulgur" (server too new) and "update Fulgurant" (server too old).
591///
592/// ### Arguments
593/// - `profile`: The profile to check.
594/// - `master_on`: Whether the master switch is on.
595/// - `cx`: The application context.
596///
597/// ### Returns
598/// - `Some(String)`: A tooltip with one or both warnings, newline-separated.
599/// - `None`: No version mismatch to surface.
600fn get_profile_version_warning(
601    profile: &ServerProfile,
602    master_on: bool,
603    cx: &App,
604) -> Option<String> {
605    let mut parts = Vec::new();
606    if let Some(warning) = get_profile_fulgur_update_warning(profile, master_on, cx) {
607        parts.push(warning);
608    }
609    if let Some(warning) = get_profile_fulgurant_update_warning(profile, master_on, cx) {
610        parts.push(warning);
611    }
612    (!parts.is_empty()).then(|| parts.join("\n"))
613}

615/// Resolve the foreground/background colors for a status pill.
616///
617/// ### Arguments
618/// - `status`: The profile's status enum value.
619///
620/// ### Returns
621/// - `(gpui::Hsla, gpui::Hsla)`: Background and foreground colors.
622fn pill_colors(status: SynchronizationStatus, cx: &App) -> (gpui::Hsla, gpui::Hsla) {
623    let theme = cx.theme();
624    match status {
625        SynchronizationStatus::Connected => (theme.success, theme.success),
626        SynchronizationStatus::Connecting => (theme.warning, theme.warning),
627        SynchronizationStatus::AuthenticationFailed
628        | SynchronizationStatus::ConnectionFailed
629        | SynchronizationStatus::NotActivated
630        | SynchronizationStatus::Other => (theme.danger, theme.danger),
631        SynchronizationStatus::Disconnected => (theme.info, theme.info),
632    }
633}

638/// - `entity`: The Fulgur entity.
639///
640/// ### Returns
641/// - `SettingItem`: The button row.
642fn render_add_server_button(entity: &Entity<Fulgur>) -> SettingItem {
643    let entity = entity.clone();

645        let entity = entity.clone();
646        h_flex()
647            .w_full()
648            .justify_end()

654                    .primary()

660                            .app_settings
661                            .synchronization_settings
662                            .profiles
663                            .len()

675                            return;
676                        }
677                        entity.update(cx, |this, cx| {
678                            this.open_edit_profile_sheet(None, window, cx);
679                        });
680                    }),

682            .into_any_element()
683    })
