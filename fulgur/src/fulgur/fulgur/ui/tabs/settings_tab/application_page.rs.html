16/// Create the Application settings page
17///
18/// ### Arguments
19/// - `entity`: The Fulgur entity
20///
21/// ### Returns
22/// - `SettingPage`: The Application settings page
23pub fn create_application_page(entity: Entity<Fulgur>) -> SettingPage {
24    let default_app_settings = AppSettings::new();
25
26    SettingPage::new("Application")
27        .default_open(true)
28        .groups(vec![
29            SettingGroup::new().title("General").items(vec![
30                SettingItem::new(
31                    "Confirm Exit",
32                    SettingField::switch(
33                        {
34                            let entity = entity.clone();
35                            move |cx: &App| entity.read(cx).settings.app_settings.confirm_exit
36                        },
37                        {
38                            let entity = entity.clone();
39                            move |val: bool, cx: &mut App| {
40                                entity.update(cx, |this, cx| {
41                                    this.settings.app_settings.confirm_exit = val;
42                                    if let Err(e) = this.update_and_propagate_settings(cx) {
43                                        log::error!("Failed to save settings: {}", e);
44                                    }
45                                });
46                            }
47                        },
48                    )
49                    .default_value(default_app_settings.confirm_exit),
50                )
51                .description("Show confirmation dialog before exiting the application."),
52                SettingItem::new(
53                    "Debug mode",
54                    SettingField::switch(
55                        {
56                            let entity = entity.clone();
57                            move |cx: &App| entity.read(cx).settings.app_settings.debug_mode
58                        },
59                        {
60                            let entity = entity.clone();
61                            move |val: bool, cx: &mut App| {
62                                entity.update(cx, |this, cx| {
63                                    this.settings.app_settings.debug_mode = val;
64                                    if let Err(e) = this.update_and_propagate_settings(cx) {
65                                        log::error!("Failed to save settings: {}", e);
66                                    }
67                                });
68                            }
69                        },
70                    )
71                    .default_value(default_app_settings.debug_mode),
72                )
73                .description("Enables debug mode, showing more info in the logs."),
74            ]),
75            SettingGroup::new().title("Synchronization").items({
76                let mut sync_items = vec![];
77                sync_items.push(SettingItem::render({
78                    move |_options, _window, cx| {
79                        let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
80                        if let Some(error_msg) = shared.sync_error.lock().as_ref() {
81                            v_flex()
82                                .w_full()
83                                .p_3()
84                                .mb_2()
85                                .bg(cx.theme().muted)
86                                .border_1()
87                                .border_color(cx.theme().border)
88                                .rounded(gpui::px(4.0))
89                                .child(
90                                    div()
91                                        .text_color(cx.theme().foreground)
92                                        .text_size(gpui::px(13.0))
93                                        .font_weight(gpui::FontWeight::SEMIBOLD)
94                                        .child(error_msg.clone()),
95                                )
96                                .into_any_element()
97                        } else {
98                            div().into_any_element()
99                        }
100                    }
101                }));
102                sync_items.push(SettingItem::new(
103                    "Activate Synchronization",
104                    SettingField::switch(
105                        {
106                            let entity = entity.clone();
107                            move |cx: &App| {
108                                entity
109                                    .read(cx)
110                                    .settings
111                                    .app_settings
112                                    .synchronization_settings
113                                    .is_synchronization_activated
114                            }
115                        },
116                        {
117                            let entity = entity.clone();
118                            move |val: bool, cx: &mut App| {
119                                entity.update(cx, |this, cx| {
120                                        this.settings
121                                            .app_settings
122                                            .synchronization_settings
123                                            .is_synchronization_activated = val;
124                                        if val && let Err(e) =
125                                                crypto_helper::check_private_public_keys(&mut this.settings)
126                                        {
127                                                log::error!("Failed to check private/public keys: {}", e);
128                                        }
129                                        if let Err(e) = this.update_and_propagate_settings(cx) {
130                                            log::error!("Failed to save settings: {}", e);
131                                        }
132                                    });
133                                if val {
134                                    perform_initial_synchronization(entity.clone(), cx);
135                                }
136                            }
137                        },
138                    )
139                    .default_value(default_app_settings.synchronization_settings.is_synchronization_activated),
140                )
141                .description("Activate synchronization with the server and saves the relevant keys in the system's keychain."));
142                sync_items.push(SettingItem::new(
143                    "Deduplication",
144                    SettingField::switch(
145                        {
146                            let entity = entity.clone();
147                            move |cx: &App| {
148                                entity
149                                    .read(cx)
150                                    .settings
151                                    .app_settings
152                                    .synchronization_settings
153                                    .is_deduplication
154                            }
155                        },
156                        {
157                            let entity = entity.clone();
158                            move |val: bool, cx: &mut App| {
159                                entity.update(cx, |this, cx| {
160                                    this.settings
161                                        .app_settings
162                                        .synchronization_settings
163                                        .is_deduplication = val;
164                                    if let Err(e) = this.update_and_propagate_settings(cx) {
165                                        log::error!("Failed to save settings: {}", e);
166                                    }
167                                });
168                            }
169                        },
170                    )
171                    .default_value(default_app_settings.synchronization_settings.is_deduplication),
172                )
173                .description("Avoid duplicate shares of the same file on the server."));
174                sync_items.push(SettingItem::new(
175                    "Server URL",
176                    SettingField::input(
177                        {
178                            let entity = entity.clone();
179                            move |cx: &App| {
180                                entity
181                                    .read(cx)
182                                    .settings
183                                    .app_settings
184                                    .synchronization_settings
185                                    .server_url
186                                    .as_ref()
187                                    .map(|s| SharedString::from(s.clone()))
188                                    .unwrap_or_default()
189                            }
190                        },
191                        {
192                            let entity = entity.clone();
193                            move |val: SharedString, cx: &mut App| {
194                                entity.update(cx, |this, cx| {
195                                    let url = if val.is_empty() {
196                                        None
197                                    } else {
198                                        Some(val.to_string())
199                                    };
200                                    this.settings
201                                        .app_settings
202                                        .synchronization_settings
203                                        .server_url = url;
204                                    if let Err(e) = this.update_and_propagate_settings(cx) {
205                                        log::error!("Failed to save settings: {}", e);
206                                    }
207                                    this.restart_sse_connection(cx);
208                                });
209                            }
210                        },
211                    )
212                    .default_value(
213                        default_app_settings
214                            .synchronization_settings
215                            .server_url
216                            .clone()
217                            .map(SharedString::from)
218                            .unwrap_or_default(),
219                    ),
220                )
221                .description("URL of the synchronization server."));
222                sync_items.push(SettingItem::new(
223                    "Email",
224                    SettingField::input(
225                        {
226                            let entity = entity.clone();
227                            move |cx: &App| {
228                                entity
229                                    .read(cx)
230                                    .settings
231                                    .app_settings
232                                    .synchronization_settings
233                                    .email
234                                    .as_ref()
235                                    .map(|s| SharedString::from(s.clone()))
236                                    .unwrap_or_default()
237                            }
238                        },
239                        {
240                            let entity = entity.clone();
241                            move |val: SharedString, cx: &mut App| {
242                                entity.update(cx, |this, cx| {
243                                    let email = if val.is_empty() {
244                                        None
245                                    } else {
246                                        Some(val.to_string())
247                                    };
248                                    this.settings.app_settings.synchronization_settings.email =
249                                        email;
250                                    if let Err(e) = this.update_and_propagate_settings(cx) {
251                                        log::error!("Failed to save settings: {}", e);
252                                    }
253                                    this.restart_sse_connection(cx);
254                                });
255                            }
256                        },
257                    )
258                    .default_value(
259                        default_app_settings
260                            .synchronization_settings
261                            .email
262                            .clone()
263                            .map(SharedString::from)
264                            .unwrap_or_default(),
265                    ),
266                )
267                .description("Email for synchronization."));
268                sync_items.push(SettingItem::new(
269                    "Device Key",
270                    SettingField::input(
271                        move |_cx: &App| SharedString::from(DEVICE_KEY_PLACEHOLDER),
272                        {
273                            let entity = entity.clone();
274                            move |val: SharedString, cx: &mut App| {
275                                entity.update(cx, |this, cx| {
276                                    let key = if val.is_empty() {
277                                        None
278                                    } else if val == DEVICE_KEY_PLACEHOLDER {
279                                        return;
280                                    } else {
281                                        Some(val.to_string())
282                                    };
283                                    if let Err(e) =
284                                        crypto_helper::save_device_api_key_to_keychain(key)
285                                    {
286                                        log::error!("Failed to save device API key: {}", e);
287                                    } else {
288                                        log::info!("Device API key saved successfully");
289                                        // Clear cached token to force re-authentication with new device key
290                                        this.shared_state(cx).sync_state.token_state.clear_token();
291                                    }
292                                    this.restart_sse_connection(cx);
293                                });
294                            }
295                        },
296                    ),
297                )
298                .description(
299                    "Device Key for synchronization (stored in keychain).",
300                ));
301                sync_items.push(SettingItem::render({
302                    let entity = entity.clone();
303                    move |_options, _window, cx| {
304                        let is_connecting = cx
305                            .global::<crate::fulgur::shared_state::SharedAppState>()
306                            .sync_state
307                            .connection_status
308                            .lock()
309                            .is_connecting();
310                        let label = if is_connecting {
311                            "Connecting..."
312                        } else {
313                            "Begin Synchronization"
314                        };
315                        h_flex()
316                            .w_full()
317                            .justify_end()
318                            .mt_2()
319                            .child({
320                                let mut btn = Button::new("begin-synchronization-button")
321                                    .label(label)
322                                    .primary()
323                                    .small()
324                                    .loading(is_connecting);
325                                if !is_connecting {
326                                    btn = btn.cursor_pointer().on_click({
327                                        let entity = entity.clone();
328                                        move |_, _window, cx| {
329                                            let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
330                                            shared.sync_state.token_state.clear_token();

333                                                cx,
334                                            );
335                                        }
336                                    });
337                                }
338                                btn
339                            })
340                            .into_any_element()
341                    }
342                }));
343                sync_items
344            }),
345        ])
