21/// Create the Application settings page
22///
23/// ### Arguments
24/// - `entity`: The Fulgur entity
25///
26/// ### Returns
27/// - `SettingPage`: The Application settings page
28pub fn create_application_page(entity: &Entity<Fulgur>) -> SettingPage {
29    let default_app_settings = AppSettings::new();
30
31    SettingPage::new("Application")
32        .default_open(true)
33        .groups(vec![
34            SettingGroup::new().title("General").items(vec![
35                SettingItem::new(
36                    "Confirm Exit",
37                    SettingField::switch(
38                        {
39                            let entity = entity.clone();
40                            move |cx: &App| entity.read(cx).settings.app_settings.confirm_exit
41                        },
42                        {
43                            let entity = entity.clone();
44                            move |val: bool, cx: &mut App| {
45                                entity.update(cx, |this, cx| {
46                                    this.settings.app_settings.confirm_exit = val;
47                                    if let Err(e) = this.update_and_propagate_settings(cx) {
48                                        log::error!("Failed to save settings: {e}");
49                                    }
50                                });
51                            }
52                        },
53                    )
54                    .default_value(default_app_settings.confirm_exit),
55                )
56                .description("Show confirmation dialog before exiting the application."),
57                SettingItem::new(
58                    "Debug mode",
59                    SettingField::switch(
60                        {
61                            let entity = entity.clone();
62                            move |cx: &App| entity.read(cx).settings.app_settings.debug_mode
63                        },
64                        {
65                            let entity = entity.clone();
66                            move |val: bool, cx: &mut App| {
67                                entity.update(cx, |this, cx| {
68                                    this.settings.app_settings.debug_mode = val;
69                                    if let Err(e) = this.update_and_propagate_settings(cx) {
70                                        log::error!("Failed to save settings: {e}");
71                                    }
72                                });
73                            }
74                        },
75                    )
76                    .default_value(default_app_settings.debug_mode),
77                )
78                .description("Enables debug mode, showing more info in the logs."),
79            ]),
80            SettingGroup::new().title("Synchronization").items(vec![
81                render_sync_error_banner(),
82                render_master_switch(entity),
83                render_profiles_table(entity),
84                render_add_server_button(entity),
85            ]),
86        ])
87}

89/// Render the inline error banner shown when key initialization failed.
90///
91/// ### Returns
92/// - `SettingItem`: The banner element wrapped as a `SettingItem`.
93fn render_sync_error_banner() -> SettingItem {
94    SettingItem::render(move |_options, _window, cx| {
95        let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
96        if let Some(error_msg) = shared.sync_error.lock().as_ref() {
97            v_flex()
98                .w_full()
99                .p_3()
100                .mb_2()
101                .bg(cx.theme().muted)
102                .border_1()
103                .border_color(cx.theme().border)
104                .rounded(gpui::px(4.0))
105                .child(
106                    div()
107                        .text_color(cx.theme().foreground)
108                        .text_size(gpui::px(13.0))
109                        .font_weight(FontWeight::SEMIBOLD)
110                        .child(error_msg.clone()),
111                )
112                .into_any_element()
113        } else {
114            div().into_any_element()
115        }
116    })
117}

167/// Apply a master switch toggle and propagate the side effects.
168///
169/// ### Arguments
170/// - `entity`: The Fulgur entity.
171/// - `value`: The new master switch value.
172/// - `cx`: The application context.
173fn handle_master_toggle(entity: &Entity<Fulgur>, value: bool, cx: &mut App) {
174    let active_profile_ids = entity.update(cx, |this, cx| {
175        this.settings
176            .app_settings
177            .synchronization_settings
178            .is_synchronization_activated = value;
179        if value && let Err(e) = crypto_helper::check_private_public_keys(&mut this.settings) {
180            log::error!("Failed to check private/public keys: {e}");
181        }
182        if let Err(e) = this.update_and_propagate_settings(cx) {
183            log::error!("Failed to save settings: {e}");
184        }
185        let ids: Vec<String> = this
186            .settings
187            .app_settings
188            .synchronization_settings
189            .profiles
190            .iter()
191            .filter(|p| p.is_active)
192            .map(|p| p.id.clone())
193            .collect();
194        for profile_id in &ids {
195            this.restart_sse_connection_for(profile_id, cx);
196        }
197        ids
198    });
199    log::debug!(
200        "Master switch toggled to {value}; refreshed SSE for {} active profile(s)",
201        active_profile_ids.len()
202    );
203}

205/// Render the list of configured profiles as a table-like element.
206///
207/// ### Arguments
208/// - `entity`: The Fulgur entity.
209///
210/// ### Returns
211/// - `SettingItem`: The profiles table row.
212fn render_profiles_table(entity: &Entity<Fulgur>) -> SettingItem {
213    let entity = entity.clone();
214    SettingItem::render(move |_options, _window, cx| {
215        let profiles = entity
216            .read(cx)
217            .settings
218            .app_settings
219            .synchronization_settings
220            .profiles
221            .clone();
222        let master_on = entity
223            .read(cx)
224            .settings
225            .app_settings
226            .synchronization_settings
227            .is_synchronization_activated;
228        if profiles.is_empty() {
229            return v_flex()
230                .w_full()
231                .gap_1()
232                .child(div().text_sm().child("Fulgurant instances"))
233                .child(table_header(cx))
234                .child(
235                    div()
236                        .w_full()
237                        .p_3()
238                        .text_sm()
239                        .text_color(cx.theme().muted_foreground)
240                        .child("No server configured. Click \"Add server\" below to add your first one."),
241                )
242                .into_any_element();
243        }
244        let entity_for_rows = entity.clone();
245        v_flex()
246            .w_full()
247            .gap_1()
248            .child(div().text_sm().child("Fulgurant instances"))
249            .child(table_header(cx))
250            .children(
251                profiles
252                    .iter()
253                    .map(|profile| {
254                        render_profile_row(entity_for_rows.clone(), profile, master_on, cx)
255                    })
256                    .collect::<Vec<_>>(),
257            )
258            .into_any_element()
259    })
260}

287/// Render a single profile row in the profiles table.
288///
289/// ### Arguments
290/// - `entity`: The Fulgur entity (used to open the edit sheet).
291/// - `profile`: The profile to render.
292/// - `master_on`: Whether the master switch is on.
293/// - `cx`: The application context.
294///
295/// ### Returns
296/// - `impl IntoElement`: The row element.
297fn render_profile_row(
298    entity: Entity<Fulgur>,
299    profile: &ServerProfile,
300    master_on: bool,
301    cx: &App,
302) -> impl IntoElement {
303    let display_url = profile
304        .server_url
305        .clone()
306        .unwrap_or_else(|| "-".to_string());
307    let version_label = get_profile_version_display(profile, master_on, cx);
308    let pill = render_status_pill(profile, master_on, cx);
309    let row_id = SharedString::from(format!("profile-row-{}", profile.id));
310    let edit_id = SharedString::from(format!("profile-row-edit-{}", profile.id));
311    let profile_id_for_edit = profile.id.clone();
312    let profile_name = profile.name.clone();
313    h_flex()
314        .id(row_id)
315        .w_full()
316        .px_2()
317        .py_2()
318        .gap_2()
319        .items_center()
320        .border_b_1()
321        .border_color(cx.theme().border)
322        .child(
323            div()
324                .flex_1()
325                .text_sm()
326                .text_color(cx.theme().foreground)
327                .child(profile_name),
328        )
329        .child(
330            div()
331                .w(gpui::px(90.0))
332                .text_sm()
333                .text_color(cx.theme().muted_foreground)
334                .child(version_label),
335        )
336        .child(
337            div()
338                .flex_1()
339                .text_sm()
340                .text_color(cx.theme().muted_foreground)
341                .child(display_url),
342        )
343        .child(div().w(gpui::px(110.0)).child(pill))
344        .child(
345            div().w(gpui::px(80.0)).child(
346                Button::new(edit_id)
347                    .child("Edit")
348                    .small()
349                    .cursor_pointer()
350                    .on_click(move |_, window, cx| {
351                        let id = profile_id_for_edit.clone();
352                        entity.update(cx, |this, cx| {
353                            this.open_edit_profile_sheet(Some(&id), window, cx);
354                        });
355                    }),
356            ),
357        )
358}

360/// Render a status pill for a profile.
361///
362/// ### Arguments
363/// - `profile`: The profile to render the pill for.
364/// - `master_on`: Whether the master switch is on (overrides the per-profile
365///   status to Inactive when off).
366/// - `cx`: The application context.
367///
368/// ### Returns
369/// - `impl IntoElement`: The pill element.
370fn render_status_pill(profile: &ServerProfile, master_on: bool, cx: &App) -> impl IntoElement {
371    let status = get_profile_status(profile, master_on, cx);
372    let (bg, fg) = pill_colors(status, cx);
373    Label::new(status.label())
374        .rounded_lg()
375        .border_1()
376        .border_color(bg)
377        .text_sm()
378        .text_color(fg)
379        .text_center()
380}

382/// Get the status  for a profile.
383///
384/// ### Arguments
385/// - `profile`: The profile.
386/// - `master_on`: The master switch value.
387/// - `cx`: The application context.
388///
389/// ### Returns
390/// - `SynchronizationStatus`: The profile's status.
391fn get_profile_status(profile: &ServerProfile, master_on: bool, cx: &App) -> SynchronizationStatus {
392    if !master_on || !profile.is_active {
393        return SynchronizationStatus::NotActivated;
394    }
395    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
396    let sync_states = shared.sync_states.read();
397    sync_states
398        .get(&profile.id)
399        .map_or(SynchronizationStatus::NotActivated, |state| {
400            *state.connection_status.lock()
401        })
402}

404/// Resolve the version label to display for a profile.
405///
406/// ### Arguments
407/// - `profile`: The profile to resolve the version for.
408/// - `master_on`: Whether the master switch is on.
409/// - `cx`: The application context.
410///
411/// ### Returns
412/// - `String`: The version label to render.
413fn get_profile_version_display(profile: &ServerProfile, master_on: bool, cx: &App) -> String {
414    if !matches!(
415        get_profile_status(profile, master_on, cx),
416        SynchronizationStatus::Connected
417    ) {
418        return "-".to_string();
419    }
420    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
421    let sync_states = shared.sync_states.read();
422    let Some(state) = sync_states.get(&profile.id) else {
423        return "-".to_string();
424    };
425    let version = state.server_version.lock();
426    match version.as_deref() {
427        Some(raw) => format!("v{}", raw.trim().trim_start_matches(['v', 'V'])),
428        None => "< v0.7.0".to_string(),
429    }
430}

432/// Resolve the foreground/background colors for a status pill.
433///
434/// ### Arguments
435/// - `status`: The profile's status enum value.
436///
437/// ### Returns
438/// - `(gpui::Hsla, gpui::Hsla)`: Background and foreground colors.
439fn pill_colors(status: SynchronizationStatus, cx: &App) -> (gpui::Hsla, gpui::Hsla) {
440    let theme = cx.theme();
441    match status {
442        SynchronizationStatus::Connected => (theme.success, theme.success),
443        SynchronizationStatus::Connecting => (theme.warning, theme.warning),
444        SynchronizationStatus::AuthenticationFailed
445        | SynchronizationStatus::ConnectionFailed
446        | SynchronizationStatus::NotActivated
447        | SynchronizationStatus::Other => (theme.danger, theme.danger),
448        SynchronizationStatus::Disconnected => (theme.info, theme.info),
449    }
450}

452/// Render the right-aligned "Add server" button.
453///
454/// ### Arguments
455/// - `entity`: The Fulgur entity.
456///
457/// ### Returns
458/// - `SettingItem`: The button row.
459fn render_add_server_button(entity: &Entity<Fulgur>) -> SettingItem {
460    let entity = entity.clone();
461    SettingItem::render(move |_options, _window, _cx| {
462        let entity = entity.clone();
463        h_flex()
464            .w_full()
465            .justify_end()
466            .mt_2()
467            .child(
468                Button::new("add-server-profile")
469                    .child("Add Fulgurant instance")
470                    .small()
471                    .primary()
472                    .cursor_pointer()
473                    .on_click(move |_, window, cx| {
474                        let already_at_cap = entity
475                            .read(cx)
476                            .settings
477                            .app_settings
478                            .synchronization_settings
479                            .profiles
480                            .len()
481                            >= MAX_PROFILES;
482                        if already_at_cap {
483                            window.push_notification(
484                                (
485                                    gpui_component::notification::NotificationType::Error,
486                                    SharedString::from(format!(
487                                        "Maximum of {MAX_PROFILES} Fulgurant instances reached."
488                                    )),
489                                ),
490                                cx,
491                            );
492                            return;
493                        }
494                        entity.update(cx, |this, cx| {
495                            this.open_edit_profile_sheet(None, window, cx);
496                        });
497                    }),
498            )
499            .into_any_element()
500    })
