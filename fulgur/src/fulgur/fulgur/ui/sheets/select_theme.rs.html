22/// Make a select theme item.
23///
24/// ### Parameters:
25/// - `entity`: The Fulgur entity handle.
26/// - `theme_name`: The theme name represented by this item.
27/// - `is_current_theme`: Whether this item matches the currently selected theme.
28/// - `current_theme_shared`: Shared current-theme value for sheet highlighting.
29/// - `cx`: The application context.
30///
31/// ### Returns:
32/// `Stateful<Div>`: Represents one clickable theme row.
33fn make_select_theme_item(
34    entity: Entity<Fulgur>,
35    theme_name: String,
36    is_current_theme: bool,
37    current_theme_shared: Arc<Mutex<String>>,
38    cx: &mut App,
39) -> Stateful<Div> {
40    let id = SharedString::from(format!("Select_{theme_name}"));
41    h_flex()
42        .id(id)
43        .cursor_pointer()
44        .justify_between()
45        .my_2()
46        .border_1()
47        .border_color(cx.theme().border)
48        .child(div().p_2().text_sm().child(theme_name.clone()))
49        .when(is_current_theme, |this| this.bg(cx.theme().muted))
50        .on_click(move |_this, _window, cx| {
51            let selected_theme_name = theme_name.clone();
52            let should_refresh = entity.update(cx, |fulgur, cx| {
53                fulgur.switch_active_theme_from_sheet(
54                    cx,
55                    selected_theme_name.as_str(),
56                    &current_theme_shared,
57                )
58            });
59            if should_refresh {
60                cx.refresh_windows();
61            }
62        })
63        .hover(|this| this.bg(cx.theme().muted))
64}

66/// Check whether a theme is the currently active theme.
67///
68/// ### Parameters:
69/// - `theme_name`: The theme name to evaluate.
70/// - `current_theme`: The currently active theme name.
71///
72/// ### Returns:
73/// `true` when both theme names match, otherwise `false`.
74fn is_current_theme(theme_name: &str, current_theme: &str) -> bool {
75    theme_name == current_theme
76}

78/// Make the select-theme list.
79///
80/// ### Parameters:
81/// - `entity`: The Fulgur entity handle.
82/// - `themes`: The list of available theme names.
83/// - `current_theme`: The currently active theme.
84/// - `current_theme_shared`: Shared current-theme value for sheet highlighting.
85/// - `cx`: The application context.
86///
87/// ### Returns:
88/// `Div`: Represents the rendered theme list.
89fn make_select_theme_list(
90    entity: &Entity<Fulgur>,
91    themes: &[String],
92    current_theme: String,
93    current_theme_shared: Arc<Mutex<String>>,
94    cx: &mut App,
95) -> Div {
96    let entity = entity.clone();
97    div().rounded_md().children(
98        themes
99            .iter()
100            .map(move |theme| {
101                make_select_theme_item(
102                    entity.clone(),
103                    theme.clone(),
104                    is_current_theme(theme, &current_theme),
105                    current_theme_shared.clone(),
106                    cx,
107                )
108            })
109            .collect::<Vec<Stateful<Div>>>(),
110    )
111}

114    /// Apply a theme selected from the select-theme sheet.
115    ///
116    /// ### Parameters:
117    /// - `cx`: The Fulgur context.
118    /// - `theme_name`: The selected theme name.
119    /// - `current_theme_shared`: Shared current-theme value used by the sheet.
120    ///
121    /// ### Returns:
122    /// - `true`: The theme exists and was applied.
123    /// - `false`: The theme was not found in the registry.
124    fn switch_active_theme_from_sheet(
125        &mut self,
126        cx: &mut Context<Self>,
127        theme_name: &str,
128        current_theme_shared: &Arc<Mutex<String>>,
129    ) -> bool {
130        if let Some(theme_config) = ThemeRegistry::global(cx).themes().get(theme_name).cloned() {
131            Theme::global_mut(cx).apply_config(&theme_config);
132            self.settings.app_settings.theme = theme_name.to_string().into();
133            if let Err(error) = self.update_and_propagate_settings(cx) {
134                log::error!(
135                    "Failed to propagate settings after theme '{theme_name}' selection: {error}"
136                );
137            }
138            *current_theme_shared.lock() = theme_name.to_string();
139            cx.notify();
140            true
141        } else {
142            false
143        }
144    }

146    /// Add a theme to the themes directory. Prompt the user for the path to the theme file.
147    ///
148    /// ### Arguments
149    /// - `window`: The window context
150    /// - `cx`: The application context
151    pub fn add_theme(&self, window: &mut Window, cx: &mut Context<Self>) {
152        let path_future = cx.prompt_for_paths(PathPromptOptions {
153            files: true,
154            directories: false,
155            multiple: false,
156            prompt: Some("Select theme".into()),
157        });
158        let settings = self.settings.clone();
159        let entity = cx.entity();
160        cx.spawn_in(window, async move |_view, window| {
161            let paths = path_future.await.ok()?.ok()??;
162            let theme_path = paths.first()?.clone();
163            if theme_path.extension().and_then(|s| s.to_str()) != Some("json") {
164                window
165                    .update(|window, cx| {
166                        window.push_notification(
167                            "Invalid file type. Please select a JSON theme file.".to_string(),
168                            cx,
169                        );
170                    })
171                    .ok()?;
172                return None;
173            }
174            let themes_dir = match themes_directory_path() {
175                Ok(path) => path,
176                Err(e) => {
177                    log::error!("Failed to get themes directory: {e}");
178                    window
179                        .update(|window, cx| {
180                            let notification = SharedString::from(format!(
181                                "Failed to access themes directory: {e}"
182                            ));
183                            window.push_notification((NotificationType::Error, notification), cx);
184                        })
185                        .ok()?;
186                    return None;
187                }
188            };
189            let filename = match theme_path.file_name() {
190                Some(name) => name,
191                None => {
192                    window
193                        .update(|window, cx| {
194                            let notification = SharedString::from("Invalid theme file path.");
195                            window.push_notification((NotificationType::Error, notification), cx);
196                        })
197                        .ok()?;
198                    return None;
199                }
200            };
201            let dest_path = themes_dir.join(filename);
202            match fs::copy(&theme_path, &dest_path) {
203                Ok(_) => {
204                    log::info!("Theme file copied to: {}", dest_path.display());
205                    window
206                        .update(|window, cx| {
207                            let notification = SharedString::from(format!(
208                                "Theme '{}' added successfully!",
209                                filename.to_string_lossy()
210                            ));
211                            window.push_notification((NotificationType::Success, notification), cx);
212                            reload_themes_and_update(&settings, &entity, cx);
213                        })
214                        .ok()?;
215                }
216                Err(e) => {
217                    log::error!("Failed to copy theme file: {e}");
218                    window
219                        .update(|window, cx| {
220                            let notification =
221                                SharedString::from(format!("Failed to add theme: {e}"));
222                            window.push_notification((NotificationType::Error, notification), cx);
223                        })
224                        .ok()?;
225                }
226            }
227            Some(())
228        })
229        .detach();
230    }

232    /// Open theme selector as a sheet (sliding panel from right side)
233    ///
234    /// This is an alternative to the dialog-based theme selector
235    ///
236    /// ### Arguments
237    /// - `window`: The window context
238    /// - `cx`: The application context
239    pub fn select_theme_sheet(&self, window: &mut Window, cx: &mut Context<Self>) {
240        let entity = cx.entity();
241        let current_theme = self.settings.app_settings.theme.to_string();
242        let current_theme_shared = Arc::new(Mutex::new(current_theme.clone()));
243        let viewport_height = window.viewport_size().height;
244        window.open_sheet(cx, move |sheet, _window, cx| {
245            let themes = ThemeRegistry::global(cx).sorted_themes();
246            let light_themes: Vec<String> = themes
247                .iter()
248                .filter(|theme| theme.mode == ThemeMode::Light)
249                .map(|theme| theme.name.to_string())
250                .collect();
251            let dark_themes: Vec<String> = themes
252                .iter()
253                .filter(|theme| theme.mode == ThemeMode::Dark)
254                .map(|theme| theme.name.to_string())
255                .collect();
256
257            let entity_dark = entity.clone();
258            let entity_light = entity.clone();
259            let current_theme_shared_dark = current_theme_shared.clone();
260            let current_theme_shared_light = current_theme_shared.clone();
261            let current_theme_display = current_theme_shared.lock().clone();
262            let max_height = px((viewport_height - px(150.0)).into()); //TODO: Make this dynamic based on the content
263            sheet
264                .title("Select Theme")
265                .size(px(400.))
266                .overlay(false)
267                .child(
268                    v_flex()
269                        .overflow_y_scrollbar()
270                        .gap_2()
271                        .h(max_height)
272                        .child(div().text_lg().child("Dark themes"))
273                        .child(make_select_theme_list(
274                            &entity_dark,
275                            &dark_themes,
276                            current_theme_display.clone(),
277                            current_theme_shared_dark,
278                            cx,
279                        ))
280                        .child(div().text_lg().mt_4().child("Light themes"))
281                        .child(make_select_theme_list(
282                            &entity_light,
283                            &light_themes,
284                            current_theme_display.clone(),
285                            current_theme_shared_light,
286                            cx,
287                        )),
288                )
289                .footer({
290                    let entity_footer = entity.clone();
291                    h_flex()
292                        .justify_between()
293                        .w_full()
294                        .child(
295                            Button::new("add-theme-footer")
296                                .child("Add new theme...")
297                                .small()
298                                .cursor_pointer()
299                                .on_click(move |_, window, cx| {
300                                    entity_footer.update(cx, |this, cx| {
301                                        this.add_theme(window, cx);
302                                    });
303                                    // Close sheet after opening add theme dialog
304                                    // User can reopen to see new theme
305                                    //window.close_sheet(cx);
306                                }),
307                        )
308                        .child(
309                            Button::new("ok-footer")
310                                .child("OK")
311                                .small()
312                                .primary()
313                                .cursor_pointer()
314                                .on_click(|_, window, cx| {
315                                    window.close_sheet(cx);
316                                }),
317                        )
318                })
319        });
320    }

379    #[cfg(feature = "gpui-test-support")]
380    #[gpui::test]
381    fn test_switch_active_theme_from_sheet_updates_current_theme(cx: &mut TestAppContext) {
382        let (fulgur, mut visual_cx) = setup_fulgur(cx);
383        let current_theme = fulgur.read_with(&visual_cx, |this, _| {
384            this.settings.app_settings.theme.to_string()
385        });
386
387        let current_theme_for_lookup = current_theme.clone();
388        let target_theme = visual_cx.update(|_window, cx| {
389            let themes = ThemeRegistry::global(cx).sorted_themes();
390            themes
391                .iter()
392                .map(|theme| theme.name.to_string())
393                .find(|name| name != &current_theme_for_lookup)
394                .or_else(|| themes.first().map(|theme| theme.name.to_string()))
395                .expect("expected at least one registered theme")
396        });
397
398        let current_theme_shared = Arc::new(Mutex::new(current_theme.clone()));
399        let target_theme_for_apply = target_theme.clone();
400        let current_theme_shared_for_apply = current_theme_shared.clone();
401        let applied = visual_cx.update(|_window, cx| {
402            fulgur.update(cx, |this, cx| {
403                this.switch_active_theme_from_sheet(
404                    cx,
405                    target_theme_for_apply.as_str(),
406                    &current_theme_shared_for_apply,
407                )
408            })
409        });
410
411        assert!(applied);
412
413        let switched_theme = fulgur.read_with(&visual_cx, |this, _| {
414            this.settings.app_settings.theme.to_string()
415        });
416        assert_eq!(switched_theme, target_theme);
417        assert_eq!(*current_theme_shared.lock(), target_theme);
418    }
