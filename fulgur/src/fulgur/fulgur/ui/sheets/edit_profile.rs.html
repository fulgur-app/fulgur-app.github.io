42/// Build a profile draft from the current form values.
43///
44/// ### Arguments
45/// - `state`: The shared form state.
46/// - `existing_public_key`: Existing public key to carry over for Edit mode;
47///   `None` for Add mode (key generation happens later if needed).
48/// - `cx`: The application context (used to read input values).
49///
50/// ### Returns
51/// - `ServerProfile`: A profile struct populated from the form.
52fn build_profile_from_form(
53    state: &ProfileFormState,
54    existing_public_key: Option<String>,
55    cx: &App,
56) -> ServerProfile {
57    let name = state.name_input.read(cx).value().trim().to_string();
58    let server_url = optional_text(&state.server_url_input.read(cx).value());
59    let email = optional_text(&state.email_input.read(cx).value());
60    ServerProfile {
61        id: state.profile_id.clone(),
62        name,
63        is_active: *state.is_active.lock(),
64        server_url,
65        email,
66        public_key: existing_public_key,
67        is_deduplication: *state.is_deduplication.lock(),
68    }
69}

71/// Convert an input value into the optional string used by `ServerProfile`.
72///
73/// ### Arguments
74/// - `value`: The trimmed/untrimmed input string.
75///
76/// ### Returns
77/// - `Some(String)`: The trimmed value when non-empty.
78/// - `None`: When the trimmed value is empty.
79fn optional_text(value: &str) -> Option<String> {
80    let trimmed = value.trim();
81    if trimmed.is_empty() {
82        None
83    } else {
84        Some(trimmed.to_string())
85    }
86}

88/// Validate a server URL string from the form.
89///
90/// ### Arguments
91/// - `value`: The form value (already trimmed by the caller).
92///
93/// ### Returns
94/// - `Ok(())`: The URL is empty (allowed) or parses successfully.
95/// - `Err(SharedString)`: A user-facing error message describing the failure.
96fn validate_url(value: &str) -> Result<(), SharedString> {
97    if value.is_empty() {
98        return Ok(());
99    }
100    url::Url::parse(value)
101        .map(|_| ())
102        .map_err(|_| SharedString::from("Server URL is not a valid URL."))
103}

105/// Validate an email string from the form using the same heuristic the
106/// settings validator uses (presence of `@` and a `.` after it).
107///
108/// ### Arguments
109/// - `value`: The form value (already trimmed by the caller).
110///
111/// ### Returns
112/// - `Ok(())`: The email is empty (allowed) or passes the heuristic.
113/// - `Err(SharedString)`: A user-facing error message describing the failure.
114fn validate_email(value: &str) -> Result<(), SharedString> {
115    if value.is_empty() {
116        return Ok(());
117    }
118    let at_pos = value.find('@');
119    let is_valid = at_pos
120        .is_some_and(|pos| pos > 0 && pos < value.len() - 1 && value[pos + 1..].contains('.'));
121    if is_valid {
122        Ok(())
123    } else {
124        Err(SharedString::from("Email address is not valid."))
125    }
126}

128/// Read the current device key value from the form.
129///
130/// ### Arguments
131/// - `state`: The shared form state.
132/// - `cx`: The application context.
133///
134/// ### Returns
135/// - `DeviceKeyEdit::Untouched`: The user has not modified the placeholder.
136/// - `DeviceKeyEdit::Clear`: The user erased the field.
137/// - `DeviceKeyEdit::Set(value)`: The user typed a new key.
138enum DeviceKeyEdit {
139    Untouched,
140    Clear,
141    Set(String),
142}
143
144fn read_device_key_edit(state: &ProfileFormState, cx: &App) -> DeviceKeyEdit {
145    let raw = state.device_key_input.read(cx).value().to_string();
146    if raw == DEVICE_KEY_PLACEHOLDER {
147        DeviceKeyEdit::Untouched
148    } else if raw.is_empty() {
149        DeviceKeyEdit::Clear
150    } else {
151        DeviceKeyEdit::Set(raw)
152    }
153}

155/// Apply a `DeviceKeyEdit` to the keychain for a profile.
156///
157/// ### Arguments
158/// - `profile_id`: The id whose keychain entries should be updated.
159/// - `edit`: The user's edit, as decoded by `read_device_key_edit`.
160///
161/// ### Returns
162/// - `Ok(true)`: A keychain write happened (set or clear).
163/// - `Ok(false)`: The user did not modify the field; nothing was written.
164/// - `Err(anyhow::Error)`: The keychain operation failed.
165fn apply_device_key_edit(profile_id: &str, edit: &DeviceKeyEdit) -> anyhow::Result<bool> {
166    match edit {
167        DeviceKeyEdit::Untouched => Ok(false),
168        DeviceKeyEdit::Clear => {
169            save_device_api_key_to_keychain(profile_id, None)?;
170            Ok(true)
171        }
172        DeviceKeyEdit::Set(value) => {
173            save_device_api_key_to_keychain(profile_id, Some(value))?;
174            Ok(true)
175        }
176    }
177}

424                .child(
425                    Switch::new("edit-profile-activate")

481/// Handle the Test connection button.
482///
483/// ### Description
484/// Saves the device key from the form to the keychain (if changed), then
485/// pings the Fulgurant server with a fresh JWT token. The result is shown
486/// as a success or error notification. No profile changes are persisted.
487///
488/// ### Arguments
489/// - `entity`: The Fulgur entity (used to look up the existing public key).
490/// - `state`: The shared form state.
491/// - `window`: The window to attach the progress notification to.
492/// - `cx`: The application context.
493fn handle_test_connection(
494    entity: &Entity<Fulgur>,
495    state: &Arc<ProfileFormState>,
496    window: &mut Window,
497    cx: &mut App,
498) {
499    let url_raw = state.server_url_input.read(cx).value().trim().to_string();
500    let Some(server_url) = optional_text(&url_raw) else {
501        window.push_notification(
502            (
503                NotificationType::Error,
504                SharedString::from("Enter a server URL to test the connection."),
505            ),
506            cx,
507        );
508        return;
509    };
510    if let Err(msg) = validate_url(&server_url) {
511        window.push_notification((NotificationType::Error, msg), cx);
512        return;
513    }
514    let device_key_edit = read_device_key_edit(state, cx);
515    if let Err(e) = apply_device_key_edit(&state.profile_id, &device_key_edit) {
516        log::error!("Failed to save device API key for ping: {e}");
517        window.push_notification(
518            (
519                NotificationType::Error,
520                SharedString::from(format!("Failed to save device key: {e}")),
521            ),
522            cx,
523        );
524        return;
525    }
526    if state.is_new && matches!(device_key_edit, DeviceKeyEdit::Set(_)) {
527        state
528            .device_key_written_for_add
529            .store(true, Ordering::Relaxed);
530    }
531    let existing_public_key = entity
532        .read(cx)
533        .settings
534        .app_settings
535        .synchronization_settings
536        .find_profile(&state.profile_id)
537        .and_then(|p| p.public_key.clone());
538    let profile = build_profile_from_form(state, existing_public_key, cx);
539    cx.global::<crate::fulgur::shared_state::SharedAppState>()
540        .sync_state_for(&profile.id)
541        .token_state
542        .clear_token();
543    let display_name = if profile.name.is_empty() {
544        server_url
545    } else {
546        profile.name.clone()
547    };
548    perform_ping_with_progress(profile, display_name, window, cx);
549}

551/// Handle the Save button.
552///
553/// ### Arguments
554/// - `entity`: The Fulgur entity.
555/// - `state`: The shared form state.
556/// - `window`: The window to attach notifications to.
557/// - `cx`: The application context.
558fn handle_save(
559    entity: &Entity<Fulgur>,
560    state: &Arc<ProfileFormState>,
561    window: &mut Window,
562    cx: &mut App,
563) {
564    if let Err(message) = validate_form(state, entity, cx) {
565        window.push_notification((NotificationType::Error, message), cx);
566        return;
567    }
568    let device_key_edit = read_device_key_edit(state, cx);
569    if let Err(e) = apply_device_key_edit(&state.profile_id, &device_key_edit) {
570        log::error!("Failed to save device API key: {e}");
571        window.push_notification(
572            (
573                NotificationType::Error,
574                SharedString::from(format!("Failed to save device key: {e}")),
575            ),
576            cx,
577        );
578        return;
579    }
580    let existing_public_key = entity
581        .read(cx)
582        .settings
583        .app_settings
584        .synchronization_settings
585        .find_profile(&state.profile_id)
586        .and_then(|p| p.public_key.clone());
587    let profile = build_profile_from_form(state, existing_public_key, cx);
588    let profile_id = profile.id.clone();
589    let is_new = state.is_new;
590    let result = entity.update(cx, |this, cx| {
591        if is_new {
592            this.add_profile(profile, cx).map(|()| true)
593        } else {
594            this.update_profile(
595                &profile_id,
596                |existing| {
597                    existing.name.clone_from(&profile.name);
598                    existing.is_active = profile.is_active;
599                    existing.server_url.clone_from(&profile.server_url);
600                    existing.email.clone_from(&profile.email);
601                    existing.is_deduplication = profile.is_deduplication;
602                },
603                cx,
604            )
605        }
606    });
607    match result {
608        Ok(true) => {
609            // Clear the cached JWT so the next call uses the latest key/email.
610            cx.global::<crate::fulgur::shared_state::SharedAppState>()
611                .sync_state_for(&profile_id)
612                .token_state
613                .clear_token();
614            entity.update(cx, |this, cx| {
615                this.restart_sse_connection_for_with_progress(&profile_id, window, cx);
616            });
617            window.close_sheet(cx);
618        }
619        Ok(false) => {
620            log::warn!(
621                "Profile '{profile_id}' could not be updated (no longer in settings); closing sheet"
622            );
623            window.close_sheet(cx);
624        }
625        Err(e) => {
626            log::error!("Failed to save profile: {e}");
627            window.push_notification(
628                (
629                    NotificationType::Error,
630                    SharedString::from(format!("Failed to save profile: {e}")),
631                ),
632                cx,
633            );
634        }
635    }
636}

663/// Validate every field in the form, returning the first failure as a
664/// user-facing message.
665///
666/// ### Arguments
667/// - `state`: The shared form state.
668/// - `entity`: The Fulgur entity (used to check name uniqueness against the
669///   current profile list).
670/// - `cx`: The application context.
671///
672/// ### Returns
673/// - `Ok(())`: All fields pass validation.
674/// - `Err(SharedString)`: A user-facing error message.
675fn validate_form(
676    state: &ProfileFormState,
677    entity: &Entity<Fulgur>,
678    cx: &App,
679) -> Result<(), SharedString> {
680    let name = state.name_input.read(cx).value().trim().to_string();
681    if name.is_empty() {
682        return Err(SharedString::from("Server name cannot be empty."));
683    }
684    let collides = entity
685        .read(cx)
686        .settings
687        .app_settings
688        .synchronization_settings
689        .name_collides(&name, Some(&state.profile_id));
690    if collides {
691        return Err(SharedString::from("Another server already uses this name."));
692    }
693    let url_value = state.server_url_input.read(cx).value().trim().to_string();
694    validate_url(&url_value)?;
695    let email_value = state.email_input.read(cx).value().trim().to_string();
696    validate_email(&email_value)?;
697    Ok(())
698}

706///
707/// ### Arguments
708/// - `entity`: The Fulgur entity.
709/// - `state`: The shared form state (read for profile id and name).
710/// - `window`: The window to attach the dialog to.
711/// - `cx`: The application context.
712fn confirm_delete_profile(
713    entity: &Entity<Fulgur>,
714    state: &Arc<ProfileFormState>,
715    window: &mut Window,
716    cx: &mut App,
717) {
718    let profile_name = state.name_input.read(cx).value().trim().to_string();
719    let display_name = if profile_name.is_empty() {
720        "this server".to_string()
721    } else {
722        format!("'{profile_name}'")
723    };
724    let entity_for_confirm = entity.clone();
725    let state_for_confirm = Arc::clone(state);
726    window.open_alert_dialog(cx, move |modal, _, _| {
727        let entity_ok = entity_for_confirm.clone();
728        let state_ok = Arc::clone(&state_for_confirm);
729        modal
730            .title(div().text_size(px(16.)).child("Delete Server"))
731            .keyboard(true)
732            .button_props(
733                DialogButtonProps::default()
734                    .show_cancel(true)
735                    .cancel_text("Cancel")
736                    .cancel_variant(ButtonVariant::Secondary)
737                    .ok_text("Delete")
738                    .ok_variant(ButtonVariant::Danger),
739            )
740            .overlay_closable(false)
741            .close_button(false)
742            .child(
743                v_flex()
744                    .gap_2()
745                    .child(format!("Delete {display_name}?"))
746                    .child("This cannot be undone."),
747            )
748            .on_ok(move |_, window, cx| {
749                let profile_id = state_ok.profile_id.clone();
750                let outcome = entity_ok.update(cx, |this, cx| this.delete_profile(&profile_id, cx));
751                match outcome {
752                    Ok(_) => {
753                        window.close_sheet(cx);
754                    }
755                    Err(e) => {
756                        log::error!("Failed to delete profile '{profile_id}': {e}");
757                        window.push_notification(
758                            (
759                                NotificationType::Error,
760                                SharedString::from(format!("Failed to delete server: {e}")),
761                            ),
762                            cx,
763                        );
764                    }
765                }
766                true
767            })
768            .on_cancel(|_, _, _| true)
769    });
770}
