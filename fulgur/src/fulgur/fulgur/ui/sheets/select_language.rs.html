15/// Create a select language item
16///
17/// ### Parameters:
18/// - `entity`: The Fulgur entity handle.
19/// - `language`: The language to create the item for.
20/// - `is_current_language`: Whether the language is the current language.
21/// - `cx`: The application context.
22///
23/// ### Returns:
24/// `Div`: Represents a select language item.
25fn make_select_language_item(
26    entity: Entity<Fulgur>,
27    language: SupportedLanguage,
28    is_current_language: bool,
29    cx: &App,
30) -> impl Element {
31    let pretty_name = pretty_name(&language);
32    let id = format!("Select_{}", pretty_name.clone().replace(' ', "_"));
33    let id = SharedString::from(id);
34    h_flex()
35        .id(id)
36        .justify_between()
37        .my_2()
38        .cursor_pointer()
39        .border_1()
40        .border_color(cx.theme().border)
41        .child(div().p_2().text_sm().child(pretty_name))
42        .when(is_current_language, |this| this.bg(cx.theme().muted))
43        .hover(|this| this.bg(cx.theme().muted))
44        .on_click(move |_event, window, cx| {
45            entity.update(cx, |this, cx| {
46                this.switch_active_tab_language(window, cx, language);
47                window.close_sheet(cx);
48            });
49        })
50}

52/// Check if the given language is the current language.
53///
54/// ### Parameters:
55/// - `language`: The language to check.
56/// - `current_language`: The current language.
57///
58/// ### Returns:
59/// `true` if the given language is the current language, otherwise `false`.
60fn is_current_language(
61    language: SupportedLanguage,
62    current_language: Option<SupportedLanguage>,
63) -> bool {
64    current_language == Some(language)
65}

67/// Create a select language list.
68///
69/// ### Parameters:
70/// - `entity`: The Fulgur entity handle.
71/// - `current_language`: The current language.
72/// - `cx`: The application context.
73///
74/// ### Returns:
75/// `Div`: Represents a select language list.
76fn make_select_language_list(
77    entity: Entity<Fulgur>,
78    current_language: Option<SupportedLanguage>,
79    cx: &App,
80) -> Div {
81    div().gap_2().children(
82        SupportedLanguage::all()
83            .into_iter()
84            .map(move |language| {
85                make_select_language_item(
86                    entity.clone(),
87                    language,
88                    is_current_language(language, current_language),
89                    cx,
90                )
91            })
92            .collect::<Vec<_>>(),
93    )
94}

96impl Fulgur {
97    /// Get the language that should be highlighted as current in the select-language sheet.
98    ///
99    /// ### Parameters:
100    /// - `cx`: The application context.
101    ///
102    /// ### Returns:
103    /// - `Some(SupportedLanguage)`: Active editor tab language, or `Plain` for non-editor tabs.
104    /// - `None`: If there is no active tab.
105    fn current_sheet_language(&self, cx: &gpui::App) -> Option<SupportedLanguage> {
106        self.active_tab(cx).map(|tab| {
107            tab.as_editor()
108                .map_or(SupportedLanguage::Plain, |editor_tab| editor_tab.language)
109        })
110    }
111
112    /// Force the active editor tab language from the select-language sheet.
113    ///
114    /// ### Parameters:
115    /// - `window`: The window context.
116    /// - `cx`: The application context.
117    /// - `language`: The language to apply to the active editor tab.
118    fn switch_active_tab_language(
119        &mut self,
120        window: &mut Window,
121        cx: &mut Context<Self>,
122        language: SupportedLanguage,
123    ) {
124        if let Some(tab_entity) = self.active_tab_entity(cx) {
125            let settings = self.settings.editor_settings.clone();
126            tab_entity.update(cx, |tab, cx| {
127                tab.force_language(window, cx, language, &settings);
128            });
129        }
130    }
131
132    /// Render the select language sheet.
133    ///
134    /// ### Parameters:
135    /// - `window`: The window to render the sheet in.
136    /// - `cx`: The context to render the sheet in.
137    pub fn render_select_language_sheet(&self, window: &mut Window, cx: &mut Context<Self>) {
138        let entity = cx.entity();
139        let current_language = self.current_sheet_language(cx);
140        let viewport_height = window.viewport_size().height;
141        let max_height = px((viewport_height - px(100.0)).into()); //TODO: Make this dynamic based on the content
142        window.open_sheet_at(Placement::Left, cx, move |sheet, _window, cx| {
143            sheet
144                .title("Select Language")
145                .size(px(400.))
146                .overlay(true)
147                .child(v_flex().overflow_y_scrollbar().gap_2().h(max_height).child(
148                    make_select_language_list(entity.clone(), current_language, cx),
149                ))
150        });
151    }
152}
