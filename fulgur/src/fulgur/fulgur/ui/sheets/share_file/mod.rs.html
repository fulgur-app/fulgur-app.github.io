20    /// Return the list of profile ids whose devices should populate the share sheet.
21    ///
22    /// ### Returns
23    /// - `Vec<ProfileId>`: Active only profile ids when the master sync switch is on, in declaration order.
24    fn collect_active_profiles(&self) -> Vec<ServerProfile> {
25        if !self
26            .settings
27            .app_settings
28            .synchronization_settings
29            .is_synchronization_activated
30        {
31            return Vec::new();
32        }
33        self.settings
34            .app_settings
35            .synchronization_settings
36            .profiles
37            .iter()
38            .filter(|p| p.is_active)
39            .cloned()
40            .collect()
41    }

43    /// Open the share file sheet immediately and fetch each active profile's devices in the background.
44    ///
45    /// ### Arguments
46    /// - `window`: The window context.
47    /// - `cx`: The application context.
48    pub fn open_share_file_sheet(&mut self, window: &mut Window, cx: &mut Context<Self>) {
49        let active_profiles = self.collect_active_profiles();
50        if active_profiles.is_empty() {
51            log::warn!("Share aborted: no active profile or master sync is off");
52            window.push_notification(
53                (
54                    NotificationType::Warning,
55                    SharedString::from(
56                        "No active sync profile. Enable a profile in Settings, Synchronization.",
57                    ),
58                ),
59                cx,
60            );
61            return;
62        }
63        if let Some(previous) = self.share_sheet_state.take() {
64            previous
65                .active
66                .store(false, std::sync::atomic::Ordering::Release);
67        }
68        let state = Arc::new(ShareSheetState::new(active_profiles.clone()));
69        self.share_sheet_state = Some(Arc::clone(&state));
70        let shared = Fulgur::shared_state(cx);
71        let http_agent = Arc::clone(&shared.http_agent);
72        for profile in active_profiles {
73            let sync_state = shared.sync_state_for(&profile.id);
74            spawn_profile_device_fetch(
75                profile,
76                Arc::clone(&state),
77                sync_state,
78                Arc::clone(&http_agent),
79            );
80        }
81        Self::start_share_sheet_render_pump(Arc::clone(&state), window, cx);
82        Self::show_share_sheet(state, window, cx);
83    }

85    /// Drive periodic re-renders while the share sheet is open and at least one profile is still loading.
86    ///
87    /// ### Arguments
88    /// - `state`: Shared sheet state.
89    /// - `window`: The window context.
90    /// - `cx`: The application context.
91    fn start_share_sheet_render_pump(
92        state: Arc<ShareSheetState>,
93        window: &mut Window,
94        cx: &mut Context<Self>,
95    ) {
96        let entity = cx.entity();
97        window
98            .spawn(cx, async move |async_cx| {
99                loop {
100                    async_cx
101                        .background_executor()
102                        .timer(Duration::from_millis(100))
103                        .await;
104                    if !state.active.load(std::sync::atomic::Ordering::Acquire) {
105                        return;
106                    }
107                    let _ = async_cx.update(|_, cx| {
108                        entity.update(cx, |_, cx| cx.notify());
109                    });
110                    if state.all_settled() {
111                        return;
112                    }
113                }
114            })
115            .detach();
116    }

118    /// Drain pending SSE restart requests posted by share-sheet device fetches.
119    ///
120    /// ### Arguments
121    /// - `_window`: The window context.
122    /// - `cx`: The application context.
123    pub fn process_pending_share_sheet(&mut self, _window: &mut Window, cx: &mut Context<Self>) {
124        let Some(state) = self.share_sheet_state.as_ref().map(Arc::clone) else {
125            return;
126        };
127        let to_restart: Vec<ProfileId> = std::mem::take(&mut *state.pending_sse_restarts.lock());
128        for profile_id in to_restart {
129            self.restart_sse_connection_for(&profile_id, cx);
130        }
131    }

133    /// Open the share sheet UI for the given shared state. Each profile section reads live state on every render.
134    ///
135    /// ### Arguments
136    /// - `state`: Shared sheet state.
137    /// - `window`: The window context.
138    /// - `cx`: The application context.
139    fn show_share_sheet(state: Arc<ShareSheetState>, window: &mut Window, cx: &mut Context<Self>) {
140        let entity = cx.entity();
141        let viewport_height = window.viewport_size().height;
142        window.open_sheet(cx, move |sheet, _window, cx| {
143            #[cfg(target_os = "linux")]
144            let sheet_overhead = px(200.0);
145            #[cfg(not(target_os = "linux"))]
146            let sheet_overhead = px(150.0);
147            let max_height = px((viewport_height - sheet_overhead).into());
148            let state_for_list = Arc::clone(&state);
149            let state_for_footer = Arc::clone(&state);
150            sheet
151                .title("Share with...")
152                .size(px(400.))
153                .overlay(true)
154                .child(
155                    v_flex()
156                        .overflow_y_scrollbar()
157                        .gap_2()
158                        .h(max_height)
159                        .child(make_device_list(&state_for_list, cx)),
160                )
161                .footer(render_footer(state_for_footer, entity.clone()))
162        });
163    }

180    /// Initialize globals and open a test window with a Root-mounted `Fulgur`.
181    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
182        cx.update(|cx| {
183            gpui_component::init(cx);
184            let mut settings = Settings::new();
185            settings.editor_settings.watch_files = false;
186            let pending_files: Arc<Mutex<Vec<PathBuf>>> = Arc::new(Mutex::new(Vec::new()));
187            cx.set_global(SharedAppState::new(settings, pending_files, None));
188            cx.set_global(WindowManager::new());
189        });
190        let fulgur_slot: RefCell<Option<Entity<Fulgur>>> = RefCell::new(None);
191        let window = cx
192            .update(|cx| {
193                cx.open_window(WindowOptions::default(), |window, cx| {
194                    let window_id = window.window_handle().window_id();
195                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
196                    *fulgur_slot.borrow_mut() = Some(fulgur.clone());
197                    cx.new(|cx| gpui_component::Root::new(fulgur, window, cx))
198                })
199            })
200            .expect("failed to open test window");
201        let visual_cx = VisualTestContext::from_window(window.into(), cx);
202        visual_cx.run_until_parked();
203        let fulgur = fulgur_slot
204            .into_inner()
205            .expect("failed to capture Fulgur entity");
206        (fulgur, visual_cx)
207    }

222        let profile_a = ServerProfile::new("A");
223        let profile_b = ServerProfile::new("B");

266    fn test_process_pending_share_sheet_drains_sse_restart_queue(cx: &mut TestAppContext) {
267        let (fulgur, mut visual_cx) = setup_fulgur(cx);
268        visual_cx.update(|window, cx| {
269            fulgur.update(cx, |this, cx| {
270                let profile = ServerProfile::new("queued");
271                let id = profile.id.clone();
272                this.settings
273                    .app_settings
274                    .synchronization_settings
275                    .profiles
276                    .push(profile.clone());
277                let state = Arc::new(ShareSheetState::new(vec![profile]));
278                state.pending_sse_restarts.lock().push(id.clone());
279                this.share_sheet_state = Some(Arc::clone(&state));
280                this.process_pending_share_sheet(window, cx);
281                assert!(
282                    state.pending_sse_restarts.lock().is_empty(),
283                    "queue must be drained on each render"
284                );
285            });
286        });
