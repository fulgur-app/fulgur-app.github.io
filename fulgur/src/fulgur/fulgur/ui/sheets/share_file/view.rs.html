24/// Create a single device row in the share sheet.
25///
26/// ### Arguments
27/// - `profile_id`: The id of the profile that owns this device.
28/// - `device`: The device to display.
29/// - `is_selected`: Whether the device is currently selected.
30/// - `selected_keys`: Shared mutable selection state, keyed by `(profile_id, device_id)`.
31/// - `idx`: Stable index used to disambiguate the GPUI element id.
32/// - `cx`: The application context.
33///
34/// ### Returns
35/// - `impl Element`: The device row element.
36fn make_device_item(
37    profile_id: &ProfileId,
38    device: &Device,
39    is_selected: bool,
40    selected_keys: Arc<Mutex<Vec<(ProfileId, String)>>>,
41    idx: usize,
42    cx: &App,
43) -> impl Element {
44    let profile_id = profile_id.clone();
45    let device_id = device.id.clone();
46    let device_name = device.name.clone();
47    let device_expires = device.expires_at.clone();
48    let device_for_icon = device;
49    let has_public_key = device.public_key.is_some();
50    h_flex()
51        .id(("share-device-sheet", idx))
52        .items_center()
53        .justify_between()
54        .w_full()
55        .p_2()
56        .my_2()
57        .rounded_sm()
58        .border_color(cx.theme().border)
59        .border_1()
60        .when(has_public_key, gpui::Styled::cursor_pointer)
61        .when(!has_public_key, |this| this.opacity(0.5))
62        .when(has_public_key, |this| {
63            this.hover(|hover| hover.bg(cx.theme().muted))
64        })
65        .when(is_selected && has_public_key, |this| {
66            this.bg(cx.theme().accent)
67                .text_color(cx.theme().accent_foreground)
68        })
69        .child(
70            v_flex()
71                .gap_1()
72                .child(
73                    h_flex()
74                        .items_center()
75                        .justify_start()
76                        .gap_2()
77                        .child(get_icon(device_for_icon))
78                        .child(div().child(device_name))
79                        .child(div().text_xs().child(format!("Expires: {device_expires}"))),
80                )
81                .when(!has_public_key, |this| {
82                    this.child(
83                        div()
84                            .text_xs()
85                            .text_color(cx.theme().muted_foreground)
86                            .child("No public key for this device"),
87                    )
88                }),
89        )
90        .when(is_selected && has_public_key, |this| {
91            this.child(Icon::new(CustomIcon::Zap))
92        })
93        .when(has_public_key, |this| {
94            this.on_click(move |_event, _window, _cx| {
95                let mut keys = selected_keys.lock();
96                let key = (profile_id.clone(), device_id.clone());
97                if let Some(pos) = keys.iter().position(|existing| existing == &key) {
98                    keys.remove(pos);
99                } else {
100                    keys.push(key);
101                }
102            })
103        })
104}

106/// Build a profile section header used to delimit per-profile device rows.
107///
108/// ### Arguments
109/// - `profile`: The profile to label.
110/// - `cx`: The application context (used to read theme tokens).
111///

115    let url_label = profile

118        .unwrap_or_else(|| "(no URL)".to_string());
119    v_flex()
120        .gap_0p5()
121        .pt_2()
122        .child(
123            div()
124                .text_sm()
125                .font_weight(FontWeight::SEMIBOLD)

127                .child(profile.name.clone()),
128        )
129        .child(
130            div()
131                .text_xs()
132                .text_color(cx.theme().muted_foreground)
133                .child(url_label),
134        )
135}

189/// Build the grouped device list reflecting the current per-profile state.
190///
191/// Each profile (header + its devices, loading placeholder, or error message)
192/// is wrapped in its own `v_flex` so the outer container can apply a wider
193/// gap between profile groups than between rows inside a group.
194///
195/// ### Arguments
196/// - `state`: Shared sheet state.
197/// - `cx`: The application context.
198///
199/// ### Returns
200/// - `Div`: The grouped list element.
201pub(super) fn make_device_list(state: &Arc<ShareSheetState>, cx: &App) -> Div {
202    let mut container = v_flex().gap_6();
203    let mut row_idx: usize = 0;
204    let map = state.per_profile.read();
205    let selected_guard = state.selected.lock();
206    let selected_keys: HashSet<(&str, &str)> = selected_guard
207        .iter()
208        .map(|(profile_id, device_id)| (profile_id.as_str(), device_id.as_str()))
209        .collect();
210    for profile in &state.profiles {
211        let mut group = div().child(make_profile_header(profile, cx));
212        match map.get(&profile.id) {
213            None | Some(ProfileFetchState::Loading) => {
214                group = group.child(make_loading_section(cx));
215            }
216            Some(ProfileFetchState::Failed(message)) => {
217                group = group.child(make_error_section(message, cx));
218            }
219            Some(ProfileFetchState::Loaded(devices)) => {
220                if devices.is_empty() {
221                    group = group.child(make_empty_section(cx));
222                } else {
223                    for device in devices.iter() {
224                        let is_selected =
225                            selected_keys.contains(&(profile.id.as_str(), device.id.as_str()));
226                        group = group.child(make_device_item(
227                            &profile.id,
228                            device,
229                            is_selected,
230                            state.selected.clone(),
231                            row_idx,
232                            cx,
233                        ));
234                        row_idx += 1;
235                    }
236                }
237            }
238        }
239        container = container.child(group);
240    }
241    container
242}

244/// Render the share sheet footer with Cancel and Share buttons.
245///
246/// ### Arguments
247/// - `state`: Shared sheet state.
248/// - `entity`: The Fulgur entity.
249///
250/// ### Returns
251/// - `Div`: The footer element.
252pub(super) fn render_footer(state: Arc<ShareSheetState>, entity: Entity<Fulgur>) -> Div {
253    let state_for_cancel = Arc::clone(&state);
254    let entity_for_cancel = entity.clone();
255    let state_for_share = state;
256    let entity_for_share = entity;
257    h_flex()
258        .justify_end()
259        .w_full()
260        .gap_2()
261        .child(
262            Button::new("cancel-share")
263                .child("Cancel")
264                .small()
265                .cursor_pointer()
266                .on_click(move |_, window, cx| {
267                    state_for_cancel
268                        .active
269                        .store(false, std::sync::atomic::Ordering::Release);
270                    entity_for_cancel.update(cx, |this, _| {
271                        this.share_sheet_state = None;
272                    });
273                    window.close_sheet(cx);
274                }),
275        )
276        .child(
277            Button::new("ok-share")
278                .child("Share")
279                .small()
280                .primary()
281                .cursor_pointer()
282                .on_click(move |_, window, cx| {
283                    handle_share_file(&state_for_share, &entity_for_share, window, cx);
284                }),
285        )
