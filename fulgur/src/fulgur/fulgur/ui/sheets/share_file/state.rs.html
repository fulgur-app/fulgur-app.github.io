11/// Per-profile fetch state shown in the share sheet.
12pub enum ProfileFetchState {
13    /// The device list is being fetched.
14    Loading,
15    /// Device list arrived from the server.
16    Loaded(Arc<Vec<Device>>),
17    /// The fetch failed; the message describes why.
18    Failed(String),
19}
20
21/// Shared state owning everything the share sheet needs while it is open.
22pub struct ShareSheetState {
23    /// Active profiles when the sheet was opened, in declaration order.
24    pub profiles: Vec<ServerProfile>,
25    /// Per-profile fetch progress keyed by `ProfileId`.
26    pub per_profile: Arc<RwLock<HashMap<ProfileId, ProfileFetchState>>>,
27    /// User selection across all profiles, keyed by `(profile_id, device_id)`.
28    pub selected: Arc<Mutex<Vec<(ProfileId, String)>>>,
29    /// Profile ids whose SSE worker should be restarted because we had to
30    /// reconnect during the device fetch. Drained by the render loop.
31    pub pending_sse_restarts: Arc<Mutex<Vec<ProfileId>>>,
32    /// Cleared to `false` while the sheet is open; flipped on Cancel/Share so
33    /// background tasks can stop polling.
34    pub active: Arc<AtomicBool>,
35}
36
37impl ShareSheetState {
38    /// Build a fresh shared state with every profile starting in the `Loading` state.
39    ///
40    /// ### Arguments
41    /// - `profiles`: Active profiles to render in the sheet.
42    ///
43    /// ### Returns
44    /// - `Self`: An initialized state ready for fetch threads to populate.
45    pub(super) fn new(profiles: Vec<ServerProfile>) -> Self {
46        let map: HashMap<ProfileId, ProfileFetchState> = profiles
47            .iter()
48            .map(|p| (p.id.clone(), ProfileFetchState::Loading))
49            .collect();
50        Self {
51            profiles,
52            per_profile: Arc::new(RwLock::new(map)),
53            selected: Arc::new(Mutex::new(Vec::new())),
54            pending_sse_restarts: Arc::new(Mutex::new(Vec::new())),
55            active: Arc::new(AtomicBool::new(true)),
56        }
57    }
58
59    /// Whether every profile has finished its fetch (Loaded or Failed).
60    ///
61    /// ### Returns
62    /// - `true`: No profile remains in the `Loading` state.
63    /// - `false`: At least one profile is still being fetched.
64    pub(super) fn all_settled(&self) -> bool {
65        self.per_profile
66            .read()
67            .values()
68            .all(|state| !matches!(state, ProfileFetchState::Loading))
69    }
70}
