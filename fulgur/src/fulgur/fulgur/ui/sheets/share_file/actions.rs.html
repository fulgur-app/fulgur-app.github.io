18/// Snapshot of the file content captured from the active editor tab when Share is pressed. Reused across each per-profile share request.
19struct ShareContext {
20    content: Arc<str>,
21    file_name: String,
22    file_path: Option<std::path::PathBuf>,
23}
24
25/// Resources required to share to a single profile inside the fan-out.
26struct ProfileShareResources {
27    profile: ServerProfile,
28    devices: Arc<Vec<Device>>,
29    device_ids: Vec<String>,
30    token_state: Arc<crate::fulgur::sync::access_token::TokenStateManager>,
31    max_file_size_bytes: u64,
32}
33
34/// Capture the file content snapshot from the active editor tab.
35///
36/// ### Arguments
37/// - `entity`: The Fulgur entity owning the active tab.
38/// - `cx`: The application context.
39///
40/// ### Returns
41/// - `ShareContext`: The captured file content, name, and optional file path.
42fn capture_share_context(entity: &Entity<Fulgur>, cx: &mut App) -> ShareContext {
43    entity.update(cx, |this, cx| {
44        let active_tab = this.get_active_editor_tab(cx);
45        let content: Arc<str> = active_tab.as_ref().map_or_else(
46            || Arc::from(""),
47            |tab| Arc::from(tab.content.read(cx).value().as_str()),
48        );
49        let file_path = active_tab.as_ref().and_then(|tab| tab.file_path().cloned());
50        let file_name = file_path
51            .as_ref()
52            .and_then(|path| path.file_name())
53            .and_then(|name| name.to_str())
54            .unwrap_or("Untitled")
55            .to_string();
56        ShareContext {
57            content,
58            file_name,
59            file_path,
60        }
61    })
62}

64/// Group selected `(profile_id, device_id)` pairs into per-profile resource bundles.
65///
66/// ### Arguments
67/// - `selected_keys`: The user's selection across all groups.
68/// - `state`: Shared sheet state holding per-profile loaded devices.
69/// - `entity`: The Fulgur entity (used to read shared state for tokens and limits).
70/// - `cx`: The application context.
71///
72/// ### Returns
73/// - `Vec<ProfileShareResources>`: One bundle per profile with at least one selected device.
74fn build_profile_share_resources(
75    selected_keys: &[(ProfileId, String)],
76    state: &Arc<ShareSheetState>,
77    entity: &Entity<Fulgur>,
78    cx: &mut App,
79) -> Vec<ProfileShareResources> {
80    let mut bundles: Vec<ProfileShareResources> = Vec::new();
81    let map = state.per_profile.read();
82    for profile in &state.profiles {
83        let device_ids: Vec<String> = selected_keys
84            .iter()
85            .filter(|(pid, _)| pid == &profile.id)
86            .map(|(_, did)| did.clone())
87            .collect();
88        if device_ids.is_empty() {
89            continue;
90        }
91        let devices = if let Some(ProfileFetchState::Loaded(devices)) = map.get(&profile.id) {
92            Arc::clone(devices)
93        } else {
94            log::warn!(
95                "Profile '{}': selection present but no loaded devices, skipping",
96                profile.name
97            );
98            continue;
99        };
100        let (token_state, max_file_size_bytes) = entity.update(cx, |_, cx| {
101            let sync_state = Fulgur::shared_state(cx).sync_state_for(&profile.id);
102            (
103                Arc::clone(&sync_state.token_state),
104                sync_state
105                    .max_file_size_bytes
106                    .load(std::sync::atomic::Ordering::Acquire),
107            )
108        });
109        bundles.push(ProfileShareResources {
110            profile: profile.clone(),
111            devices,
112            device_ids,
113            token_state,
114            max_file_size_bytes,
115        });
116    }
117    bundles
118}

120/// Run a single profile's share request and translate the result into a
121/// `ProfileShareOutcome` for aggregation.
122///
123/// ### Arguments
124/// - `resources`: The profile and its selected device subset.
125/// - `request`: The shared request payload (content, filename, `file_path`).
126/// - `http_agent`: Shared HTTP agent.
127///
128/// ### Returns
129/// - `ProfileShareOutcome`: `Completed` for a per-device dispatch result;
130///   `Aborted` when validation prevented any per-device call.
131fn execute_profile_share(
132    resources: &ProfileShareResources,
133    request: &ShareFileRequest,
134    http_agent: &ureq::Agent,
135) -> ProfileShareOutcome {
136    let result = share_file(
137        &resources.profile,
138        request,
139        &resources.devices,
140        &resources.token_state,
141        http_agent,
142        resources.max_file_size_bytes,
143    );
144    match result {
145        Ok(share_result) => ProfileShareOutcome::Completed(share_result),
146        Err(e) => {
147            log::error!(
148                "Profile '{}': share aborted: {e}",
149                resources.profile.name.as_str()
150            );
151            ProfileShareOutcome::Aborted(e)
152        }
153    }
154}

156/// Pick the right notification type for an aggregated multi-profile share.
157///
158/// ### Arguments
159/// - `outcomes`: Per-profile outcomes after fan-out.
160///
161/// ### Returns
162/// - `NotificationType`: `Success` if every profile completely succeeded,
163///   `Error` if no profile had any success, otherwise `Warning`.
164fn aggregate_notification_type(outcomes: &[(String, ProfileShareOutcome)]) -> NotificationType {
165    let any_success = outcomes.iter().any(|(_, o)| o.has_success());
166    let any_failure = outcomes.iter().any(|(_, o)| o.has_failure());
167    match (any_success, any_failure) {
168        (true, false) => NotificationType::Success,
169        (true, true) => NotificationType::Warning,
170        _ => NotificationType::Error,
171    }
172}

174/// Handle the Share button: validate selection, fan out per profile, and queue an aggregated notification when every profile has reported back.
175///
176/// ### Arguments
177/// - `state`: Shared sheet state.
178/// - `entity`: The Fulgur entity.
179/// - `window`: The current window context.
180/// - `cx`: The application context.
181pub(super) fn handle_share_file(
182    state: &Arc<ShareSheetState>,
183    entity: &Entity<Fulgur>,
184    window: &mut Window,
185    cx: &mut App,
186) {
187    let keys = state.selected.lock().clone();
188    if keys.is_empty() {
189        window.push_notification(
190            (
191                NotificationType::Warning,
192                SharedString::from("Please select at least one device to share with."),
193            ),
194            cx,
195        );
196        return;
197    }
198    let bundles = build_profile_share_resources(&keys, state, entity, cx);
199    if bundles.is_empty() {
200        log::warn!("Share aborted: no loaded profiles in selection");
201        window.push_notification(
202            (
203                NotificationType::Warning,
204                SharedString::from(
205                    "No devices available to share with. Wait for the device list to load and try again.",
206                ),
207            ),
208            cx,
209        );
210        return;
211    }
212    let share_context = capture_share_context(entity, cx);
213    let http_agent = entity.update(cx, |_, cx| Arc::clone(&Fulgur::shared_state(cx).http_agent));
214    let notification_tx =
215        entity.update(cx, |_, cx| Fulgur::shared_state(cx).notification_tx.clone());
216    state
217        .active
218        .store(false, std::sync::atomic::Ordering::Release);
219    entity.update(cx, |this, _| {
220        this.share_sheet_state = None;
221    });
222    window.close_sheet(cx);
223    let progress_label = format!("Sharing {}...", share_context.file_name);
224    // No Cancel button: every per-device upload is dispatched in parallel via
225    // nested `thread::scope` and `ureq` offers no in-flight abort, so a cancel
226    // flag could not actually stop a share once started. See improvement 025.
227    let progress = start_progress(window, cx, progress_label.into(), None);
228    std::thread::spawn(move || {
229        let _progress = progress;
230        let outcomes: Vec<(String, ProfileShareOutcome)> = std::thread::scope(|scope| {
231            let handles: Vec<_> = bundles
232                .iter()
233                .map(|bundle| {
234                    let request = ShareFileRequest {
235                        content: Arc::clone(&share_context.content),
236                        file_name: share_context.file_name.clone(),
237                        device_ids: bundle.device_ids.clone(),
238                        file_path: share_context.file_path.clone(),
239                    };
240                    let http_agent_ref = &http_agent;
241                    scope.spawn(move || {
242                        let outcome = execute_profile_share(bundle, &request, http_agent_ref);
243                        (bundle.profile.name.clone(), outcome)
244                    })
245                })
246                .collect();
247            handles
248                .into_iter()
249                .map(|h| {
250                    h.join().unwrap_or_else(|e| {
251                        log::error!("Per-profile share thread panicked: {e:?}");
252                        (
253                            String::from("(unknown)"),
254                            ProfileShareOutcome::Aborted(
255                                crate::fulgur::sync::synchronization::SynchronizationError::Other(
256                                    "Internal issue".to_string(),
257                                ),
258                            ),
259                        )
260                    })
261                })
262                .collect()
263        });
264        let summary = format_multi_profile_summary(&outcomes);
265        let notification_type = aggregate_notification_type(&outcomes);
266        let _ = notification_tx.unbounded_send((notification_type, SharedString::from(summary)));
267    });
268}

270/// Spawn a background thread that fetches one profile's devices and writes its outcome to the shared sheet state.
271///
272/// ### Arguments
273/// - `profile`: Profile to fetch from.
274/// - `state`: Shared sheet state to update.
275/// - `sync_state`: Per-profile shared sync state.
276/// - `http_agent`: Shared HTTP agent.
277pub(super) fn spawn_profile_device_fetch(
278    profile: ServerProfile,
279    state: Arc<ShareSheetState>,
280    sync_state: Arc<crate::fulgur::shared_state::SyncState>,
281    http_agent: Arc<ureq::Agent>,
282) {
283    let needs_reconnect = !sync_state.connection_status.lock().is_connected();
284    crate::fulgur::sync::synchronization::set_sync_server_connection_status(
285        &sync_state.connection_status,
286        SynchronizationStatus::Connecting,
287    );
288    *sync_state.connecting_since.lock() = Some(std::time::Instant::now());
289    std::thread::spawn(move || {
290        if !state.active.load(std::sync::atomic::Ordering::Acquire) {
291            return;
292        }
293        if needs_reconnect {
294            match crate::fulgur::sync::synchronization::initial_synchronization(
295                &profile,
296                &sync_state.token_state,
297                &http_agent,
298                &sync_state.pending_ack_share_ids,
299            ) {
300                Ok(crate::fulgur::sync::synchronization::InitialSyncOutcome {
301                    begin: begin_response,
302                    min_fulgur_version,
303                    fulgurant_version,
304                }) => {
305                    *sync_state.device_name.lock() = Some(begin_response.device_name.clone());
306                    *sync_state.pending_shared_files.lock() = begin_response.shares;
307                    crate::fulgur::sync::synchronization::store_server_max_file_size(
308                        &sync_state.max_file_size_bytes,
309                        begin_response.max_file_size_bytes,
310                    );
311                    // Refresh the stored versions so the profiles-list warnings
312                    // stay current after a reconnect triggered from sharing.
313                    let _ = crate::fulgur::sync::synchronization::record_server_min_fulgur_version(
314                        &sync_state.server_min_fulgur_version,
315                        &profile.name,
316                        min_fulgur_version,
317                    );
318                    let _ = crate::fulgur::sync::synchronization::record_fulgurant_version(
319                        &sync_state.server_version,
320                        &profile.name,
321                        fulgurant_version,
322                    );
323                    crate::fulgur::sync::synchronization::set_sync_server_connection_status(
324                        &sync_state.connection_status,
325                        SynchronizationStatus::Connected,
326                    );
327                }
328                Err(e) => {
329                    let status = SynchronizationStatus::from_error(&e);
330                    crate::fulgur::sync::synchronization::set_sync_server_connection_status(
331                        &sync_state.connection_status,
332                        status,
333                    );
334                    *sync_state.connecting_since.lock() = None;
335                    state.per_profile.write().insert(
336                        profile.id.clone(),
337                        ProfileFetchState::Failed(format!("{e}")),
338                    );
339                    return;
340                }
341            }
342        }
343        let result = get_devices(&profile, &sync_state.token_state, &http_agent);
344        *sync_state.connecting_since.lock() = None;
345        match result {
346            Ok((devices, server_max_size)) => {
347                crate::fulgur::sync::synchronization::store_server_max_file_size(
348                    &sync_state.max_file_size_bytes,
349                    server_max_size,
350                );
351                crate::fulgur::sync::synchronization::set_sync_server_connection_status(
352                    &sync_state.connection_status,
353                    SynchronizationStatus::Connected,
354                );
355                state.per_profile.write().insert(
356                    profile.id.clone(),
357                    ProfileFetchState::Loaded(Arc::new(devices)),
358                );
359                if needs_reconnect {
360                    state.pending_sse_restarts.lock().push(profile.id.clone());
361                }
362            }
363            Err(e) => {
364                state.per_profile.write().insert(
365                    profile.id.clone(),
366                    ProfileFetchState::Failed(format!("{e}")),
367                );
368            }
369        }
370    });
