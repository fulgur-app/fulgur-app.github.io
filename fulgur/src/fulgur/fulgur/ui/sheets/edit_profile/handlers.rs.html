15/// Handle the Test connection button.
16///
17/// ### Arguments
18/// - `entity`: The Fulgur entity (used to look up the existing public key).
19/// - `state`: The shared form state.
20/// - `window`: The window to attach the progress notification to.
21/// - `cx`: The application context.
22pub(super) fn handle_test_connection(
23    entity: &Entity<Fulgur>,
24    state: &Arc<ProfileFormState>,
25    window: &mut Window,
26    cx: &mut App,
27) {
28    let url_raw = state.server_url_input.read(cx).value().trim().to_string();
29    let Some(server_url) = optional_text(&url_raw) else {
30        window.push_notification(
31            (
32                NotificationType::Error,
33                SharedString::from("Enter a server URL to test the connection."),
34            ),
35            cx,
36        );
37        return;
38    };
39    if let Err(msg) = validate_url(&server_url) {
40        window.push_notification((NotificationType::Error, msg), cx);
41        return;
42    }
43    let device_key_edit = read_device_key_edit(state, cx);
44    // Snapshot the existing key before any write so Cancel can restore it.
45    if matches!(
46        device_key_edit,
47        DeviceKeyEdit::Set(_) | DeviceKeyEdit::Clear
48    ) {
49        snapshot_device_key_for_rollback(state);
50    }
51    if let Err(e) = apply_device_key_edit(&state.profile_id, &device_key_edit) {
52        log::error!("Failed to save device API key for ping: {e}");
53        window.push_notification(
54            (
55                NotificationType::Error,
56                SharedString::from(format!("Failed to save device key: {e}")),
57            ),
58            cx,
59        );
60        return;
61    }
62    let existing_public_key = entity
63        .read(cx)
64        .settings
65        .app_settings
66        .synchronization_settings
67        .find_profile(&state.profile_id)
68        .and_then(|p| p.public_key.clone());
69    let profile = build_profile_from_form(state, existing_public_key, cx);
70    cx.global::<crate::fulgur::shared_state::SharedAppState>()
71        .sync_state_for(&profile.id)
72        .token_state
73        .clear_token();
74    let display_name = if profile.name.is_empty() {
75        server_url
76    } else {
77        profile.name.clone()
78    };
79    perform_ping_with_progress(profile, display_name, window, cx);
80}

98/// Handle profile persistence for the Save button, with optional HTTP warning.
99///
100/// ### Arguments
101/// - `entity`: The Fulgur entity.
102/// - `state`: The shared form state.
103/// - `window`: The window to attach notifications/modals to.
104/// - `cx`: The application context.
105/// - `warn_on_http`: Whether to show an HTTP confirmation before saving.
106fn handle_save_internal(
107    entity: &Entity<Fulgur>,
108    state: &Arc<ProfileFormState>,
109    window: &mut Window,
110    cx: &mut App,
111    warn_on_http: bool,
112) {
113    if let Err(message) = validate_form(state, entity, cx) {
114        window.push_notification((NotificationType::Error, message), cx);
115        return;
116    }
117    let server_url_value = state.server_url_input.read(cx).value().trim().to_string();
118    if warn_on_http && should_warn_for_http_url(&server_url_value) {
119        confirm_http_save(entity, state, window, cx);
120        return;
121    }
122    let device_key_edit = read_device_key_edit(state, cx);
123    if let Err(e) = apply_device_key_edit(&state.profile_id, &device_key_edit) {
124        log::error!("Failed to save device API key: {e}");
125        window.push_notification(
126            (
127                NotificationType::Error,
128                SharedString::from(format!("Failed to save device key: {e}")),
129            ),
130            cx,
131        );
132        return;
133    }
134    let existing_public_key = entity
135        .read(cx)
136        .settings
137        .app_settings
138        .synchronization_settings
139        .find_profile(&state.profile_id)
140        .and_then(|p| p.public_key.clone());
141    let profile = build_profile_from_form(state, existing_public_key, cx);
142    let profile_id = profile.id.clone();
143    let is_new = state.is_new;
144    let result = entity.update(cx, |this, cx| {
145        if is_new {
146            this.add_profile(profile, cx).map(|()| true)
147        } else {
148            this.update_profile(
149                &profile_id,
150                |existing| {
151                    existing.name.clone_from(&profile.name);
152                    existing.is_active = profile.is_active;
153                    existing.server_url.clone_from(&profile.server_url);
154                    existing.email.clone_from(&profile.email);
155                    existing.is_deduplication = profile.is_deduplication;
156                },
157                cx,
158            )
159        }
160    });
161    match result {
162        Ok(true) => {
163            // Clear the cached JWT so the next call uses the latest key/email.
164            cx.global::<crate::fulgur::shared_state::SharedAppState>()
165                .sync_state_for(&profile_id)
166                .token_state
167                .clear_token();
168            entity.update(cx, |this, cx| {
169                this.restart_sse_connection_for_with_progress(&profile_id, window, cx);
170            });
171            window.close_sheet(cx);
172        }
173        Ok(false) => {
174            log::warn!(
175                "Profile '{profile_id}' could not be updated (no longer in settings); closing sheet"
176            );
177            window.close_sheet(cx);
178        }
179        Err(e) => {
180            log::error!("Failed to save profile: {e}");
181            window.push_notification(
182                (
183                    NotificationType::Error,
184                    SharedString::from(format!("Failed to save profile: {e}")),
185                ),
186                cx,
187            );
188        }
189    }
190}

192/// Ask for explicit confirmation before saving an `http://` server URL.
193///
194/// ### Arguments
195/// - `entity`: The Fulgur entity.
196/// - `state`: The shared form state.
197/// - `window`: The window to attach the alert dialog to.
198/// - `cx`: The application context.
199fn confirm_http_save(
200    entity: &Entity<Fulgur>,
201    state: &Arc<ProfileFormState>,
202    window: &mut Window,
203    cx: &mut App,
204) {
205    let server_url = state.server_url_input.read(cx).value().trim().to_string();
206    let entity_for_confirm = entity.clone();
207    let state_for_confirm = Arc::clone(state);
208    window.open_alert_dialog(cx, move |modal, _, _| {
209        let entity_ok = entity_for_confirm.clone();
210        let state_ok = Arc::clone(&state_for_confirm);
211        modal
212            .title(div().text_size(px(16.)).child("Insecure HTTP connection"))
213            .keyboard(true)
214            .button_props(
215                DialogButtonProps::default()
216                    .show_cancel(true)
217                    .cancel_text("Cancel")
218                    .cancel_variant(ButtonVariant::Secondary)
219                    .ok_text("Continue")
220                    .ok_variant(ButtonVariant::Danger),
221            )
222            .overlay_closable(false)
223            .close_button(false)
224            .child(
225                v_flex()
226                    .gap_2()
227                    .child(format!(
228                        "The server URL \"{server_url}\" uses HTTP and can expose credentials in transit."
229                    ))
230                    .child("Are you sure you want to continue saving this server?"),
231            )
232            .on_ok(move |_, window, cx| {
233                handle_save_internal(&entity_ok, &state_ok, window, cx, false);
234                true
235            })
236            .on_cancel(|_, _, _| true)
237    });
238}

240/// Handle the Cancel button.
241///
242/// ### Arguments
243/// - `_entity`: The Fulgur entity (unused; reserved for future cleanup).
244/// - `state`: The shared form state.
245/// - `window`: The window the sheet is attached to.
246/// - `cx`: The application context.
247pub(super) fn handle_cancel(
248    _entity: &Entity<Fulgur>,
249    state: &Arc<ProfileFormState>,
250    window: &mut Window,
251    cx: &mut App,
252) {
253    if let Err(e) = rollback_device_key(state) {
254        log::warn!(
255            "Failed to roll back device key for profile '{}': {e}",
256            state.profile_id
257        );
258    }
259    window.close_sheet(cx);
260}

262/// Open a confirmation dialog before deleting a profile.
263///
264/// ### Arguments
265/// - `entity`: The Fulgur entity.
266/// - `state`: The shared form state (read for profile id and name).
267/// - `window`: The window to attach the dialog to.
268/// - `cx`: The application context.
269pub(super) fn confirm_delete_profile(
270    entity: &Entity<Fulgur>,
271    state: &Arc<ProfileFormState>,
272    window: &mut Window,
273    cx: &mut App,
274) {
275    let profile_name = state.name_input.read(cx).value().trim().to_string();
276    let display_name = if profile_name.is_empty() {
277        "this server".to_string()
278    } else {
279        format!("'{profile_name}'")
280    };
281    let entity_for_confirm = entity.clone();
282    let state_for_confirm = Arc::clone(state);
283    window.open_alert_dialog(cx, move |modal, _, _| {
284        let entity_ok = entity_for_confirm.clone();
285        let state_ok = Arc::clone(&state_for_confirm);
286        modal
287            .title(div().text_size(px(16.)).child("Delete Server"))
288            .keyboard(true)
289            .button_props(
290                DialogButtonProps::default()
291                    .show_cancel(true)
292                    .cancel_text("Cancel")
293                    .cancel_variant(ButtonVariant::Secondary)
294                    .ok_text("Delete")
295                    .ok_variant(ButtonVariant::Danger),
296            )
297            .overlay_closable(false)
298            .close_button(false)
299            .child(
300                v_flex()
301                    .gap_2()
302                    .child(format!("Delete {display_name}?"))
303                    .child("This cannot be undone."),
304            )
305            .on_ok(move |_, window, cx| {
306                let profile_id = state_ok.profile_id.clone();
307                let outcome = entity_ok.update(cx, |this, cx| this.delete_profile(&profile_id, cx));
308                match outcome {
309                    Ok(_) => {
310                        window.close_sheet(cx);
311                    }
312                    Err(e) => {
313                        log::error!("Failed to delete profile '{profile_id}': {e}");
314                        window.push_notification(
315                            (
316                                NotificationType::Error,
317                                SharedString::from(format!("Failed to delete server: {e}")),
318                            ),
319                            cx,
320                        );
321                    }
322                }
323                true
324            })
325            .on_cancel(|_, _, _| true)
326    });
