24/// Build a profile draft from the current form values.
25///
26/// ### Arguments
27/// - `state`: The shared form state.
28/// - `existing_public_key`: Existing public key to carry over for Edit mode;
29///   `None` for Add mode (key generation happens later if needed).
30/// - `cx`: The application context (used to read input values).
31///
32/// ### Returns
33/// - `ServerProfile`: A profile struct populated from the form.
34pub(super) fn build_profile_from_form(
35    state: &ProfileFormState,
36    existing_public_key: Option<String>,
37    cx: &App,
38) -> ServerProfile {
39    let name = state.name_input.read(cx).value().trim().to_string();
40    let server_url = optional_text(&state.server_url_input.read(cx).value());
41    let email = optional_text(&state.email_input.read(cx).value());
42    ServerProfile {
43        id: state.profile_id.clone(),
44        name,
45        is_active: *state.is_active.lock(),
46        server_url,
47        email,
48        public_key: existing_public_key,
49        is_deduplication: *state.is_deduplication.lock(),
50    }
51}

53/// Convert an input value into the optional string used by `ServerProfile`.
54///
55/// ### Arguments
56/// - `value`: The trimmed/untrimmed input string.
57///
58/// ### Returns
59/// - `Some(String)`: The trimmed value when non-empty.
60/// - `None`: When the trimmed value is empty.
61pub(super) fn optional_text(value: &str) -> Option<String> {
62    let trimmed = value.trim();
63    if trimmed.is_empty() {
64        None
65    } else {
66        Some(trimmed.to_string())
67    }
68}

70pub(super) enum DeviceKeyEdit {
71    Untouched,
72    Clear,
73    Set(String),
74}

76/// Read the current device key value from the form.
77///
78/// ### Arguments
79/// - `state`: The shared form state.
80/// - `cx`: The application context.
81///
82/// ### Returns
83/// - `DeviceKeyEdit::Untouched`: The user has not modified the placeholder.
84/// - `DeviceKeyEdit::Clear`: The user erased the field.
85/// - `DeviceKeyEdit::Set(value)`: The user typed a new key.
86pub(super) fn read_device_key_edit(state: &ProfileFormState, cx: &App) -> DeviceKeyEdit {
87    let raw = state.device_key_input.read(cx).value().to_string();
88    if raw == DEVICE_KEY_PLACEHOLDER {
89        DeviceKeyEdit::Untouched
90    } else if raw.is_empty() {
91        DeviceKeyEdit::Clear
92    } else {
93        DeviceKeyEdit::Set(raw)
94    }
95}

97/// Apply a `DeviceKeyEdit` to the keychain for a profile.
98///
99/// ### Arguments
100/// - `profile_id`: The id whose keychain entries should be updated.
101/// - `edit`: The user's edit, as decoded by `read_device_key_edit`.
102///
103/// ### Returns
104/// - `Ok(true)`: A keychain write happened (set or clear).
105/// - `Ok(false)`: The user did not modify the field; nothing was written.
106/// - `Err(anyhow::Error)`: The keychain operation failed.
107pub(super) fn apply_device_key_edit(
108    profile_id: &str,
109    edit: &DeviceKeyEdit,
110) -> anyhow::Result<bool> {
111    match edit {
112        DeviceKeyEdit::Untouched => Ok(false),
113        DeviceKeyEdit::Clear => {
114            save_device_api_key_to_keychain(profile_id, None)?;
115            Ok(true)
116        }
117        DeviceKeyEdit::Set(value) => {
118            save_device_api_key_to_keychain(profile_id, Some(value))?;
119            Ok(true)
120        }
121    }
122}
