37/// Per-profile fetch state shown in the share sheet.
38pub enum ProfileFetchState {
39    /// The device list is being fetched.
40    Loading,
41    /// Device list arrived from the server.
42    Loaded(Arc<Vec<Device>>),
43    /// The fetch failed; the message describes why.
44    Failed(String),
45}
46
47/// Shared state owning everything the share sheet needs while it is open.
48pub struct ShareSheetState {
49    /// Active profiles when the sheet was opened, in declaration order.
50    pub profiles: Vec<ServerProfile>,
51    /// Per-profile fetch progress keyed by `ProfileId`.
52    pub per_profile: Arc<RwLock<HashMap<ProfileId, ProfileFetchState>>>,
53    /// User selection across all profiles, keyed by `(profile_id, device_id)`.
54    pub selected: Arc<Mutex<Vec<(ProfileId, String)>>>,
55    /// Profile ids whose SSE worker should be restarted because we had to
56    /// reconnect during the device fetch. Drained by the render loop.
57    pub pending_sse_restarts: Arc<Mutex<Vec<ProfileId>>>,
58    /// Cleared to `false` while the sheet is open; flipped on Cancel/Share so
59    /// background tasks can stop polling.
60    pub active: Arc<AtomicBool>,
61}
62
63impl ShareSheetState {
64    /// Build a fresh shared state with every profile starting in the `Loading` state.
65    ///
66    /// ### Arguments
67    /// - `profiles`: Active profiles to render in the sheet.
68    ///
69    /// ### Returns
70    /// - `Self`: An initialized state ready for fetch threads to populate.
71    fn new(profiles: Vec<ServerProfile>) -> Self {
72        let map: HashMap<ProfileId, ProfileFetchState> = profiles
73            .iter()
74            .map(|p| (p.id.clone(), ProfileFetchState::Loading))
75            .collect();
76        Self {
77            profiles,
78            per_profile: Arc::new(RwLock::new(map)),
79            selected: Arc::new(Mutex::new(Vec::new())),
80            pending_sse_restarts: Arc::new(Mutex::new(Vec::new())),
81            active: Arc::new(AtomicBool::new(true)),
82        }
83    }
84
85    /// Whether every profile has finished its fetch (Loaded or Failed).
86    ///
87    /// ### Returns
88    /// - `true`: No profile remains in the `Loading` state.
89    /// - `false`: At least one profile is still being fetched.
90    fn all_settled(&self) -> bool {
91        self.per_profile
92            .read()
93            .values()
94            .all(|state| !matches!(state, ProfileFetchState::Loading))
95    }
96}

114/// Create a single device row in the share sheet.
115///
116/// ### Arguments
117/// - `profile_id`: The id of the profile that owns this device.
118/// - `device`: The device to display.
119/// - `is_selected`: Whether the device is currently selected.
120/// - `selected_keys`: Shared mutable selection state, keyed by `(profile_id, device_id)`.
121/// - `idx`: Stable index used to disambiguate the GPUI element id.
122/// - `cx`: The application context.
123///
124/// ### Returns
125/// - `impl Element`: The device row element.
126fn make_device_item(
127    profile_id: &ProfileId,
128    device: &Device,
129    is_selected: bool,
130    selected_keys: Arc<Mutex<Vec<(ProfileId, String)>>>,
131    idx: usize,
132    cx: &App,
133) -> impl Element {
134    let profile_id = profile_id.clone();
135    let device_id = device.id.clone();
136    let device_name = device.name.clone();
137    let device_expires = device.expires_at.clone();
138    let device_for_icon = device;
139    let has_public_key = device.public_key.is_some();
140    h_flex()
141        .id(("share-device-sheet", idx))
142        .items_center()
143        .justify_between()
144        .w_full()
145        .p_2()
146        .my_2()
147        .rounded_sm()
148        .border_color(cx.theme().border)
149        .border_1()
150        .when(has_public_key, |this| this.cursor_pointer())
151        .when(!has_public_key, |this| this.opacity(0.5))
152        .when(has_public_key, |this| {
153            this.hover(|hover| hover.bg(cx.theme().muted))
154        })
155        .when(is_selected && has_public_key, |this| {
156            this.bg(cx.theme().accent)
157                .text_color(cx.theme().accent_foreground)
158        })
159        .child(
160            v_flex()
161                .gap_1()
162                .child(
163                    h_flex()
164                        .items_center()
165                        .justify_start()
166                        .gap_2()
167                        .child(get_icon(device_for_icon))
168                        .child(div().child(device_name))
169                        .child(div().text_xs().child(format!("Expires: {device_expires}"))),
170                )
171                .when(!has_public_key, |this| {
172                    this.child(
173                        div()
174                            .text_xs()
175                            .text_color(cx.theme().muted_foreground)
176                            .child("No public key for this device"),
177                    )
178                }),
179        )
180        .when(is_selected && has_public_key, |this| {
181            this.child(Icon::new(CustomIcon::Zap))
182        })
183        .when(has_public_key, |this| {
184            this.on_click(move |_event, _window, _cx| {
185                let mut keys = selected_keys.lock();
186                let key = (profile_id.clone(), device_id.clone());
187                if let Some(pos) = keys.iter().position(|existing| existing == &key) {
188                    keys.remove(pos);
189                } else {
190                    keys.push(key);
191                }
192            })
193        })
194}

279/// Build the grouped device list reflecting the current per-profile state.
280///
281/// Each profile (header + its devices, loading placeholder, or error message)
282/// is wrapped in its own `v_flex` so the outer container can apply a wider
283/// gap between profile groups than between rows inside a group.
284///
285/// ### Arguments
286/// - `state`: Shared sheet state.
287/// - `cx`: The application context.
288///
289/// ### Returns
290/// - `Div`: The grouped list element.
291fn make_device_list(state: &Arc<ShareSheetState>, cx: &App) -> Div {
292    let mut container = v_flex().gap_6();
293    let mut row_idx: usize = 0;
294    let map = state.per_profile.read();
295    for profile in &state.profiles {
296        let mut group = div().child(make_profile_header(profile, cx));
297        match map.get(&profile.id) {
298            None | Some(ProfileFetchState::Loading) => {
299                group = group.child(make_loading_section(cx));
300            }
301            Some(ProfileFetchState::Failed(message)) => {
302                group = group.child(make_error_section(message, cx));
303            }
304            Some(ProfileFetchState::Loaded(devices)) => {
305                if devices.is_empty() {
306                    group = group.child(make_empty_section(cx));
307                } else {
308                    for device in devices.iter() {
309                        let is_selected = state
310                            .selected
311                            .lock()
312                            .iter()
313                            .any(|(pid, did)| pid == &profile.id && did == &device.id);
314                        group = group.child(make_device_item(
315                            &profile.id,
316                            device,
317                            is_selected,
318                            state.selected.clone(),
319                            row_idx,
320                            cx,
321                        ));
322                        row_idx += 1;
323                    }
324                }
325            }
326        }
327        container = container.child(group);
328    }
329    container
330}

332/// Capture the file content snapshot from the active editor tab.
333///
334/// ### Arguments
335/// - `entity`: The Fulgur entity owning the active tab.
336/// - `cx`: The application context.
337///
338/// ### Returns
339/// - `ShareContext`: The captured file content, name, and optional file path.
340fn capture_share_context(entity: &Entity<Fulgur>, cx: &mut App) -> ShareContext {
341    entity.update(cx, |this, cx| {
342        let active_tab = this.get_active_editor_tab();
343        let content: Arc<str> = active_tab
344            .as_ref()
345            .map(|tab| Arc::from(tab.content.read(cx).value().as_str()))
346            .unwrap_or_else(|| Arc::from(""));
347        let file_path = active_tab.as_ref().and_then(|tab| tab.file_path().cloned());
348        let file_name = file_path
349            .as_ref()
350            .and_then(|path| path.file_name())
351            .and_then(|name| name.to_str())
352            .unwrap_or("Untitled")
353            .to_string();
354        ShareContext {
355            content,
356            file_name,
357            file_path,
358        }
359    })
360}

362/// Group selected `(profile_id, device_id)` pairs into per-profile resource bundles.
363///
364/// ### Arguments
365/// - `selected_keys`: The user's selection across all groups.
366/// - `state`: Shared sheet state holding per-profile loaded devices.
367/// - `entity`: The Fulgur entity (used to read shared state for tokens and limits).
368/// - `cx`: The application context.
369///
370/// ### Returns
371/// - `Vec<ProfileShareResources>`: One bundle per profile with at least one selected device.
372fn build_profile_share_resources(
373    selected_keys: &[(ProfileId, String)],
374    state: &Arc<ShareSheetState>,
375    entity: &Entity<Fulgur>,
376    cx: &mut App,
377) -> Vec<ProfileShareResources> {
378    let mut bundles: Vec<ProfileShareResources> = Vec::new();
379    let map = state.per_profile.read();
380    for profile in &state.profiles {
381        let device_ids: Vec<String> = selected_keys
382            .iter()
383            .filter(|(pid, _)| pid == &profile.id)
384            .map(|(_, did)| did.clone())
385            .collect();
386        if device_ids.is_empty() {
387            continue;
388        }
389        let devices = match map.get(&profile.id) {
390            Some(ProfileFetchState::Loaded(devices)) => Arc::clone(devices),
391            _ => {
392                log::warn!(
393                    "Profile '{}': selection present but no loaded devices, skipping",
394                    profile.name
395                );
396                continue;
397            }
398        };
399        let (token_state, max_file_size_bytes) = entity.update(cx, |this, cx| {
400            let sync_state = this.shared_state(cx).sync_state_for(&profile.id);
401            (
402                Arc::clone(&sync_state.token_state),
403                sync_state
404                    .max_file_size_bytes
405                    .load(std::sync::atomic::Ordering::Acquire),
406            )
407        });
408        bundles.push(ProfileShareResources {
409            profile: profile.clone(),
410            devices,
411            device_ids,
412            token_state,
413            max_file_size_bytes,
414        });
415    }
416    bundles
417}

419/// Run a single profile's share request and translate the result into a
420/// `ProfileShareOutcome` for aggregation.
421///
422/// ### Arguments
423/// - `resources`: The profile and its selected device subset.
424/// - `request`: The shared request payload (content, filename, `file_path`).
425/// - `http_agent`: Shared HTTP agent.
426///
427/// ### Returns
428/// - `ProfileShareOutcome`: `Completed` for a per-device dispatch result;
429///   `Aborted` when validation prevented any per-device call.
430fn execute_profile_share(
431    resources: &ProfileShareResources,
432    request: &ShareFileRequest,
433    http_agent: &ureq::Agent,
434) -> ProfileShareOutcome {
435    let result = share_file(
436        &resources.profile,
437        request,
438        &resources.devices,
439        &resources.token_state,
440        http_agent,
441        resources.max_file_size_bytes,
442    );
443    match result {
444        Ok(share_result) => ProfileShareOutcome::Completed(share_result),
445        Err(e) => {
446            log::error!(
447                "Profile '{}': share aborted: {e}",
448                resources.profile.name.as_str()
449            );
450            ProfileShareOutcome::Aborted(e)
451        }
452    }
453}

455/// Pick the right notification type for an aggregated multi-profile share.
456///
457/// ### Arguments
458/// - `outcomes`: Per-profile outcomes after fan-out.
459///
460/// ### Returns
461/// - `NotificationType`: `Success` if every profile completely succeeded,
462///   `Error` if no profile had any success, otherwise `Warning`.
463fn aggregate_notification_type(outcomes: &[(String, ProfileShareOutcome)]) -> NotificationType {
464    let any_success = outcomes.iter().any(|(_, o)| o.has_success());
465    let any_failure = outcomes.iter().any(|(_, o)| o.has_failure());
466    match (any_success, any_failure) {
467        (true, false) => NotificationType::Success,
468        (true, true) => NotificationType::Warning,
469        _ => NotificationType::Error,
470    }
471}

473/// Handle the Share button: validate selection, fan out per profile, and queue an aggregated notification when every profile has reported back.
474///
475/// ### Arguments
476/// - `state`: Shared sheet state.
477/// - `entity`: The Fulgur entity.
478/// - `window`: The current window context.
479/// - `cx`: The application context.
480fn handle_share_file(
481    state: &Arc<ShareSheetState>,
482    entity: &Entity<Fulgur>,
483    window: &mut Window,
484    cx: &mut App,
485) {
486    let keys = state.selected.lock().clone();
487    if keys.is_empty() {
488        window.push_notification(
489            (
490                NotificationType::Warning,
491                SharedString::from("Please select at least one device to share with."),
492            ),
493            cx,
494        );
495        return;
496    }
497    let bundles = build_profile_share_resources(&keys, state, entity, cx);
498    if bundles.is_empty() {
499        log::warn!("Share aborted: no loaded profiles in selection");
500        return;
501    }
502    let share_context = capture_share_context(entity, cx);
503    let http_agent = entity.update(cx, |this, cx| Arc::clone(&this.shared_state(cx).http_agent));
504    let pending_notification = entity.update(cx, |this, cx| {
505        let first_profile_id = bundles
506            .first()
507            .map(|b| b.profile.id.clone())
508            .unwrap_or_default();
509        Arc::clone(
510            &this
511                .shared_state(cx)
512                .sync_state_for(&first_profile_id)
513                .pending_notification,
514        )
515    });
516    state
517        .active
518        .store(false, std::sync::atomic::Ordering::Release);
519    entity.update(cx, |this, _| {
520        this.share_sheet_state = None;
521    });
522    window.close_sheet(cx);
523    let progress_label = format!("Sharing {}...", share_context.file_name);
524    let cancel_callback: Option<CancelCallback> = Some(Box::new(|_, _| {}));
525    let progress = start_progress(window, cx, progress_label.into(), cancel_callback);
526    let cancel_flag = progress.cancel_flag();
527    std::thread::spawn(move || {
528        let _progress = progress;
529        let outcomes: Vec<(String, ProfileShareOutcome)> = std::thread::scope(|scope| {
530            let handles: Vec<_> = bundles
531                .iter()
532                .map(|bundle| {
533                    let request = ShareFileRequest {
534                        content: Arc::clone(&share_context.content),
535                        file_name: share_context.file_name.clone(),
536                        device_ids: bundle.device_ids.clone(),
537                        file_path: share_context.file_path.clone(),
538                    };
539                    let http_agent_ref = &http_agent;
540                    scope.spawn(move || {
541                        let outcome = execute_profile_share(bundle, &request, http_agent_ref);
542                        (bundle.profile.name.clone(), outcome)
543                    })
544                })
545                .collect();
546            handles
547                .into_iter()
548                .map(|h| {
549                    h.join().unwrap_or_else(|e| {
550                        log::error!("Per-profile share thread panicked: {e:?}");
551                        (
552                            String::from("(unknown)"),
553                            ProfileShareOutcome::Aborted(
554                                crate::fulgur::sync::synchronization::SynchronizationError::Other(
555                                    "Internal issue".to_string(),
556                                ),
557                            ),
558                        )
559                    })
560                })
561                .collect()
562        });
563        if cancel_flag.load(std::sync::atomic::Ordering::Acquire) {
564            return;
565        }
566        let summary = format_multi_profile_summary(&outcomes);
567        let notification_type = aggregate_notification_type(&outcomes);
568        *pending_notification.lock() = Some((notification_type, SharedString::from(summary)));
569    });
570}

572/// Spawn a background thread that fetches one profile's devices and writes its outcome to the shared sheet state.
573///
574/// ### Arguments
575/// - `profile`: Profile to fetch from.
576/// - `state`: Shared sheet state to update.
577/// - `sync_state`: Per-profile shared sync state.
578/// - `http_agent`: Shared HTTP agent.
579fn spawn_profile_device_fetch(
580    profile: ServerProfile,
581    state: Arc<ShareSheetState>,
582    sync_state: Arc<crate::fulgur::shared_state::SyncState>,
583    http_agent: Arc<ureq::Agent>,
584) {
585    let needs_reconnect = !sync_state.connection_status.lock().is_connected();
586    crate::fulgur::sync::synchronization::set_sync_server_connection_status(
587        &sync_state.connection_status,
588        SynchronizationStatus::Connecting,
589    );
590    *sync_state.connecting_since.lock() = Some(std::time::Instant::now());
591    std::thread::spawn(move || {
592        if !state.active.load(std::sync::atomic::Ordering::Acquire) {
593            return;
594        }
595        if needs_reconnect {
596            match crate::fulgur::sync::synchronization::initial_synchronization(
597                &profile,
598                &sync_state.token_state,
599                &http_agent,
600            ) {
601                Ok(begin_response) => {
602                    *sync_state.device_name.lock() = Some(begin_response.device_name.clone());
603                    *sync_state.pending_shared_files.lock() = begin_response.shares;
604                    crate::fulgur::sync::synchronization::store_server_max_file_size(
605                        &sync_state.max_file_size_bytes,
606                        begin_response.max_file_size_bytes,
607                    );
608                    crate::fulgur::sync::synchronization::set_sync_server_connection_status(
609                        &sync_state.connection_status,
610                        SynchronizationStatus::Connected,
611                    );
612                }
613                Err(e) => {
614                    let status = SynchronizationStatus::from_error(&e);
615                    crate::fulgur::sync::synchronization::set_sync_server_connection_status(
616                        &sync_state.connection_status,
617                        status,
618                    );
619                    *sync_state.connecting_since.lock() = None;
620                    state.per_profile.write().insert(
621                        profile.id.clone(),
622                        ProfileFetchState::Failed(format!("{e}")),
623                    );
624                    return;
625                }
626            }
627        }
628        let result = get_devices(&profile, &sync_state.token_state, &http_agent);
629        *sync_state.connecting_since.lock() = None;
630        match result {
631            Ok((devices, server_max_size)) => {
632                crate::fulgur::sync::synchronization::store_server_max_file_size(
633                    &sync_state.max_file_size_bytes,
634                    server_max_size,
635                );
636                crate::fulgur::sync::synchronization::set_sync_server_connection_status(
637                    &sync_state.connection_status,
638                    SynchronizationStatus::Connected,
639                );
640                state.per_profile.write().insert(
641                    profile.id.clone(),
642                    ProfileFetchState::Loaded(Arc::new(devices)),
643                );
644                if needs_reconnect {
645                    state.pending_sse_restarts.lock().push(profile.id.clone());
646                }
647            }
648            Err(e) => {
649                state.per_profile.write().insert(
650                    profile.id.clone(),
651                    ProfileFetchState::Failed(format!("{e}")),
652                );
653            }
654        }
655    });
656}

662/// - `entity`: The Fulgur entity.

664/// ### Returns
665/// - `Div`: The footer element.
666fn render_footer(state: Arc<ShareSheetState>, entity: Entity<Fulgur>) -> Div {

668    let entity_for_cancel = entity.clone();
669    let state_for_share = state;

702impl Fulgur {
703    /// Return the list of profile ids whose devices should populate the share sheet.
704    ///
705    /// ### Returns
706    /// - `Vec<ProfileId>`: Active only profile ids when the master sync switch is on, in declaration order.
707    fn collect_active_profiles(&self) -> Vec<ServerProfile> {
708        if !self
709            .settings
710            .app_settings
711            .synchronization_settings
712            .is_synchronization_activated
713        {
714            return Vec::new();
715        }
716        self.settings
717            .app_settings
718            .synchronization_settings
719            .profiles
720            .iter()
721            .filter(|p| p.is_active)
722            .cloned()
723            .collect()
724    }
725
726    /// Open the share file sheet immediately and fetch each active profile's devices in the background.
727    ///
728    /// ### Arguments
729    /// - `window`: The window context.
730    /// - `cx`: The application context.
731    pub fn open_share_file_sheet(&mut self, window: &mut Window, cx: &mut Context<Self>) {
732        let active_profiles = self.collect_active_profiles();
733        if active_profiles.is_empty() {
734            log::warn!("Share aborted: no active profile or master sync is off");
735            return;
736        }
737        if let Some(previous) = self.share_sheet_state.take() {
738            previous
739                .active
740                .store(false, std::sync::atomic::Ordering::Release);
741        }
742        let state = Arc::new(ShareSheetState::new(active_profiles.clone()));
743        self.share_sheet_state = Some(Arc::clone(&state));
744        let shared = self.shared_state(cx);
745        let http_agent = Arc::clone(&shared.http_agent);
746        for profile in active_profiles {
747            let sync_state = shared.sync_state_for(&profile.id);
748            spawn_profile_device_fetch(
749                profile,
750                Arc::clone(&state),
751                sync_state,
752                Arc::clone(&http_agent),
753            );
754        }
755        self.start_share_sheet_render_pump(Arc::clone(&state), window, cx);
756        self.show_share_sheet(state, window, cx);
757    }
758
759    /// Drive periodic re-renders while the share sheet is open and at least one profile is still loading.
760    ///
761    /// ### Arguments
762    /// - `state`: Shared sheet state.
763    /// - `window`: The window context.
764    /// - `cx`: The application context.
765    fn start_share_sheet_render_pump(
766        &mut self,
767        state: Arc<ShareSheetState>,
768        window: &mut Window,
769        cx: &mut Context<Self>,
770    ) {
771        let entity = cx.entity();
772        window
773            .spawn(cx, async move |async_cx| {
774                loop {
775                    async_cx
776                        .background_executor()
777                        .timer(Duration::from_millis(100))
778                        .await;
779                    if !state.active.load(std::sync::atomic::Ordering::Acquire) {
780                        return;
781                    }
782                    let _ = async_cx.update(|_, cx| {
783                        entity.update(cx, |_, cx| cx.notify());
784                    });
785                    if state.all_settled() {
786                        return;
787                    }
788                }
789            })
790            .detach();
791    }
792
793    /// Drain pending SSE restart requests posted by share-sheet device fetches.
794    ///
795    /// ### Arguments
796    /// - `_window`: The window context.
797    /// - `cx`: The application context.
798    pub fn process_pending_share_sheet(&mut self, _window: &mut Window, cx: &mut Context<Self>) {
799        let Some(state) = self.share_sheet_state.as_ref().map(Arc::clone) else {
800            return;
801        };
802        let to_restart: Vec<ProfileId> = std::mem::take(&mut *state.pending_sse_restarts.lock());
803        for profile_id in to_restart {
804            self.restart_sse_connection_for(&profile_id, cx);
805        }
806    }
807
808    /// Open the share sheet UI for the given shared state. Each profile section reads live state on every render.
809    ///
810    /// ### Arguments
811    /// - `state`: Shared sheet state.
812    /// - `window`: The window context.
813    /// - `cx`: The application context.
814    fn show_share_sheet(
815        &mut self,
816        state: Arc<ShareSheetState>,
817        window: &mut Window,
818        cx: &mut Context<Self>,
819    ) {
820        let entity = cx.entity();
821        let viewport_height = window.viewport_size().height;
822        window.open_sheet(cx, move |sheet, _window, cx| {
823            #[cfg(target_os = "linux")]
824            let sheet_overhead = px(200.0);
825            #[cfg(not(target_os = "linux"))]
826            let sheet_overhead = px(150.0);
827            let max_height = px((viewport_height - sheet_overhead).into());
828            let state_for_list = Arc::clone(&state);
829            let state_for_footer = Arc::clone(&state);
830            sheet
831                .title("Share with...")
832                .size(px(400.))
833                .overlay(true)
834                .child(
835                    v_flex()
836                        .overflow_y_scrollbar()
837                        .gap_2()
838                        .h(max_height)
839                        .child(make_device_list(&state_for_list, cx)),
840                )
841                .footer(render_footer(state_for_footer, entity.clone()))
842        });
843    }
