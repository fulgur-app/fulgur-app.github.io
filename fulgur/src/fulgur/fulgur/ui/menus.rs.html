53/// A single tab entry for the dock menu, carrying the display name and the action to fire

55pub enum DockMenuTab {

65/// Build the menus for the Fulgur instance
66///
67/// ### Arguments
68/// - `recent_files`: The list of recent files to display
69/// - `update_link`: The optional link to the update
70///
71/// ### Returns
72/// - `Vec<Menu>`: The menus for the Fulgur instance
73pub fn build_menus(recent_files: &[PathBuf], update_link: Option<String>) -> Vec<Menu> {
74    let recent_files_items = if recent_files.is_empty() {
75        vec![MenuItem::action("No recent files", NoneAction)]
76    } else {
77        let mut items: Vec<MenuItem> = recent_files
78            .iter()
79            .map(|file| {
80                MenuItem::action(
81                    file.display().to_string(),
82                    OpenRecentFile(file.to_path_buf()),
83                )
84            })
85            .collect();
86        items.push(MenuItem::Separator);
87        items.push(MenuItem::action("Clear recent files", ClearRecentFiles));
88        items
89    };
90    vec![
91        Menu {
92            name: "Fulgur".into(),
93            items: vec![
94                MenuItem::action("About Fulgur", About),
95                if update_link.is_some() {
96                    MenuItem::action("Update available", CheckForUpdates)
97                } else {
98                    MenuItem::action("Check for updates", CheckForUpdates)
99                },
100                MenuItem::Separator,
101                MenuItem::action("Settings", SettingsTab),
102                MenuItem::action("Select theme", SelectTheme),
103                MenuItem::action("Get more themes...", GetTheme),
104                MenuItem::Separator,
105                MenuItem::action("Close Window", CloseWindow),
106                MenuItem::action("Quit", Quit),
107            ],
108        },
109        Menu {
110            name: "File".into(),
111            items: vec![
112                MenuItem::action("New", NewFile),
113                MenuItem::action("New Window", NewWindow),
114                MenuItem::action("Open...", OpenFile),
115                MenuItem::action("Open from path...", OpenPath),
116                MenuItem::Submenu(Menu {
117                    name: "Recent Files".into(),
118                    items: recent_files_items,
119                }),
120                MenuItem::separator(),
121                MenuItem::action("Save", SaveFile),
122                MenuItem::action("Save as...", SaveFileAs),
123                MenuItem::separator(),
124                MenuItem::action("Print...", PrintFile),
125                MenuItem::separator(),
126                MenuItem::action("Close file", CloseFile),
127                MenuItem::action("Close all files", CloseAllFiles),
128            ],
129        },
130        Menu {
131            name: "Edit".into(),
132            items: vec![
133                MenuItem::action("Undo", gpui_component::input::Undo),
134                MenuItem::action("Redo", gpui_component::input::Redo),
135                MenuItem::separator(),
136                MenuItem::action("Cut", gpui_component::input::Cut),
137                MenuItem::action("Copy", gpui_component::input::Copy),
138                MenuItem::action("Paste", gpui_component::input::Paste),
139                MenuItem::separator(),
140                MenuItem::action("Find & Replace", FindInFile),
141            ],
142        },
143        Menu {
144            name: "Go".into(),
145            items: vec![
146                MenuItem::action("Next Tab", NextTab),
147                MenuItem::action("Previous Tab", PreviousTab),
148                MenuItem::Separator,
149                MenuItem::action("Jump to line", JumpToLine),
150            ],
151        },
152    ]

155/// Build the macOS dock menu (right-click on dock icon)
156///
157/// Shows recent files in a submenu, then open tabs grouped by window (separated by dividers),
158/// then new tab/window actions.
159///
160/// ### Arguments
161/// - `windows`: Open tabs grouped by window; each inner slice represents one window's tabs
162/// - `recent_files`: List of recent file paths (most recent first)
163///
164/// ### Returns
165/// - `Vec<MenuItem>`: The dock menu items
166#[cfg(target_os = "macos")]
167pub fn build_dock_menu(windows: &[Vec<DockMenuTab>], recent_files: &[PathBuf]) -> Vec<MenuItem> {
168    let mut items = Vec::new();
169    if !recent_files.is_empty() {
170        let recent_items: Vec<MenuItem> = recent_files
171            .iter()
172            .map(|file| {
173                MenuItem::action(
174                    file.display().to_string(),
175                    OpenRecentFile(file.to_path_buf()),
176                )
177            })
178            .collect();
179        items.push(MenuItem::Submenu(Menu {
180            name: "Recent Files".into(),
181            items: recent_items,
182        }));
183        items.push(MenuItem::Separator);
184    }
185    let non_empty_windows: Vec<&Vec<DockMenuTab>> =
186        windows.iter().filter(|w| !w.is_empty()).collect();
187    if !non_empty_windows.is_empty() {
188        for (i, window_tabs) in non_empty_windows.iter().enumerate() {
189            if i > 0 {
190                items.push(MenuItem::Separator);
191            }
192            for tab in *window_tabs {
193                let menu_item = match tab {
194                    DockMenuTab::File { name, path } => {
195                        MenuItem::action(name.clone(), DockActivateTab(path.clone()))
196                    }
197                    DockMenuTab::Titled { name, title } => {
198                        MenuItem::action(name.clone(), DockActivateTabByTitle(title.clone()))
199                    }
200                };
201                items.push(menu_item);
202            }
203        }
204        items.push(MenuItem::Separator);
205    }
206    items.push(MenuItem::action("New Tab", NewFile));
207    items.push(MenuItem::action("New Window", NewWindow));
208    items
209}

211impl Fulgur {
212    /// Set the application menus and sync them to the AppMenuBar on non-macOS platforms.
213    ///
214    /// ### Arguments
215    /// - `menus`: The menus to set
216    /// - `cx`: The application context
217    pub fn update_menus(&mut self, menus: Vec<Menu>, cx: &mut Context<Self>) {
218        cx.set_menus(menus);
219        #[cfg(not(target_os = "macos"))]
220        {
221            if let Some(owned_menus) = cx.get_menus() {
222                GlobalState::global_mut(cx).set_app_menus(owned_menus);
223            }
224            self.title_bar
225                .update(cx, |tb, cx| tb.reload_app_menu_bar(cx));
226        }
227    }

229    /// Check for updates, open the download page in the browser if an update is available, update the menus to show the update available action and trigger notifications
230    ///
231    /// ### Arguments
232    /// - `window`: The window context
233    /// - `cx`: The application context
234    pub fn check_for_updates(&self, window: &mut Window, cx: &mut Context<Self>) {
235        if let Some(update_info) = self.shared_state(cx).update_info.lock().as_ref() {
236            match open::that(update_info.download_url.clone()) {
237                Ok(_) => {
238                    log::debug!("Successfully opened browser");
239                }
240                Err(e) => {
241                    log::error!("Failed to open browser: {}", e);
242                }
243            }
244            return;
245        }
246        let bg = cx.background_executor().clone();
247        cx.spawn_in(window, async move |view, window| {
248            log::debug!("Checking for updates");
249            let current_version = env!("CARGO_PKG_VERSION");
250            log::debug!("Current version: {}", current_version);
251            let update_info = bg
252                .spawn(async move {
253                    check_for_updates(current_version.to_string())
254                        .ok()
255                        .flatten()
256                })
257                .await;
258            window
259                .update(|window, cx| {
260                    if let Some(new_update_info) = update_info {
261                        let current_ver = new_update_info.current_version.clone();
262                        let latest_ver = new_update_info.latest_version.clone();
263                        let download_url = new_update_info.download_url.clone();
264                        let _ = view.update(cx, |this, cx| {
265                            {
266                                let mut update_info = this.shared_state(cx).update_info.lock();
267                                *update_info = Some(new_update_info);
268                            }
269                            let menus = build_menus(
270                                this.settings.recent_files.get_files(),
271                                Some(download_url),
272                            );
273                            this.update_menus(menus, cx);
274                            cx.notify();
275                        });
276                        log::info!("Update available: {} -> {}", current_ver, latest_ver);
277                    } else {
278                        let notification = SharedString::from("No update found");
279                        window.push_notification((NotificationType::Info, notification), cx);
280                    }
281                })
282                .ok();
283        })
284        .detach();
285    }
