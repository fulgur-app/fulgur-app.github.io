13    /// Check if any profile is currently connected to its sync server.
14    ///
15    /// ### Returns
16    /// - `true`: At least one profile reports a connected status.
17    /// - `false`: No profile is connected.
18    pub fn is_connected(&self, cx: &App) -> bool {
19        let shared = Fulgur::shared_state(cx);
20        let states = shared.sync_states.read();
21        states
22            .values()
23            .any(|s| s.connection_status.lock().is_connected())
24    }

26    /// Spawn the app-scope task that consumes one profile's SSE events.
27    ///
28    /// ### Arguments
29    /// - `profile_id`: The profile whose events should be consumed.
30    /// - `cx`: The application context.
31    pub fn spawn_sse_event_consumer(profile_id: &str, cx: &mut App) {
32        let sync_state = Fulgur::shared_state(cx).sync_state_for(profile_id);
33        let Some(mut events) = sync_state.sse.lock().sse_events.take() else {
34            return;
35        };
36        let profile_id: ProfileId = profile_id.to_string();
37        cx.spawn(async move |cx| {
38            while let Some(event) = events.next().await {
39                cx.update(|cx| Self::handle_sse_event_for_profile(&profile_id, event, cx));
40            }
41        })
42        .detach();
43    }

45    /// Handle a single SSE event for a specific profile on the UI thread.
46    ///
47    /// Runs at application scope: state updates go to the profile's shared sync
48    /// state, user-facing notifications go through the app notification channel,
49    /// and windows are re-rendered when the event changed something they display.
50    ///
51    /// ### Arguments
52    /// - `profile_id`: The profile that produced the event.
53    /// - `event`: The SSE event to handle.
54    /// - `cx`: The application context.
55    pub fn handle_sse_event_for_profile(profile_id: &ProfileId, event: SseEvent, cx: &mut App) {
56        let now = Instant::now();
57        let sync_state = Fulgur::shared_state(cx).sync_state_for(profile_id);
58        {
59            let mut sse = sync_state.sse.lock();
60            if let Some(last_time) = sse.last_sse_event
61                && now.duration_since(last_time) < Duration::from_millis(500)
62            {
63                return;
64            }
65            sse.last_sse_event = Some(now);
66        }
67        match event {
68            SseEvent::Heartbeat { timestamp } => {
69                log::debug!("SSE heartbeat received for profile '{profile_id}': {timestamp}");
70                let was_disconnected = !sync_state.connection_status.lock().is_connected();
71                *sync_state.last_heartbeat.lock() = Some(now);
72                if was_disconnected {
73                    *sync_state.connection_status.lock() = SynchronizationStatus::Connected;
74                    log::info!(
75                        "Profile '{profile_id}': connection restored on heartbeat after timeout"
76                    );
77                    Self::notify_all_windows(cx);
78                }
79            }
80            SseEvent::ShareAvailable(notification) => {
81                log::debug!(
82                    "Share doorbell received on consumer task (share_id={})",
83                    notification.share_id
84                );
85                Fulgur::shared_state(cx).notify((
86                    NotificationType::Info,
87                    SharedString::from("New file received"),
88                ));
89                // Wake every window so the queued share is decrypted and opened.
90                Self::notify_all_windows(cx);
91            }
92            SseEvent::Error(err) => {
93                log::error!("SSE error for profile '{profile_id}': {err}");
94            }
95        }
96    }
97
98    /// Request a re-render of every open window.
99    ///
100    /// ### Arguments
101    /// - `cx`: The application context.
102    fn notify_all_windows(cx: &mut App) {
103        let windows = match cx.try_global::<WindowManager>() {
104            Some(manager) => manager.get_all_windows(),
105            None => return,
106        };
107        for weak in windows {
108            if let Some(entity) = weak.upgrade() {
109                entity.update(cx, |_, cx| cx.notify());
110            }
111        }
112    }
