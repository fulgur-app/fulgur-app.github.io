26/// Maximum time to wait for the previous SSE thread to exit before starting a new one
27const SSE_THREAD_SHUTDOWN_TIMEOUT: Duration = Duration::from_secs(5);
28
29/// Resources produced by the common SSE restart setup phase.
30struct SseRestartSetup {
31    old_handle: Option<thread::JoinHandle<()>>,
32    sse_tx: Sender<SseEvent>,
33    sse_shutdown_flag: Arc<AtomicBool>,
34    handle_storage: Arc<Mutex<Option<thread::JoinHandle<()>>>>,
35    profile: ServerProfile,
36}
37
38/// Wait for a previous SSE background thread to stop before proceeding.
39///
40/// ### Arguments
41/// - `old_handle`: The join handle from the previous SSE thread, if any.
42fn wait_for_previous_sse_thread(old_handle: Option<thread::JoinHandle<()>>) {
43    let Some(handle) = old_handle else { return };
44    let deadline = Instant::now() + SSE_THREAD_SHUTDOWN_TIMEOUT;
45    while !handle.is_finished() && Instant::now() < deadline {
46        thread::sleep(Duration::from_millis(100));
47    }
48    if handle.is_finished() {
49        let _ = handle.join();
50        log::info!("Previous SSE thread exited");
51    } else {
52        log::warn!(
53            "Previous SSE thread still running after {SSE_THREAD_SHUTDOWN_TIMEOUT:?}, proceeding with new connection"
54        );
55    }
56}

59    /// Check if any profile is currently connected to its sync server.
60    ///
61    /// ### Returns
62    /// - `true`: At least one profile reports a connected status.
63    /// - `false`: No profile is connected.
64    pub fn is_connected(&self, cx: &App) -> bool {
65        let shared = self.shared_state(cx);
66        let states = shared.sync_states.read();
67        states
68            .values()
69            .any(|s| s.connection_status.lock().is_connected())
70    }

72    /// Signal the old SSE worker's shutdown flag, allocate a fresh SSE state
73    /// slot, and validate that the profile is ready to connect.
74    ///
75    /// ### Arguments
76    /// - `profile_id`: The profile to restart.
77    ///
78    /// ### Returns
79    /// - `Some(SseRestartSetup)`: Setup resources ready for the caller to spawn a thread.
80    /// - `None`: The profile was not found, is inactive, or the master switch is off.
81    fn prepare_sse_restart(&mut self, profile_id: &str) -> Option<SseRestartSetup> {
82        if let Some(state) = self.sse_states.get(profile_id)
83            && let Some(ref shutdown_flag) = state.sse_shutdown_flag
84        {
85            log::info!("Profile '{profile_id}': signaling SSE shutdown");
86            shutdown_flag.store(true, Ordering::Relaxed);
87        }
88        let old_handle = self
89            .sse_states
90            .get(profile_id)
91            .map(|s| s.sse_thread_handle.lock().take())
92            .unwrap_or(None);
93        let (sse_tx, sse_rx) = std::sync::mpsc::channel();
94        let sse_shutdown_flag = Arc::new(AtomicBool::new(false));
95        let mut new_state = SseState::new();
96        new_state.sse_events = Some(sse_rx);
97        new_state.sse_event_tx = Some(sse_tx.clone());
98        new_state.sse_shutdown_flag = Some(sse_shutdown_flag.clone());
99        let handle_storage = Arc::clone(&new_state.sse_thread_handle);
100        self.sse_states.insert(profile_id.to_string(), new_state);
101
102        let profile = match self
103            .settings
104            .app_settings
105            .synchronization_settings
106            .profiles
107            .iter()
108            .find(|p| p.id == profile_id)
109        {
110            Some(p) => p.clone(),
111            None => {
112                log::warn!("prepare_sse_restart: profile id '{profile_id}' not found in settings");
113                return None;
114            }
115        };
116        let master_on = self
117            .settings
118            .app_settings
119            .synchronization_settings
120            .is_synchronization_activated;
121        if !master_on || !profile.is_active {
122            log::info!(
123                "Profile '{}' not active or master switch off, SSE connection not started",
124                profile.name
125            );
126            return None;
127        }
128        Some(SseRestartSetup {
129            old_handle,
130            sse_tx,
131            sse_shutdown_flag,
132            handle_storage,
133            profile,
134        })
135    }

137    /// Restart the SSE connection for a single profile.
138    ///
139    /// ### Arguments
140    /// - `profile_id`: The profile whose SSE worker should be restarted.
141    /// - `cx`: The context of the application.
142    pub fn restart_sse_connection_for(&mut self, profile_id: &str, cx: &mut Context<Self>) {
143        let Some(SseRestartSetup {
144            old_handle,
145            sse_tx,
146            sse_shutdown_flag,
147            handle_storage,
148            profile,
149        }) = self.prepare_sse_restart(profile_id)
150        else {
151            return;
152        };
153        let shared = self.shared_state(cx);
154        let sync_state = shared.sync_state_for(&profile.id);
155        let sync_status = sync_state.connection_status.clone();
156        let token_state = Arc::clone(&sync_state.token_state);
157        let http_agent = Arc::clone(&shared.http_agent);
158        let pending_shared_files = Arc::clone(&sync_state.pending_shared_files);
159        thread::spawn(move || {
160            wait_for_previous_sse_thread(old_handle);
161            thread::sleep(Duration::from_millis(200));
162            match initial_synchronization(&profile, &token_state, &http_agent) {
163                Ok(_) => {
164                    log::info!(
165                        "Profile '{}': initial sync succeeded, starting new SSE",
166                        profile.name
167                    );
168                    match connect_sse(
169                        &profile,
170                        sse_tx,
171                        sse_shutdown_flag,
172                        sync_status,
173                        &token_state,
174                        &http_agent,
175                        &pending_shared_files,
176                    ) {
177                        Ok(new_handle) => {
178                            *handle_storage.lock() = Some(new_handle);
179                        }
180                        Err(e) => {
181                            log::error!("Profile '{}': failed to start SSE: {e}", profile.name);
182                        }
183                    }
184                }
185                Err(e) => {
186                    log::error!(
187                        "Profile '{}': initial sync failed, not starting SSE: {e}",
188                        profile.name
189                    );
190                }
191            }
192        });
193    }

195    /// Restart the SSE connection for a single profile, showing a progress
196    /// indicator and a success/error notification when the connection attempt
197    /// completes.
198    ///
199    /// ### Arguments
200    /// - `profile_id`: The profile whose SSE worker should be restarted.
201    /// - `window`: The window to attach the progress indicator to.
202    /// - `cx`: The context of the application.
203    pub fn restart_sse_connection_for_with_progress(
204        &mut self,
205        profile_id: &str,
206        window: &mut Window,
207        cx: &mut Context<Self>,
208    ) {
209        let Some(SseRestartSetup {
210            old_handle,
211            sse_tx,
212            sse_shutdown_flag,
213            handle_storage,
214            profile,
215        }) = self.prepare_sse_restart(profile_id)
216        else {
217            return;
218        };
219        let shared = self.shared_state(cx);
220        let sync_state = shared.sync_state_for(&profile.id);
221        let connection_status = sync_state.connection_status.clone();
222        let connecting_since = sync_state.connecting_since.clone();
223        let token_state = Arc::clone(&sync_state.token_state);
224        let http_agent = Arc::clone(&shared.http_agent);
225        let pending_shared_files = Arc::clone(&sync_state.pending_shared_files);
226        let pending_notification = Arc::clone(&sync_state.pending_notification);
227        let device_name = sync_state.device_name.clone();
228        let max_file_size_bytes = Arc::clone(&sync_state.max_file_size_bytes);
229        let profile_name = profile.name.clone();
230
231        set_sync_server_connection_status(&connection_status, SynchronizationStatus::Connecting);
232        *connecting_since.lock() = Some(Instant::now());
233
234        let done = Arc::new(AtomicBool::new(false));
235        let done_for_thread = Arc::clone(&done);
236
237        let cancel_status = connection_status.clone();
238        let cancel_connecting_since = connecting_since.clone();
239        let cancel_callback: Option<CancelCallback> = Some(Box::new(move |_window, _cx| {
240            set_sync_server_connection_status(&cancel_status, SynchronizationStatus::Disconnected);
241            *cancel_connecting_since.lock() = None;
242        }));
243
244        let progress = start_progress(
245            window,
246            cx,
247            format!("Connecting to {profile_name}...").into(),
248            cancel_callback,
249        );
250        let cancel_flag = progress.cancel_flag();
251        let cancel_flag_for_thread = Arc::clone(&cancel_flag);
252
253        thread::spawn(move || {
254            wait_for_previous_sse_thread(old_handle);
255            thread::sleep(Duration::from_millis(200));
256
257            if cancel_flag_for_thread.load(Ordering::Acquire) {
258                done_for_thread.store(true, Ordering::Release);
259                return;
260            }
261
262            let (notification, status) =
263                match initial_synchronization(&profile, &token_state, &http_agent) {
264                    Ok(begin_response) => {
265                        store_server_max_file_size(
266                            &max_file_size_bytes,
267                            begin_response.max_file_size_bytes,
268                        );
269                        *device_name.lock() = Some(begin_response.device_name.clone());
270                        *pending_shared_files.lock() = begin_response.shares;
271                        let msg = format!(
272                            "{profile_name}: Connection successful as {}",
273                            begin_response.device_name
274                        );
275                        match connect_sse(
276                            &profile,
277                            sse_tx,
278                            sse_shutdown_flag,
279                            connection_status.clone(),
280                            &token_state,
281                            &http_agent,
282                            &pending_shared_files,
283                        ) {
284                            Ok(new_handle) => {
285                                *handle_storage.lock() = Some(new_handle);
286                            }
287                            Err(e) => {
288                                log::error!("Profile '{}': failed to start SSE: {e}", profile.name);
289                            }
290                        }
291                        (
292                            (NotificationType::Success, SharedString::from(msg)),
293                            SynchronizationStatus::Connected,
294                        )
295                    }
296                    Err(e) => {
297                        let msg = format!("{profile_name}: Connection failed: {e}");
298                        (
299                            (NotificationType::Error, SharedString::from(msg)),
300                            SynchronizationStatus::from_error(&e),
301                        )
302                    }
303                };
304            set_sync_server_connection_status(&connection_status, status);
305            *connecting_since.lock() = None;
306            *pending_notification.lock() = Some(notification);
307            done_for_thread.store(true, Ordering::Release);
308        });
309
310        window
311            .spawn(cx, async move |async_cx| {
312                let _progress = progress;
313                loop {
314                    async_cx
315                        .background_executor()
316                        .timer(Duration::from_millis(100))
317                        .await;
318                    if done.load(Ordering::Acquire) || cancel_flag.load(Ordering::Acquire) {
319                        break;
320                    }
321                }
322            })
323            .detach();
324    }

326    /// Handle a single SSE event for a specific profile.
327    ///
328    /// ### Arguments
329    /// - `profile_id`: The profile that produced the event.
330    /// - `event`: The SSE event to handle.
331    /// - `window`: The window to show notifications in.
332    /// - `cx`: The application context.
333    pub fn handle_sse_event_for(
334        &mut self,
335        profile_id: &ProfileId,
336        event: SseEvent,
337        window: &mut Window,
338        cx: &mut Context<Self>,
339    ) {
340        let now = Instant::now();
341        let state = self.sse_states.entry(profile_id.clone()).or_default();
342        if let Some(last_time) = state.last_sse_event
343            && now.duration_since(last_time) < Duration::from_millis(500)
344        {
345            return;
346        }
347        state.last_sse_event = Some(now);
348        let sync_state = self.shared_state(cx).sync_state_for(profile_id);
349        match event {
350            SseEvent::Heartbeat { timestamp } => {
351                log::debug!("SSE heartbeat received for profile '{profile_id}': {timestamp}");
352                let was_disconnected = !sync_state.connection_status.lock().is_connected();
353                *sync_state.last_heartbeat.lock() = Some(now);
354                if was_disconnected {
355                    *sync_state.connection_status.lock() = SynchronizationStatus::Connected;
356                    log::info!(
357                        "Profile '{profile_id}': connection restored on heartbeat after timeout"
358                    );
359                }
360            }
361            SseEvent::ShareAvailable(notification) => {
362                log::debug!(
363                    "Share doorbell on UI tick (share_id={})",
364                    notification.share_id
365                );
366                let message = SharedString::from("New file received".to_string());
367                window.push_notification((NotificationType::Info, message), cx);
368            }
369            SseEvent::Error(err) => {
370                log::error!("SSE error for profile '{profile_id}': {err}");
371            }
372        }
373    }

375    /// Handle an SSE event using the primary profile.
376    ///
377    /// ### Arguments
378    /// - `event`: The SSE event to handle.
379    /// - `window`: The window to show notifications in.
380    /// - `cx`: The application context.
381    pub fn handle_sse_event(
382        &mut self,
383        event: SseEvent,
384        window: &mut Window,
385        cx: &mut Context<Self>,
386    ) {
387        let primary_id = self
388            .settings
389            .app_settings
390            .synchronization_settings
391            .primary_profile()
392            .map(|p| p.id.clone())
393            .unwrap_or_default();
394        self.handle_sse_event_for(&primary_id, event, window, cx);
395    }

397    /// Drain pending SSE events from every profile's channel and dispatch them.
398    ///
399    /// ### Arguments
400    /// - `window`: The window to handle events in.
401    /// - `cx`: The application context.
402    pub fn process_sse_events(&mut self, window: &mut Window, cx: &mut Context<Self>) {
403        let profile_ids: Vec<ProfileId> = self.sse_states.keys().cloned().collect();
404        for profile_id in profile_ids {
405            let events = self
406                .sse_states
407                .get(&profile_id)
408                .map(|s| collect_events(&s.sse_events))
409                .unwrap_or_default();
410            for event in events {
411                self.handle_sse_event_for(&profile_id, event, window, cx);
412            }
413        }
414    }

428    /// Initialize globals and open a test window with a `gpui_component::Root`-mounted `Fulgur`.
429    ///
430    /// The root must be a `gpui_component::Root` (not a bare `EmptyView`) because
431    /// `window.push_notification(...)` asserts that the first layer is a Root.
432    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
433        cx.update(|cx| {
434            gpui_component::init(cx);
435            let mut settings = Settings::new();
436            settings.editor_settings.watch_files = false;
437            let pending_files: Arc<Mutex<Vec<PathBuf>>> = Arc::new(Mutex::new(Vec::new()));
438            cx.set_global(SharedAppState::new(settings, pending_files));
439            cx.set_global(WindowManager::new());
440        });
441        let fulgur_slot: RefCell<Option<Entity<Fulgur>>> = RefCell::new(None);
442        let window = cx
443            .update(|cx| {
444                cx.open_window(WindowOptions::default(), |window, cx| {
445                    let window_id = window.window_handle().window_id();
446                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
447                    *fulgur_slot.borrow_mut() = Some(fulgur.clone());
448                    cx.new(|cx| gpui_component::Root::new(fulgur, window, cx))
449                })
450            })
451            .expect("failed to open test window");
452        let visual_cx = VisualTestContext::from_window(window.into(), cx);
453        visual_cx.run_until_parked();
454        let fulgur = fulgur_slot
455            .into_inner()
456            .expect("failed to capture Fulgur entity");
457        (fulgur, visual_cx)
458    }

686    /// Insert (or replace) the SSE channel for the empty profile id used by
687    /// the Phase 1 single-profile tests. Returns the `Sender` for the test to
688    /// emit events through.
689    fn install_test_sse_channel(this: &mut Fulgur) -> std::sync::mpsc::Sender<SseEvent> {
690        let (tx, rx) = std::sync::mpsc::channel();
691        let mut state = SseState::new();
692        state.sse_events = Some(rx);
693        this.sse_states.insert(String::new(), state);
694        tx
