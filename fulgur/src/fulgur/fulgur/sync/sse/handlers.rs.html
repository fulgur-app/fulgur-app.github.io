19/// Maximum time to wait for the previous SSE thread to exit before starting a new one
20const SSE_THREAD_SHUTDOWN_TIMEOUT: Duration = Duration::from_secs(5);

23    /// Check if the Fulgur is connected to the sync server
24    ///
25    /// ### Returns
26    /// - `true` if Fulgur is connected to the sync server, `false` otherwise
27    pub fn is_connected(&self, cx: &App) -> bool {
28        self.shared_state(cx)
29            .sync_state
30            .connection_status
31            .lock()
32            .is_connected()
33    }

35    /// Restart the SSE connection with new settings
36    ///
37    /// ### Description
38    /// Stops the current SSE connection and starts a new one with the updated settings.
39    /// Waits for the previous SSE thread to exit (bounded by `SSE_THREAD_SHUTDOWN_TIMEOUT`)
40    /// in a background thread to avoid blocking the UI.
41    ///
42    /// Should be called when synchronization settings (server URL, email, or key) change.
43    ///
44    /// ### Arguments
45    /// - `cx`: The context of the application.
46    pub fn restart_sse_connection(&mut self, cx: &mut Context<Self>) {
47        if let Some(ref shutdown_flag) = self.sse_state.sse_shutdown_flag {
48            log::info!("Signaling SSE connection to shutdown...");
49            shutdown_flag.store(true, Ordering::Relaxed);
50        }
51        let old_handle = self.sse_state.sse_thread_handle.lock().take();
52        let (sse_tx, sse_rx) = std::sync::mpsc::channel();
53        let sse_shutdown_flag = Arc::new(AtomicBool::new(false));
54        self.sse_state.sse_events = Some(sse_rx);
55        self.sse_state.sse_event_tx = Some(sse_tx.clone());
56        self.sse_state.sse_shutdown_flag = Some(sse_shutdown_flag.clone());
57        if self
58            .settings
59            .app_settings
60            .synchronization_settings
61            .is_synchronization_activated
62        {
63            let settings = self.settings.clone();
64            let sync_status = self.shared_state(cx).sync_state.connection_status.clone();
65            let token_state = Arc::clone(&self.shared_state(cx).sync_state.token_state);
66            let http_agent = Arc::clone(&self.shared_state(cx).http_agent);
67            let pending_shared_files =
68                Arc::clone(&self.shared_state(cx).sync_state.pending_shared_files);
69            let handle_storage = Arc::clone(&self.sse_state.sse_thread_handle);
70            thread::spawn(move || {
71                if let Some(handle) = old_handle {
72                    let deadline = Instant::now() + SSE_THREAD_SHUTDOWN_TIMEOUT;
73                    while !handle.is_finished() && Instant::now() < deadline {
74                        thread::sleep(Duration::from_millis(100));
75                    }
76                    if handle.is_finished() {
77                        let _ = handle.join();
78                        log::info!("Previous SSE thread exited");
79                    } else {
80                        log::warn!(
81                            "Previous SSE thread still running after {:?}, proceeding with new connection",
82                            SSE_THREAD_SHUTDOWN_TIMEOUT
83                        );
84                    }
85                }
86                thread::sleep(Duration::from_millis(200));
87                match initial_synchronization(
88                    &settings.app_settings.synchronization_settings,
89                    Arc::clone(&token_state),
90                    &http_agent,
91                ) {
92                    Ok(_) => {
93                        log::info!("Initial sync succeeded, starting new SSE connection");
94                        match connect_sse(
95                            &settings.app_settings.synchronization_settings,
96                            sse_tx,
97                            sse_shutdown_flag,
98                            sync_status,
99                            token_state,
100                            http_agent,
101                            pending_shared_files,
102                        ) {
103                            Ok(new_handle) => {
104                                *handle_storage.lock() = Some(new_handle);
105                            }
106                            Err(e) => {
107                                log::error!("Failed to start new SSE connection: {}", e);
108                            }
109                        }
110                    }
111                    Err(e) => {
112                        log::error!("Initial sync failed, not starting SSE: {}", e);
113                    }
114                }
115            });
116        } else {
117            log::info!("Synchronization is not activated, SSE connection not started");
118        }
119    }

121    /// Handle SSE (Server-Sent Events) from the sync server
122    ///
123    /// ### Arguments
124    /// - `event`: The SSE event to handle
125    /// - `window`: The window to show notifications in
126    /// - `cx`: The application context
127    pub fn handle_sse_event(
128        &mut self,
129        event: SseEvent,
130        window: &mut Window,
131        cx: &mut Context<Self>,
132    ) {
133        let now = Instant::now();
134        if let Some(last_time) = self.sse_state.last_sse_event
135            && now.duration_since(last_time) < Duration::from_millis(500)
136        {
137            return;
138        }
139        self.sse_state.last_sse_event = Some(now);
140        match event {
141            SseEvent::Heartbeat { timestamp } => {
142                log::debug!("SSE heartbeat received: {}", timestamp);
143                let was_disconnected = !self
144                    .shared_state(cx)
145                    .sync_state
146                    .connection_status
147                    .lock()
148                    .is_connected();
149                {
150                    let mut last_heartbeat = self.shared_state(cx).sync_state.last_heartbeat.lock();
151                    *last_heartbeat = Some(now);
152                }
153                if was_disconnected {
154                    *self.shared_state(cx).sync_state.connection_status.lock() =
155                        SynchronizationStatus::Connected;
156                    log::info!("Connection restored - heartbeat received after timeout");
157                }
158            }
159            SseEvent::ShareAvailable(notification) => {
160                log::debug!(
161                    "Share doorbell on UI tick (share_id={})",
162                    notification.share_id
163                );
164                let message = SharedString::from("New file received".to_string());
165                window.push_notification((NotificationType::Info, message), cx);
166            }
167            SseEvent::Error(err) => {
168                log::error!("SSE error: {}", err);
169            }
170        }
171    }

173    /// Collect and process Server-Sent Events from the sync server:
174    /// - Heartbeat: Periodic keepalive messages to detect connection timeouts
175    /// - ShareAvailable: Another device has shared a file (triggers file download and decryption)
176    /// - Error: Connection or server errors (updates connection status in UI)
177    ///
178    /// ### Arguments
179    /// - `window`: The window to handle events in
180    /// - `cx`: The application context
181    pub fn process_sse_events(&mut self, window: &mut Window, cx: &mut Context<Self>) {
182        let sse_events = collect_events(&self.sse_state.sse_events);
183        for event in sse_events {
184            self.handle_sse_event(event, window, cx);
185        }
186    }
187}

200    /// Initialize globals and open a test window with a `gpui_component::Root`-mounted `Fulgur`.
201    ///
202    /// The root must be a `gpui_component::Root` (not a bare `EmptyView`) because
203    /// `window.push_notification(...)` asserts that the first layer is a Root.
204    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
205        cx.update(|cx| {
206            gpui_component::init(cx);
207            let mut settings = Settings::new();
208            settings.editor_settings.watch_files = false;
209            let pending_files: Arc<Mutex<Vec<PathBuf>>> = Arc::new(Mutex::new(Vec::new()));
210            cx.set_global(SharedAppState::new(settings, pending_files));
211            cx.set_global(WindowManager::new());
212        });
213        let fulgur_slot: RefCell<Option<Entity<Fulgur>>> = RefCell::new(None);
214        let window = cx
215            .update(|cx| {
216                cx.open_window(Default::default(), |window, cx| {
217                    let window_id = window.window_handle().window_id();
218                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
219                    *fulgur_slot.borrow_mut() = Some(fulgur.clone());
220                    cx.new(|cx| gpui_component::Root::new(fulgur, window, cx))
221                })
222            })
223            .expect("failed to open test window");
224        let visual_cx = VisualTestContext::from_window(window.into(), cx);
225        visual_cx.run_until_parked();
226        let fulgur = fulgur_slot
227            .into_inner()
228            .expect("failed to capture Fulgur entity");
229        (fulgur, visual_cx)
230    }
