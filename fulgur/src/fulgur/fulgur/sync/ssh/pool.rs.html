12/// Time after which an idle SSH session is considered stale and discarded.
13pub const SSH_SESSION_IDLE_TTL: Duration = Duration::from_secs(60 * 60);
14
15/// Maximum number of idle sessions held by the pool at once.
16const SSH_SESSION_POOL_CAPACITY: usize = 64;
17
18/// Identifier for a pooled session, derived from connection coordinates and password digest.
19///
20/// Including a SHA-256 digest of the password ensures rotated credentials never
21/// silently reuse a session authenticated under the previous password.
22#[derive(Debug, Clone, PartialEq, Eq, Hash)]
23pub struct SshSessionKey {
24    pub host: String,
25    pub port: u16,
26    pub user: String,
27    pub password_hash: [u8; 32],
28}
29
30impl SshSessionKey {
31    /// Build a pool key from a `(host, port, user)` triple and a password.
32    ///
33    /// ### Arguments
34    /// - `host`: Hostname or IP of the remote server.
35    /// - `port`: SSH port.
36    /// - `user`: Username used for authentication.
37    /// - `password`: Password used for authentication; its SHA-256 digest is
38    ///   stored in the key.
39    ///
40    /// ### Returns
41    /// - `Self`: The composite key used to look up cached sessions.
42    pub fn new(
43        host: impl Into<String>,
44        port: u16,
45        user: impl Into<String>,
46        password: &Zeroizing<String>,
47    ) -> Self {
48        let mut hasher = Sha256::new();
49        hasher.update(password.as_bytes());
50        let mut password_hash = [0u8; 32];
51        password_hash.copy_from_slice(&hasher.finalize());
52        Self {
53            host: host.into(),
54            port,
55            user: user.into(),
56            password_hash,
57        }
58    }
59}

61/// Pool slot holding an idle SSH session and its last-used timestamp.
62struct SlotEntry {
63    session: SshSession,
64    last_used: Instant,
65}
66
67/// Process-wide pool of authenticated SSH sessions keyed by `(host, port, user, password-hash)`.
68///
69/// Idle sessions are checked out on demand and returned on guard drop. Stale entries are
70/// evicted lazily on every `take`/`put` based on a configurable idle TTL.
71pub struct SshSessionPool {
72    entries: Mutex<HashMap<SshSessionKey, SlotEntry>>,
73    idle_ttl: Duration,
74}
75
76impl SshSessionPool {
77    /// Create an empty pool with the default idle TTL.
78    ///
79    /// ### Returns
80    /// - `Self`: Empty pool ready to accept sessions.
81    pub fn new() -> Self {
82        Self::with_idle_ttl(SSH_SESSION_IDLE_TTL)
83    }
84
85    /// Create an empty pool with a custom idle TTL (used by tests).
86    ///
87    /// ### Arguments
88    /// - `idle_ttl`: Maximum time an idle session may remain in the pool.
89    ///
90    /// ### Returns
91    /// - `Self`: Empty pool with the supplied TTL.
92    pub fn with_idle_ttl(idle_ttl: Duration) -> Self {
93        Self {
94            entries: Mutex::new(HashMap::new()),
95            idle_ttl,
96        }
97    }
98
99    /// Drop every cached session immediately.
100    pub fn clear(&self) {
101        self.entries.lock().clear();
102    }
103
104    /// Drop sessions for a `(host, port, user)` triple regardless of stored password digest.
105    ///
106    /// ### Arguments
107    /// - `cred_key`: Credential identifier whose sessions should be discarded.
108    pub fn invalidate_by_credential(&self, cred_key: &SshCredKey) {
109        let mut entries = self.entries.lock();
110        entries.retain(|key, _| {
111            !(key.host == cred_key.host && key.port == cred_key.port && key.user == cred_key.user)
112        });
113    }
114
115    /// Number of idle sessions currently held by the pool (used by tests).
116    ///
117    /// ### Returns
118    /// - `usize`: Count of cached entries after stale eviction.
119    #[cfg(test)]
120    pub fn len(&self) -> usize {
121        let mut entries = self.entries.lock();
122        Self::evict_stale(&mut entries, self.idle_ttl);
123        entries.len()
124    }
125
126    /// Whether the pool currently holds no idle sessions (used by tests).
127    ///
128    /// ### Returns
129    /// - `true`: No cached entries remain after stale eviction.
130    /// - `false`: At least one session is cached.
131    #[cfg(test)]
132    pub fn is_empty(&self) -> bool {
133        self.len() == 0
134    }
135
136    /// Try to remove and return a session for the given key, evicting stale entries first.
137    ///
138    /// ### Arguments
139    /// - `key`: Composite key identifying the session.
140    ///
141    /// ### Returns
142    /// - `Some(SshSession)`: A fresh enough idle session was available.
143    /// - `None`: No session is cached for this key, or the cached one had expired.
144    fn take(&self, key: &SshSessionKey) -> Option<SshSession> {
145        let mut entries = self.entries.lock();
146        Self::evict_stale(&mut entries, self.idle_ttl);
147        entries.remove(key).map(|slot| slot.session)
148    }
149
150    /// Insert a session into the pool, evicting the oldest entry if capacity is reached.
151    ///
152    /// ### Arguments
153    /// - `key`: Composite key identifying the session.
154    /// - `session`: Session to cache.
155    fn put(&self, key: SshSessionKey, session: SshSession) {
156        let mut entries = self.entries.lock();
157        Self::evict_stale(&mut entries, self.idle_ttl);
158        if entries.len() >= SSH_SESSION_POOL_CAPACITY
159            && let Some(oldest_key) = entries
160                .iter()
161                .min_by_key(|(_, slot)| slot.last_used)
162                .map(|(k, _)| k.clone())
163        {
164            entries.remove(&oldest_key);
165        }
166        entries.insert(
167            key,
168            SlotEntry {
169                session,
170                last_used: Instant::now(),
171            },
172        );
173    }
174
175    /// Drop entries whose last-used timestamp is older than `idle_ttl`.
176    ///
177    /// ### Arguments
178    /// - `entries`: Locked pool map to mutate in place.
179    /// - `idle_ttl`: Maximum allowed idle duration.
180    fn evict_stale(entries: &mut HashMap<SshSessionKey, SlotEntry>, idle_ttl: Duration) {
181        let now = Instant::now();
182        entries.retain(|_, slot| now.saturating_duration_since(slot.last_used) < idle_ttl);
183    }
184
185    /// Reuse a cached session if one is available, otherwise establish a new one.
186    /// ### Arguments
187    /// - `spec`: Parsed remote specification supplying host and port.
188    /// - `user`: Resolved username; must not be empty.
189    /// - `password`: Session-scoped password used to compute the pool key and, on
190    ///   cache miss, to authenticate.
191    /// - `host_key_cb`: Called only on cache miss when the host key is unknown.
192    ///
193    /// ### Errors
194    /// Returns an `SshError` on connect, handshake, host-key check, authentication,
195    /// or SFTP initialization failures when establishing a new session.
196    ///
197    /// ### Returns
198    /// - `Ok(PooledSession)`: A guard owning the checked-out session. Drop returns
199    ///   it to the pool; `invalidate()` discards it instead.
200    /// - `Err(SshError)`: Any failure during connect, handshake, host-key check,
201    ///   auth, or SFTP init when establishing a new session.
202    pub fn checkout_or_connect(
203        self: &Arc<Self>,
204        spec: &RemoteSpec,
205        user: &str,
206        password: &Zeroizing<String>,
207        host_key_cb: impl FnOnce(&str, &str, u16) -> HostKeyDecision,
208    ) -> Result<PooledSession, SshError> {
209        let key = SshSessionKey::new(&spec.host, spec.port, user, password);
210
211        if let Some(session) = self.take(&key)
212            && session.is_authenticated()
213        {
214            return Ok(PooledSession {
215                inner: Some(SessionWithKey { key, session }),
216                pool: Arc::clone(self),
217            });
218        }
219
220        let session = connect(spec, user, password, host_key_cb)?;
221        Ok(PooledSession {
222            inner: Some(SessionWithKey { key, session }),
223            pool: Arc::clone(self),
224        })
225    }
226}

234/// Internal pairing of a checked-out session with the key it should return to.
235struct SessionWithKey {
236    key: SshSessionKey,
237    session: SshSession,
238}
239
240/// RAII guard owning an SSH session checked out from an `SshSessionPool`.
241///
242/// Drop returns the session to the pool. Call `invalidate()` after a transport
243/// failure to discard the session instead of recycling a broken connection.
244pub struct PooledSession {
245    inner: Option<SessionWithKey>,
246    pool: Arc<SshSessionPool>,
247}
248
249impl PooledSession {
250    /// Borrow the underlying SSH session for SFTP operations.
251    ///
252    /// ### Returns
253    /// - `&SshSession`: Reference to the live SSH session and its SFTP subsystem.
254    ///
255    /// ### Panics
256    /// Panics if the session has already been consumed (e.g. via `discard` or `Drop`).
257    pub fn session(&self) -> &SshSession {
258        &self
259            .inner
260            .as_ref()
261            .expect("pooled session already consumed") //TODO: remove this
262            .session
263    }
264
265    /// Discard the session without returning it to the pool.
266    ///
267    /// Use this after any transport-level error so a broken session is never
268    /// served again.
269    pub fn invalidate(mut self) {
270        self.inner = None;
271    }
272}
273
274impl Drop for PooledSession {
275    fn drop(&mut self) {
276        if let Some(SessionWithKey { key, session }) = self.inner.take() {
277            self.pool.put(key, session);
278        }
279    }
280}

287    #[test]
288    fn key_includes_password_digest() {
289        let pwd_a = Zeroizing::new("hunter2".to_string());
290        let pwd_b = Zeroizing::new("trustno1".to_string());
291        let key_a = SshSessionKey::new("example.com", 22, "alice", &pwd_a);
292        let key_b = SshSessionKey::new("example.com", 22, "alice", &pwd_b);
293        assert_ne!(key_a, key_b);
294        assert_ne!(key_a.password_hash, [0u8; 32]);
295    }
296
297    #[test]
298    fn key_is_stable_for_same_password() {
299        let pwd = Zeroizing::new("hunter2".to_string());
300        let key_a = SshSessionKey::new("example.com", 22, "alice", &pwd);
301        let key_b = SshSessionKey::new("example.com", 22, "alice", &pwd);
302        assert_eq!(key_a, key_b);
303    }

306    fn invalidate_by_credential_drops_matching_entries() {

309        // Without an actual session we cannot exercise put/take here; just confirm
310        // the API is callable on an empty pool without panicking.
