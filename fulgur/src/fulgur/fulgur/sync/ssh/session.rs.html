59/// Connect to a remote host over SSH and open an SFTP subsystem.
60///
61/// ### Description
62/// Performs TCP connect (10 s timeout), SSH handshake, host-key verification against
63/// `~/.ssh/known_hosts` (TOFU policy), password authentication, and SFTP subsystem init.
64/// `host_key_cb` is called synchronously when the server's key is not yet in `known_hosts`;
65/// on `Accept` the key is appended to the file.
66///
67/// ### Arguments
68/// - `spec`: Parsed remote specification supplying host and port.
69/// - `user`: Resolved username; must not be empty.
70/// - `password`: Session-scoped password, zeroed on drop by the `Zeroizing` wrapper.
71/// - `host_key_cb`: Called with `(fingerprint_sha256_hex, host, port)` when the host key is unknown.
72///
73/// ### Errors
74/// Returns an `SshError` on TCP connect failure, SSH handshake failure, host-key
75/// rejection, password authentication failure, or SFTP subsystem init failure.
76///
77/// ### Returns
78/// - `Ok(SshSession)`: Ready session with an open SFTP subsystem.
79/// - `Err(SshError)`: Any failure during TCP connect, handshake, host-key check, auth, or SFTP init.
80pub fn connect(
81    spec: &RemoteSpec,
82    user: &str,
83    password: &Zeroizing<String>,
84    host_key_cb: impl FnOnce(&str, &str, u16) -> HostKeyDecision,
85) -> Result<SshSession, SshError> {
86    let addr_str = format!("{}:{}", spec.host, spec.port);
87    let addrs: Vec<_> = addr_str
88        .to_socket_addrs()
89        .map_err(|e| SshError::ConnectionFailed(format!("Cannot resolve {addr_str}: {e}")))?
90        .collect();
91    if addrs.is_empty() {
92        return Err(SshError::ConnectionFailed(format!(
93            "No addresses found for {addr_str}"
94        )));
95    }
96    // Try each resolved address in order (DNS may return IPv6 before IPv4; fall through on
97    // EHOSTUNREACH so the connection succeeds even when only one address family is routable).
98    let mut last_err = None;
99    let tcp = addrs
100        .into_iter()
101        .find_map(|addr| {
102            match TcpStream::connect_timeout(&addr, Duration::from_secs(CONNECT_TIMEOUT_SECS)) {
103                Ok(stream) => Some(stream),
104                Err(e) => {
105                    last_err = Some(e);
106                    None
107                }
108            }
109        })
110        .ok_or_else(|| {
111            SshError::ConnectionFailed(
112                last_err.map_or_else(|| "unknown error".to_string(), |e| e.to_string()),
113            )
114        })?;
115
116    let mut session = Session::new().map_err(|e| SshError::ConnectionFailed(e.to_string()))?;
117    if let Some(hostkey_prefs) = hostkey_method_preferences_from_known_hosts(&spec.host, spec.port)
118    {
119        let _ = session.method_pref(ssh2::MethodType::HostKey, &hostkey_prefs);
120    }
121    session.set_timeout(SESSION_TIMEOUT_MS);
122    session.set_tcp_stream(tcp);
123    session
124        .handshake()
125        .map_err(|e| SshError::ConnectionFailed(e.to_string()))?;
126
127    check_host_key(&session, &spec.host, spec.port, host_key_cb)?;
128
129    session
130        .userauth_password(user, password.as_str())
131        .map_err(|e| map_password_auth_error(&e))?;
132
133    if !session.authenticated() {
134        return Err(SshError::AuthFailed);
135    }
136
137    let sftp = session
138        .sftp()
139        .map_err(|e| SshError::SftpError(e.to_string()))?;
140
141    Ok(SshSession { session, sftp })
142}

144/// Classify `userauth_password` failures as credential rejection or transport/session errors.
145///
146/// ### Arguments
147/// - `error`: Raw ssh2 error returned from `userauth_password`.
148///
149/// ### Returns
150/// - `SshError::AuthFailed`: Password authentication was explicitly rejected by the server.
151/// - `SshError::ConnectionFailed`: Any non-auth failure during the authentication request.
152fn map_password_auth_error(error: &ssh2::Error) -> SshError {
153    match error.code() {
154        ssh2::ErrorCode::Session(code) if code == LIBSSH2_AUTHENTICATION_FAILED_CODE => {
155            SshError::AuthFailed
156        }
157        _ => SshError::ConnectionFailed(format!("SSH authentication request failed: {error}")),
158    }
159}

161/// Verify the server's host key against `~/.ssh/known_hosts`.
162///
163/// ### Description
164/// Match → proceed silently. Mismatch → hard error (user must edit `known_hosts` manually).
165/// NotFound/Failure → call `host_key_cb`; on `Accept` append the entry to the file.
166///
167/// ### Arguments
168/// - `session`: Active SSH session after handshake, used to retrieve the server's host key.
169/// - `host`: Hostname string, used for `known_hosts` lookup and callback.
170/// - `port`: SSH port, used for `known_hosts` lookup and callback.
171/// - `host_key_cb`: Called with `(fingerprint_sha256_hex, host, port)` when the key is not found.
172///
173/// ### Returns
174/// - `Ok(())`: Host key verified or accepted by the user.
175/// - `Err(SshError::HostKeyMismatch)`: Key in `known_hosts` does not match the server.
176/// - `Err(SshError::ConnectionFailed)`: Key rejected by the user or I/O error on `known_hosts`.
177fn check_host_key(
178    session: &Session,
179    host: &str,
180    port: u16,
181    host_key_cb: impl FnOnce(&str, &str, u16) -> HostKeyDecision,
182) -> Result<(), SshError> {
183    let kh_path = known_hosts_path();
184    let mut known_hosts = session
185        .known_hosts()
186        .map_err(|e| SshError::ConnectionFailed(e.to_string()))?;
187
188    if kh_path.exists() {
189        known_hosts
190            .read_file(&kh_path, ssh2::KnownHostFileKind::OpenSSH)
191            .map_err(|e| SshError::ConnectionFailed(format!("Failed to read known_hosts: {e}")))?;
192    }
193
194    let (key, key_type) = session
195        .host_key()
196        .ok_or_else(|| SshError::ConnectionFailed("Server provided no host key".to_string()))?;
197
198    match resolve_known_host_check_result_with_known_hosts_fallback(&known_hosts, host, port, key) {
199        ssh2::CheckResult::Match => Ok(()),
200        ssh2::CheckResult::Mismatch => Err(SshError::HostKeyMismatch {
201            host: host.to_string(),
202            port,
203        }),
204        ssh2::CheckResult::NotFound | ssh2::CheckResult::Failure => {
205            let fingerprint = sha256_fingerprint(key);
206            match host_key_cb(&fingerprint, host, port) {
207                HostKeyDecision::Reject => Err(SshError::ConnectionFailed(format!(
208                    "Host key rejected for {host}:{port}"
209                ))),
210                HostKeyDecision::Accept => {
211                    ensure_ssh_dir()?;
212                    let known_host = known_hosts_entry_host(host, port);
213                    known_hosts
214                        .add(&known_host, key, "", host_key_type_to_format(key_type))
215                        .map_err(|e| {
216                            SshError::ConnectionFailed(format!("Failed to add host key: {e}"))
217                        })?;
218                    known_hosts
219                        .write_file(&kh_path, ssh2::KnownHostFileKind::OpenSSH)
220                        .map_err(|e| {
221                            SshError::ConnectionFailed(format!("Failed to write known_hosts: {e}"))
222                        })?;
223                    set_file_permissions_600(&kh_path);
224                    Ok(())
225                }
226            }
227        }
228    }
229}

231/// Resolve host-key check result with a pure-Rust `known_hosts` fallback.
232///
233/// ### Description
234/// The primary check uses `libssh2` APIs. On some real-world `known_hosts` layouts, those APIs can
235/// return non-match states even when OpenSSH would accept the same host key. For non-match results,
236/// this method parses `known_hosts` with `ssh-key` and refines the decision when it can.
237///
238/// ### Arguments
239/// - `known_hosts`: Loaded known-hosts collection.
240/// - `host`: Hostname or IP used for the SSH connection.
241/// - `port`: SSH port used for the SSH connection.
242/// - `key`: Raw server host key returned by libssh2.
243///
244/// ### Returns
245/// - `ssh2::CheckResult`: Resolved check result after optional fallback refinement.
246fn resolve_known_host_check_result_with_known_hosts_fallback(
247    known_hosts: &ssh2::KnownHosts,
248    host: &str,
249    port: u16,
250    key: &[u8],
251) -> ssh2::CheckResult {
252    let primary = resolve_known_host_check_result(known_hosts, host, port, key);
253    if matches!(primary, ssh2::CheckResult::Match) {
254        return primary;
255    }
256
257    let fallback = check_known_hosts_with_parser(host, port, key, &known_hosts_path());
258    match fallback {
259        Some(ssh2::CheckResult::Match) => ssh2::CheckResult::Match,
260        Some(ssh2::CheckResult::Mismatch) => ssh2::CheckResult::Mismatch,
261        Some(ssh2::CheckResult::NotFound) => {
262            if matches!(primary, ssh2::CheckResult::Failure) {
263                ssh2::CheckResult::NotFound
264            } else {
265                primary
266            }
267        }
268        Some(ssh2::CheckResult::Failure) | None => primary,
269    }
270}

272/// Resolve host-key check result across host representations used by OpenSSH.
273///
274/// ### Description
275/// `known_hosts` entries may exist as plain hostnames/IPs, bracketed host:port forms,
276/// or be interpreted differently by `check` vs `check_port`. To stay compatible with
277/// OpenSSH behavior, this method checks all relevant forms and accepts on any match.
278///
279/// ### Arguments
280/// - `known_hosts`: Loaded known-hosts collection.
281/// - `host`: Hostname or IP used for the SSH connection.
282/// - `port`: SSH port used for the SSH connection.
283/// - `key`: Raw server host key returned by libssh2.
284///
285/// ### Returns
286/// - `ssh2::CheckResult::Match`: Any representation matched.
287/// - `ssh2::CheckResult::Mismatch`: No matches and at least one representation mismatched.
288/// - `ssh2::CheckResult::NotFound`: No matches/mismatches and at least one representation was missing.
289/// - `ssh2::CheckResult::Failure`: All checks failed unexpectedly.
290fn resolve_known_host_check_result(
291    known_hosts: &ssh2::KnownHosts,
292    host: &str,
293    port: u16,
294    key: &[u8],
295) -> ssh2::CheckResult {
296    let bracket_host = format!("[{host}]:{port}");
297    if port == 22 {
298        aggregate_check_results([
299            known_hosts.check_port(host, port, key),
300            known_hosts.check(host, key),
301            known_hosts.check(&bracket_host, key),
302        ])
303    } else {
304        aggregate_check_results([
305            known_hosts.check_port(host, port, key),
306            known_hosts.check(&bracket_host, key),
307        ])
308    }
309}

311/// Parse `known_hosts` with `ssh-key` and compare keys with the server key.
312///
313/// ### Arguments
314/// - `host`: Hostname or IP used for the SSH connection.
315/// - `port`: SSH port used for the SSH connection.
316/// - `key`: Raw server host key returned by libssh2.
317/// - `known_hosts_path`: Path to the `known_hosts` file.
318///
319/// ### Returns
320/// - `Some(ssh2::CheckResult)`: Parsed result from known-host entries.
321/// - `None`: The file could not be parsed or the key format is unsupported.
322fn check_known_hosts_with_parser(
323    host: &str,
324    port: u16,
325    key: &[u8],
326    known_hosts_path: &Path,
327) -> Option<ssh2::CheckResult> {
328    let entries = KnownHosts::read_file(known_hosts_path).ok()?;
329    let server_key = PublicKey::from_bytes(key).ok()?;
330    Some(resolve_known_host_check_result_from_entries(
331        &entries,
332        host,
333        port,
334        &server_key,
335    ))
336}

338/// Resolve host-key check result from parsed `known_hosts` entries.
339///
340/// ### Arguments
341/// - `entries`: Parsed `known_hosts` entries.
342/// - `host`: Hostname or IP used for the SSH connection.
343/// - `port`: SSH port used for the SSH connection.
344/// - `server_key`: Server key parsed from libssh2 raw bytes.
345///
346/// ### Returns
347/// - `ssh2::CheckResult::Match`: A matching host entry with an identical key was found.
348/// - `ssh2::CheckResult::Mismatch`: Host entry exists but key differs.
349/// - `ssh2::CheckResult::NotFound`: No host entry matched.
350fn resolve_known_host_check_result_from_entries(
351    entries: &[KnownHostEntry],
352    host: &str,
353    port: u16,
354    server_key: &PublicKey,
355) -> ssh2::CheckResult {
356    let mut saw_host_entry = false;
357    for entry in entries {
358        if !known_host_entry_matches_target(entry, host, port) {
359            continue;
360        }
361
362        saw_host_entry = true;
363        if entry.public_key().key_data() == server_key.key_data() {
364            return ssh2::CheckResult::Match;
365        }
366    }
367
368    if saw_host_entry {
369        ssh2::CheckResult::Mismatch
370    } else {
371        ssh2::CheckResult::NotFound
372    }
373}

375/// Build an SSH host-key algorithm preference from existing `known_hosts` entries.
376///
377/// ### Description
378/// Some servers offer multiple host-key algorithms. OpenSSH tends to prefer the algorithm already
379/// present in `known_hosts`, while libssh2 may negotiate a different one and then fail host-key
380/// validation. This function extracts known key types for the target host and builds a preference
381/// list that places those algorithms first.
382///
383/// ### Arguments
384/// - `host`: Hostname or IP used for the SSH connection.
385/// - `port`: SSH port used for the SSH connection.
386///
387/// ### Returns
388/// - `Some(String)`: Comma-separated `MethodType::HostKey` preference string.
389/// - `None`: No known key types were found or `known_hosts` could not be parsed.
390fn hostkey_method_preferences_from_known_hosts(host: &str, port: u16) -> Option<String> {
391    let known_types = known_host_algorithms_for_target(host, port, &known_hosts_path())?;
392    if known_types.is_empty() {
393        return None;
394    }
395
396    let mut merged: Vec<String> = Vec::new();
397    for known in known_types {
398        push_unique_hostkey_alg(&mut merged, &known);
399    }
400    for alg in default_hostkey_algorithms() {
401        push_unique_hostkey_alg(&mut merged, alg);
402    }
403    Some(merged.join(","))
404}

406/// Extract matching host-key algorithms from parsed `known_hosts` entries.
407///
408/// ### Arguments
409/// - `host`: Hostname or IP used for lookup.
410/// - `port`: SSH port used for lookup.
411/// - `known_hosts_path`: Path to the `known_hosts` file.
412///
413/// ### Returns
414/// - `Some(Vec<String>)`: Distinct algorithm names in first-seen order.
415/// - `None`: The file could not be parsed.
416fn known_host_algorithms_for_target(
417    host: &str,
418    port: u16,
419    known_hosts_path: &Path,
420) -> Option<Vec<String>> {
421    let entries = KnownHosts::read_file(known_hosts_path).ok()?;
422    let mut key_types: Vec<String> = Vec::new();
423    for entry in entries {
424        if known_host_entry_matches_target(&entry, host, port) {
425            push_unique_hostkey_alg(&mut key_types, entry.public_key().algorithm().as_ref());
426        }
427    }
428    Some(key_types)
429}

445/// Evaluate known-host host-pattern matching for a target host and port.
446///
447/// ### Arguments
448/// - `patterns`: Parsed host pattern set for one known-host entry.
449/// - `host`: Hostname or IP used for the SSH connection.
450/// - `port`: SSH port used for the SSH connection.
451///
452/// ### Returns
453/// - `true`: Pattern set matches this target host.
454/// - `false`: Pattern set does not match.
455fn known_host_patterns_match(patterns: &HostPatterns, host: &str, port: u16) -> bool {
456    let candidates = host_match_candidates(host, port);
457    match patterns {
458        HostPatterns::Patterns(patterns) => match_known_host_patterns(patterns, &candidates),
459        HostPatterns::HashedName { salt, hash } => candidates
460            .iter()
461            .any(|candidate| hashed_host_matches(salt, hash, candidate)),
462    }
463}

481/// Match known-host pattern list with OpenSSH negation semantics.
482///
483/// ### Arguments
484/// - `patterns`: Comma-separated host patterns from one known-host entry.
485/// - `candidates`: Candidate host representations for the current connection.
486///
487/// ### Returns
488/// - `true`: At least one positive pattern matched and no negated match rejected it.
489/// - `false`: No positive match, or a negated pattern matched.
490fn match_known_host_patterns(patterns: &[String], candidates: &[String]) -> bool {
491    let mut matched = false;
492    for pattern in patterns {
493        let (negated, token) = if let Some(stripped) = pattern.strip_prefix('!') {
494            (true, stripped)
495        } else {
496            (false, pattern.as_str())
497        };
498
499        if candidates
500            .iter()
501            .any(|candidate| wildcard_pattern_matches(token, candidate))
502        {
503            if negated {
504                return false;
505            }
506            matched = true;
507        }
508    }
509    matched
510}

512/// Match a target string against a known-host wildcard pattern.
513///
514/// ### Arguments
515/// - `pattern`: Wildcard pattern using `*` and `?`.
516/// - `target`: Candidate host string.
517///
518/// ### Returns
519/// - `true`: Pattern matches the target.
520/// - `false`: Pattern does not match.
521fn wildcard_pattern_matches(pattern: &str, target: &str) -> bool {
522    let pattern_chars: Vec<char> = pattern.chars().collect();
523    let target_chars: Vec<char> = target.chars().collect();
524    let mut pattern_index = 0usize;
525    let mut target_index = 0usize;
526    let mut star_pattern_index = None;
527    let mut star_target_index = 0usize;
528
529    while target_index < target_chars.len() {
530        if pattern_index < pattern_chars.len()
531            && (pattern_chars[pattern_index] == '?'
532                || pattern_chars[pattern_index] == target_chars[target_index])
533        {
534            pattern_index += 1;
535            target_index += 1;
536        } else if pattern_index < pattern_chars.len() && pattern_chars[pattern_index] == '*' {
537            star_pattern_index = Some(pattern_index);
538            pattern_index += 1;
539            star_target_index = target_index;
540        } else if let Some(star_idx) = star_pattern_index {
541            pattern_index = star_idx + 1;
542            star_target_index += 1;
543            target_index = star_target_index;
544        } else {
545            return false;
546        }
547    }
548
549    while pattern_index < pattern_chars.len() && pattern_chars[pattern_index] == '*' {
550        pattern_index += 1;
551    }
552
553    pattern_index == pattern_chars.len()
554}

556/// Validate a hashed known-host entry against a candidate host string.
557///
558/// ### Arguments
559/// - `salt`: HMAC salt from the known-hosts hashed entry.
560/// - `expected_hash`: SHA-1 HMAC output from the known-hosts hashed entry.
561/// - `candidate`: Candidate host representation to verify.
562///
563/// ### Returns
564/// - `true`: Candidate matches the hashed entry.
565/// - `false`: Candidate does not match.
566fn hashed_host_matches(salt: &[u8], expected_hash: &[u8; 20], candidate: &str) -> bool {
567    type HmacSha1 = Hmac<Sha1>;
568    let Ok(mut mac) = HmacSha1::new_from_slice(salt) else {
569        return false;
570    };
571    mac.update(candidate.as_bytes());
572    mac.verify_slice(expected_hash).is_ok()
573}

602/// Aggregate several `CheckResult` values into a single decision.
603///
604/// ### Description
605/// Priority order is `Match` > `Mismatch` > `NotFound` > `Failure`.
606/// This ensures we accept if any representation is an exact match, while still treating
607/// genuine mismatches as errors when no representation matches.
608///
609/// ### Arguments
610/// - `results`: Check results from different host representations.
611///
612/// ### Returns
613/// - `ssh2::CheckResult`: Combined result following the documented priority order.
614fn aggregate_check_results(
615    results: impl IntoIterator<Item = ssh2::CheckResult>,
616) -> ssh2::CheckResult {
617    let mut saw_mismatch = false;
618    let mut saw_not_found = false;
619    for result in results {
620        match result {
621            ssh2::CheckResult::Match => return ssh2::CheckResult::Match,
622            ssh2::CheckResult::Mismatch => saw_mismatch = true,
623            ssh2::CheckResult::NotFound => saw_not_found = true,
624            ssh2::CheckResult::Failure => {}
625        }
626    }
627    if saw_mismatch {
628        ssh2::CheckResult::Mismatch
629    } else if saw_not_found {
630        ssh2::CheckResult::NotFound
631    } else {
632        ssh2::CheckResult::Failure
633    }
634}

636/// Build the host string format used when storing entries in `known_hosts`.
637///
638/// ### Arguments
639/// - `host`: Hostname or IP.
640/// - `port`: SSH port.
641///
642/// ### Returns
643/// - `String`: Plain host for port 22, or bracketed `[host]:port` for non-default ports.
644fn known_hosts_entry_host(host: &str, port: u16) -> String {
645    if port == 22 {
646        host.to_string()
647    } else {
648        format!("[{host}]:{port}")
649    }
650}

652/// Compute a colon-separated SHA-256 hex fingerprint from raw host-key bytes.
653///
654/// ### Arguments
655/// - `key`: Raw bytes of the server's host key.
656///
657/// ### Returns
658/// - `String`: Hex pairs joined by colons, e.g. `"ab:cd:ef:…"`.
659fn sha256_fingerprint(key: &[u8]) -> String {
660    let hash = Sha256::digest(key);
661    hash.iter()
662        .map(|b| format!("{b:02x}"))
663        .collect::<Vec<_>>()
664        .join(":")
665}

667/// Map a `ssh2::HostKeyType` to the `KnownHostKeyFormat` required by `known_hosts.add`.
668///
669/// ### Arguments
670/// - `key_type`: Key type reported by libssh2 after handshake.
671///
672/// ### Returns
673/// - `ssh2::KnownHostKeyFormat`: Corresponding format constant; `Unknown` falls back to `SshRsa`.
674fn host_key_type_to_format(key_type: ssh2::HostKeyType) -> ssh2::KnownHostKeyFormat {
675    match key_type {
676        ssh2::HostKeyType::Dss => ssh2::KnownHostKeyFormat::SshDss,
677        ssh2::HostKeyType::Ecdsa256 => ssh2::KnownHostKeyFormat::Ecdsa256,
678        ssh2::HostKeyType::Ecdsa384 => ssh2::KnownHostKeyFormat::Ecdsa384,
679        ssh2::HostKeyType::Ecdsa521 => ssh2::KnownHostKeyFormat::Ecdsa521,
680        ssh2::HostKeyType::Ed25519 => ssh2::KnownHostKeyFormat::Ed25519,
681        ssh2::HostKeyType::Rsa | ssh2::HostKeyType::Unknown => ssh2::KnownHostKeyFormat::SshRsa,
682    }
683}

693/// Create `~/.ssh` with mode `0700` on Unix if it does not already exist.
694///
695/// ### Returns
696/// - `Ok(())`: Directory exists or was created successfully.
697/// - `Err(SshError::IoError)`: Directory could not be created.
698fn ensure_ssh_dir() -> Result<(), SshError> {
699    let ssh_dir = home_dir().join(".ssh");
700    if !ssh_dir.exists() {
701        std::fs::create_dir_all(&ssh_dir)
702            .map_err(|e| SshError::IoError(format!("Failed to create ~/.ssh: {e}")))?;
703        set_dir_permissions_700(&ssh_dir);
704    }
705    Ok(())
706}

708/// Return the current user's home directory from platform-specific environment variables.
709///
710/// ### Returns
711/// - `PathBuf`: Home directory path; falls back to `/tmp` on Unix or `C:\Users\User` on Windows
712///   if the environment variable is missing.
713pub fn home_dir() -> PathBuf {
714    #[cfg(windows)]
715    {
716        std::env::var("USERPROFILE")
717            .or_else(|_| {
718                std::env::var("HOMEDRIVE")
719                    .and_then(|d| std::env::var("HOMEPATH").map(|p| format!("{d}{p}")))
720            })
721            .map_or_else(|_| PathBuf::from(r"C:\Users\User"), PathBuf::from)
722    }
723    #[cfg(not(windows))]
724    {
725        std::env::var("HOME").map_or_else(|_| PathBuf::from("/tmp"), PathBuf::from)
726    }
727}

730///
731/// ### Arguments
732/// - `path`: Path to the file whose permissions are to be set.
733fn set_file_permissions_600(path: &Path) {

736    #[cfg(not(unix))]
737    let _ = path;

740/// Set directory permissions to `0700` on Unix; no-op on Windows.
741///
742/// ### Arguments
743/// - `path`: Path to the directory whose permissions are to be set.
744fn set_dir_permissions_700(path: &Path) {
745    #[cfg(unix)]
746    apply_unix_mode(path, 0o700);
747    #[cfg(not(unix))]
748    let _ = path;
749}

751/// Apply a Unix permission mode to a file or directory.
752///
753/// ### Arguments
754/// - `path`: Path to the filesystem entry.
755/// - `mode`: Unix permission bits, e.g. `0o600` or `0o700`.
756#[cfg(unix)]
757fn apply_unix_mode(path: &Path, mode: u32) {
758    use std::os::unix::fs::PermissionsExt;
759    if let Ok(meta) = std::fs::metadata(path) {
760        let mut perms = meta.permissions();
761        perms.set_mode(mode);
762        let _ = std::fs::set_permissions(path, perms);
763    }
764}
