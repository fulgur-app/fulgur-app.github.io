1use std::fmt;
2
3/// All errors that can arise from SSH/SFTP operations.
4#[derive(Debug)]
5pub enum SshError {
6    /// A user-supplied URL could not be parsed into a `RemoteSpec`.
7    ParseError(String),
8    /// TCP connection or SSH handshake failed.
9    ConnectionFailed(String),
10    /// Server's host key is in `known_hosts` but does not match.
11    HostKeyMismatch { host: String, port: u16 },
12    /// Server's host key is not in `known_hosts`; contains the SHA-256 fingerprint for display.
13    UnknownHost {
14        host: String,
15        port: u16,
16        fingerprint: String,
17    },
18    /// Password authentication was rejected by the server.
19    AuthFailed,
20    /// An SFTP-level error (open, read, write, rename, or unlink).
21    SftpError(String),
22    /// Local filesystem I/O error, e.g. when reading or writing `known_hosts`.
23    IoError(String),
24}
25
26impl fmt::Display for SshError {
27    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
28        match self {
29            SshError::ParseError(msg) => write!(f, "Invalid remote URL: {msg}"),
30            SshError::ConnectionFailed(msg) => write!(f, "Connection failed: {msg}"),
31            SshError::HostKeyMismatch { host, port } => write!(
32                f,
33                "Host key mismatch for {host}:{port} — possible MITM. \
34                 Verify or update ~/.ssh/known_hosts."
35            ),
36            SshError::UnknownHost {
37                host,
38                port,
39                fingerprint,
40            } => write!(f, "Unknown host {host}:{port} (fingerprint: {fingerprint})"),
41            SshError::AuthFailed => {
42                write!(f, "Authentication failed — check username and password.")
43            }
44            SshError::SftpError(msg) => write!(f, "SFTP error: {msg}"),
45            SshError::IoError(msg) => write!(f, "I/O error: {msg}"),
46        }
47    }
48}
49
50impl SshError {
51    /// Return the error as a human-readable string suitable for displaying in a notification.
52    ///
53    /// ### Returns
54    /// - `String`: User-facing message; delegates to the `Display` implementation.
55    pub fn user_message(&self) -> String {
56        self.to_string()
57    }
58}
