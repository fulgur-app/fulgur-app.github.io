1use std::collections::HashMap;
2use zeroize::Zeroizing;

4/// Unique key for a (host, port, user) triple used to look up cached passwords.
5#[derive(Debug, Clone, PartialEq, Eq, Hash)]
6pub struct SshCredKey {
7    pub host: String,
8    pub port: u16,
9    pub user: String,
10}
11
12impl SshCredKey {
13    /// Create a new credential key from its components.
14    ///
15    /// ### Arguments
16    /// - `host`: Hostname or IP address of the remote server.
17    /// - `port`: SSH port.
18    /// - `user`: Username for authentication.
19    pub fn new(host: impl Into<String>, port: u16, user: impl Into<String>) -> Self {
20        Self {
21            host: host.into(),
22            port,
23            user: user.into(),
24        }
25    }
26
27    /// Return a display string in `user@host:port` format.
28    ///
29    /// ### Returns
30    /// - `String`: Human-readable representation, e.g. `"alice@example.com:22"`.
31    #[must_use]
32    pub fn display(&self) -> String {
33        format!("{}@{}:{}", self.user, self.host, self.port)
34    }
35}
36
37/// Session-scoped in-memory password cache.
38///
39/// Stored inside `SharedAppState` behind an `Arc<Mutex<…>>`. Passwords are held as
40/// `Zeroizing<String>` so memory is zeroed on drop. The map is never persisted to disk.
41pub type SshCredentialCache = HashMap<SshCredKey, Zeroizing<String>>;
