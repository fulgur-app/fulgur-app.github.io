13/// Maximum per-share decryption attempts before the share is quarantined.
14const MAX_DECRYPT_ATTEMPTS: u32 = 5;
15
16/// Base delay for the exponential decryption-retry backoff (doubles per attempt).
17const DECRYPT_RETRY_BASE_DELAY: Duration = Duration::from_secs(2);
18
19/// Fixed retry delay applied when the private key cannot be loaded from the  keychain.
20const KEY_RETRY_DELAY: Duration = Duration::from_secs(5);
21
22/// A shared file that has been decrypted and decompressed into plaintext,
23/// ready for the render loop to open as a new tab.
24pub struct DecryptedShare {
25    pub file_name: String,
26    pub content: String,
27}

29/// Retry bookkeeping for a pending share that failed to decode.
30pub struct ShareRetryState {
31    /// Number of failed decryption attempts so far.
32    pub attempts: u32,
33    /// Earliest time the next decryption attempt may run.
34    pub next_retry_at: Instant,
35}

37/// Compute the exponential backoff delay for a given failed-attempt count.
38///
39/// ### Arguments
40/// - `attempts`: The number of failed decryption attempts so far (at least 1).
41///
42/// ### Returns
43/// - `Duration`: `DECRYPT_RETRY_BASE_DELAY * 2^(attempts - 1)`.
44fn backoff_delay(attempts: u32) -> Duration {
45    DECRYPT_RETRY_BASE_DELAY.saturating_mul(1u32 << (attempts.saturating_sub(1)).min(16))
46}

48/// Tallies collected during a single decryption pass, used to decide which
49/// deduplicated error notification (if any) to surface to the user.
50#[derive(Default)]
51struct DecryptionOutcome {
52    /// The keychain held no private key entry for the profile.
53    key_unavailable: bool,
54    /// Loading the private key from the keychain errored.
55    key_load_failed: bool,
56    /// Number of files that failed to decode this pass.
57    decrypt_failures: usize,
58    /// Number of files decoded successfully this pass.
59    decrypted_count: usize,
60    /// Number of files re-queued for a later attempt.
61    retry_count: usize,
62    /// Number of files dropped after exhausting the decryption attempt cap.
63    quarantined_count: usize,
64}

66/// Start a one-shot background decryption pass for a profile when encrypted
67/// shares are due for an attempt and no pass is already running.
68///
69/// Shares in their retry-backoff window do not trigger a pass, so a queue
70/// holding only failing shares stays cheap to poll from the render loop.
71///
72/// ### Arguments
73/// - `profile`: The server profile whose pending shares should be decoded;
74///   also used to acknowledge v2 shares once they decrypt successfully.
75/// - `sync_state`: The shared sync state holding the queues and guard flag.
76/// - `http_agent`: Shared HTTP agent used to acknowledge downloaded v2 shares.
77pub fn start_decryption_if_idle(
78    profile: &ServerProfile,
79    sync_state: &Arc<SyncState>,
80    http_agent: &Arc<ureq::Agent>,
81) {
82    if !any_share_due(sync_state, Instant::now()) {
83        return;
84    }
85    if sync_state
86        .decrypt_in_flight
87        .compare_exchange(false, true, Ordering::AcqRel, Ordering::Acquire)
88        .is_err()
89    {
90        return;
91    }
92    let profile = profile.clone();
93    let sync_state = Arc::clone(sync_state);
94    let http_agent = Arc::clone(http_agent);
95    thread::spawn(move || {
96        run_decryption_pass(&profile, &sync_state, &http_agent);
97        sync_state.decrypt_in_flight.store(false, Ordering::Release);
98    });
99}

101/// Check whether at least one pending share is outside its retry-backoff window.
102///
103/// ### Arguments
104/// - `sync_state`: The shared sync state holding the queue and retry bookkeeping.
105/// - `now`: The instant to compare retry deadlines against.
106///
107/// ### Returns
108/// - `true`: At least one pending share is due for a decryption attempt.
109/// - `false`: The queue is empty or every pending share is backed off.
110fn any_share_due(sync_state: &SyncState, now: Instant) -> bool {
111    let pending = sync_state.pending_shared_files.lock();
112    if pending.is_empty() {
113        return false;
114    }
115    let retry_state = sync_state.share_retry_state.lock();
116    pending.iter().any(|share| {
117        retry_state
118            .get(&share.id)
119            .is_none_or(|retry| retry.next_retry_at <= now)
120    })
121}

123/// Decode every currently due pending share for a profile on a background thread.
124///
125/// ### Arguments
126/// - `profile`: The server profile whose shares are being decoded; also used to
127///   acknowledge v2 shares that decrypt successfully.
128/// - `sync_state`: The shared sync state holding the queues and notification slots.
129/// - `http_agent`: Shared HTTP agent used to acknowledge downloaded v2 shares.
130fn run_decryption_pass(profile: &ServerProfile, sync_state: &SyncState, http_agent: &ureq::Agent) {
131    let profile_id = &profile.id;
132    let profile_name = profile.name.as_str();
133    let server_max_size = sync_state.max_file_size_bytes.load(Ordering::Acquire);
134    let now = Instant::now();
135    let shared_files: Vec<fulgur_common::api::shares::SharedFileResponse> = {
136        let mut pending = sync_state.pending_shared_files.lock();
137        let retry_state = sync_state.share_retry_state.lock();
138        let (due, deferred): (Vec<_>, Vec<_>) = std::mem::take(&mut *pending)
139            .into_iter()
140            .partition(|share| {
141                retry_state
142                    .get(&share.id)
143                    .is_none_or(|retry| retry.next_retry_at <= now)
144            });
145        *pending = deferred;
146        due
147    };
148    if shared_files.is_empty() {
149        return;
150    }
151
152    let mut retry_queue = Vec::new();
153    let mut outcome = DecryptionOutcome::default();
154    let mut decrypted_ids: Vec<String> = Vec::new();
155
156    match load_private_key_from_keychain(profile_id) {
157        Ok(Some(encryption_key)) => {
158            for shared_file in shared_files {
159                if server_max_size != u64::MAX
160                    && shared_file.content.len() as u64 > server_max_size.saturating_mul(2)
161                {
162                    log::warn!(
163                        "Skipping shared file '{}' from device {}: encrypted payload ({} bytes) exceeds the server max ({} bytes)",
164                        shared_file.file_name,
165                        shared_file.source_device_id,
166                        shared_file.content.len(),
167                        server_max_size
168                    );
169                    continue;
170                }
171
172                let decoded =
173                    crypto_helper::decrypt_bytes(&shared_file.content, encryption_key.as_str())
174                        .and_then(|compressed_bytes| {
175                            share::decompress_content(&compressed_bytes, server_max_size)
176                        });
177
178                match decoded {
179                    Ok(content) => {
180                        sync_state
181                            .pending_decrypted_files
182                            .lock()
183                            .push(DecryptedShare {
184                                file_name: shared_file.file_name.clone(),
185                                content,
186                            });
187                        sync_state.share_retry_state.lock().remove(&shared_file.id);
188                        outcome.decrypted_count += 1;
189                        decrypted_ids.push(shared_file.id.clone());
190                        log::info!("Decrypted shared file: {}", shared_file.file_name);
191                    }
192                    Err(e) => {
193                        let attempts = {
194                            let mut retry_state = sync_state.share_retry_state.lock();
195                            let entry = retry_state.entry(shared_file.id.clone()).or_insert(
196                                ShareRetryState {
197                                    attempts: 0,
198                                    next_retry_at: now,
199                                },
200                            );
201                            entry.attempts += 1;
202                            entry.next_retry_at = Instant::now() + backoff_delay(entry.attempts);
203                            entry.attempts
204                        };
205                        if attempts >= MAX_DECRYPT_ATTEMPTS {
206                            sync_state.share_retry_state.lock().remove(&shared_file.id);
207                            outcome.quarantined_count += 1;
208                            log::warn!(
209                                "Quarantining shared file '{}' for profile {profile_id}: decryption failed {attempts} time(s), giving up ({e}). \
210                                 The share was likely encrypted for a different device and will expire server-side.",
211                                shared_file.file_name
212                            );
213                        } else {
214                            outcome.decrypt_failures += 1;
215                            log::warn!(
216                                "Deferring shared file '{}' for profile {profile_id}: decryption failed ({e}), attempt {attempts} of {MAX_DECRYPT_ATTEMPTS}",
217                                shared_file.file_name
218                            );
219                            retry_queue.push(shared_file);
220                        }
221                    }
222                }
223            }
224            drop(encryption_key);
225        }
226        Ok(None) => {
227            outcome.key_unavailable = true;
228            log::warn!(
229                "Deferring {} shared file(s) for profile {profile_id}: encryption key is unavailable",
230                shared_files.len()
231            );
232            defer_batch_for_key_retry(sync_state, &shared_files);
233            retry_queue.extend(shared_files);
234        }
235        Err(e) => {
236            outcome.key_load_failed = true;
237            log::warn!(
238                "Deferring {} shared file(s) for profile {profile_id}: failed to load encryption key from keychain: {e}",
239                shared_files.len()
240            );
241            defer_batch_for_key_retry(sync_state, &shared_files);
242            retry_queue.extend(shared_files);
243        }
244    }
245
246    outcome.retry_count = retry_queue.len();
247    if outcome.retry_count > 0 {
248        let mut pending = sync_state.pending_shared_files.lock();
249        retry_queue.extend(std::mem::take(&mut *pending));
250        *pending = retry_queue;
251        log::warn!(
252            "Re-queued {} shared file(s) for profile {profile_id} for retry",
253            outcome.retry_count
254        );
255    }
256
257    acknowledge_downloaded_shares(profile, sync_state, http_agent, &decrypted_ids);
258
259    outcome.publish_notification(profile_name, sync_state);
260}

262/// Push every share's next retry out by `KEY_RETRY_DELAY` after a keychain
263/// failure, without counting a decryption attempt.
264///
265/// ### Arguments
266/// - `sync_state`: The shared sync state holding the retry bookkeeping.
267/// - `shared_files`: The batch of shares deferred by the keychain failure.
268fn defer_batch_for_key_retry(
269    sync_state: &SyncState,
270    shared_files: &[fulgur_common::api::shares::SharedFileResponse],
271) {
272    let next_retry_at = Instant::now() + KEY_RETRY_DELAY;
273    let mut retry_state = sync_state.share_retry_state.lock();
274    for share in shared_files {
275        retry_state
276            .entry(share.id.clone())
277            .and_modify(|retry| retry.next_retry_at = next_retry_at)
278            .or_insert(ShareRetryState {
279                attempts: 0,
280                next_retry_at,
281            });
282    }
283}

285/// Acknowledge successfully-downloaded v2 shares, consuming them server-side.
286///
287/// ### Arguments
288/// - `profile`: The server profile to acknowledge against.
289/// - `sync_state`: The shared sync state holding the ack set and token manager.
290/// - `http_agent`: Shared HTTP agent for connection pooling.
291/// - `decrypted_ids`: IDs of shares decrypted successfully in this pass.
292fn acknowledge_downloaded_shares(
293    profile: &ServerProfile,
294    sync_state: &SyncState,
295    http_agent: &ureq::Agent,
296    decrypted_ids: &[String],
297) {
298    if decrypted_ids.is_empty() {
299        return;
300    }
301    let ids_to_ack: Vec<String> = {
302        let pending_ack = sync_state.pending_ack_share_ids.lock();
303        decrypted_ids
304            .iter()
305            .filter(|id| pending_ack.contains(*id))
306            .cloned()
307            .collect()
308    };
309    for id in ids_to_ack {
310        match share::acknowledge_share_download(profile, &sync_state.token_state, http_agent, &id) {
311            Ok(()) => {
312                sync_state.pending_ack_share_ids.lock().remove(&id);
313            }
314            Err(e) => {
315                log::warn!(
316                    "Failed to acknowledge downloaded share {id}: {e}; duplicate delivery is suppressed and the share will expire server-side"
317                );
318            }
319        }
320    }
321}

323impl DecryptionOutcome {
324    /// Publish or clear the deduplicated share-receive error notification for this pass.
325    ///
326    /// ### Arguments
327    /// - `profile_name`: The human-readable profile name used in messages.
328    /// - `sync_state`: The shared sync state holding the notification slots.
329    fn publish_notification(&self, profile_name: &str, sync_state: &SyncState) {
330        let error_notification = if self.key_unavailable {
331            Some((
332                "missing-keychain-private-key",
333                SharedString::from(format!(
334                    "{profile_name}: Cannot receive shared files because the encryption key is unavailable in the keychain. Fulgur will retry automatically."
335                )),
336            ))
337        } else if self.key_load_failed {
338            Some((
339                "failed-to-load-keychain-private-key",
340                SharedString::from(format!(
341                    "{profile_name}: Cannot receive shared files because the encryption key could not be loaded from the keychain. Fulgur will retry automatically."
342                )),
343            ))
344        } else if self.quarantined_count > 0 {
345            Some((
346                "share-decryption-quarantined",
347                SharedString::from(format!(
348                    "{profile_name}: Could not decrypt {} shared file(s) after {MAX_DECRYPT_ATTEMPTS} attempts. They may have been encrypted for a different device and will not be retried.",
349                    self.quarantined_count
350                )),
351            ))
352        } else if self.decrypt_failures > 0 {
353            Some((
354                "share-decryption-failed",
355                SharedString::from(format!(
356                    "{profile_name}: Failed to decrypt {} shared file(s). Fulgur will retry automatically.",
357                    self.decrypt_failures
358                )),
359            ))
360        } else {
361            None
362        };
363
364        if let Some((signature, message)) = error_notification {
365            let mut last_signature = sync_state.last_share_receive_error_signature.lock();
366            if last_signature.as_deref() != Some(signature) {
367                sync_state.notify(AppNotification::background(
368                    NotificationType::Error,
369                    message,
370                ));
371                *last_signature = Some(signature.to_string());
372            }
373        } else if self.decrypted_count > 0 || self.retry_count == 0 {
374            *sync_state.last_share_receive_error_signature.lock() = None;
375        }
376    }
