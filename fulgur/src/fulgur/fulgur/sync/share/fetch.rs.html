13/// Atomically fetch and delete all pending shares for the current device on
14/// a single profile's server.
15///
16/// ### Arguments
17/// - `profile`: The server profile to fetch from
18/// - `token_state`: Per-profile token state manager
19/// - `http_agent`: Shared HTTP agent for connection pooling
20///
21/// ### Returns
22/// - `Ok(Vec<SharedFileResponse>)`: The shares that were drained from the server
23/// - `Err(SynchronizationError)`: If the request failed or the response was invalid
24pub fn fetch_pending_shares(
25    profile: &ServerProfile,
26    token_state: &Arc<TokenStateManager>,
27    http_agent: &ureq::Agent,
28) -> Result<Vec<SharedFileResponse>, SynchronizationError> {
29    let Some(server_url) = profile.server_url.clone() else {
30        return Err(SynchronizationError::ServerUrlMissing);
31    };
32    let token = get_valid_token(profile, token_state, http_agent)?;
33    let shares_url = format!("{server_url}/api/shares");
34    let mut response = http_agent
35        .get(&shares_url)
36        .header("Authorization", &format!("Bearer {token}"))
37        .call()
38        .map_err(|e| handle_ureq_error(e, "Failed to fetch pending shares"))?;
39    let shares: Vec<SharedFileResponse> = response
40        .body_mut()
41        .read_json::<Vec<SharedFileResponse>>()
42        .map_err(|e| {
43            log::error!("Failed to read pending shares: {e}");
44            SynchronizationError::InvalidResponse(e.to_string())
45        })?;
46    log::debug!("Fetched {} pending share(s) from server", shares.len());
47    Ok(shares)
48}

50/// Fetch a single pending share by ID and remove it from the server.
51///
52/// ### Arguments
53/// - `profile`: The server profile to fetch from
54/// - `token_state`: Per-profile token state manager
55/// - `http_agent`: Shared HTTP agent for connection pooling
56/// - `id`: The share identifier returned by the v2 begin endpoint
57///
58/// ### Returns
59/// - `Ok(SharedFileResponse)`: The share content
60/// - `Err(SynchronizationError)`: If the request failed, the share is gone,
61///   or the response was invalid or too large
62pub fn fetch_share_by_id(
63    profile: &ServerProfile,
64    token_state: &Arc<TokenStateManager>,
65    http_agent: &ureq::Agent,
66    id: &str,
67) -> Result<SharedFileResponse, SynchronizationError> {
68    let Some(server_url) = profile.server_url.clone() else {
69        return Err(SynchronizationError::ServerUrlMissing);
70    };
71    let token = get_valid_token(profile, token_state, http_agent)?;
72    let share_url = format!("{server_url}/api/shares/{id}");
73    let mut response = http_agent
74        .get(&share_url)
75        .header("Authorization", &format!("Bearer {token}"))
76        .call()
77        .map_err(|e| handle_ureq_error(e, "Failed to fetch share by id"))?;
78    let body = response
79        .body_mut()
80        .with_config()
81        .limit(MAX_HTTP_SINGLE_SHARE_RESPONSE_BYTES)
82        .read_to_string()
83        .map_err(|e| {
84            log::error!("Failed to read share response body for id {id}: {e}");
85            SynchronizationError::Other(e.to_string())
86        })?;
87    serde_json::from_str::<SharedFileResponse>(&body).map_err(|e| {
88        log::error!("Failed to parse share response body for id {id}: {e}");
89        SynchronizationError::InvalidResponse(e.to_string())
90    })
91}
