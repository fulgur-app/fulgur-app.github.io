15/// Atomically fetch and delete all pending shares for the current device on
16/// a single profile's server.
17///
18/// Will be deprecated in 0.11.0.
19///
20/// ### Arguments
21/// - `profile`: The server profile to fetch from
22/// - `token_state`: Per-profile token state manager
23/// - `http_agent`: Shared HTTP agent for connection pooling
24/// - `server_max_file_size`: The server-advertised max file size
25///
26/// ### Errors
27/// Returns a `SynchronizationError` if the profile has no server URL, the
28/// authentication token cannot be obtained, the HTTP request fails, the
29/// response exceeds the per-server bulk cap, or the response cannot be
30/// deserialized.
31///
32/// ### Returns
33/// - `Ok(Vec<SharedFileResponse>)`: The shares that were drained from the
34///   server, with each `file_name` sanitized against path traversal and
35///   control characters
36/// - `Err(SynchronizationError)`: If the request failed or the response was invalid
37pub fn fetch_pending_shares(
38    profile: &ServerProfile,
39    token_state: &Arc<TokenStateManager>,
40    http_agent: &ureq::Agent,
41    server_max_file_size: u64,
42) -> Result<Vec<SharedFileResponse>, SynchronizationError> {
43    let Some(server_url) = profile.server_url.clone() else {
44        return Err(SynchronizationError::ServerUrlMissing);
45    };
46    let token = get_valid_token(profile, token_state, http_agent)?;
47    let shares_url = format!("{server_url}/api/shares");
48    let mut response = http_agent
49        .get(&shares_url)
50        .header("Authorization", &format!("Bearer {token}"))
51        .call()
52        .map_err(|e| handle_ureq_error(e, "Failed to fetch pending shares"))?;
53    let body = response
54        .body_mut()
55        .with_config()
56        .limit(max_http_bulk_shares_response_bytes(server_max_file_size))
57        .read_to_string()
58        .map_err(|e| {
59            log::error!("Failed to read pending shares: {e}");
60            SynchronizationError::Other(e.to_string())
61        })?;
62    let mut shares: Vec<SharedFileResponse> = serde_json::from_str(&body).map_err(|e| {
63        log::error!("Failed to parse pending shares: {e}");
64        SynchronizationError::InvalidResponse(e.to_string())
65    })?;
66    for share in &mut shares {
67        share.file_name = sanitize_filename(&share.file_name);
68    }
69    log::debug!("Fetched {} pending share(s) from server", shares.len());
70    Ok(shares)
71}

73/// Fetch a single pending share by ID and remove it from the server.
74///
75/// ### Arguments
76/// - `profile`: The server profile to fetch from
77/// - `token_state`: Per-profile token state manager
78/// - `http_agent`: Shared HTTP agent for connection pooling
79/// - `id`: The share identifier returned by the v2 begin endpoint
80/// - `server_max_file_size`: The server-advertised max file size, or `u64::MAX`,
81///   used to size the response body cap
82///
83/// ### Errors
84/// Returns a `SynchronizationError` if the profile has no server URL, the
85/// authentication token cannot be obtained, the HTTP request fails, the share
86/// is missing, or the response is invalid or too large.
87///
88/// ### Returns
89/// - `Ok(SharedFileResponse)`: The share content, with `file_name` sanitized
90///   against path traversal and control characters
91/// - `Err(SynchronizationError)`: If the request failed, the share is gone,
92///   or the response was invalid or too large
93pub fn fetch_share_by_id(
94    profile: &ServerProfile,
95    token_state: &Arc<TokenStateManager>,
96    http_agent: &ureq::Agent,
97    id: &str,
98    server_max_file_size: u64,
99) -> Result<SharedFileResponse, SynchronizationError> {
100    let Some(server_url) = profile.server_url.clone() else {
101        return Err(SynchronizationError::ServerUrlMissing);
102    };
103    let token = get_valid_token(profile, token_state, http_agent)?;
104    let share_url = format!("{server_url}/api/shares/{id}");
105    let mut response = http_agent
106        .get(&share_url)
107        .header("Authorization", &format!("Bearer {token}"))
108        .call()
109        .map_err(|e| handle_ureq_error(e, "Failed to fetch share by id"))?;
110    let body = response
111        .body_mut()
112        .with_config()
113        .limit(max_http_single_share_response_bytes(server_max_file_size))
114        .read_to_string()
115        .map_err(|e| {
116            log::error!("Failed to read share response body for id {id}: {e}");
117            SynchronizationError::Other(e.to_string())
118        })?;
119    let mut share = serde_json::from_str::<SharedFileResponse>(&body).map_err(|e| {
120        log::error!("Failed to parse share response body for id {id}: {e}");
121        SynchronizationError::InvalidResponse(e.to_string())
122    })?;
123    share.file_name = sanitize_filename(&share.file_name);
124    Ok(share)
125}
