1use crate::fulgur::Fulgur;
2use crate::fulgur::sync::share;
3use crate::fulgur::ui::tabs::editor_tab;
4use crate::fulgur::ui::tabs::tab::Tab;
5use gpui::{Context, Window};
6use std::sync::Arc;
7
8impl Fulgur {
9    /// Process shared files received from every active sync profile.
10    ///
11    /// ### Arguments
12    /// - `window`: The window to create new tabs in
13    /// - `cx`: The application context
14    pub fn process_shared_files_from_sync(&mut self, window: &mut Window, cx: &mut Context<Self>) {
15        let http_agent = Arc::clone(&Fulgur::shared_state(cx).http_agent);
16
17        let mut decrypted_files: Vec<share::DecryptedShare> = Vec::new();
18        for profile in self
19            .settings
20            .app_settings
21            .synchronization_settings
22            .profiles
23            .iter()
24            .filter(|p| p.is_active)
25        {
26            let sync_state = Fulgur::shared_state(cx).sync_state_for(&profile.id);
27
28            // Decryption and decompression are CPU-bound; kick them off on a
29            // background worker so a batch of incoming shares never freezes the
30            // UI thread. The worker pushes plaintext into `pending_decrypted_files`
31            // and acknowledges v2 shares once they decrypt successfully.
32            share::start_decryption_if_idle(profile, &sync_state, &http_agent);
33
34            let mut pending = sync_state.pending_decrypted_files.lock();
35            decrypted_files.append(&mut pending);
36        }
37
38        // Open everything that has finished decrypting. This is the only part
39        // that needs `window`/`cx` and mutable tab state, and it is cheap (no crypto).
40        for decrypted in decrypted_files {
41            let tab_id = self.allocate_tab_id();
42            let new_tab = Tab::Editor(editor_tab::EditorTab::from_content(
43                tab_id,
44                &decrypted.content,
45                decrypted.file_name.clone(),
46                window,
47                cx,
48                &self.settings.editor_settings,
49            ));
50            self.tabs.push(new_tab.into_entity(cx));
51            self.active_tab_id = Some(tab_id);
52            self.request_tab_scroll(tab_id, cx);
53            log::info!("Opened shared file: {}", decrypted.file_name);
54        }
55    }
56}
