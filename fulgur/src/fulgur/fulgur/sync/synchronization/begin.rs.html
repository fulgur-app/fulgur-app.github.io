17/// Initial synchronization with the server.
18///
19/// ### Description
20/// Attempts the v2 begin flow first (`POST /api/v2/begin` + per-share fetch).
21/// If the server returns HTTP 404 (endpoint not deployed yet), falls back to
22/// the legacy v1 flow (`POST /api/begin`) for compatibility with Fulgurant
23/// servers that have not been upgraded.
24///
25/// ### Arguments
26/// - `profile`: The server profile to synchronize with
27/// - `token_state`: Per-profile JWT token state manager
28/// - `http_agent`: Shared HTTP agent for connection pooling
29/// - `pending_ack_share_ids`: Ack set the v2 read/ack flow registers fetched ids
30///   into so the decryption pass acknowledges them after a successful download
31///
32/// ### Errors
33/// Returns a `SynchronizationError` if both the v2 and the legacy v1 begin
34/// requests fail (network failure, authentication failure, or invalid response).
35///
36/// ### Returns
37/// - `Ok(InitialSyncOutcome)`: Begin response plus the server's advertised
38///   minimum supported Fulgur version (v2 only)
39/// - `Err(SynchronizationError)`: If both v2 and v1 begin calls failed
40// The ack set is always the default-hasher `HashSet<String>` owned by `SyncState`;
41// generalizing over `BuildHasher` would only add noise.
42#[allow(clippy::implicit_hasher)]
43pub fn initial_synchronization(
44    profile: &ServerProfile,
45    token_state: &Arc<TokenStateManager>,
46    http_agent: &ureq::Agent,
47    pending_ack_share_ids: &Arc<Mutex<HashSet<String>>>,
48) -> Result<InitialSyncOutcome, SynchronizationError> {
49    match initial_synchronization_v2(profile, token_state, http_agent, pending_ack_share_ids) {
50        Ok(response) => Ok(response),
51        Err(SynchronizationError::ServerError(404)) => {
52            log::warn!(
53                "Server does not support /api/v2/begin (404); falling back to legacy /api/begin"
54            );
55            initial_synchronization_v1(profile, token_state, http_agent)
56        }
57        Err(e) => Err(e),
58    }
59}

61/// Outcome of an initial synchronization.
62pub struct InitialSyncOutcome {
63    /// Device name, max file size, and successfully fetched pending shares.
64    pub begin: BeginResponse,
65    /// The `min_fulgur_version` advertised by the v2 begin response, if any.

68    /// Raw `x-fulgurant-version` header advertised by the server, if any.
69    /// `None` for Fulgurant before 0.7.0, which does not advertise it.
70    pub fulgurant_version: Option<String>,
71}

81/// Perform the `POST /api/v2/begin` request and parse its response.
82///
83/// ### Arguments
84/// - `profile`: The server profile to synchronize with
85/// - `token_state`: Per-profile JWT token state manager
86/// - `http_agent`: Shared HTTP agent for connection pooling
87///
88/// ### Errors
89/// Returns a `SynchronizationError` if the request fails, the response cannot be
90/// read or parsed, or the server announces more pending shares than the client allows.
91///
92/// ### Returns
93/// - `Ok(BeginV2Outcome)`: The parsed begin response and advertised version
94/// - `Err(SynchronizationError)`: If the begin call failed or returned an invalid response
95fn perform_begin_v2(
96    profile: &ServerProfile,
97    token_state: &Arc<TokenStateManager>,
98    http_agent: &ureq::Agent,
99) -> Result<BeginV2Outcome, SynchronizationError> {
100    let Some(server_url) = profile.server_url.clone() else {
101        return Err(SynchronizationError::ServerUrlMissing);
102    };
103    let Some(public_key) = profile.public_key.clone() else {
104        return Err(SynchronizationError::MissingEncryptionKey);
105    };
106    let token = get_valid_token(profile, token_state, http_agent)?;
107    let begin_url = format!("{server_url}/api/v2/begin");
108    let payload = InitialSynchronizationPayload { public_key };
109    let mut response = http_agent
110        .post(begin_url)
111        .header("Authorization", &format!("Bearer {token}"))
112        .send_json(payload)
113        .map_err(|e| handle_ureq_error(e, "Failed to begin synchronization (v2)"))?;
114    let version_header = response
115        .headers()
116        .get(FULGURANT_VERSION_HEADER)
117        .and_then(|value| value.to_str().ok())
118        .map(str::to_owned);
119    let body = match response
120        .body_mut()
121        .with_config()
122        .limit(MAX_HTTP_SMALL_RESPONSE_BYTES)
123        .read_to_string()
124    {
125        Ok(body) => body,
126        Err(e) => {
127            log::error!("Failed to read v2 begin response body: {e}");
128            return Err(SynchronizationError::Other(e.to_string()));
129        }
130    };
131    let begin_v2: BeginV2Response = match serde_json::from_str(&body) {
132        Ok(response) => response,
133        Err(e) => {
134            log::error!("Failed to parse v2 begin response body: {e}");
135            return Err(SynchronizationError::InvalidResponse(e.to_string()));
136        }
137    };
138    if begin_v2.share_ids.len() > share::MAX_PENDING_SHARES_PER_RESPONSE {
139        log::error!(
140            "Server returned {} pending share ids, exceeding the client limit of {}",
141            begin_v2.share_ids.len(),
142            share::MAX_PENDING_SHARES_PER_RESPONSE
143        );
144        return Err(SynchronizationError::InvalidResponse(format!(
145            "Server returned too many pending share ids ({} > {})",
146            begin_v2.share_ids.len(),
147            share::MAX_PENDING_SHARES_PER_RESPONSE
148        )));
149    }
150    Ok(BeginV2Outcome {
151        response: begin_v2,
152        version_header,
153    })
154}

156/// List the IDs of the device's pending shares via `POST /api/v2/begin` without consuming them.
157///
158/// ### Arguments
159/// - `profile`: The server profile to synchronize with
160/// - `token_state`: Per-profile JWT token state manager
161/// - `http_agent`: Shared HTTP agent for connection pooling
162///
163/// ### Errors
164/// Returns a `SynchronizationError` if the begin call fails or returns an invalid response.
165///
166/// ### Returns
167/// - `Ok(Vec<String>)`: The IDs of the device's pending shares
168/// - `Err(SynchronizationError)`: If the begin call failed
169pub fn list_pending_share_ids_v2(
170    profile: &ServerProfile,
171    token_state: &Arc<TokenStateManager>,

179/// Maximum number of concurrent share fetch worker threads.
180const MAX_FETCH_WORKERS: usize = 8;
181
182/// Fetch each announced share by id in parallel, choosing the retrieval endpoint
183/// based on whether the server supports the v2 read/ack flow.
184///
185/// ### Arguments
186/// - `profile`: The server profile to fetch from
187/// - `token_state`: Per-profile JWT token state manager
188/// - `http_agent`: Shared HTTP agent for connection pooling
189/// - `share_ids`: The announced pending share ids to retrieve
190/// - `server_max_file_size`: Server-advertised max file size used to bound each response
191/// - `use_v2_flow`: When `true`, fetch via the non-consuming `GET /api/v2/shares/:id`;
192///   otherwise via the consuming `GET /api/shares/:id`
193///
194/// ### Returns
195/// - `Vec<SharedFileResponse>`: The successfully fetched shares; failures are logged and skipped
196fn fetch_shares_for_ids(
197    profile: &ServerProfile,
198    token_state: &Arc<TokenStateManager>,
199    http_agent: &ureq::Agent,
200    share_ids: &[String],
201    server_max_file_size: u64,
202    use_v2_flow: bool,
203) -> Vec<SharedFileResponse> {
204    if share_ids.is_empty() {
205        return Vec::new();
206    }
207    let worker_count = MAX_FETCH_WORKERS.min(share_ids.len());
208    let next_index = AtomicUsize::new(0);
209    let results = Mutex::new(Vec::with_capacity(share_ids.len()));
210    std::thread::scope(|scope| {
211        for _ in 0..worker_count {
212            scope.spawn(|| {
213                loop {
214                    let index = next_index.fetch_add(1, Ordering::Relaxed);
215                    let Some(id) = share_ids.get(index) else {
216                        break;
217                    };
218                    let result = if use_v2_flow {
219                        share::fetch_share_by_id_v2(
220                            profile,
221                            token_state,
222                            http_agent,
223                            id,
224                            server_max_file_size,
225                        )
226                    } else {
227                        share::fetch_share_by_id(
228                            profile,
229                            token_state,
230                            http_agent,
231                            id,
232                            server_max_file_size,
233                        )
234                    };
235                    match result {
236                        Ok(s) => results.lock().push(s),
237                        Err(e) => log::warn!("Skipping share id {id}: {e}"),
238                    }
239                }
240            });
241        }
242    });
243    results.into_inner()
244}

246/// Initial synchronization via the v2 begin flow.
247///
248/// ### Arguments
249/// - `profile`: The server profile to synchronize with
250/// - `token_state`: Per-profile JWT token state manager
251/// - `http_agent`: Shared HTTP agent for connection pooling
252/// - `pending_ack_share_ids`: Ack set the v2 read/ack flow registers fetched ids into
253///
254/// ### Returns
255/// - `Ok(InitialSyncOutcome)`: Begin response and advertised minimum Fulgur version
256/// - `Err(SynchronizationError)`: If the v2 begin call failed or returned an invalid response
257fn initial_synchronization_v2(
258    profile: &ServerProfile,
259    token_state: &Arc<TokenStateManager>,
260    http_agent: &ureq::Agent,
261    pending_ack_share_ids: &Arc<Mutex<HashSet<String>>>,
262) -> Result<InitialSyncOutcome, SynchronizationError> {
263    let BeginV2Outcome {
264        response: begin_v2,
265        version_header,
266    } = perform_begin_v2(profile, token_state, http_agent)?;
267    let min_fulgur_version = begin_v2.min_fulgur_version.clone();
268    let use_v2_flow = version_supports_v2_share_flow(version_header.as_deref());
269    let server_max_file_size = resolve_server_max_file_size(begin_v2.max_file_size_bytes);
270    let shares = fetch_shares_for_ids(
271        profile,
272        token_state,
273        http_agent,
274        &begin_v2.share_ids,
275        server_max_file_size,
276        use_v2_flow,
277    );
278    if use_v2_flow {
279        let mut ack_set = pending_ack_share_ids.lock();
280        for share in &shares {
281            ack_set.insert(share.id.clone());
282        }
283    }
284    log::info!(
285        "Initial synchronization (v2) successful: {} announced, {} retrieved (read/ack flow: {use_v2_flow})",
286        begin_v2.share_ids.len(),
287        shares.len()
288    );
289    Ok(InitialSyncOutcome {
290        begin: BeginResponse {
291            device_name: begin_v2.device_name,
292            shares,
293            max_file_size_bytes: begin_v2.max_file_size_bytes,
294        },
295        min_fulgur_version,
296        fulgurant_version: version_header,
297    })
298}

300/// Initial synchronization via the legacy v1 begin flow.
301///
302/// ### Description
303/// Calls `POST /api/begin`, which returns the device name, max file size and
304/// pending shares inline. Used as a fallback when the server does not yet
305/// expose `/api/v2/begin`.
306///
307/// ### Arguments
308/// - `profile`: The server profile to synchronize with
309/// - `token_state`: Per-profile JWT token state manager
310/// - `http_agent`: Shared HTTP agent for connection pooling
311///
312/// ### Returns
313/// - `Ok(InitialSyncOutcome)`: Begin response with no advertised minimum Fulgur
314///   version (legacy v1 servers do not advertise one)
315/// - `Err(SynchronizationError)`: If the v1 begin call failed or returned an invalid response
316fn initial_synchronization_v1(
317    profile: &ServerProfile,
318    token_state: &Arc<TokenStateManager>,
319    http_agent: &ureq::Agent,
320) -> Result<InitialSyncOutcome, SynchronizationError> {
321    let Some(server_url) = profile.server_url.clone() else {
322        return Err(SynchronizationError::ServerUrlMissing);
323    };
324    let Some(public_key) = profile.public_key.clone() else {
325        return Err(SynchronizationError::MissingEncryptionKey);
326    };
327    let token = get_valid_token(profile, token_state, http_agent)?;
328    let begin_url = format!("{server_url}/api/begin");
329    let payload = InitialSynchronizationPayload { public_key };
330    let mut response = http_agent
331        .post(begin_url)
332        .header("Authorization", &format!("Bearer {token}"))
333        .send_json(payload)
334        .map_err(|e| handle_ureq_error(e, "Failed to begin synchronization (v1)"))?;
335    let version_header = response
336        .headers()
337        .get(FULGURANT_VERSION_HEADER)
338        .and_then(|value| value.to_str().ok())
339        .map(str::to_owned);
340    let body = match response
341        .body_mut()
342        .with_config()
343        .limit(MAX_HTTP_V1_BEGIN_RESPONSE_BYTES)
344        .read_to_string()
345    {
346        Ok(body) => body,
347        Err(e) => {
348            log::error!("Failed to read v1 begin response body: {e}");
349            return Err(SynchronizationError::Other(e.to_string()));
350        }
351    };
352    let mut begin_response: BeginResponse = match serde_json::from_str(&body) {
353        Ok(response) => response,
354        Err(e) => {
355            log::error!("Failed to parse v1 begin response body: {e}");
356            return Err(SynchronizationError::InvalidResponse(e.to_string()));
357        }
358    };
359    for share in &mut begin_response.shares {
360        share.file_name = sanitize_filename(&share.file_name);
361    }
362    log::info!(
363        "Initial synchronization (v1) successful with {} shared files",
364        begin_response.shares.len()
365    );
366    Ok(InitialSyncOutcome {
367        begin: begin_response,
368        min_fulgur_version: None,
369        fulgurant_version: version_header,
370    })
