44/// Perform the `POST /api/v2/begin` request and parse its response.
45///
46/// ### Arguments
47/// - `profile`: The server profile to synchronize with
48/// - `token_state`: Per-profile JWT token state manager
49/// - `http_agent`: Shared HTTP agent for connection pooling
50///
51/// ### Errors
52/// Returns a `SynchronizationError` if the request fails, the server is older
53/// than the minimum supported Fulgurant version, the response cannot be read or
54/// parsed, or the server announces more pending shares than the client allows.
55///
56/// ### Returns
57/// - `Ok(BeginResponseOutcome)`: The parsed begin response and advertised version
58/// - `Err(SynchronizationError)`: If the begin call failed or returned an invalid response
59fn perform_begin(
60    profile: &ServerProfile,
61    token_state: &Arc<TokenStateManager>,
62    http_agent: &ureq::Agent,
63) -> Result<BeginResponseOutcome, SynchronizationError> {
64    let Some(server_url) = profile.server_url.clone() else {
65        return Err(SynchronizationError::ServerUrlMissing);
66    };
67    let Some(public_key) = profile.public_key.clone() else {
68        return Err(SynchronizationError::MissingEncryptionKey);
69    };
70    let token = get_valid_token(profile, token_state, http_agent)?;
71    let begin_url = format!("{server_url}/api/v2/begin");
72    let payload = InitialSynchronizationPayload { public_key };
73    let mut response = http_agent
74        .post(begin_url)
75        .header("Authorization", &format!("Bearer {token}"))
76        .send_json(payload)
77        .map_err(|e| handle_ureq_error(e, "Failed to begin synchronization"))?;
78    let version_header = response
79        .headers()
80        .get(FULGURANT_VERSION_HEADER)
81        .and_then(|value| value.to_str().ok())
82        .map(str::to_owned);
83    if !server_meets_minimum_version(version_header.as_deref()) {
84        log::error!(
85            "Server advertises Fulgurant version {}, which is below the required {}",
86            version_header.as_deref().unwrap_or("<none>"),
87            MIN_SUPPORTED_FULGURANT_VERSION_DISPLAY
88        );
89        return Err(SynchronizationError::ServerTooOld {
90            required: MIN_SUPPORTED_FULGURANT_VERSION_DISPLAY,
91        });
92    }
93    let body = match response
94        .body_mut()
95        .with_config()
96        .limit(MAX_HTTP_SMALL_RESPONSE_BYTES)
97        .read_to_string()
98    {
99        Ok(body) => body,
100        Err(e) => {
101            log::error!("Failed to read begin response body: {e}");
102            return Err(SynchronizationError::Other(e.to_string()));
103        }
104    };
105    let begin: BeginV2Response = match serde_json::from_str(&body) {
106        Ok(response) => response,
107        Err(e) => {
108            log::error!("Failed to parse begin response body: {e}");
109            return Err(SynchronizationError::InvalidResponse(e.to_string()));
110        }
111    };
112    if begin.share_ids.len() > share::MAX_PENDING_SHARES_PER_RESPONSE {
113        log::error!(
114            "Server returned {} pending share ids, exceeding the client limit of {}",
115            begin.share_ids.len(),
116            share::MAX_PENDING_SHARES_PER_RESPONSE
117        );
118        return Err(SynchronizationError::InvalidResponse(format!(
119            "Server returned too many pending share ids ({} > {})",
120            begin.share_ids.len(),
121            share::MAX_PENDING_SHARES_PER_RESPONSE
122        )));
123    }
124    Ok(BeginResponseOutcome {
125        response: begin,
126        version_header,
127    })
128}

130/// List the IDs of the device's pending shares without consuming them.
131///
132/// ### Arguments
133/// - `profile`: The server profile to synchronize with
134/// - `token_state`: Per-profile JWT token state manager
135/// - `http_agent`: Shared HTTP agent for connection pooling
136///
137/// ### Errors
138/// Returns a `SynchronizationError` if the begin call fails or returns an invalid response.
139///
140/// ### Returns
141/// - `Ok(Vec<String>)`: The IDs of the device's pending shares
142/// - `Err(SynchronizationError)`: If the begin call failed
143pub fn list_pending_share_ids(
144    profile: &ServerProfile,
145    token_state: &Arc<TokenStateManager>,
146    http_agent: &ureq::Agent,
147) -> Result<Vec<String>, SynchronizationError> {
148    Ok(perform_begin(profile, token_state, http_agent)?
149        .response
150        .share_ids)
151}

153/// Queue freshly fetched shares for decryption, skipping ones already pending.
154///
155/// ### Arguments
156/// - `pending_shared_files`: The profile's queue of shares awaiting decryption
157/// - `shares`: The shares retrieved by the latest initial synchronization
158pub fn queue_pending_shares(
159    pending_shared_files: &Mutex<Vec<SharedFileResponse>>,
160    shares: Vec<SharedFileResponse>,
161) {
162    if shares.is_empty() {
163        return;
164    }
165    let mut pending = pending_shared_files.lock();
166    let already_queued: HashSet<&str> = pending.iter().map(|share| share.id.as_str()).collect();
167    let unseen: Vec<SharedFileResponse> = shares
168        .into_iter()
169        .filter(|share| !already_queued.contains(share.id.as_str()))
170        .collect();
171    drop(already_queued);
172    pending.extend(unseen);
173}

175/// Maximum number of concurrent share fetch worker threads.
176const MAX_FETCH_WORKERS: usize = 8;
177
178/// Fetch each announced share by id in parallel.
179///
180/// ### Arguments
181/// - `profile`: The server profile to fetch from
182/// - `token_state`: Per-profile JWT token state manager
183/// - `http_agent`: Shared HTTP agent for connection pooling
184/// - `share_ids`: The announced pending share ids to retrieve
185/// - `server_max_file_size`: Server-advertised max file size used to bound each response
186///
187/// ### Returns
188/// - `Vec<SharedFileResponse>`: The successfully fetched shares; failures are logged and skipped
189fn fetch_shares_for_ids(
190    profile: &ServerProfile,
191    token_state: &Arc<TokenStateManager>,
192    http_agent: &ureq::Agent,
193    share_ids: &[String],
194    server_max_file_size: u64,
195) -> Vec<SharedFileResponse> {
196    if share_ids.is_empty() {
197        return Vec::new();
198    }
199    let worker_count = MAX_FETCH_WORKERS.min(share_ids.len());
200    let next_index = AtomicUsize::new(0);
201    let results = Mutex::new(Vec::with_capacity(share_ids.len()));
202    std::thread::scope(|scope| {
203        for _ in 0..worker_count {
204            scope.spawn(|| {
205                loop {
206                    let index = next_index.fetch_add(1, Ordering::Relaxed);
207                    let Some(id) = share_ids.get(index) else {
208                        break;
209                    };
210                    match share::fetch_share_by_id(
211                        profile,
212                        token_state,
213                        http_agent,
214                        id,
215                        server_max_file_size,
216                    ) {
217                        Ok(s) => results.lock().push(s),
218                        Err(e) => log::warn!("Skipping share id {id}: {e}"),
219                    }
220                }
221            });
222        }
223    });
224    results.into_inner()
225}

227/// Initial synchronization with the server.
228///
229/// ### Arguments
230/// - `profile`: The server profile to synchronize with
231/// - `token_state`: Per-profile JWT token state manager
232/// - `http_agent`: Shared HTTP agent for connection pooling
233/// - `pending_ack_share_ids`: Ack set the fetched ids are registered into
234///
235/// ### Errors
236/// - Returns a `SynchronizationError` if the begin request fails (network failure,
237///   authentication failure, server older than the minimum supported Fulgurant
238///   version, or invalid response).
239///
240/// ### Returns
241/// - `Ok(InitialSyncOutcome)`: Begin response plus the server's advertised
242///   minimum supported Fulgur version
243/// - `Err(SynchronizationError)`: If the begin call failed
244#[allow(clippy::implicit_hasher)]
245pub fn initial_synchronization(
246    profile: &ServerProfile,
247    token_state: &Arc<TokenStateManager>,
248    http_agent: &ureq::Agent,
249    pending_ack_share_ids: &Arc<Mutex<HashSet<String>>>,
250) -> Result<InitialSyncOutcome, SynchronizationError> {
251    let BeginResponseOutcome {
252        response: begin,
253        version_header,
254    } = perform_begin(profile, token_state, http_agent)?;
255    let min_fulgur_version = begin.min_fulgur_version.clone();
256    let server_max_file_size = resolve_server_max_file_size(begin.max_file_size_bytes);
257    let shares = fetch_shares_for_ids(
258        profile,
259        token_state,
260        http_agent,
261        &begin.share_ids,
262        server_max_file_size,
263    );
264    {
265        let mut ack_set = pending_ack_share_ids.lock();
266        for share in &shares {
267            ack_set.insert(share.id.clone());
268        }
269    }
270    log::info!(
271        "Initial synchronization successful: {} announced, {} retrieved",
272        begin.share_ids.len(),
273        shares.len()
274    );
275    Ok(InitialSyncOutcome {
276        begin: BeginOutcome {
277            device_name: begin.device_name,
278            shares,
279            max_file_size_bytes: begin.max_file_size_bytes,
280        },
281        min_fulgur_version,
282        fulgurant_version: version_header,
283    })
284}

290    fn share(id: &str) -> SharedFileResponse {
291        SharedFileResponse {
292            id: id.to_string(),

298            expires_at: String::new(),

302    fn queued_ids(pending: &Mutex<Vec<SharedFileResponse>>) -> Vec<String> {
303        pending.lock().iter().map(|s| s.id.clone()).collect()
304    }
305
306    #[test]
307    fn queue_pending_shares_fills_an_empty_queue() {
308        let pending = Mutex::new(Vec::new());
309        queue_pending_shares(&pending, vec![share("a"), share("b")]);
310        assert_eq!(queued_ids(&pending), vec!["a", "b"]);
311    }
312
313    #[test]
314    fn queue_pending_shares_keeps_shares_the_server_no_longer_announces() {
315        let pending = Mutex::new(vec![share("a")]);
316        queue_pending_shares(&pending, vec![share("b")]);
317        assert_eq!(queued_ids(&pending), vec!["a", "b"]);
318    }
319
320    #[test]
321    fn queue_pending_shares_does_not_duplicate_already_queued_shares() {
322        let pending = Mutex::new(vec![share("a"), share("b")]);
323        queue_pending_shares(&pending, vec![share("b"), share("c")]);
324        assert_eq!(queued_ids(&pending), vec!["a", "b", "c"]);
325    }
326
327    #[test]
328    fn queue_pending_shares_leaves_the_queue_untouched_when_nothing_was_fetched() {
329        let pending = Mutex::new(vec![share("a")]);
330        queue_pending_shares(&pending, Vec::new());
331        assert_eq!(queued_ids(&pending), vec!["a"]);
