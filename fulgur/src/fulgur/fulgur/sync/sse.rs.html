72/// Error type for line reading with shutdown support
73enum ReadError {
74    /// I/O error during reading
75    Io(std::io::Error),
76    /// Shutdown was requested
77    Shutdown,
78}
79
80/// Read a line from a buffered reader with periodic shutdown checks
81///
82/// ### Arguments
83/// - `reader`: The buffered reader to read from
84/// - `shutdown_flag`: Atomic flag to check for shutdown requests
85///
86/// ### Returns
87/// - `Ok(Some(String))`: A line was read successfully
88/// - `Ok(None)`: End of stream reached
89/// - `Err(ReadError::Shutdown)`: Shutdown was requested
90/// - `Err(ReadError::Io)`: I/O error occurred
91fn read_line_with_timeout<R: Read>(
92    reader: &mut BufReader<R>,
93    shutdown_flag: &Arc<AtomicBool>,
94) -> Result<Option<String>, ReadError> {
95    let mut line = String::new();
96    let mut byte = [0u8; 1];
97
98    loop {
99        if shutdown_flag.load(Ordering::Relaxed) {
100            return Err(ReadError::Shutdown);
101        }
102        match reader.read(&mut byte) {
103            Ok(0) => {
104                // End of stream
105                if line.is_empty() {
106                    return Ok(None);
107                } else {
108                    return Ok(Some(line));
109                }
110            }
111            Ok(_) => {
112                if byte[0] == b'\n' {
113                    // End of line (handle both \n and \r\n)
114                    if line.ends_with('\r') {
115                        line.pop();
116                    }
117                    return Ok(Some(line));
118                } else {
119                    line.push(byte[0] as char);
120                }
121            }
122            Err(e)
123                if e.kind() == std::io::ErrorKind::WouldBlock
124                    || e.kind() == std::io::ErrorKind::TimedOut =>
125            {
126                thread::sleep(Duration::from_millis(10));
127                continue;
128            }
129            Err(e) if e.kind() == std::io::ErrorKind::Interrupted => {
130                continue;
131            }
132            Err(e) => {
133                return Err(ReadError::Io(e));
134            }
135        }
136    }
137}

139/// Connect to SSE (Server-Sent Events) endpoint on the sync serverfor real-time notifications
140///
141/// ### Description
142/// Establishes a persistent connection to the server's SSE endpoint to receive:
143/// - Heartbeat events to keep connection alive
144/// - Share notifications when files are shared from other devices
145///
146/// The connection runs in a background thread and automatically reconnects on failure.
147///
148/// ### Arguments
149/// - `synchronization_settings`: The synchronization settings containing server URL, email, and key
150/// - `event_tx`: Channel sender for sending SSE events to the main thread
151/// - `shutdown_flag`: Atomic boolean flag to signal the SSE thread to shutdown
152/// - `sync_server_connection_status`: Arc-wrapped connection status to update on connection/disconnection
153/// - `http_agent`: Shared HTTP agent for connection pooling
154///
155/// ### Returns
156/// - `Ok(())`: If the SSE connection thread was spawned successfully
157/// - `Err(SynchronizationError)`: If required settings are missing
158pub fn connect_sse(
159    synchronization_settings: &SynchronizationSettings,
160    event_tx: Sender<SseEvent>,
161    shutdown_flag: Arc<AtomicBool>,
162    sync_server_connection_status: Arc<Mutex<SynchronizationStatus>>,
163    token_state: Arc<TokenStateManager>,
164    http_agent: Arc<ureq::Agent>,
165) -> Result<thread::JoinHandle<()>, SynchronizationError> {
166    let server_url = synchronization_settings
167        .server_url
168        .clone()
169        .ok_or(SynchronizationError::ServerUrlMissing)?;
170    let sse_url = format!("{}/api/sse", server_url);
171    let settings_clone = synchronization_settings.clone();
172    let token_state_clone = Arc::clone(&token_state);
173    let http_agent_clone = Arc::clone(&http_agent);
174    let handle = thread::spawn(move || {
175        // Exponential backoff for reconnection attempts (1s, 2s, 4s, 8s... up to 5 minutes)
176        let mut backoff = BackoffCalculator::default_settings();
177
178        loop {
179            if shutdown_flag.load(Ordering::Relaxed) {
180                log::info!("SSE connection shutdown requested, stopping...");
181                break;
182            }
183            let token = match get_valid_token(
184                &settings_clone,
185                Arc::clone(&token_state_clone),
186                &http_agent_clone,
187            ) {
188                Ok(t) => t,
189                Err(e) => {
190                    log::error!("Failed to get valid token for SSE: {}", e);
191                    set_sync_server_connection_status(
192                        sync_server_connection_status.clone(),
193                        SynchronizationStatus::AuthenticationFailed,
194                    );
195                    let delay = backoff.record_failure();
196                    log::info!("Retrying SSE connection after {:?}", delay);
197                    thread::sleep(delay);
198                    continue;
199                }
200            };
201            log::info!("Connecting to SSE endpoint: {}", sse_url);
202            let response = match http_agent_clone
203                .get(&sse_url)
204                .header("Authorization", &format!("Bearer {}", token))
205                .header("Accept", "text/event-stream")
206                .call()
207            {
208                Ok(resp) => {
209                    set_sync_server_connection_status(
210                        sync_server_connection_status.clone(),
211                        SynchronizationStatus::Connected,
212                    );
213                    log::info!("SSE connection established");
214                    backoff.record_success(); // Reset backoff on successful connection
215                    resp
216                }
217                Err(e) => {
218                    log::error!("SSE connection failed: {}", e);
219                    set_sync_server_connection_status(
220                        sync_server_connection_status.clone(),
221                        SynchronizationStatus::Disconnected,
222                    );
223                    event_tx.send(SseEvent::Error(e.to_string())).ok();
224                    if shutdown_flag.load(Ordering::Relaxed) {
225                        log::info!("SSE connection shutdown requested, stopping...");
226                        break;
227                    }
228                    let delay = backoff.record_failure();
229                    log::info!("Retrying SSE connection after {:?}", delay);
230                    thread::sleep(delay);
231                    continue;
232                }
233            };
234            let mut response = response;
235            let mut reader = std::io::BufReader::new(response.body_mut().as_reader());
236            let mut current_event_type = String::new();
237            let mut current_data = String::new();
238
239            loop {
240                if shutdown_flag.load(Ordering::Relaxed) {
241                    log::info!(
242                        "SSE connection shutdown requested during event reading, stopping..."
243                    );
244                    break;
245                }
246
247                let line_result = read_line_with_timeout(&mut reader, &shutdown_flag);
248                match line_result {
249                    Ok(Some(line)) => {
250                        if line.starts_with("event:") {
251                            current_event_type =
252                                line.trim_start_matches("event:").trim().to_string();
253                        } else if line.starts_with("data:") {
254                            let fragment = line.trim_start_matches("data:").trim();
255                            if current_data.len() + fragment.len() > MAX_SSE_EVENT_DATA_BYTES {
256                                log::warn!(
257                                    "SSE event data exceeds size limit ({} bytes), discarding",
258                                    MAX_SSE_EVENT_DATA_BYTES
259                                );
260                                current_data.clear();
261                                current_event_type.clear();
262                                continue;
263                            }
264                            current_data.push_str(fragment);
265                        } else if line.is_empty() && !current_data.is_empty() {
266                            log::info!("SSE event type: {}", current_event_type);
267                            log::debug!("SSE event received ({} bytes)", current_data.len());
268                            let event = SseEvent::parse(&current_event_type, &current_data);
269                            if let Err(e) = event_tx.send(event) {
270                                log::error!("Failed to send SSE event: {}", e);
271                                break;
272                            }
273                            current_event_type.clear();
274                            current_data.clear();
275                        }
276                    }
277                    Ok(None) => {
278                        // End of stream
279                        log::info!("SSE stream ended");
280                        break;
281                    }
282                    Err(ReadError::Shutdown) => {
283                        log::info!("SSE connection shutdown requested");
284                        break;
285                    }
286                    Err(ReadError::Io(e)) => {
287                        log::error!("SSE stream error: {}", e);
288                        set_sync_server_connection_status(
289                            sync_server_connection_status.clone(),
290                            SynchronizationStatus::Disconnected,
291                        );
292                        event_tx.send(SseEvent::Error(e.to_string())).ok();
293                        break;
294                    }
295                }
296            }
297            if shutdown_flag.load(Ordering::Relaxed) {
298                log::info!("SSE connection shutdown requested, stopping...");
299                break;
300            }
301            // When the connection is lost
302            let delay = backoff.record_failure();
303            log::warn!("SSE connection closed, reconnecting after {:?}", delay);
304            set_sync_server_connection_status(
305                sync_server_connection_status.clone(),
306                SynchronizationStatus::Disconnected,
307            );
308            thread::sleep(delay);
309        }
310    });
311    Ok(handle)
312}

334/// SSE Event types that can be received from the server
335#[derive(Debug, Clone)]
336pub enum SseEvent {
337    /// Heartbeat event to keep connection alive (sent every ~30s)
338    Heartbeat { timestamp: String },
339    /// File share notification with full share details
340    ShareAvailable(ShareNotification),
341    /// Error event for connection or parsing errors
342    Error(String),
343}

345impl SseEvent {
346    /// Parse an SSE event from the event type and data
347    ///
348    /// ### Arguments
349    /// - `event_type`: The SSE event type (e.g., "heartbeat", "share_available")
350    /// - `data`: The JSON data for the event
351    ///
352    /// ### Returns
353    /// - `SseEvent`: The parsed event
354    pub fn parse(event_type: &str, data: &str) -> Self {
355        match event_type {
356            "heartbeat" => match serde_json::from_str::<HeartbeatData>(data) {
357                Ok(hb) => SseEvent::Heartbeat {
358                    timestamp: hb.timestamp,
359                },
360                Err(e) => {
361                    log::warn!("Failed to parse heartbeat: {}", e);
362                    SseEvent::Heartbeat {
363                        timestamp: String::new(),
364                    }
365                }
366            },
367            "share_available" => match serde_json::from_str::<ShareNotification>(data) {
368                Ok(notification) => SseEvent::ShareAvailable(notification),
369                Err(e) => {
370                    log::error!("Failed to parse share notification: {}", e);
371                    SseEvent::Error(format!("Invalid share notification: {}", e))
372                }
373            },
374            "" => {
375                log::error!("Unknown event with no event type");
376                SseEvent::Error("Unknown event with no event type".to_string())
377            }
378            _ => {
379                log::warn!("Unknown SSE event type: {}", event_type);
380                SseEvent::Error(format!("Unknown event type: {}", event_type))
381            }
382        }
383    }
384}

399    /// Restart the SSE connection with new settings
400    ///
401    /// ### Description
402    /// Stops the current SSE connection and starts a new one with the updated settings.
403    /// Waits for the previous SSE thread to exit (bounded by `SSE_THREAD_SHUTDOWN_TIMEOUT`)
404    /// in a background thread to avoid blocking the UI.
405    ///
406    /// Should be called when synchronization settings (server URL, email, or key) change.
407    ///
408    /// ### Arguments
409    /// - `cx`: The context of the application.
410    pub fn restart_sse_connection(&mut self, cx: &mut Context<Self>) {
411        if let Some(ref shutdown_flag) = self.sse_state.sse_shutdown_flag {
412            log::info!("Signaling SSE connection to shutdown...");
413            shutdown_flag.store(true, Ordering::Relaxed);
414        }
415        let old_handle = self.sse_state.sse_thread_handle.lock().take();
416        let (sse_tx, sse_rx) = std::sync::mpsc::channel();
417        let sse_shutdown_flag = Arc::new(AtomicBool::new(false));
418        self.sse_state.sse_events = Some(sse_rx);
419        self.sse_state.sse_event_tx = Some(sse_tx.clone());
420        self.sse_state.sse_shutdown_flag = Some(sse_shutdown_flag.clone());
421        if self
422            .settings
423            .app_settings
424            .synchronization_settings
425            .is_synchronization_activated
426        {
427            let settings = self.settings.clone();
428            let sync_status = self.shared_state(cx).sync_state.connection_status.clone();
429            let token_state = Arc::clone(&self.shared_state(cx).sync_state.token_state);
430            let http_agent = Arc::clone(&self.shared_state(cx).http_agent);
431            let handle_storage = Arc::clone(&self.sse_state.sse_thread_handle);
432            thread::spawn(move || {
433                // Wait for previous SSE thread to exit before starting new connection
434                if let Some(handle) = old_handle {
435                    let deadline = Instant::now() + SSE_THREAD_SHUTDOWN_TIMEOUT;
436                    while !handle.is_finished() && Instant::now() < deadline {
437                        thread::sleep(Duration::from_millis(100));
438                    }
439                    if handle.is_finished() {
440                        let _ = handle.join();
441                        log::info!("Previous SSE thread exited");
442                    } else {
443                        log::warn!(
444                            "Previous SSE thread still running after {:?}, proceeding with new connection",
445                            SSE_THREAD_SHUTDOWN_TIMEOUT
446                        );
447                    }
448                }
449                // Small delay to ensure old connection is fully stopped
450                thread::sleep(Duration::from_millis(200));
451                match super::synchronization::initial_synchronization(
452                    &settings.app_settings.synchronization_settings,
453                    Arc::clone(&token_state),
454                    &http_agent,
455                ) {
456                    Ok(_) => {
457                        log::info!("Initial sync succeeded, starting new SSE connection");
458                        match connect_sse(
459                            &settings.app_settings.synchronization_settings,
460                            sse_tx,
461                            sse_shutdown_flag,
462                            sync_status,
463                            token_state,
464                            http_agent,
465                        ) {
466                            Ok(new_handle) => {
467                                *handle_storage.lock() = Some(new_handle);
468                            }
469                            Err(e) => {
470                                log::error!("Failed to start new SSE connection: {}", e);
471                            }
472                        }
473                    }
474                    Err(e) => {
475                        log::error!("Initial sync failed, not starting SSE: {}", e);
476                    }
477                }
478            });
479        } else {
480            log::info!("Synchronization is not activated, SSE connection not started");
481        }
482    }

484    /// Handle SSE (Server-Sent Events) from the sync server
485    ///
486    /// ### Arguments
487    /// - `event`: The SSE event to handle
488    /// - `window`: The window to show notifications in
489    /// - `cx`: The application context
490    pub fn handle_sse_event(
491        &mut self,
492        event: SseEvent,
493        window: &mut Window,
494        cx: &mut Context<Self>,
495    ) {
496        // Debounce: ignore events within 500ms of last event
497        let now = Instant::now();
498        if let Some(last_time) = self.sse_state.last_sse_event
499            && now.duration_since(last_time) < Duration::from_millis(500)
500        {
501            return;
502        }
503        self.sse_state.last_sse_event = Some(now);
504        match event {
505            SseEvent::Heartbeat { timestamp } => {
506                log::debug!("SSE heartbeat received: {}", timestamp);
507                let was_disconnected = !self
508                    .shared_state(cx)
509                    .sync_state
510                    .connection_status
511                    .lock()
512                    .is_connected();
513                {
514                    let mut last_heartbeat = self.shared_state(cx).sync_state.last_heartbeat.lock();
515                    *last_heartbeat = Some(now);
516                }
517                if was_disconnected {
518                    *self.shared_state(cx).sync_state.connection_status.lock() =
519                        SynchronizationStatus::Connected;
520                    log::info!("Connection restored - heartbeat received after timeout");
521                }
522            }
523            SseEvent::ShareAvailable(notification) => {
524                if notification.content.len() > MAX_SYNC_SHARE_PAYLOAD_BYTES {
525                    log::warn!(
526                        "Dropping shared file '{}' from device {}: encrypted payload exceeds {} bytes",
527                        notification.file_name,
528                        notification.source_device_id,
529                        MAX_SYNC_SHARE_PAYLOAD_BYTES
530                    );
531                    return;
532                }
533                let safe_filename = sanitize_filename(&notification.file_name);
534                log::info!(
535                    "File shared from device {}: {} (sanitized: {})",
536                    notification.source_device_id,
537                    notification.file_name,
538                    safe_filename
539                );
540                {
541                    let mut files = self.shared_state(cx).sync_state.pending_shared_files.lock();
542                    let shared_file = fulgur_common::api::shares::SharedFileResponse {
543                        id: notification.share_id,
544                        source_device_id: notification.source_device_id.clone(),
545                        file_name: safe_filename.clone(),
546                        file_size: notification.file_size as i32,
547                        content: notification.content,
548                        created_at: notification.created_at,
549                        expires_at: notification.expires_at,
550                    };
551                    files.push(shared_file);
552                }
553                let message = SharedString::from(format!("New file received: {}", safe_filename));
554                window.push_notification((NotificationType::Info, message), cx);
555            }
556            SseEvent::Error(err) => {
557                log::error!("SSE error: {}", err);
558            }
559        }
560    }

562    /// Collect and process Server-Sent Events from the sync server:
563    /// - Heartbeat: Periodic keepalive messages to detect connection timeouts
564    /// - ShareAvailable: Another device has shared a file (triggers file download and decryption)
565    /// - Error: Connection or server errors (updates connection status in UI)
566    ///
567    /// ### Arguments
568    /// - `window`: The window to handle events in
569    /// - `cx`: The application context
570    pub fn process_sse_events(&mut self, window: &mut Window, cx: &mut Context<Self>) {
571        let sse_events = collect_events(&self.sse_state.sse_events);
572        for event in sse_events {
573            self.handle_sse_event(event, window, cx);
574        }
575    }

745                        timestamp: "ts1".to_string(),
746                    },
747                    window,
748                    cx,
749                );
750                // Force Disconnected so we can detect whether the second heartbeat
751                // is processed (it would restore Connected if not debounced).
752                *this.shared_state(cx).sync_state.connection_status.lock() =
753                    SynchronizationStatus::Disconnected;
754                // Second heartbeat arrives immediately — within the 500ms debounce window.
755                this.handle_sse_event(
756                    SseEvent::Heartbeat {
757                        timestamp: "ts2".to_string(),
758                    },
759                    window,
760                    cx,
761                );
762                // Status must still be Disconnected: the rapid second event was debounced.
763                assert!(
764                    !this
765                        .shared_state(cx)
766                        .sync_state
767                        .connection_status
768                        .lock()
769                        .is_connected(),
770                    "Second event within the 500ms debounce window must be ignored"
771                );
772            });
773        });
