10impl Fulgur {
11    /// Add a new server profile to the configuration.
12    ///
13    /// ### Arguments
14    /// - `profile`: The fully formed profile to register.
15    /// - `cx`: The Fulgur context.
16    ///
17    /// ### Errors
18    /// Returns an error if persisting the updated settings fails.
19    ///
20    /// ### Returns
21    /// - `Ok(())`: The profile was added and settings persisted.
22    /// - `Err(anyhow::Error)`: The settings could not be saved; the in-memory
23    ///   profile list is left mutated but the on-disk store is unchanged.
24    pub fn add_profile(
25        &mut self,
26        mut profile: ServerProfile,
27        cx: &mut Context<Self>,
28    ) -> anyhow::Result<()> {
29        if profile.is_active
30            && let Err(e) = ensure_profile_keypair(&mut profile)
31        {
32            log::error!(
33                "Failed to ensure keypair for new profile '{}': {e}",
34                profile.name
35            );
36        }
37        let profile_id = profile.id.clone();
38        self.settings
39            .app_settings
40            .synchronization_settings
41            .profiles
42            .push(profile);
43        let _ = Fulgur::shared_state(cx).sync_state_for(&profile_id);
44        self.update_and_propagate_settings(cx)
45    }
46
47    /// Update an existing profile and propagate the change.
48    ///
49    /// ### Arguments
50    /// - `profile_id`: The id of the profile to mutate.
51    /// - `mutator`: Closure invoked with the mutable profile reference.
52    /// - `cx`: The Fulgur context.
53    ///
54    /// ### Errors
55    /// Returns an error if persisting the updated settings fails.
56    ///
57    /// ### Returns
58    /// - `Ok(true)`: The profile was found, mutated, and settings persisted.
59    /// - `Ok(false)`: No profile with the given id exists.
60    /// - `Err(anyhow::Error)`: Persistence failed.
61    pub fn update_profile<F>(
62        &mut self,
63        profile_id: &str,
64        mutator: F,
65        cx: &mut Context<Self>,
66    ) -> anyhow::Result<bool>
67    where
68        F: FnOnce(&mut ServerProfile),
69    {
70        let found = self
71            .settings
72            .app_settings
73            .synchronization_settings
74            .find_profile_mut(profile_id)
75            .is_some_and(|profile| {
76                mutator(profile);
77                true
78            });
79        if !found {
80            return Ok(false);
81        }
82
83        if let Some(profile) = self
84            .settings
85            .app_settings
86            .synchronization_settings
87            .find_profile_mut(profile_id)
88            && profile.is_active
89            && let Err(e) = ensure_profile_keypair(profile)
90        {
91            log::error!(
92                "Failed to ensure keypair for profile '{}': {e}",
93                profile.name
94            );
95        }
96        self.update_and_propagate_settings(cx)?;
97        Ok(true)
98    }
99
100    /// Delete a server profile and tear down its associated state.
101    ///
102    /// ### Arguments
103    /// - `profile_id`: The id of the profile to delete.
104    /// - `cx`: The Fulgur context.
105    ///
106    /// ### Errors
107    /// Returns an error if persisting the updated settings fails.
108    ///
109    /// ### Returns
110    /// - `Ok(true)`: The profile was found and removed.
111    /// - `Ok(false)`: No profile with the given id exists.
112    /// - `Err(anyhow::Error)`: Persistence failed; in-memory state is still
113    ///   updated, on-disk settings may diverge until the next save.
114    pub fn delete_profile(
115        &mut self,
116        profile_id: &str,
117        cx: &mut Context<Self>,
118    ) -> anyhow::Result<bool> {
119        let exists = self
120            .settings
121            .app_settings
122            .synchronization_settings
123            .find_profile(profile_id)
124            .is_some();
125        if !exists {
126            return Ok(false);
127        }
128        let sse_worker = Fulgur::shared_state(cx)
129            .sync_state_for(profile_id)
130            .sse
131            .lock()
132            .worker
133            .take();
134        if let Some(worker) = sse_worker {
135            worker.signal_shutdown();
136            // Drop off the UI thread: Worker::drop joins with a bounded timeout.
137            cx.background_executor()
138                .spawn(async move { drop(worker) })
139                .detach();
140        }
141        if let Err(e) = save_private_key_to_keychain(profile_id, None) {
142            log::warn!("Failed to remove private key for profile '{profile_id}': {e}");
143        }
144        if let Err(e) = save_device_api_key_to_keychain(profile_id, None) {
145            log::warn!("Failed to remove device API key for profile '{profile_id}': {e}");
146        }
147        let _ = Fulgur::shared_state(cx).remove_sync_state(profile_id);
148        self.settings
149            .app_settings
150            .synchronization_settings
151            .profiles
152            .retain(|p| p.id != profile_id);
153        self.update_and_propagate_settings(cx)?;
154        Ok(true)
155    }
156}

175    /// Initialize globals and open a test window with a Root-mounted Fulgur.
176    fn setup_fulgur(cx: &mut TestAppContext) -> (Entity<Fulgur>, VisualTestContext) {
177        cx.update(|cx| {
178            gpui_component::init(cx);
179            let mut settings = Settings::new();
180            settings.editor_settings.watch_files = false;
181            let pending_files: Arc<Mutex<Vec<PathBuf>>> = Arc::new(Mutex::new(Vec::new()));
182            cx.set_global(SharedAppState::new(settings, pending_files, None));
183            cx.set_global(WindowManager::new());
184        });
185        let fulgur_slot: RefCell<Option<Entity<Fulgur>>> = RefCell::new(None);
186        let window = cx
187            .update(|cx| {
188                cx.open_window(WindowOptions::default(), |window, cx| {
189                    let window_id = window.window_handle().window_id();
190                    let fulgur = Fulgur::new(window, cx, window_id, usize::MAX);
191                    *fulgur_slot.borrow_mut() = Some(fulgur.clone());
192                    cx.new(|cx| gpui_component::Root::new(fulgur, window, cx))
193                })
194            })
195            .expect("failed to open test window");
196        let visual_cx = VisualTestContext::from_window(window.into(), cx);
197        visual_cx.run_until_parked();
198        let fulgur = fulgur_slot
199            .into_inner()
200            .expect("failed to capture Fulgur entity");
201        (fulgur, visual_cx)
202    }

240    fn test_add_profile_generates_keypair_for_active_profile(cx: &mut TestAppContext) {
241        // Reproduces the user-reported "Missing encryption key" path: a
242        // freshly added active profile must have its keypair generated as
243        // part of `add_profile`, otherwise `initial_synchronization` bails
244        // out before any network call.
245        let (fulgur, mut visual_cx) = setup_fulgur(cx);
246        visual_cx.update(|_window, cx| {
247            fulgur.update(cx, |this, cx| {
248                let mut profile = ServerProfile::new("Active-On-Add");
249                profile.is_active = true;
250                let id = profile.id.clone();
251                assert!(profile.public_key.is_none());
252                this.add_profile(profile, cx).expect("add should persist");
253
254                let stored = this
255                    .settings
256                    .app_settings
257                    .synchronization_settings
258                    .find_profile(&id)
259                    .expect("profile should be in settings after add");
260                assert!(
261                    stored.public_key.is_some(),
262                    "public_key must be set on an active profile after add_profile"
263                );
264                assert!(
265                    load_private_key_from_keychain(&id)
266                        .expect("keychain access should succeed")
267                        .is_some(),
268                    "private key must be persisted in keychain for an active profile"
269                );
270                // Cleanup: drop the keychain entry so other tests start fresh.
271                let _ = save_private_key_to_keychain(&id, None);
272            });
273        });
274    }
