31/// Maximum number of bytes accepted from small JSON HTTP responses (token, ping).
32pub const MAX_HTTP_SMALL_RESPONSE_BYTES: u64 = 64 * 1024;
33
34/// Maximum number of bytes accepted from the devices listing response
35/// (`GET /api/devices`).
36pub const MAX_HTTP_DEVICES_RESPONSE_BYTES: u64 = 1024 * 1024;
37
38/// Top-level JSON framing overhead allowance for a single-share response
39/// (object braces, sibling fields like `id`, `file_name`, timestamps).
40const SHARE_RESPONSE_FRAMING_BYTES: u64 = 4 * 1024;
41
42/// Maximum number of bytes accepted from a single-share HTTP response
43/// (`GET /api/shares/:id`).
44pub const MAX_HTTP_SINGLE_SHARE_RESPONSE_BYTES: u64 = (share::MAX_SYNC_SHARE_PAYLOAD_BYTES as u64)
45    + (share::JSON_OVERHEAD_PER_SHARE_BYTES as u64)
46    + SHARE_RESPONSE_FRAMING_BYTES;
47
48/// Maximum number of bytes accepted from the legacy `POST /api/begin` response. Sized to fit the worst-case bundle.
49const MAX_HTTP_V1_BEGIN_RESPONSE_BYTES: u64 = (share::MAX_PENDING_SHARES_PER_RESPONSE as u64)
50    * ((share::MAX_SYNC_SHARE_PAYLOAD_BYTES as u64)
51        + (share::JSON_OVERHEAD_PER_SHARE_BYTES as u64))
52    + SHARE_RESPONSE_FRAMING_BYTES;
53
54/// Compute the wire-size cap for a bulk `GET /api/shares` drain, derived from
55/// the server's advertised maximum file size.
56///
57/// ### Arguments
58/// - `server_max_file_size`: The server-advertised max file size in bytes, or
59///   `u64::MAX` when the server reports no limit.
60///
61/// ### Returns
62/// - `u64`: The maximum number of bytes to accept from the bulk drain response.
63pub fn max_http_bulk_shares_response_bytes(server_max_file_size: u64) -> u64 {
64    if server_max_file_size == u64::MAX {
65        return MAX_HTTP_V1_BEGIN_RESPONSE_BYTES;
66    }
67    let per_share_wire = server_max_file_size
68        .saturating_mul(2)
69        .saturating_add(share::JSON_OVERHEAD_PER_SHARE_BYTES as u64);
70    (share::MAX_PENDING_SHARES_PER_RESPONSE as u64)
71        .saturating_mul(per_share_wire)
72        .saturating_add(SHARE_RESPONSE_FRAMING_BYTES)
73}

75/// Handle ureq errors and convert them to `SynchronizationError` with appropriate logging
76///
77/// ### Description
78/// Centralizes ureq error handling logic that was duplicated across sync modules.
79/// Maps all ureq error variants to appropriate `SynchronizationError` types and logs
80/// the error with context.
81///
82/// ### Arguments
83/// - `error`: The ureq error to handle
84/// - `context`: Human-readable context for logging (e.g., "Failed to get devices")
85///
86/// ### Returns
87/// - `SynchronizationError`: The mapped synchronization error
88pub fn handle_ureq_error(error: ureq::Error, context: &str) -> SynchronizationError {
89    match error {
90        ureq::Error::StatusCode(code) => {
91            log::error!("{context}: HTTP status {code}");
92            if code == 401 || code == 403 {
93                SynchronizationError::AuthenticationFailed
94            } else if code == 400 {
95                SynchronizationError::BadRequest
96            } else if code == 413 {
97                SynchronizationError::ContentTooLarge {
98                    file_size: 0,
99                    max_size: 0,
100                }
101            } else {
102                SynchronizationError::ServerError(code)
103            }
104        }
105        ureq::Error::Io(io_error) => {
106            log::error!("{context} (IO): {io_error}");
107            match io_error.kind() {
108                std::io::ErrorKind::ConnectionRefused => SynchronizationError::ConnectionFailed,
109                std::io::ErrorKind::TimedOut => SynchronizationError::Timeout(io_error.to_string()),
110                std::io::ErrorKind::ConnectionReset | std::io::ErrorKind::ConnectionAborted => {
111                    SynchronizationError::ConnectionFailed
112                }
113                std::io::ErrorKind::AddrNotAvailable => SynchronizationError::HostNotFound,
114                _ => SynchronizationError::Other(io_error.to_string()),
115            }
116        }
117        ureq::Error::ConnectionFailed => {
118            log::error!("{context}: Connection failed");
119            SynchronizationError::ConnectionFailed
120        }
121        ureq::Error::HostNotFound => {
122            log::error!("{context}: Host not found");
123            SynchronizationError::HostNotFound
124        }
125        ureq::Error::Timeout(timeout) => {
126            log::error!("{context}: Timeout ({timeout})");
127            SynchronizationError::Timeout(timeout.to_string())
128        }
129        e => {
130            log::error!("{context}: {e}");
131            SynchronizationError::Other(e.to_string())
132        }
133    }
134}

136/// Validate and persist the server-advertised `max_file_size_bytes`.
137///
138/// ### Arguments
139/// - `atomic`: The shared atomic holding the current cap
140/// - `advertised`: The `Option<u64>` received from the server's response
141pub fn store_server_max_file_size(atomic: &std::sync::atomic::AtomicU64, advertised: Option<u64>) {
142    let value = match advertised {
143        None => {
144            log::info!("Server max file size: no limit");
145            u64::MAX
146        }
147        Some(0) => {
148            log::warn!(
149                "Server advertised max_file_size_bytes = 0 (would disable sharing); falling back to {} bytes",
150                share::MAX_SYNC_SHARE_PAYLOAD_BYTES
151            );
152            share::MAX_SYNC_SHARE_PAYLOAD_BYTES as u64
153        }
154        Some(n) => {
155            log::info!("Server max file size: {n} bytes");
156            n
157        }
158    };
159    atomic.store(value, std::sync::atomic::Ordering::Release);
160}

162/// Initial synchronization with the server.
163///
164/// ### Description
165/// Attempts the v2 begin flow first (`POST /api/v2/begin` + per-share fetch).
166/// If the server returns HTTP 404 (endpoint not deployed yet), falls back to
167/// the legacy v1 flow (`POST /api/begin`) for compatibility with Fulgurant
168/// servers that have not been upgraded.
169///
170/// ### Arguments
171/// - `profile`: The server profile to synchronize with
172/// - `token_state`: Per-profile JWT token state manager
173/// - `http_agent`: Shared HTTP agent for connection pooling
174///
175/// ### Errors
176/// Returns a `SynchronizationError` if both the v2 and the legacy v1 begin
177/// requests fail (network failure, authentication failure, or invalid response).
178///
179/// ### Returns
180/// - `Ok(BeginResponse)`: Device name, max file size, and pending shares
181/// - `Err(SynchronizationError)`: If both v2 and v1 begin calls failed
182pub fn initial_synchronization(
183    profile: &ServerProfile,
184    token_state: &Arc<TokenStateManager>,
185    http_agent: &ureq::Agent,
186) -> Result<BeginResponse, SynchronizationError> {
187    match initial_synchronization_v2(profile, token_state, http_agent) {
188        Ok(response) => Ok(response),
189        Err(SynchronizationError::ServerError(404)) => {
190            log::warn!(
191                "Server does not support /api/v2/begin (404); falling back to legacy /api/begin"
192            );
193            initial_synchronization_v1(profile, token_state, http_agent)
194        }
195        Err(e) => Err(e),
196    }
197}

199/// Initial synchronization via the v2 begin flow.
200///
201/// ### Description
202/// Calls `POST /api/v2/begin` to update the device's encryption key and obtain
203/// the list of pending share IDs, then fetches each share individually via
204/// `GET /api/shares/:id`.
205///
206/// ### Arguments
207/// - `profile`: The server profile to synchronize with
208/// - `token_state`: Per-profile JWT token state manager
209/// - `http_agent`: Shared HTTP agent for connection pooling
210///
211/// ### Returns
212/// - `Ok(BeginResponse)`: Device name, max file size, and successfully fetched pending shares
213/// - `Err(SynchronizationError)`: If the v2 begin call failed or returned an invalid response
214fn initial_synchronization_v2(
215    profile: &ServerProfile,
216    token_state: &Arc<TokenStateManager>,
217    http_agent: &ureq::Agent,
218) -> Result<BeginResponse, SynchronizationError> {
219    let Some(server_url) = profile.server_url.clone() else {
220        return Err(SynchronizationError::ServerUrlMissing);
221    };
222    let Some(public_key) = profile.public_key.clone() else {
223        return Err(SynchronizationError::MissingEncryptionKey);
224    };
225    let token = get_valid_token(profile, token_state, http_agent)?;
226    let begin_url = format!("{server_url}/api/v2/begin");
227    let payload = InitialSynchronizationPayload { public_key };
228    let mut response = http_agent
229        .post(begin_url)
230        .header("Authorization", &format!("Bearer {token}"))
231        .send_json(payload)
232        .map_err(|e| handle_ureq_error(e, "Failed to begin synchronization (v2)"))?;
233    let body = match response
234        .body_mut()
235        .with_config()
236        .limit(MAX_HTTP_SMALL_RESPONSE_BYTES)
237        .read_to_string()
238    {
239        Ok(body) => body,
240        Err(e) => {
241            log::error!("Failed to read v2 begin response body: {e}");
242            return Err(SynchronizationError::Other(e.to_string()));
243        }
244    };
245    let begin_v2: BeginV2Response = match serde_json::from_str(&body) {
246        Ok(response) => response,
247        Err(e) => {
248            log::error!("Failed to parse v2 begin response body: {e}");
249            return Err(SynchronizationError::InvalidResponse(e.to_string()));
250        }
251    };
252    if begin_v2.share_ids.len() > share::MAX_PENDING_SHARES_PER_RESPONSE {
253        log::error!(
254            "Server returned {} pending share ids, exceeding the client limit of {}",
255            begin_v2.share_ids.len(),
256            share::MAX_PENDING_SHARES_PER_RESPONSE
257        );
258        return Err(SynchronizationError::InvalidResponse(format!(
259            "Server returned too many pending share ids ({} > {})",
260            begin_v2.share_ids.len(),
261            share::MAX_PENDING_SHARES_PER_RESPONSE
262        )));
263    }
264    let shares: Vec<_> = std::thread::scope(|scope| {
265        let handles: Vec<_> = begin_v2
266            .share_ids
267            .iter()
268            .map(|id| {
269                scope.spawn(move || {
270                    (
271                        id.as_str(),
272                        share::fetch_share_by_id(profile, token_state, http_agent, id),
273                    )
274                })
275            })
276            .collect();
277        handles
278            .into_iter()
279            .filter_map(|h| match h.join() {
280                Ok((_, Ok(s))) => Some(s),
281                Ok((id, Err(e))) => {
282                    log::warn!("Skipping share id {id}: {e}");
283                    None
284                }
285                Err(_) => {
286                    log::error!("Fetch share worker thread panicked");
287                    None
288                }
289            })
290            .collect()
291    });
292    log::info!(
293        "Initial synchronization (v2) successful: {} announced, {} retrieved",
294        begin_v2.share_ids.len(),
295        shares.len()
296    );
297    Ok(BeginResponse {
298        device_name: begin_v2.device_name,
299        shares,
300        max_file_size_bytes: begin_v2.max_file_size_bytes,
301    })
302}

304/// Initial synchronization via the legacy v1 begin flow.
305///

318/// - `Err(SynchronizationError)`: If the v1 begin call failed or returned an invalid response
319fn initial_synchronization_v1(
320    profile: &ServerProfile,
321    token_state: &Arc<TokenStateManager>,
322    http_agent: &ureq::Agent,
323) -> Result<BeginResponse, SynchronizationError> {
324    let Some(server_url) = profile.server_url.clone() else {
325        return Err(SynchronizationError::ServerUrlMissing);
326    };
327    let Some(public_key) = profile.public_key.clone() else {
328        return Err(SynchronizationError::MissingEncryptionKey);
329    };
330    let token = get_valid_token(profile, token_state, http_agent)?;
331    let begin_url = format!("{server_url}/api/begin");
332    let payload = InitialSynchronizationPayload { public_key };
333    let mut response = http_agent
334        .post(begin_url)
335        .header("Authorization", &format!("Bearer {token}"))
336        .send_json(payload)
337        .map_err(|e| handle_ureq_error(e, "Failed to begin synchronization (v1)"))?;
338    let body = match response
339        .body_mut()
340        .with_config()
341        .limit(MAX_HTTP_V1_BEGIN_RESPONSE_BYTES)
342        .read_to_string()
343    {
344        Ok(body) => body,
345        Err(e) => {
346            log::error!("Failed to read v1 begin response body: {e}");
347            return Err(SynchronizationError::Other(e.to_string()));
348        }
349    };
350    let mut begin_response: BeginResponse = match serde_json::from_str(&body) {

363    );

367/// Ping an authenticated Fulgurant server endpoint to test connectivity and credentials.
368///
369/// ### Arguments
370/// - `server_url`: The base URL of the server (e.g. `https://example.com`).
371/// - `token`: A valid JWT Bearer token.
372/// - `http_agent`: Shared HTTP agent.
373///
374/// ### Errors
375/// Returns a `SynchronizationError` if the server is unreachable, the token is
376/// rejected, or the response cannot be parsed as the expected `ok: true` body.
377///
378/// ### Returns
379/// - `Ok(())`: Server responded with `ok: true`.
380/// - `Err(SynchronizationError)`: Server is unreachable, auth failed, or returned an unexpected response.
381pub fn ping_server(
382    server_url: &str,
383    token: &str,
384    http_agent: &ureq::Agent,
385) -> Result<(), SynchronizationError> {
386    let ping_url = format!("{server_url}/api/ping");
387    let mut response = http_agent
388        .get(&ping_url)
389        .header("Authorization", &format!("Bearer {token}"))
390        .call()
391        .map_err(|e| handle_ureq_error(e, "Ping failed"))?;
392    let body = response
393        .body_mut()
394        .with_config()
395        .limit(MAX_HTTP_SMALL_RESPONSE_BYTES)
396        .read_to_string()
397        .map_err(|e| SynchronizationError::Other(e.to_string()))?;
398    let ping_response: PingResponse = serde_json::from_str(&body)
399        .map_err(|e| SynchronizationError::InvalidResponse(e.to_string()))?;
400    if ping_response.ok {
401        Ok(())
402    } else {
403        Err(SynchronizationError::Other(
404            "Server returned ok: false".to_string(),
405        ))
406    }
407}

409/// Ping a Fulgurant server with a progress indicator and a result notification.
410///
411/// ### Arguments
412/// - `profile`: The server profile to authenticate and ping.
413/// - `display_name`: Human-readable label shown in the progress/result notifications.
414/// - `window`: The window to attach the progress indicator to.
415/// - `cx`: The application context.
416pub fn perform_ping_with_progress(
417    profile: ServerProfile,
418    display_name: String,
419    window: &mut Window,
420    cx: &mut App,
421) {
422    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
423    let sync_state = shared.sync_state_for(&profile.id);
424    let pending_notification = sync_state.pending_notification.clone();
425    let token_state = Arc::clone(&sync_state.token_state);
426    let http_agent = Arc::clone(&shared.http_agent);
427
428    let done = Arc::new(AtomicBool::new(false));
429    let done_for_thread = Arc::clone(&done);
430
431    let progress = start_progress(
432        window,
433        cx,
434        format!("Testing connection to {display_name}...").into(),
435        None,
436    );
437
438    thread::spawn(move || {
439        let result = get_valid_token(&profile, &token_state, &http_agent).and_then(|token| {
440            match profile.server_url.as_deref() {
441                Some(url) => ping_server(url, &token, &http_agent),
442                None => Err(SynchronizationError::ServerUrlMissing),
443            }
444        });
445        let notification = match result {
446            Ok(()) => (
447                NotificationType::Success,
448                SharedString::from(format!("{display_name}: Server is reachable")),
449            ),
450            Err(e) => (
451                NotificationType::Error,
452                SharedString::from(format!("{display_name}: Ping failed: {e}")),
453            ),
454        };
455        *pending_notification.lock() = Some(notification);
456        done_for_thread.store(true, Ordering::Release);
457    });
458
459    window
460        .spawn(cx, async move |async_cx| {
461            let _progress = progress;
462            loop {
463                async_cx
464                    .background_executor()
465                    .timer(Duration::from_millis(100))
466                    .await;
467                if done.load(Ordering::Acquire) {
468                    break;
469                }
470            }
471        })
472        .detach();
473}

475/// Fetches shared files from each active profile's server and starts SSE
476/// connections for real-time updates.
477///
478/// ### Arguments
479/// - `entity`: The Fulgur entity
480/// - `cx`: The application context.
481pub fn begin_synchronization(entity: &gpui::Entity<crate::fulgur::Fulgur>, cx: &gpui::App) {
482    if !entity
483        .read(cx)
484        .settings
485        .app_settings
486        .synchronization_settings
487        .is_synchronization_activated
488    {
489        return;
490    }
491    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
492    if shared
493        .sync_initialized
494        .swap(true, std::sync::atomic::Ordering::SeqCst)
495    {
496        log::debug!("Sync already initialized by another window");
497        return;
498    }
499    log::info!("Initializing sync system");
500    let active_profiles: Vec<ServerProfile> = entity
501        .read(cx)
502        .settings
503        .app_settings
504        .synchronization_settings
505        .profiles
506        .iter()
507        .filter(|p| p.is_active)
508        .cloned()
509        .collect();
510    let http_agent = Arc::clone(&shared.http_agent);
511    let sse_http_agent = Arc::clone(&shared.sse_http_agent);
512    for profile in active_profiles {
513        let sync_state = shared.sync_state_for(&profile.id);
514        let (sse_tx, sse_shutdown_flag, sse_thread_handle) = {
515            let mut sse = sync_state.sse.lock();
516            let sse_tx = sse.sse_event_tx.clone();
517            let shutdown_flag = Arc::new(AtomicBool::new(false));
518            sse.sse_shutdown_flag = Some(Arc::clone(&shutdown_flag));
519            let thread_handle = Arc::clone(&sse.sse_thread_handle);
520            (sse_tx, Some(shutdown_flag), Some(thread_handle))
521        };
522        let http_agent_clone = Arc::clone(&http_agent);
523        let sse_http_agent_clone = Arc::clone(&sse_http_agent);
524        thread::spawn(move || {
525            run_profile_bootstrap(
526                &profile,
527                &sync_state,
528                sse_tx,
529                sse_shutdown_flag,
530                sse_thread_handle,
531                &http_agent_clone,
532                &sse_http_agent_clone,
533            );
534        });
535    }
536}

538/// Run the bootstrap sequence for a single profile in a background thread.
539///
540/// ### Arguments
541/// - `profile`: The profile being bootstrapped.
542/// - `sync_state`: Shared per-profile sync state.
543/// - `sse_tx`: Optional SSE event sender; `None` skips the SSE step.
544/// - `sse_shutdown_flag`: Shutdown flag signalled by `restart_sse_connection`.
545/// - `sse_thread_handle`: Slot for the SSE worker thread handle.
546/// - `http_agent`: Shared HTTP agent for short-lived REST calls.
547/// - `sse_http_agent`: Dedicated long-timeout HTTP agent for the SSE stream.
548fn run_profile_bootstrap(
549    profile: &ServerProfile,
550    sync_state: &Arc<SyncState>,
551    sse_tx: Option<std::sync::mpsc::Sender<crate::fulgur::sync::sse::SseEvent>>,
552    sse_shutdown_flag: Option<Arc<AtomicBool>>,
553    sse_thread_handle: Option<Arc<Mutex<Option<thread::JoinHandle<()>>>>>,
554    http_agent: &Arc<ureq::Agent>,
555    sse_http_agent: &Arc<ureq::Agent>,
556) {
557    // Small delay to ensure app initialization doesn't block
558    thread::sleep(Duration::from_millis(100));
559    let key = match load_device_api_key_from_keychain(&profile.id) {
560        Ok(value) => value,
561        Err(e) => {
562            log::error!(
563                "Profile '{}': failed to load device API key from keychain: {e}",
564                profile.name
565            );
566            set_sync_server_connection_status(
567                &sync_state.connection_status,
568                SynchronizationStatus::Disconnected,
569            );
570            return;
571        }
572    };
573    if profile.server_url.is_none() || profile.email.is_none() || key.is_none() {
574        set_sync_server_connection_status(
575            &sync_state.connection_status,
576            SynchronizationStatus::Disconnected,
577        );
578        return;
579    }
580    match initial_synchronization(profile, &sync_state.token_state, http_agent) {
581        Ok(begin_response) => {
582            log::info!("Profile '{}': connected to sync server", profile.name);
583            set_sync_server_connection_status(
584                &sync_state.connection_status,
585                SynchronizationStatus::Connected,
586            );
587            store_server_max_file_size(
588                &sync_state.max_file_size_bytes,
589                begin_response.max_file_size_bytes,
590            );
591            {
592                let mut device_name = sync_state.device_name.lock();
593                *device_name = Some(begin_response.device_name);
594            }
595            {
596                let mut files = sync_state.pending_shared_files.lock();
597                *files = begin_response.shares;
598            }
599            if let (Some(tx), Some(shutdown), Some(handle_storage)) =
600                (sse_tx, sse_shutdown_flag, sse_thread_handle)
601            {
602                log::info!(
603                    "Profile '{}': starting SSE connection for real-time updates",
604                    profile.name
605                );
606                let agents = SseAgents {
607                    rest: Arc::clone(http_agent),
608                    stream: Arc::clone(sse_http_agent),
609                };
610                let share_state = SseShareState {
611                    pending_shared_files: Arc::clone(&sync_state.pending_shared_files),
612                    max_file_size_bytes: Arc::clone(&sync_state.max_file_size_bytes),
613                };
614                match connect_sse(
615                    profile,
616                    tx,
617                    shutdown,
618                    sync_state.connection_status.clone(),
619                    &sync_state.token_state,
620                    &agents,
621                    &share_state,
622                ) {
623                    Ok(handle) => {
624                        *handle_storage.lock() = Some(handle);
625                    }
626                    Err(e) => {
627                        log::error!("Profile '{}': failed to start SSE: {e}", profile.name);
628                    }
629                }
630            } else {
631                log::warn!(
632                    "Profile '{}': SSE channels not available, skipping SSE start",
633                    profile.name
634                );
635            }
636        }
637        Err(e) => {
638            log::error!(
639                "Profile '{}': initial synchronization failed: {e}",
640                profile.name
641            );
642            set_sync_server_connection_status(
643                &sync_state.connection_status,
644                SynchronizationStatus::Disconnected,
645            );
646        }
647    }
648}

662/// Perform initial synchronization with a single profile's server in a
663/// background thread.
664///
665/// Sets the connection status to `Connecting` immediately, then spawns a background
666/// thread to perform the actual network call. The UI remains responsive while the
667/// connection is in progress.
668///
669/// ### Arguments
670/// - `profile`: The server profile to synchronize with.
671/// - `cx`: The application context (used to obtain shared state).
672pub fn perform_initial_synchronization(profile: ServerProfile, cx: &mut App) {
673    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
674    let sync_state = shared.sync_state_for(&profile.id);
675    set_sync_server_connection_status(
676        &sync_state.connection_status,
677        SynchronizationStatus::Connecting,
678    );
679    *sync_state.connecting_since.lock() = Some(Instant::now());
680    let token_state = Arc::clone(&sync_state.token_state);
681    let http_agent = Arc::clone(&shared.http_agent);
682    let profile_name = profile.name.clone();
683    let connection_status = sync_state.connection_status.clone();
684    let connecting_since = sync_state.connecting_since.clone();
685    let device_name = sync_state.device_name.clone();
686    let pending_shared_files = sync_state.pending_shared_files.clone();
687    let pending_notification = sync_state.pending_notification.clone();
688    let max_file_size_bytes = sync_state.max_file_size_bytes.clone();
689    thread::spawn(move || {
690        let result = initial_synchronization(&profile, &token_state, &http_agent);
691        let (notification, status) = match result {
692            Ok(begin_response) => {
693                store_server_max_file_size(
694                    &max_file_size_bytes,
695                    begin_response.max_file_size_bytes,
696                );
697                {
698                    let mut name = device_name.lock();
699                    *name = Some(begin_response.device_name.clone());
700                }
701                {
702                    let mut files = pending_shared_files.lock();
703                    *files = begin_response.shares;
704                }
705                (
706                    (
707                        NotificationType::Success,
708                        SharedString::from(format!(
709                            "{profile_name}: Connection successful as {}",
710                            begin_response.device_name
711                        )),
712                    ),
713                    SynchronizationStatus::Connected,
714                )
715            }
716            Err(e) => (
717                (
718                    NotificationType::Error,
719                    SharedString::from(format!("{profile_name}: Connection failed: {e}")),
720                ),
721                SynchronizationStatus::from_error(&e),
722            ),
723        };
724        set_sync_server_connection_status(&connection_status, status);
725        *connecting_since.lock() = None;
726        *pending_notification.lock() = Some(notification);
727    });
728}

730/// Perform initial synchronization with a single profile's server, showing
731/// a progress spinner.
732///
733/// ### Arguments
734/// - `profile`: The server profile to synchronize with.
735/// - `window`: Target window for the progress notification.
736/// - `cx`: The application context.
737pub fn perform_initial_synchronization_with_progress(
738    profile: ServerProfile,
739    window: &mut Window,
740    cx: &mut App,
741) {
742    let shared = cx.global::<crate::fulgur::shared_state::SharedAppState>();
743    let sync_state = shared.sync_state_for(&profile.id);
744    set_sync_server_connection_status(
745        &sync_state.connection_status,
746        SynchronizationStatus::Connecting,
747    );
748    *sync_state.connecting_since.lock() = Some(Instant::now());
749    let token_state = Arc::clone(&sync_state.token_state);
750    let http_agent = Arc::clone(&shared.http_agent);
751    let profile_name = profile.name.clone();
752    let connection_status = sync_state.connection_status.clone();
753    let connecting_since = sync_state.connecting_since.clone();
754    let device_name = sync_state.device_name.clone();
755    let pending_shared_files = sync_state.pending_shared_files.clone();
756    let pending_notification = sync_state.pending_notification.clone();
757    let max_file_size_bytes = sync_state.max_file_size_bytes.clone();
758
759    let done = Arc::new(AtomicBool::new(false));
760    let done_for_thread = Arc::clone(&done);
761
762    let cancel_status = connection_status.clone();
763    let cancel_connecting_since = connecting_since.clone();
764    let cancel_callback: Option<CancelCallback> = Some(Box::new(move |_window, _cx| {
765        set_sync_server_connection_status(&cancel_status, SynchronizationStatus::Disconnected);
766        *cancel_connecting_since.lock() = None;
767    }));
768
769    let progress = start_progress(
770        window,
771        cx,
772        format!("Connecting to {profile_name}...").into(),
773        cancel_callback,
774    );
775    let cancel_flag = progress.cancel_flag();
776    let cancel_flag_for_thread = Arc::clone(&cancel_flag);
777
778    thread::spawn(move || {
779        let result = initial_synchronization(&profile, &token_state, &http_agent);
780
781        if cancel_flag_for_thread.load(Ordering::Acquire) {
782            done_for_thread.store(true, Ordering::Release);
783            return;
784        }
785
786        let (notification, status) = match result {
787            Ok(begin_response) => {
788                store_server_max_file_size(
789                    &max_file_size_bytes,
790                    begin_response.max_file_size_bytes,
791                );
792                {
793                    let mut name = device_name.lock();
794                    *name = Some(begin_response.device_name.clone());
795                }
796                {
797                    let mut files = pending_shared_files.lock();
798                    *files = begin_response.shares;
799                }
800                (
801                    (
802                        NotificationType::Success,
803                        SharedString::from(format!(
804                            "{profile_name}: Connection successful as {}",
805                            begin_response.device_name
806                        )),
807                    ),
808                    SynchronizationStatus::Connected,
809                )
810            }
811            Err(e) => (
812                (
813                    NotificationType::Error,
814                    SharedString::from(format!("{profile_name}: Connection failed: {e}")),
815                ),
816                SynchronizationStatus::from_error(&e),
817            ),
818        };
819        set_sync_server_connection_status(&connection_status, status);
820        *connecting_since.lock() = None;
821        *pending_notification.lock() = Some(notification);
822        done_for_thread.store(true, Ordering::Release);
823    });
824
825    window
826        .spawn(cx, async move |async_cx| {
827            let _progress = progress;
828            loop {
829                async_cx
830                    .background_executor()
831                    .timer(Duration::from_millis(100))
832                    .await;
833                if done.load(Ordering::Acquire) || cancel_flag.load(Ordering::Acquire) {
834                    break;
835                }
836            }
837        })
838        .detach();
839}

911#[derive(Clone, Copy)]
912pub enum SynchronizationStatus {
913    Connected,
914    Connecting,
915    Disconnected,
916    AuthenticationFailed,
917    ConnectionFailed,
918    Other,
919    NotActivated,
920}
921
922impl SynchronizationStatus {
923    /// Convert the error to a synchronization status
924    ///
925    /// ### Arguments
926    /// - `error`: The error
927    ///
928    /// ### Returns
929    /// - `SynchronizationStatus`: The synchronization status
930    pub fn from_error(error: &SynchronizationError) -> SynchronizationStatus {
931        match error {
932            SynchronizationError::AuthenticationFailed => {
933                SynchronizationStatus::AuthenticationFailed
934            }
935            SynchronizationError::HostNotFound
936            | SynchronizationError::ConnectionFailed
937            | SynchronizationError::Timeout(_) => SynchronizationStatus::ConnectionFailed,
938            _ => SynchronizationStatus::Other,
939        }
940    }
941
942    /// Check if the synchronization status is connected
943    ///
944    /// ### Returns
945    /// - `true` if the synchronization status is connected, `false` otherwise
946    pub fn is_connected(&self) -> bool {
947        match self {
948            SynchronizationStatus::Connected => true,
949            SynchronizationStatus::Connecting
950            | SynchronizationStatus::Disconnected
951            | SynchronizationStatus::AuthenticationFailed
952            | SynchronizationStatus::ConnectionFailed
953            | SynchronizationStatus::Other
954            | SynchronizationStatus::NotActivated => false,
955        }
956    }
957
958    /// Check if the synchronization status is connecting
959    ///
960    /// ### Returns
961    /// - `true` if the synchronization status is connecting, `false` otherwise
962    pub fn is_connecting(&self) -> bool {
963        matches!(self, SynchronizationStatus::Connecting)
964    }
965
966    /// Return a short human-readable label for display in tooltips and status pills.
967    ///
968    /// ### Returns
969    /// - `&'static str`: One of "Connected", "Connecting", "Disconnected", "Inactive", or "Error".
970    pub fn label(self) -> &'static str {
971        match self {
972            SynchronizationStatus::Connected => "Connected",
973            SynchronizationStatus::Connecting => "Connecting",
974            SynchronizationStatus::Disconnected => "Disconnected",
975            SynchronizationStatus::NotActivated => "Inactive",
976            SynchronizationStatus::AuthenticationFailed
977            | SynchronizationStatus::ConnectionFailed
978            | SynchronizationStatus::Other => "Error",
979        }
980    }
981}

983impl Fulgur {
984    /// Process shared files received from every active sync profile.
985    ///
986    /// ### Arguments
987    /// - `window`: The window to create new tabs in
988    /// - `cx`: The application context
989    pub fn process_shared_files_from_sync(&mut self, window: &mut Window, cx: &mut Context<Self>) {
990        let active_profiles: Vec<(crate::fulgur::settings::ProfileId, String)> = self
991            .settings
992            .app_settings
993            .synchronization_settings
994            .profiles
995            .iter()
996            .filter(|p| p.is_active)
997            .map(|p| (p.id.clone(), p.name.clone()))
998            .collect();
999        for (profile_id, profile_name) in active_profiles {
1000            let sync_state = Fulgur::shared_state(cx).sync_state_for(&profile_id);
1001            let shared_files_to_open =
1002                if let Some(mut pending) = sync_state.pending_shared_files.try_lock() {
1003                    if pending.is_empty() {
1004                        Vec::new()
1005                    } else {
1006                        log::info!(
1007                            "Processing {} shared file(s) for profile {profile_id}",
1008                            pending.len()
1009                        );
1010                        pending.drain(..).collect()
1011                    }
1012                } else {
1013                    Vec::new()
1014                };
1015            if shared_files_to_open.is_empty() {
1016                *sync_state.last_share_receive_error_signature.lock() = None;
1017                continue;
1018            }
1019            let server_max_size = sync_state
1020                .max_file_size_bytes
1021                .load(std::sync::atomic::Ordering::Acquire);
1022            let mut shared_files_iter = shared_files_to_open.into_iter();
1023            let mut retry_queue = Vec::new();
1024            let mut key_unavailable = false;
1025            let mut key_load_failed = false;
1026            let mut decrypt_failures = 0usize;
1027            let mut opened_files = 0usize;
1028
1029            while let Some(shared_file) = shared_files_iter.next() {
1030                if server_max_size != u64::MAX
1031                    && shared_file.content.len() as u64 > server_max_size.saturating_mul(2)
1032                {
1033                    log::warn!(
1034                        "Skipping shared file '{}' from device {}: encrypted payload ({} bytes) exceeds the server max ({} bytes)",
1035                        shared_file.file_name,
1036                        shared_file.source_device_id,
1037                        shared_file.content.len(),
1038                        server_max_size
1039                    );
1040                    continue;
1041                }
1042
1043                let encryption_key = match load_private_key_from_keychain(&profile_id) {
1044                    Ok(Some(key)) => key,
1045                    Ok(None) => {
1046                        key_unavailable = true;
1047                        log::warn!(
1048                            "Deferring {} shared file(s) for profile {profile_id}: encryption key is unavailable",
1049                            1 + shared_files_iter.len()
1050                        );
1051                        retry_queue.push(shared_file);
1052                        retry_queue.extend(shared_files_iter);
1053                        break;
1054                    }
1055                    Err(e) => {
1056                        key_load_failed = true;
1057                        log::warn!(
1058                            "Deferring {} shared file(s) for profile {profile_id}: failed to load encryption key from keychain: {e}",
1059                            1 + shared_files_iter.len()
1060                        );
1061                        retry_queue.push(shared_file);
1062                        retry_queue.extend(shared_files_iter);
1063                        break;
1064                    }
1065                };
1066
1067                let decrypted_result =
1068                    crypto_helper::decrypt_bytes(&shared_file.content, encryption_key.as_str())
1069                        .and_then(|compressed_bytes| {
1070                            share::decompress_content(&compressed_bytes, server_max_size)
1071                        });
1072                drop(encryption_key);
1073
1074                match decrypted_result {
1075                    Ok(decrypted_content) => {
1076                        let tab_id = self.next_tab_id;
1077                        self.next_tab_id += 1;
1078                        let new_tab = Tab::Editor(editor_tab::EditorTab::from_content(
1079                            tab_id,
1080                            &decrypted_content,
1081                            shared_file.file_name.clone(),
1082                            window,
1083                            cx,
1084                            &self.settings.editor_settings,
1085                        ));
1086                        self.tabs.push(new_tab);
1087                        self.active_tab_index = Some(self.tabs.len() - 1);
1088                        self.pending_tab_scroll = Some(self.tabs.len() - 1);
1089                        opened_files += 1;
1090                        log::info!("Opened shared file: {}", shared_file.file_name);
1091                    }
1092                    Err(e) => {
1093                        decrypt_failures += 1;
1094                        log::warn!(
1095                            "Deferring shared file '{}' for profile {profile_id}: decryption failed ({e})",
1096                            shared_file.file_name
1097                        );
1098                        retry_queue.push(shared_file);
1099                    }
1100                }
1101            }
1102
1103            let mut retry_count = 0usize;
1104            if !retry_queue.is_empty() {
1105                retry_count = retry_queue.len();
1106                let mut pending = sync_state.pending_shared_files.lock();
1107                retry_queue.extend(std::mem::take(&mut *pending));
1108                *pending = retry_queue;
1109                log::warn!(
1110                    "Re-queued {retry_count} shared file(s) for profile {profile_id} for retry"
1111                );
1112            }
1113
1114            let error_notification = if key_unavailable {
1115                Some((
1116                    "missing-keychain-private-key",
1117                    SharedString::from(format!(
1118                        "{profile_name}: Cannot receive shared files because the encryption key is unavailable in the keychain. Fulgur will retry automatically."
1119                    )),
1120                ))
1121            } else if key_load_failed {
1122                Some((
1123                    "failed-to-load-keychain-private-key",
1124                    SharedString::from(format!(
1125                        "{profile_name}: Cannot receive shared files because the encryption key could not be loaded from the keychain. Fulgur will retry automatically."
1126                    )),
1127                ))
1128            } else if decrypt_failures > 0 {
1129                Some((
1130                    "share-decryption-failed",
1131                    SharedString::from(format!(
1132                        "{profile_name}: Failed to decrypt {decrypt_failures} shared file(s). Fulgur will retry automatically."
1133                    )),
1134                ))
1135            } else {
1136                None
1137            };
1138
1139            if let Some((signature, message)) = error_notification {
1140                let mut last_signature = sync_state.last_share_receive_error_signature.lock();
1141                if last_signature.as_deref() != Some(signature) {
1142                    *sync_state.pending_notification.lock() =
1143                        Some((NotificationType::Error, message));
1144                    *last_signature = Some(signature.to_string());
1145                }
1146            } else if opened_files > 0 || retry_count == 0 {
1147                *sync_state.last_share_receive_error_signature.lock() = None;
1148            }
1149        }
1150    }
1151}
