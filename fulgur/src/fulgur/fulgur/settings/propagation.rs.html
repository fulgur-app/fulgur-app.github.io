1use crate::fulgur::{Fulgur, shared_state::SharedAppState};
2use gpui::{BorrowAppContext, Context, Window};
3use gpui_component::notification::NotificationType;
4
5impl Fulgur {
6    /// Update settings and propagate to all windows
7    ///
8    /// This method should be called whenever settings are changed. It will:
9    /// 1. Save settings to disk
10    /// 2. Publish the settings to `SharedAppState` via `cx.update_global`
11    ///
12    /// Every window (including this one) observes the global with
13    /// `observe_global_in` and applies the new editor settings to its tabs
14    /// from that observer.
15    ///
16    /// ### Arguments
17    /// - `cx`: The application context
18    ///
19    /// ### Errors
20    /// Returns an error if persisting the settings to disk fails. The shared
21    /// state update and re-render still happen even when saving fails.
22    ///
23    /// ### Returns
24    /// - `anyhow::Result<()>`: Result of the operation
25    pub fn update_and_propagate_settings(&mut self, cx: &mut Context<Self>) -> anyhow::Result<()> {
26        // Apply log level immediately so the change takes effect in this session.
27        crate::fulgur::utils::logger::set_debug_mode(self.settings.app_settings.debug_mode);
28
29        // Save settings to disk
30        if let Err(e) = self.settings.save() {
31            log::error!("Failed to save settings: {e}");
32            Fulgur::shared_state(cx).notify((
33                NotificationType::Error,
34                format!("Failed to save settings: {e}").into(),
35            ));
36            return Err(e);
37        }
38
39        let settings = self.settings.clone();
40        cx.update_global::<SharedAppState, _>(|shared, _| {
41            shared.settings = settings;
42        });
43
44        Ok(())
45    }
46
47    /// Apply the current editor settings to every tab in this window
48    ///
49    /// ### Arguments
50    /// - `window`: The window containing the tabs
51    /// - `cx`: The application context
52    pub fn apply_editor_settings_to_tabs(&mut self, window: &mut Window, cx: &mut Context<Self>) {
53        let settings = self.settings.editor_settings.clone();
54        for tab in self.tabs.clone() {
55            tab.update(cx, |tab, cx| {
56                tab.update_settings(window, cx, &settings);
57            });
58        }
59    }
60}
