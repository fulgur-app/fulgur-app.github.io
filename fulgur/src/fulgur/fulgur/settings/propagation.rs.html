1use crate::fulgur::{Fulgur, shared_state::SharedAppState, ui::tabs::tab::Tab};
2use gpui::{BorrowAppContext, Context, Window};
3use gpui_component::notification::NotificationType;
4
5impl Fulgur {
6    /// Update settings and propagate to all windows
7    ///
8    /// This method should be called whenever settings are changed. It will:
9    /// 1. Save settings to disk
10    /// 2. Publish the settings to `SharedAppState` via `cx.update_global`
11    ///
12    /// ### Arguments
13    /// - `cx`: The application context
14    ///
15    /// ### Errors
16    /// Returns an error if persisting the settings to disk fails. The shared
17    /// state update and re-render still happen even when saving fails.
18    ///
19    /// ### Returns
20    /// - `anyhow::Result<()>`: Result of the operation
21    pub fn update_and_propagate_settings(&mut self, cx: &mut Context<Self>) -> anyhow::Result<()> {
22        // Apply log level immediately so the change takes effect in this session.
23        crate::fulgur::utils::logger::set_debug_mode(self.settings.app_settings.debug_mode);
24
25        // Save settings to disk
26        if let Err(e) = self.settings.save() {
27            log::error!("Failed to save settings: {e}");
28            Fulgur::shared_state(cx).notify((
29                NotificationType::Error,
30                format!("Failed to save settings: {e}").into(),
31            ));
32            return Err(e);
33        }
34
35        // Mark settings as changed for this window so the next render pushes
36        // the new editor settings into this window's tabs.
37        self.settings_changed = true;
38
39        let settings = self.settings.clone();
40        cx.update_global::<SharedAppState, _>(|shared, _| {
41            shared.settings = settings;
42        });
43
44        Ok(())
45    }
46
47    /// Propagate settings changes to tabs
48    ///
49    /// ### Arguments
50    /// - `window`: The window containing the tabs
51    /// - `cx`: The application context
52    pub fn propagate_settings_to_tabs(&mut self, window: &mut Window, cx: &mut Context<Self>) {
53        if !self.tabs_pending_update.is_empty() {
54            let settings = self.settings.editor_settings.clone();
55            for tab_id in self.tabs_pending_update.drain().collect::<Vec<_>>() {
56                if let Some(Tab::Editor(editor_tab)) =
57                    self.tabs.iter_mut().find(|tab| tab.id() == tab_id)
58                {
59                    editor_tab.update_settings(window, cx, &settings);
60                }
61            }
62        }
63        if self.settings_changed {
64            let settings = self.settings.editor_settings.clone();
65            for tab_id in self.rendered_tabs.iter().copied().collect::<Vec<_>>() {
66                if let Some(Tab::Editor(editor_tab)) =
67                    self.tabs.iter_mut().find(|tab| tab.id() == tab_id)
68                {
69                    editor_tab.update_settings(window, cx, &settings);
70                }
71            }
72            self.settings_changed = false;
73        }
74    }
75
76    /// Track newly rendered tabs and mark them for settings update
77    ///
78    /// ### Arguments
79    /// - `cx`: The application context
80    pub fn track_newly_rendered_tabs(&mut self, cx: &mut Context<Self>) {
81        if let Some(tab_id) = self.active_tab_id {
82            let is_newly_rendered = !self.rendered_tabs.contains(&tab_id);
83            self.rendered_tabs.insert(tab_id);
84            if is_newly_rendered {
85                self.tabs_pending_update.insert(tab_id);
86                cx.notify();
87            }
88        }
89    }
