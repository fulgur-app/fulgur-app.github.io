6    /// Update settings and propagate to all windows
7    ///
8    /// This method should be called whenever settings are changed. It will:
9    /// 1. Save settings to disk
10    /// 2. Update shared settings in SharedAppState
11    /// 3. Increment the shared settings version (so other windows detect the change)
12    /// 4. Set settings_changed flag for this window
13    /// 5. Force all windows to re-render immediately
14    ///
15    /// ### Arguments
16    /// - `cx`: The application context
17    ///
18    /// ### Returns
19    /// - `anyhow::Result<()>`: Result of the operation
20    pub fn update_and_propagate_settings(&mut self, cx: &mut Context<Self>) -> anyhow::Result<()> {
21        // Apply log level immediately so the change takes effect in this session.
22        crate::fulgur::utils::logger::set_debug_mode(self.settings.app_settings.debug_mode);
23
24        // Save settings to disk
25        if let Err(e) = self.settings.save() {
26            log::error!("Failed to save settings: {}", e);
27            self.pending_notification = Some((
28                NotificationType::Error,
29                format!("Failed to save settings: {}", e).into(),
30            ));
31            return Err(e);
32        }
33
34        // Update shared settings
35        let shared = self.shared_state(cx);
36        *shared.settings.lock() = self.settings.clone();
37
38        // Increment the version counter so other windows detect the change
39        let new_version = shared
40            .settings_version
41            .fetch_add(1, std::sync::atomic::Ordering::SeqCst)
42            + 1;
43        self.local_settings_version = new_version;
44
45        // Mark settings as changed for this window
46        self.settings_changed = true;
47
48        log::debug!(
49            "Window {:?} updated settings to version {}, notifying other windows",
50            self.window_id,
51            new_version
52        );
53
54        // Force other windows to re-render immediately
55        // (Skip the current window to avoid reentrancy issues - it will re-render naturally)
56        let current_window_id = self.window_id;
57        let window_manager = cx.global::<window_manager::WindowManager>();
58        let all_windows = window_manager.get_all_windows();
59
60        // Defer notifications to avoid reentrancy issues
61        cx.defer(move |cx| {
62            for weak_window in all_windows.iter() {
63                if let Some(window_entity) = weak_window.upgrade() {
64                    // Skip the current window (already updating)
65                    let should_notify = window_entity.read(cx).window_id != current_window_id;
66                    if should_notify {
67                        window_entity.update(cx, |_, cx| {
68                            cx.notify();
69                        });
70                    }
71                }
72            }
73        });
74
75        Ok(())
76    }

78    /// Synchronize settings from other windows
79    ///
80    /// ### Arguments
81    /// - `cx`: The application context
82    pub fn synchronize_settings_from_other_windows(&mut self, cx: &mut Context<Self>) {
83        let shared = self.shared_state(cx);
84        let shared_version = shared
85            .settings_version
86            .load(std::sync::atomic::Ordering::Relaxed);
87        if shared_version > self.local_settings_version {
88            // Settings have been updated in another window - reload them
89            let shared_settings = shared.settings.lock().clone();
90            self.settings = shared_settings;
91            self.local_settings_version = shared_version;
92            self.settings_changed = true;
93            log::debug!(
94                "Window {:?} detected settings change from another window (version {} -> {})",
95                self.window_id,
96                self.local_settings_version,
97                shared_version
98            );
99        }
100    }

102    /// Propagate settings changes to tabs
103    ///
104    /// ### Arguments
105    /// - `window`: The window containing the tabs
106    /// - `cx`: The application context
107    pub fn propagate_settings_to_tabs(&mut self, window: &mut Window, cx: &mut Context<Self>) {
108        if !self.tabs_pending_update.is_empty() {
109            let settings = self.settings.editor_settings.clone();
110            for tab_index in self.tabs_pending_update.drain() {
111                if let Some(Tab::Editor(editor_tab)) = self.tabs.get_mut(tab_index) {
112                    editor_tab.update_settings(window, cx, &settings);
113                }
114            }
115        }
116        if self.settings_changed {
117            let settings = self.settings.editor_settings.clone();
118            for tab_index in self.rendered_tabs.iter().copied().collect::<Vec<_>>() {
119                if let Some(Tab::Editor(editor_tab)) = self.tabs.get_mut(tab_index) {
120                    editor_tab.update_settings(window, cx, &settings);
121                }
122            }
123            self.settings_changed = false;
124        }
125    }

127    /// Track newly rendered tabs and mark them for settings update
128    ///
129    /// ### Arguments
130    /// - `cx`: The application context
131    pub fn track_newly_rendered_tabs(&mut self, cx: &mut Context<Self>) {
132        if let Some(index) = self.active_tab_index {
133            let is_newly_rendered = !self.rendered_tabs.contains(&index);
134            self.rendered_tabs.insert(index);
135            if is_newly_rendered {
136                self.tabs_pending_update.insert(index);
137                cx.notify();
138            }
139        }
140    }
